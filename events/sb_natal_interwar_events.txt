# Apply the restoration package from an event option only after the engine has
# finished creating the secession country. Applying it directly in
# on_secession_start leaves the inherited colonial government in place.
sb_natal_interwar.039 = {
	type = country_event
	placement = root

	title = sb_natal_interwar.039.t
	desc = sb_natal_interwar.039.d
	flavor = sb_natal_interwar.039.f

	event_image = { video = "gfx/event_pictures/africa_leader_arguing.bk2" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 3

	trigger = {
		country_definition = cd:ZUL
		is_country_alive = yes
		is_secessionist = yes
		has_variable = sb_natal_zulu_restoration_active_var
		c:NAL ?= {
			is_country_alive = yes
			has_variable = sb_natal_zulu_restoration_active_var
		}
	}

	option = {
		name = sb_natal_interwar.039.a
		default_option = yes
		custom_tooltip = sb_natal_interwar.039.a.tt
		hidden_effect = {
			sb_natal_prepare_zulu_restoration_secession = yes
			# Queue the appeal only after the engine-created government and dynasty
			# have been normalized, so the military bargain cannot race this event.
			if = {
				limit = {
					sb_natal_zulu_secession_controls_full_zululand = yes
					c:TRN ?= { is_country_alive = yes }
					c:NAL ?= {
						is_country_alive = yes
						NOT = { has_variable = sb_natal_zulu_arms_appeal_sent_var }
					}
				}
				c:NAL = {
					set_variable = sb_natal_zulu_arms_appeal_sent_var
					set_variable = {
						name = sb_natal_zulu_arms_appeal_wait_var
						days = 30
					}
				}
				c:TRN = { trigger_event = { id = sb_natal_interwar.040 days = 2 popup = yes } }
			}
		}
	}
}
