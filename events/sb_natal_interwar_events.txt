sb_natal_interwar.001 = {
	type = country_event
	placement = root

	title = sb_natal_interwar.001.t
	desc = sb_natal_interwar.001.d
	flavor = sb_natal_interwar.001.f

	event_image = { video = "gfx/event_pictures/unspecific_signed_contract.bk2" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3

	trigger = {
		has_variable = sb_natal_indenture_pending_var
		sb_natal_is_british_colony = yes
		NOT = { has_variable = sb_natal_indenture_completed_var }
		sb_natal_british_network_has_indenture_origin = yes
		s:STATE_NATAL.region_state:NAL ?= {
			state_is_eligible_as_mass_migration_target = yes
		}
	}

	cancellation_trigger = {
		OR = {
			NOT = { sb_natal_is_british_colony = yes }
			has_variable = sb_natal_indenture_completed_var
			NOT = { sb_natal_british_network_has_indenture_origin = yes }
			NOT = {
				s:STATE_NATAL.region_state:NAL ?= {
					state_is_eligible_as_mass_migration_target = yes
				}
			}
		}
	}

	immediate = {
		remove_variable = sb_natal_indenture_pending_var
		# Match vanilla South American migration events: create before the option is shown.
		sb_natal_select_indenture_origin_and_create_migration = yes
	}

	option = {
		name = sb_natal_interwar.001.a
		default_option = yes
		custom_tooltip = sb_natal_indenture_migration_tt
	}
}

sb_natal_interwar.010 = {
	type = country_event
	placement = root

	title = sb_natal_interwar.010.t
	desc = sb_natal_interwar.010.d
	flavor = sb_natal_interwar.010.f

	event_image = { video = "gfx/event_pictures/africa_public_protest.bk2" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_protest.dds"
	duration = 3

	trigger = {
		has_variable = sb_natal_civil_rights_petition_pending_var
		sb_natal_has_responsible_government = yes
		has_law = law_type:law_racial_segregation
		sb_natal_has_civil_rights_movement = yes
		OR = {
			NOT = { has_variable = sb_natal_civil_rights_petition_requires_gandhi_var }
			any_scope_character = {
				has_template = BIC_mohandas_karamchand_gandhi
				is_character_alive = yes
			}
		}
	}

	cancellation_trigger = {
		OR = {
			NOT = { sb_natal_has_responsible_government = yes }
			NOT = { has_law = law_type:law_racial_segregation }
			NOT = { sb_natal_has_civil_rights_movement = yes }
			AND = {
				has_variable = sb_natal_civil_rights_petition_requires_gandhi_var
				NOT = {
					any_scope_character = {
						has_template = BIC_mohandas_karamchand_gandhi
						is_character_alive = yes
					}
				}
			}
		}
	}

	immediate = {
		remove_variable = sb_natal_civil_rights_petition_pending_var
		if = {
			limit = { has_variable = sb_natal_civil_rights_petition_requires_gandhi_var }
			remove_variable = sb_natal_civil_rights_petition_requires_gandhi_var
		}
	}

	option = {
		name = sb_natal_interwar.010.a
		custom_tooltip = sb_natal_indian_civic_rights_accept_tt
		hidden_effect = {
			set_variable = sb_natal_indian_civic_rights_accepted_var
			set_variable = sb_natal_civil_rights_resolved_var
			active_law:lawgroup_citizenship ?= {
				add_amendment = {
					type = amendment_sb_natal_indian_civic_rights
					sponsor = PREV.ig:ig_intelligentsia
				}
			}
			sb_natal_make_indian_pops_loyalist_10 = yes
		}
		ai_chance = {
			base = 20
			modifier = {
				trigger = { sb_frontier_ai_behavior_strict_historical = yes }
				add = -1000
			}
		}
	}

	option = {
		name = sb_natal_interwar.010.b
		default_option = yes
		custom_tooltip = sb_natal_indian_civic_rights_refuse_tt
		hidden_effect = { sb_natal_make_indian_pops_radical_20 = yes }
		ai_chance = {
			base = 80
			modifier = {
				trigger = { sb_frontier_ai_behavior_strict_historical = yes }
				add = 1000
			}
		}
	}
}
