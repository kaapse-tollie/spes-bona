		c:PHL ?= {
			is_country_alive = no
		}

			is_active_in_diplomatic_play = no
		}
	}

	immediate = {
		set_variable = sb_ora_bst_1856_launch_lease_var
		create_diplomatic_play = { type = dp_annex_war target_country = c:BST }
		if = {
			limit = { has_variable = sb_ora_bst_1856_launch_lease_var }
			remove_variable = sb_ora_bst_1856_launch_lease_var
			sb_cleanup_ora_bst_1856_play_support = yes
		}

		}
	}

			create_diplomatic_play = {
				type = dp_annex_war
				target_country = c:GZA
			}

			# Keep the follow-up marker so the existing monthly route can retry.

				limit = { has_variable = sb_gaza_zulu_launch_lease_var }
				remove_variable = sb_gaza_zulu_launch_lease_var
			}
		}
	}

sb_frontier_ai_wars.100 = {
	type = country_event
	hidden = yes

	# Receipt is permissive for temporary war/play/truce states. The immediate
	# handler retries them and only cancels permanently invalid authority.
	trigger = {
		country_definition = cd:XHO
		is_country_alive = yes
		has_variable = sb_xhosa_delivery_pending_var
		has_variable = sb_xhosa_delivery_target_scope
		has_variable = sb_xhosa_delivery_generation_var
		var:sb_xhosa_delivery_generation_var = 7
	}

	immediate = {
		var:sb_xhosa_delivery_target_scope = { save_temporary_scope_as = sb_xhosa_delivery_target }
		if = {
			limit = {
				any_scope_state = { state_region = s:STATE_EASTERN_CAPE num_provinces >= 1 }
				NOT = { sb_xho_owns_all_thembu_frontier_provinces = yes }
				scope:sb_xhosa_delivery_target ?= {
					is_country_alive = yes
					OR = { country_definition = cd:CAP country_definition = cd:ABY }
					has_variable = sb_xhosa_frontier_war_7_scheduled_var
					has_technology_researched = colonization
					any_scope_state = { state_region = s:STATE_EASTERN_CAPE has_state_trait = state_trait_sb_xhosa_resistance }
				}
			}
			if = {
				limit = { scope:sb_xhosa_delivery_target ?= { sb_xhosa_frontier_war_target_is_launchable = yes } }
				set_variable = { name = sb_xhosa_story_target_scope value = scope:sb_xhosa_delivery_target }
				set_variable = { name = sb_xhosa_story_generation_var value = 7 }
				set_variable = sb_xhosa_story_launch_lease_var
				create_diplomatic_play = { type = dp_sb_xhosa_war_7 target_country = scope:sb_xhosa_delivery_target }
				if = {
					limit = { has_variable = sb_xhosa_story_launch_lease_var }
					remove_variable = sb_xhosa_story_launch_lease_var
					remove_variable = sb_xhosa_story_target_scope
					remove_variable = sb_xhosa_story_generation_var
					trigger_event = { id = sb_frontier_ai_wars.100 days = 30 }
				}
			}
			else = { trigger_event = { id = sb_frontier_ai_wars.100 days = 30 } }
		}
		else = {
			scope:sb_xhosa_delivery_target ?= {
				if = { limit = { has_variable = sb_xhosa_frontier_war_7_scheduled_var } remove_variable = sb_xhosa_frontier_war_7_scheduled_var }
				set_variable = sb_xhosa_frontier_war_7_resolved_var
			}
			remove_variable = sb_xhosa_delivery_pending_var
			remove_variable = sb_xhosa_delivery_target_scope
			remove_variable = sb_xhosa_delivery_generation_var
		}
	}
}
