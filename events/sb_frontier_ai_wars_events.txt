sb_frontier_ai_wars.010 = {
	type = country_event
	hidden = yes

	# Not needed called externally
	# trigger = {
	# 	country_definition = cd:ORA
	# 	is_ai = yes
	# 	is_country_alive = yes
	# 	is_subject = no
	# 	is_at_war = no
	# 	is_active_in_diplomatic_play = no
	# 	aggressive_diplomatic_plays_permitted = yes
	# 	sb_frontier_ai_scripting_enabled = yes
	# 	c:PHL ?= {
	# 		is_country_alive = yes
	# 		is_ai = yes
	# 		is_subject = no
	# 		is_at_war = no
	# 		is_active_in_diplomatic_play = no
	# 		aggressive_diplomatic_plays_permitted = yes
	# 	}
	# }

	immediate = {
		add_modifier = {
			name = sb_laager_defence
			months = 12
		}
		if = {
			limit = {
				NOT = { has_variable = sb_ora_phl_no_zul_infantry_spawned_var }
				NOT = {
					c:ZUL ?= {
						has_variable = sb_zulu_blood_river_resolved_var
					}
				}
			}
			set_variable = sb_ora_phl_no_zul_infantry_spawned_var
			create_military_formation = {
				type = army
				hq_region = sr:region_southern_africa
				name = ora_phl_frontier_commando

				combat_unit = {
					type = unit_type:combat_unit_type_line_infantry
					state_region = s:STATE_VRYSTAAT
					count = 1
				}
			}
		}
		if = {
			limit = {
				sb_frontier_ai_behavior_dynamic_historical = yes
			}
			random_list = {
				80 = {
					add_modifier = {
						name = sb_ora_scripted_frontier_edge
						months = 12
					}
				}
				20 = { }
			}
		}
			if = {
				limit = {
					sb_frontier_ai_behavior_strict_historical = yes
				}
				add_modifier = {
					name = sb_ora_scripted_frontier_edge
					months = 12
				}
		}
		if = {
			limit = { sb_ai_trn_available_for_ora_scripted_play = yes }
			create_diplomatic_play = {
				type = dp_return_state
				target_country = c:PHL
				target_state = s:STATE_VRYSTAAT.region_state:PHL

				add_initiator_backers = { c:TRN }

				add_war_goal = {
					holder = root
					target_country = c:PHL
					type = humiliation
				}
			}
		}
		else = {
			create_diplomatic_play = {
				type = dp_return_state
				target_country = c:PHL
				target_state = s:STATE_VRYSTAAT.region_state:PHL

				add_war_goal = {
					holder = root
					target_country = c:PHL
					type = humiliation
				}
			}
		}
		set_variable = sb_ora_phl_scripted_war_resolved_var
	}
}

sb_frontier_ai_wars.030 = {
	type = country_event
	hidden = yes

	# Not needed called externally
	# trigger = {
	# 	country_definition = cd:ORA

	# 	is_country_alive = yes
	# 	is_subject = no

	# 		is_at_war = no
	# 		is_active_in_diplomatic_play = no
	# 		aggressive_diplomatic_plays_permitted = yes
	# 	}
	# }

	immediate = {
		if = {
			limit = {
				sb_frontier_ai_behavior_dynamic_historical = yes
			}
			# Dynamic:
			# 60% push to Stalemate
			# 20% BST major comeback
			# 20% ORA kills BST
			random_list = {
				# BST defensive edge
				60 = {
					c:BST ?= {
						if = {
							limit = {
								NOT = { sb_bst_holds_vrystaat = yes }
							}
							add_modifier = {
								name = sb_bst_scripted_thaba_bosiu
								months = 12
							}
							root = {
								add_modifier = {
									name = sb_ora_scripted_annex_war_overreach
									months = 12
								}
							}
						}
					}
				}
				# BST offensive edge to win
				20 = {
					c:BST ?= {
						if = {
							limit = {
								NOT = { sb_bst_holds_vrystaat = yes }
							}
							add_modifier = {
								name = sb_bst_scripted_frontier_major_edge
								months = 12
							}
							root = {
								add_modifier = {
									name = sb_ora_scripted_annex_war_overreach
									months = 12
								}
							}
						}
					}
				}
				# ORA offensive edge
				20 = {
					add_modifier = {
						name = sb_laager_defence
						months = 12
					}
				}

			}
		}
		if = {
			limit = {
				sb_frontier_ai_behavior_strict_historical = yes
			}
			c:BST ?= {
				if = {
					limit = {
						NOT = { sb_bst_holds_vrystaat = yes }
					}
					add_modifier = {
						name = sb_bst_scripted_thaba_bosiu
						months = 12
					}
					root = {
						add_modifier = {
							name = sb_ora_scripted_annex_war_overreach
							months = 12
						}
					}
				}
			}
		}
		create_diplomatic_play = {
			type = dp_annex_war
			target_country = c:BST

			add_war_goal = {
				holder = c:BST
				target_state = s:STATE_VRYSTAAT.region_state:ORA
				type = return_state
			}
		}
		set_variable = sb_ora_bst_scripted_war_1856_resolved_var
	}
}

sb_frontier_ai_wars.040 = {
	type = country_event
	hidden = yes

	trigger = {
		sb_is_zulu_dynasty_country = yes
		is_ai = yes
		is_country_alive = yes
		is_at_war = no
		is_active_in_diplomatic_play = no
		aggressive_diplomatic_plays_permitted = yes
		sb_frontier_ai_scripting_enabled = yes
		sb_frontier_ai_behavior_strict_historical = no
		has_variable = sb_zulu_blood_river_victory_var
		has_variable = sb_zulu_dynasty_secured_var
		NOT = { has_journal_entry = je_sb_zulu_kingdom }
		ruler = {
			has_template = ZUL_dingane
		}
		c:GZA ?= {
			is_country_alive = yes
			is_ai = yes
			is_subject = no
			is_at_war = no
			is_active_in_diplomatic_play = no
			aggressive_diplomatic_plays_permitted = yes
		}
	}

	immediate = {
		if = {
			limit = { has_variable = sb_zulu_gaza_followup_pending_var }
			remove_variable = sb_zulu_gaza_followup_pending_var
		}
		create_diplomatic_play = {
			type = dp_annex_war
			target_country = c:GZA
		}
		random_diplomatic_play = {
			limit = {
				initiator = root
				target = c:GZA
				is_diplomatic_play_type = dp_annex_war
			}
			if = {
				limit = { exists = s:STATE_EAST_TRANSVAAL.region_state:ZUL }
				add_war_goal = {
					holder = c:GZA
					type = conquer_state
					target_state = s:STATE_EAST_TRANSVAAL.region_state:ZUL
				}
			}
			if = {
				limit = { exists = s:STATE_ZULULAND.region_state:ZUL }
				add_war_goal = {
					holder = c:GZA
					type = conquer_state
					target_state = s:STATE_ZULULAND.region_state:ZUL
				}
			}
		}
	}
}

sb_frontier_ai_wars.100 = {
	type = country_event
	hidden = yes

	trigger = {
		country_definition = cd:XHO
		is_country_alive = yes
		OR = {
			c:CAP ?= { is_country_alive = yes }
			c:ABY ?= { is_country_alive = yes }
		}
	}

	immediate = {
		if = {
			limit = {
				game_date < 1848.1.1
					any_scope_state = {
						state_region = s:STATE_EASTERN_CAPE
						num_provinces >= 1
					}
					NOT = { sb_xho_owns_all_thembu_frontier_provinces = yes }
				OR = {
					c:ABY ?= {
						has_variable = sb_xhosa_frontier_war_7_scheduled_var
						has_technology_researched = colonization
					}
					c:CAP ?= {
						has_variable = sb_xhosa_frontier_war_7_scheduled_var
						has_technology_researched = colonization
					}
				}
			}
			if = {
				limit = {
					c:ABY ?= {
						is_country_alive = yes
						has_variable = sb_xhosa_frontier_war_7_scheduled_var
					}
				}
				create_diplomatic_play = {
					type = dp_sb_xhosa_war_7
					target_country = c:ABY
				}
				c:ABY ?= {
					set_variable = sb_xhosa_frontier_war_7_resolved_var
					remove_variable = sb_xhosa_frontier_war_7_scheduled_var
				}
			}
			else_if = {
				limit = {
					c:CAP ?= {

		}
		else = {
			c:ABY ?= {
				if = {
					limit = { has_variable = sb_xhosa_frontier_war_7_scheduled_var }
					remove_variable = sb_xhosa_frontier_war_7_scheduled_var
				}
				if = {
					limit = { game_date >= 1848.1.1 }
					set_variable = sb_xhosa_frontier_war_7_resolved_var
				}
			}
			c:CAP ?= {
