sb_zulu_dynasty.001 = {
	type = country_event
	fire_only_once = yes
	placement = root

	title = sb_zulu_dynasty.001.t
	desc = sb_zulu_dynasty.001.d
	flavor = sb_zulu_dynasty.001.f

	event_image = {
		video = "gfx/event_pictures/africa_soldiers_breaking_protest.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	trigger = {
		country_definition = cd:ZUL
		has_journal_entry = je_sb_zulu_kingdom
		any_scope_character = {
			has_template = ZUL_dingane
			is_character_alive = yes
		}
	}

	option = {
		name = sb_zulu_dynasty.001.a
		default_option = yes
		random_scope_character = {
			limit = {
				has_template = ZUL_dingane
				is_character_alive = yes
			}
			if = {
				limit = { NOT = { has_modifier = sb_zulu_dynastic_crisis } }
				add_modifier = {
					name = sb_zulu_dynastic_crisis
					months = -1
				}
			}
		}
	}
}
