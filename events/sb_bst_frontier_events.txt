sb_bst_frontier.010 = {
	type = country_event
	placement = root

	title = sb_bst_frontier.010.t
	desc = sb_bst_frontier.010.d
	flavor = sb_bst_frontier.010.f

	event_image = {
		video = "gfx/event_pictures/africa_soldiers_breaking_protest.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_default.dds"
	duration = 3

	trigger = {
		sb_bst_oranje_frontier_actor = yes
	}

	option = {
		name = sb_bst_frontier.010.a

		custom_tooltip = sb_bst_raid_ora_tt
		hidden_effect = {
			add_treasury = -1000
			if = {
				limit = {
					exists = c:BUG
					exists = c:BST
				}
				random_scope_state = {
					limit = {
						state_region = s:STATE_VRYSTAAT
						any_scope_pop = {
							NOT = { is_pop_type = slaves }
							total_size >= 100
						}
					}
					ordered_scope_pop = {
						position = 0
						order_by = total_size
						limit = {
							NOT = { is_pop_type = slaves }
							total_size >= 100
						}
						save_scope_as = sb_bst_raid_captive_pop
						culture = { save_scope_as = sb_bst_raid_captive_culture }
						# Stage through a slavery-legal state so only the captive slice is converted.
						move_partial_pop = {
								state = c:BUG.capital
								population_ratio = {
									add = 100
									divide = scope:sb_bst_raid_captive_pop.total_size
								}
							}
					}
				}
				if = {
					limit = { exists = scope:sb_bst_raid_captive_culture }
					c:BUG.capital = {
						every_scope_pop = {
							limit = {
								NOT = { is_pop_type = slaves }
								culture = scope:sb_bst_raid_captive_culture
							}
							change_poptype = pop_type:slaves
						}
						every_scope_pop = {
							limit = {
								is_pop_type = slaves
								culture = scope:sb_bst_raid_captive_culture
							}
							move_partial_pop = {
								state = c:BST.capital
								population_ratio = 1
							}
						}
					}
				}
			}
		}
		default_option = yes
	}
}

sb_bst_frontier.020 = {
	type = country_event
	placement = root

	title = sb_bst_frontier.020.t
	desc = sb_bst_frontier.020.d
	flavor = sb_bst_frontier.020.f

	event_image = {
		video = "gfx/event_pictures/africa_soldiers_breaking_protest.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	trigger = {
		country_definition = cd:BST
		has_journal_entry = je_sb_bst_ora_frontier
		sb_bst_controls_only_drakensberg = yes
		NOT = { has_variable = sb_bst_into_the_mountains_var }
	}

	immediate = {
		set_variable = sb_bst_into_the_mountains_var
	}

	option = {
		name = sb_bst_frontier.020.a
		default_option = yes
		custom_tooltip = sb_bst_frontier_020_sotho_retreat_tt
		hidden_effect = {
			random_country = {
				limit = { sb_bst_oranje_frontier_actor = yes }
				random_scope_state = {
					limit = {
						state_region = s:STATE_VRYSTAAT
						any_scope_pop = {
							culture = cu:sotho
							total_size >= 5000
						}
					}
					ordered_scope_pop = {
						position = 0
						order_by = total_size
						limit = {
							culture = cu:sotho
							total_size >= 5000
						}
						save_scope_as = sb_bst_mountain_retreat_sotho_pop
						move_partial_pop = {
							state = c:BST.capital
							population_ratio = {
								add = 5000
								divide = scope:sb_bst_mountain_retreat_sotho_pop.total_size
							}
						}
					}
				}
			}
		}
		random_country = {
			limit = { sb_bst_oranje_frontier_actor = yes }
			save_temporary_scope_as = sb_bst_current_oranje_actor
			change_relations = {
				country = root
				value = -20
			}
			set_variable = sb_bst_frontier_claim_seeded_var
		}
		if = {
			limit = { exists = scope:sb_bst_current_oranje_actor }
			s:STATE_DRAKENSBERG = {
				add_claim = scope:sb_bst_current_oranje_actor
			}
		}
		add_modifier = {
			name = sb_bst_biding_time
			months = 60
		}
		if = {
			limit = { is_ai = yes }
			hidden_effect = {
				s:STATE_DRAKENSBERG = {
					region_state:BST ?= {
						create_building = {
							building = building_maize_farm
							add_ownership = {
								building = {
									type = building_manor_house
									country = c:BST
									levels = 1
									region = "STATE_DRAKENSBERG"
								}
							}
							reserves = 1
							activate_production_methods = { "pm_simple_farming" "pm_no_secondary" "pm_tools_disabled" }
						}
						create_building = {
							building = building_livestock_ranch
							add_ownership = {
								building = {
									type = building_manor_house
									country = c:BST
									levels = 1
									region = "STATE_DRAKENSBERG"
								}
							}
							reserves = 1
							activate_production_methods = { "pm_open_air_stockyards" "pm_simple_ranch" "pm_standard_fences" "pm_unrefrigerated" }
						}
						create_building = {
							building = building_logging_camp
							add_ownership = {
								building = {
									type = building_logging_camp
									country = c:BST
									levels = 1
									region = "STATE_DRAKENSBERG"
								}
							}
							reserves = 1
							activate_production_methods = { "pm_simple_forestry" "pm_no_hardwood" "pm_no_equipment" "pm_road_carts" }
						}
					}
				}
			}
		}
	}
}

sb_bst_frontier.200 = {
	type = country_event
	placement = root

	title = sb_bst_frontier.200.t
	desc = sb_bst_frontier.200.d
	flavor = sb_bst_frontier.200.f

	event_image = {
		video = "gfx/event_pictures/africa_construction_colony.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_default.dds"
	duration = 3

	trigger = {
		country_definition = cd:GBR
		c:BST ?= {
			has_variable = sb_bst_cap_transfer_pending_var
			is_country_alive = yes
			is_subject_of = c:GBR
			is_subject_type = subject_type_protectorate
		}
		c:CAP ?= {
			is_country_alive = yes
			is_subject_of = c:GBR
			OR = {
				is_subject_type = subject_type_dominion
				is_subject_type = subject_type_sb_dominion
				is_subject_type = subject_type_sb_responsible_colony
				is_subject_type = subject_type_sb_responsible_colony_monarchy
			}
			sb_cap_controls_bst_frontier = yes
		}
	}

	immediate = {
		c:BST ?= {
			remove_variable = sb_bst_cap_transfer_pending_var
		}
	}

	option = {
		name = sb_bst_frontier.200.a
		default_option = yes
		c:BST ?= {
			set_variable = sb_bst_cap_transfer_resolved_var
		}
		c:CAP ?= {
			transfer_subject = c:BST
		}
		c:BST ?= {
			if = {
				limit = {
					is_subject_of = c:CAP
					NOT = { is_subject_type = subject_type_puppet }
				}
				change_subject_type = subject_type_puppet
			}
		}
		ai_chance = {
			base = 100
		}
	}

	option = {
		name = sb_bst_frontier.200.b
		trigger = {
			is_player = yes
		}
		c:BST ?= {
			set_variable = sb_bst_cap_transfer_resolved_var
		}
		ai_chance = {
			base = 0
		}
	}
}
