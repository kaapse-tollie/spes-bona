sb_swazi_frontier.094 = {
	type = country_event
	placement = root

	title = sb_swazi_frontier.094.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = sb_zulu_swazi_ai_won_blood_river_var }
				desc = sb_swazi_frontier.094.d_victory
			}
			triggered_desc = {
				trigger = { has_variable = sb_zulu_swazi_ai_lost_blood_river_var }
				desc = sb_swazi_frontier.094.d_defeat
			}
			triggered_desc = {
				trigger = {
					OR = {
						has_variable = sb_zulu_swazi_ai_honour_deal_var
						has_variable = sb_zulu_swazi_ai_guns_south_var
					}
				}
				desc = sb_swazi_frontier.094.d_compromise
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = sb_swazi_frontier.094.d
			}
		}
	}
	flavor = sb_swazi_frontier.094.f

	event_image = {
		video = "gfx/event_pictures/africa_soldiers_breaking_protest.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	immediate = { set_variable = { name = sb_zulu_swazi_outcome_popup_receipt_var months = 4 } }


	trigger = {
		has_variable = sb_zulu_swazi_outcome_delivery_pending_var
		has_variable = sb_zulu_swazi_outcome_delivery_queued_var
		NOT = { has_variable = sb_zulu_swazi_outcome_delivery_received_var }
		country_definition = cd:ZUL
		is_country_alive = yes
		has_journal_entry = je_sb_zulu_kingdom
		NOT = { has_variable = sb_zulu_swazi_war_active_var }
		ruler = { has_template = ZUL_dingane }
		c:SWZ ?= { is_country_alive = yes is_subject = no any_scope_state = { state_region = s:STATE_EAST_TRANSVAAL } }
	}

	# This is the crisis-release valve for Dingane after the Natal line. It only
	# becomes available once the Natal/Blood River line has actually resolved.
	# AI weighting below is driven by how that line ended.
	option = {
		name = sb_swazi_frontier.094.a
		trigger = {
			has_variable = sb_zulu_swazi_outcome_delivery_pending_var
			has_variable = sb_zulu_swazi_outcome_popup_receipt_var
			NOT = { has_variable = sb_zulu_swazi_outcome_delivery_received_var }
			country_definition = cd:ZUL
			is_country_alive = yes
			is_at_war = no
			is_active_in_diplomatic_play = no
			aggressive_diplomatic_plays_permitted = yes
			has_journal_entry = je_sb_zulu_kingdom
			NOT = { has_variable = sb_zulu_swazi_war_active_var }
			ruler = { has_template = ZUL_dingane }
			c:SWZ ?= {
				is_country_alive = yes
				is_subject = no
				is_at_war = no
				is_active_in_diplomatic_play = no
				any_scope_state = { state_region = s:STATE_EAST_TRANSVAAL }
			}
		}
		default_option = yes
		hidden_effect = {
			if = {
				limit = {
					is_ai = yes
					sb_frontier_ai_behavior_strict_historical = yes
					c:SWZ ?= { is_country_alive = yes is_ai = yes }
				}
				sb_zulu_face_swazi_defensive_muster = yes
			}
			else_if = {
				limit = {
					is_ai = yes
					sb_frontier_ai_behavior_dynamic_historical = yes
					c:SWZ ?= { is_country_alive = yes is_ai = yes }
				}
				if = {
					limit = { has_variable = sb_zulu_swazi_ai_won_blood_river_var }
					random_list = {
						20 = { sb_zulu_face_swazi_defensive_muster = yes }
						80 = { }
					}
				}
				else_if = {
					limit = { has_variable = sb_zulu_swazi_ai_lost_blood_river_var }
					random_list = {
						80 = { sb_zulu_face_swazi_defensive_muster = yes }
						20 = { }
					}
				}
				else_if = {
					limit = { has_variable = sb_zulu_swazi_ai_honour_deal_var }
					random_list = {
						60 = { sb_zulu_face_swazi_defensive_muster = yes }
						40 = { }
					}
				}
				else_if = {
					limit = { has_variable = sb_zulu_swazi_ai_guns_south_var }
					random_list = {
						40 = { sb_zulu_face_swazi_defensive_muster = yes }
						60 = { }
					}
				}
				else = {
					sb_zulu_face_swazi_defensive_muster = yes
				}
			}
		}
		set_variable = sb_zulu_swazi_war_active_var
		set_variable = sb_zulu_swazi_launch_lease_var
		set_variable = { name = sb_zulu_swazi_target_state_scope value = s:STATE_EAST_TRANSVAAL.region_state:SWZ }
		c:SWZ ?= { set_variable = sb_zulu_swazi_war_active_var }
		create_diplomatic_play = {
			type = dp_conquer_state
			target_country = c:SWZ
			target_state = s:STATE_EAST_TRANSVAAL.region_state:SWZ
		}
		# Only the matching on-start callback consumes this lease. A residual lease
		# is explicit creation failure and must not be inferred from another play.
		if = {
			limit = { has_variable = sb_zulu_swazi_launch_lease_var }
			sb_zulu_clear_swazi_campaign_runtime = yes
			if = { limit = { has_variable = sb_zulu_swazi_outcome_delivery_queued_var } remove_variable = sb_zulu_swazi_outcome_delivery_queued_var }
			if = { limit = { has_variable = sb_zulu_swazi_outcome_popup_receipt_var } remove_variable = sb_zulu_swazi_outcome_popup_receipt_var }
		}
		else = {
			remove_variable = sb_zulu_swazi_outcome_delivery_pending_var
			if = { limit = { has_variable = sb_zulu_swazi_outcome_delivery_queued_var } remove_variable = sb_zulu_swazi_outcome_delivery_queued_var }
			if = { limit = { has_variable = sb_zulu_swazi_outcome_popup_receipt_var } remove_variable = sb_zulu_swazi_outcome_popup_receipt_var }
			set_variable = { name = sb_zulu_swazi_outcome_delivery_received_var days = 30 }
			sb_zulu_clear_swazi_question_ai_context = yes
		}
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					is_ai = yes
					sb_frontier_ai_behavior_strict_historical = yes
					c:SWZ ?= { is_country_alive = yes is_ai = yes }
				}
				add = 1000
			}
			modifier = {
				factor = 0.4
				has_variable = sb_zulu_swazi_ai_honour_deal_var
			}
			modifier = {
				factor = 2
				has_variable = sb_zulu_swazi_ai_lost_blood_river_var
			}
			modifier = {
				factor = 1.6
				has_variable = sb_zulu_swazi_ai_won_blood_river_var
			}
		}
	}

	option = {
		name = sb_swazi_frontier.094.b
		trigger = {
			has_variable = sb_zulu_swazi_outcome_delivery_pending_var
			has_variable = sb_zulu_swazi_outcome_popup_receipt_var
			NOT = { has_variable = sb_zulu_swazi_outcome_delivery_received_var }
			country_definition = cd:ZUL
			is_country_alive = yes
			has_journal_entry = je_sb_zulu_kingdom
			NOT = { has_variable = sb_zulu_swazi_war_active_var }
			ruler = { has_template = ZUL_dingane }
			c:SWZ ?= { is_country_alive = yes is_subject = no any_scope_state = { state_region = s:STATE_EAST_TRANSVAAL } }
		}
		remove_variable = sb_zulu_swazi_outcome_delivery_pending_var
		if = { limit = { has_variable = sb_zulu_swazi_outcome_delivery_queued_var } remove_variable = sb_zulu_swazi_outcome_delivery_queued_var }
		if = { limit = { has_variable = sb_zulu_swazi_outcome_popup_receipt_var } remove_variable = sb_zulu_swazi_outcome_popup_receipt_var }
		set_variable = { name = sb_zulu_swazi_outcome_delivery_received_var days = 30 }
		sb_zulu_clear_swazi_question_ai_context = yes
		custom_tooltip = sb_zulu_dynasty_lose_stability_5
		hidden_effect = {
			sb_zulu_remove_dynastic_stability_5 = yes
		}
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					is_ai = yes
					sb_frontier_ai_behavior_strict_historical = yes
					c:SWZ ?= { is_country_alive = yes is_ai = yes }
				}
				add = -1000
			}
			modifier = {
				factor = 1.6
				has_variable = sb_zulu_swazi_ai_honour_deal_var
			}
			modifier = {
				factor = 0
				has_variable = sb_zulu_swazi_ai_lost_blood_river_var
			}
			modifier = {
				factor = 0.4
				has_variable = sb_zulu_swazi_ai_won_blood_river_var
			}
		}
	}


	option = {
		name = sb_swazi_frontier.094.b
		trigger = {
			has_variable = sb_zulu_swazi_outcome_delivery_pending_var
			has_variable = sb_zulu_swazi_outcome_popup_receipt_var
			NOT = {
				AND = {
					NOT = { has_variable = sb_zulu_swazi_outcome_delivery_received_var }
					country_definition = cd:ZUL
					is_country_alive = yes
					has_journal_entry = je_sb_zulu_kingdom
					NOT = { has_variable = sb_zulu_swazi_war_active_var }
					ruler = { has_template = ZUL_dingane }
					c:SWZ ?= { is_country_alive = yes is_subject = no any_scope_state = { state_region = s:STATE_EAST_TRANSVAAL } }
				}
			}
		}
		remove_variable = sb_zulu_swazi_outcome_delivery_pending_var
		remove_variable = sb_zulu_swazi_outcome_delivery_queued_var
		if = { limit = { has_variable = sb_zulu_swazi_outcome_popup_receipt_var } remove_variable = sb_zulu_swazi_outcome_popup_receipt_var }
		set_variable = { name = sb_zulu_swazi_outcome_delivery_received_var days = 30 }
		sb_zulu_clear_swazi_question_ai_context = yes
	}
}

	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3

	trigger = {
		sb_is_zulu_dynasty_country = yes
	}

		name = sb_swazi_frontier.097.a

		}

		video = "gfx/event_pictures/africa_diplomats_negotiating.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_default.dds"
	duration = 3

	trigger = {
		country_definition = cd:GBR
		c:SWZ ?= {
			is_country_alive = yes
			is_subject = no
			has_variable = sb_swazi_boer_pressure_resolved_var
			sb_has_adjacent_british_protection_route = yes
		}
		c:NAL ?= {
			is_country_alive = yes
			is_subject_of = c:GBR
		}
		c:SWZ ?= {
			sb_all_zululand_under_british_control = yes
		}
	}

	option = {
		name = sb_swazi_frontier.100.a
		default_option = yes
		c:GBR ?= {
			create_diplomatic_pact = {
				country = c:SWZ
				type = protectorate
			}
		}
		ai_chance = {
			base = 100
		}
	}

	option = {
		name = sb_swazi_frontier.100.b
		trigger = {
			is_player = yes
		}
		ai_chance = {
			base = 0
		}
	}
}
