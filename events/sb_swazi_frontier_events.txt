# =============================================================================
# Spes Bona - Swazi Frontier Events
# =============================================================================
# Ownership: ZUL-SWZ frontier war, Boer consolidation pressure, and British accommodation.
# TOC: 094 Swazi war opening, 095 Boer frontier pressure, 096 British accommodation, 097 Zulu victory follow-up.

sb_swazi_frontier.094 = {
	type = country_event
	placement = root

	title = sb_swazi_frontier.094.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = sb_zulu_swazi_ai_won_blood_river_var }
				desc = sb_swazi_frontier.094.d_victory
			}
			triggered_desc = {
				trigger = { has_variable = sb_zulu_swazi_ai_lost_blood_river_var }
				desc = sb_swazi_frontier.094.d_defeat
			}
			triggered_desc = {
				trigger = {
					OR = {
						has_variable = sb_zulu_swazi_ai_honour_deal_var
						has_variable = sb_zulu_swazi_ai_guns_south_var
					}
				}
				desc = sb_swazi_frontier.094.d_compromise
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = sb_swazi_frontier.094.d
			}
		}
	}
	flavor = sb_swazi_frontier.094.f

	gui_window = event_window_1char_tabloid
	left_icon = scope:sb_court_frontier_sovereign

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 3

	immediate = {
		sb_court_prepare_frontier_choice = yes
		ruler ?= { save_scope_as = sb_court_frontier_sovereign }
		set_variable = { name = sb_zulu_swazi_outcome_popup_receipt_var months = 4 }
	}


	trigger = {
		sb_zulu_swazi_question_is_due = yes
		has_variable = sb_zulu_swazi_outcome_delivery_pending_var
		has_variable = sb_zulu_swazi_outcome_delivery_queued_var
		NOT = { has_variable = sb_zulu_swazi_outcome_delivery_received_var }
		country_definition = cd:ZUL
		is_country_alive = yes
		has_journal_entry = je_sb_zulu_kingdom
		NOT = { has_variable = sb_zulu_swazi_war_active_var }
		ruler = { has_template = ZUL_dingane }
		c:SWZ ?= { is_country_alive = yes is_subject = no any_scope_state = { state_region = s:STATE_EAST_TRANSVAAL } }
	}

	# This is the crisis-release valve for Dingane after the Natal line. It only
	# becomes available once the Natal/Blood River line has actually resolved.
	# AI weighting below is driven by how that line ended.
	option = {
		name = sb_swazi_frontier.094.a
		trigger = {
			has_variable = sb_zulu_swazi_outcome_delivery_pending_var
			has_variable = sb_zulu_swazi_outcome_popup_receipt_var
			NOT = { has_variable = sb_zulu_swazi_outcome_delivery_received_var }
			country_definition = cd:ZUL
			is_country_alive = yes
			is_at_war = no
			is_active_in_diplomatic_play = no
			aggressive_diplomatic_plays_permitted = yes
			has_journal_entry = je_sb_zulu_kingdom
			NOT = { has_variable = sb_zulu_swazi_war_active_var }
			ruler = { has_template = ZUL_dingane }
			c:SWZ ?= {
				is_country_alive = yes
				is_subject = no
				is_at_war = no
				is_active_in_diplomatic_play = no
				any_scope_state = { state_region = s:STATE_EAST_TRANSVAAL }
			}
		}
		default_option = yes
		hidden_effect = {
			if = {
				limit = {
					is_ai = yes
					sb_frontier_ai_behavior_strict_historical = yes
					c:SWZ ?= { is_country_alive = yes is_ai = yes }
				}
				sb_zulu_face_swazi_defensive_muster = yes
			}
			else_if = {
				limit = {
					is_ai = yes
					sb_frontier_ai_behavior_dynamic_historical = yes
					c:SWZ ?= { is_country_alive = yes is_ai = yes }
				}
				if = {
					limit = { has_variable = sb_zulu_swazi_ai_won_blood_river_var }
					random_list = {
						20 = { sb_zulu_face_swazi_defensive_muster = yes }
						80 = { }
					}
				}
				else_if = {
					limit = { has_variable = sb_zulu_swazi_ai_lost_blood_river_var }
					random_list = {
						80 = { sb_zulu_face_swazi_defensive_muster = yes }
						20 = { }
					}
				}
				else_if = {
					limit = { has_variable = sb_zulu_swazi_ai_honour_deal_var }
					random_list = {
						60 = { sb_zulu_face_swazi_defensive_muster = yes }
						40 = { }
					}
				}
				else_if = {
					limit = { has_variable = sb_zulu_swazi_ai_guns_south_var }
					random_list = {
						40 = { sb_zulu_face_swazi_defensive_muster = yes }
						60 = { }
					}
				}
				else = {
					sb_zulu_face_swazi_defensive_muster = yes
				}
			}
		}
		set_variable = sb_zulu_swazi_war_active_var
		set_variable = sb_zulu_swazi_launch_lease_var
		set_variable = { name = sb_zulu_swazi_target_state_scope value = s:STATE_EAST_TRANSVAAL.region_state:SWZ }
		c:SWZ ?= { set_variable = sb_zulu_swazi_war_active_var }
		create_diplomatic_play = {
			type = dp_conquer_state
			target_country = c:SWZ
			target_state = s:STATE_EAST_TRANSVAAL.region_state:SWZ
		}
		# Only the matching on-start callback consumes this lease. A residual lease
		# is explicit creation failure and must not be inferred from another play.
		if = {
			limit = { has_variable = sb_zulu_swazi_launch_lease_var }
			sb_zulu_clear_swazi_campaign_runtime = yes
			if = { limit = { has_variable = sb_zulu_swazi_outcome_delivery_queued_var } remove_variable = sb_zulu_swazi_outcome_delivery_queued_var }
			if = { limit = { has_variable = sb_zulu_swazi_outcome_popup_receipt_var } remove_variable = sb_zulu_swazi_outcome_popup_receipt_var }
		}
		else = {
			if = { limit = { NOT = { has_variable = sb_court_swazi_launch_support_var } } set_variable = sb_court_swazi_launch_support_var sb_court_frontier_support = yes }
			remove_variable = sb_zulu_swazi_outcome_delivery_pending_var
			if = { limit = { has_variable = sb_zulu_swazi_outcome_delivery_queued_var } remove_variable = sb_zulu_swazi_outcome_delivery_queued_var }
			if = { limit = { has_variable = sb_zulu_swazi_outcome_popup_receipt_var } remove_variable = sb_zulu_swazi_outcome_popup_receipt_var }
			set_variable = { name = sb_zulu_swazi_outcome_delivery_received_var days = 30 }
			sb_zulu_clear_swazi_question_ai_context = yes
		}
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					is_ai = yes
					sb_frontier_ai_behavior_strict_historical = yes
					c:SWZ ?= { is_country_alive = yes is_ai = yes }
				}
				add = 1000
			}
			modifier = {
				factor = 0.4
				has_variable = sb_zulu_swazi_ai_honour_deal_var
			}
			modifier = {
				factor = 2
				has_variable = sb_zulu_swazi_ai_lost_blood_river_var
			}
			modifier = {
				factor = 1.6
				has_variable = sb_zulu_swazi_ai_won_blood_river_var
			}
		}
	}

	option = {
		name = sb_swazi_frontier.094.b
		trigger = {
			has_variable = sb_zulu_swazi_outcome_delivery_pending_var
			has_variable = sb_zulu_swazi_outcome_popup_receipt_var
			NOT = { has_variable = sb_zulu_swazi_outcome_delivery_received_var }
			country_definition = cd:ZUL
			is_country_alive = yes
			has_journal_entry = je_sb_zulu_kingdom
			NOT = { has_variable = sb_zulu_swazi_war_active_var }
			ruler = { has_template = ZUL_dingane }
			c:SWZ ?= { is_country_alive = yes is_subject = no any_scope_state = { state_region = s:STATE_EAST_TRANSVAAL } }
		}
		remove_variable = sb_zulu_swazi_outcome_delivery_pending_var
		if = { limit = { has_variable = sb_zulu_swazi_outcome_delivery_queued_var } remove_variable = sb_zulu_swazi_outcome_delivery_queued_var }
		if = { limit = { has_variable = sb_zulu_swazi_outcome_popup_receipt_var } remove_variable = sb_zulu_swazi_outcome_popup_receipt_var }
		set_variable = { name = sb_zulu_swazi_outcome_delivery_received_var days = 30 }
		sb_zulu_clear_swazi_question_ai_context = yes
		custom_tooltip = sb_court_frontier_opposition_tt
		if = { limit = { exists = scope:sb_court_frontier_neutral } custom_tooltip = sb_court_frontier_neutral_named_tt }
		if = { limit = { exists = scope:sb_court_frontier_rival } custom_tooltip = sb_court_frontier_rival_named_tt }
		hidden_effect = {
			sb_court_frontier_opposition = yes
		}
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					is_ai = yes
					sb_frontier_ai_behavior_strict_historical = yes
					c:SWZ ?= { is_country_alive = yes is_ai = yes }
				}
				add = -1000
			}
			modifier = {
				factor = 1.6
				has_variable = sb_zulu_swazi_ai_honour_deal_var
			}
			modifier = {
				factor = 0
				has_variable = sb_zulu_swazi_ai_lost_blood_river_var
			}
			modifier = {
				factor = 0.4
				has_variable = sb_zulu_swazi_ai_won_blood_river_var
			}
		}
	}


	option = {
		name = sb_swazi_frontier.094.b
		trigger = {
			has_variable = sb_zulu_swazi_outcome_delivery_pending_var
			has_variable = sb_zulu_swazi_outcome_popup_receipt_var
			NOT = {
				AND = {
					NOT = { has_variable = sb_zulu_swazi_outcome_delivery_received_var }
					country_definition = cd:ZUL
					is_country_alive = yes
					has_journal_entry = je_sb_zulu_kingdom
					NOT = { has_variable = sb_zulu_swazi_war_active_var }
					ruler = { has_template = ZUL_dingane }
					c:SWZ ?= { is_country_alive = yes is_subject = no any_scope_state = { state_region = s:STATE_EAST_TRANSVAAL } }
				}
			}
		}
		remove_variable = sb_zulu_swazi_outcome_delivery_pending_var
		remove_variable = sb_zulu_swazi_outcome_delivery_queued_var
		if = { limit = { has_variable = sb_zulu_swazi_outcome_popup_receipt_var } remove_variable = sb_zulu_swazi_outcome_popup_receipt_var }
		set_variable = { name = sb_zulu_swazi_outcome_delivery_received_var days = 30 }
		sb_zulu_clear_swazi_question_ai_context = yes
	}
}

sb_swazi_frontier.097 = {
	type = country_event
	placement = root

	title = sb_swazi_frontier.097.t
	desc = sb_swazi_frontier.097.d
	flavor = sb_swazi_frontier.097.f

	gui_window = event_window_1char_tabloid
	left_icon = scope:sb_court_outcome_portrait

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 3

	trigger = {
		sb_is_zulu_dynasty_country = yes
		var:sb_court_swazi_outcome_pending_var ?= 1
		NOR = { has_variable = sb_court_swazi_result_var has_variable = sb_court_swazi_outcome_open_var }
	}

	immediate = {
		sb_court_prepare_campaign_result = { CAMPAIGN = swazi WON = yes }
		set_variable = { name = sb_court_swazi_outcome_open_var months = 4 }
		save_scope_as = sb_court_country
		var:sb_court_swazi_patron_scope ?= { save_scope_as = sb_court_campaign_patron }
		ruler ?= { save_scope_as = sb_court_outcome_portrait }
		var:sb_court_swazi_ruler_scope ?= {
			save_scope_as = sb_court_campaign_ruler
			if = { limit = { is_character_alive = yes } save_scope_as = sb_court_outcome_portrait }
		}
	}
	option = {
		name = sb_swazi_frontier.097.a
		default_option = yes
		if = {
			limit = { var:sb_court_swazi_outcome_pending_var ?= 1 NOT = { has_variable = sb_court_swazi_result_var } }
			if = { limit = { scope:sb_court_campaign_ruler ?= { is_character_alive = yes } } custom_tooltip = sb_court_campaign_ruler_tt }
			show_as_tooltip = { every_in_list = { variable = sb_court_swazi_changes limit = { sb_court_episode_neutral = yes scope:sb_court_campaign_patron ?= { is_character_alive = yes } } custom_tooltip = sb_court_campaign_supporter_tt } }
			hidden_effect = {
				sb_zulu_apply_swazi_campaign_victory = yes
				remove_variable = sb_court_swazi_outcome_pending_var
				remove_variable = sb_court_swazi_outcome_open_var
			}
		}
		ai_chance = {
			base = 100
		}
	}
}

		NOR = { has_variable = sb_court_swazi_result_var has_variable = sb_court_swazi_outcome_open_var }
	}
	immediate = {
		sb_court_prepare_campaign_result = { CAMPAIGN = swazi WON = no }
		set_variable = { name = sb_court_swazi_outcome_open_var months = 4 }
		save_scope_as = sb_court_country
		ruler ?= { save_scope_as = sb_court_outcome_portrait }
		var:sb_court_swazi_ruler_scope ?= {
			save_scope_as = sb_court_campaign_ruler
			if = { limit = { is_character_alive = yes } save_scope_as = sb_court_outcome_portrait }
		}
		var:sb_court_swazi_patron_scope ?= { save_scope_as = sb_court_campaign_patron }
	}
	option = {
		name = sb_swazi_frontier.098.a
		default_option = yes
		if = {
			limit = { var:sb_court_swazi_outcome_pending_var ?= 2 NOT = { has_variable = sb_court_swazi_result_var } }
			if = { limit = { scope:sb_court_campaign_ruler ?= { is_character_alive = yes } } custom_tooltip = sb_court_campaign_defeated_ruler_tt }
			if = {
				limit = { has_variable = sb_zulu_blood_river_defeat_var }
				custom_tooltip = sb_swazi_frontier.098.prerogative_after_blood_river
			}
			else = { custom_tooltip = sb_swazi_frontier.098.prerogative }
			show_as_tooltip = {
				every_in_list = { variable = sb_court_swazi_changes limit = { sb_court_episode_neutral = yes scope:sb_court_campaign_patron ?= { is_character_alive = yes NOT = { sb_court_mpande_son_under_king = yes } } } custom_tooltip = sb_court_campaign_opposition_tt }
				every_in_list = {
					variable = sb_court_swazi_defectors
					limit = { sb_court_campaign_defector = yes scope:sb_court_country = { has_variable = sb_institutions_active_var } scope:sb_court_campaign_ruler ?= { is_character_alive = yes is_ruler_of_own_country = yes owner = scope:sb_court_country } scope:sb_court_campaign_patron ?= { is_character_alive = yes NOT = { sb_court_mpande_son_under_king = yes } } }
					custom_tooltip = sb_court_campaign_defector_tt
				}
			}
			hidden_effect = {
				sb_zulu_apply_swazi_campaign_defeat = yes
				remove_variable = sb_court_swazi_outcome_pending_var
				remove_variable = sb_court_swazi_outcome_open_var
			}
		}
		ai_chance = { base = 100 }
	}
}

	icon = "gfx/interface/icons/event_icons/event_default.dds"
	duration = 3

	trigger = {
		country_definition = cd:GBR

			is_subject = no
			has_variable = sb_swazi_boer_pressure_resolved_var
			sb_has_adjacent_british_protection_route = yes
		}
		c:NAL ?= {
			is_country_alive = yes

		}
	}

	option = {
		name = sb_swazi_frontier.100.b

	}
