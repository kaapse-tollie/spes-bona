namespace = sb_law_journal

# One deferred attempt after native country creation. Recheck current eligibility and
# duplicates in immediate, so scheduling is allowed while the country is unfinished.
# Destroyed countries and superseded law changes do not retry.
sb_law_journal.001 = {
	type = country_event
	hidden = yes
	immediate = {
		if = {
			limit = {
				sb_law_journal_country_ready = yes
				has_law_or_variant = law_type:law_technocracy
				NOT = { has_journal_entry = je_technocracy }
			}
			add_journal_entry = { type = je_technocracy }
		}
	}
}

				NOT = { has_journal_entry = je_king_in_parliament }
			}

		}
	}
}

sb_law_journal.005 = {
	type = country_event
	hidden = yes
	immediate = {
		if = {
			limit = {
				sb_law_journal_country_ready = yes
				NOT = { has_journal_entry = je_abolish_monarchy }
				OR = {
					# Enactment-start route: retain the native monarchy/enacting-law contract.
					AND = {
						has_law = law_type:law_monarchy
						OR = {
							is_enacting_law = law_type:law_corporate_state
							is_enacting_law = law_type:law_presidential_republic
							is_enacting_law = law_type:law_parliamentary_republic
							is_enacting_law = law_type:law_council_republic
						}
					}
					# Direct activation route: retain the native post-monarchy exclusions.
					AND = {
						OR = {
							has_law_or_variant = law_type:law_presidential_republic
							has_law_or_variant = law_type:law_parliamentary_republic
							has_law_or_variant = law_type:law_council_republic
							has_law_or_variant = law_type:law_corporate_state
						}
						NOR = {
							game_date <= 1836.1.2
							has_variable = monarchy_stamped_out_var
							has_variable = rcw_russian_faction
							ig:ig_landowners ?= { has_ideology = ideology:ideology_republican_paternalistic }
						}
					}
				}
			}
			add_journal_entry = { type = je_abolish_monarchy }
		}
	}
}
