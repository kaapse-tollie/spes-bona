	option = {

		default_option = yes
		if = {
			limit = {
				sb_bechuanaland_warren_decision_window_is_valid = yes
				OR = {
					container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_warren_cap_source_var }
					container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_warren_gbr_source_var }

			}
			if = {
				limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_warren_gbr_source_var } }

			}
			else = {
				c:CAP ?= {
					if = {
						limit = { c:SGO ?= { is_country_alive = yes } }
						#if = { limit = { c:SGO ?= { is_subject = yes } } transfer_subject = c:SGO } # this is redundant and crap code
						annex = c:SGO
					}
				}
			}
			sb_bechuanaland_finish_warren_attempt = yes
			sb_bechuanaland_clear_escalation_decision = yes
			container:sb_bechuanaland_corridor_state = { set_variable = sb_bechuanaland_british_victory_var }

			base = 70
			modifier = { trigger = { sb_bechuanaland_ai_swa_side_has_player = yes } add = -40 }
			modifier = { trigger = { sb_bechuanaland_ai_main_boer_is_trn_with_confederal_ora = yes } add = -20 }
			modifier = { trigger = { sb_bechuanaland_ai_main_boer_is_ora_kimberley_winner = yes } add = -20 }
			modifier = { trigger = { sb_bechuanaland_ai_main_boer_is_trn_confederal_kimberley_winner = yes } add = -10 }
			modifier = { trigger = { sb_bechuanaland_ai_main_boer_gdp_above_cap = yes } add = -5 }
			modifier = { trigger = { sb_bechuanaland_ai_main_boer_gdp_twice_cap = yes } add = -10 }
			modifier = { trigger = { sb_bechuanaland_ai_main_boer_gdp_at_most_three_quarters_cap = yes } add = 10 }
			modifier = { trigger = { sb_bechuanaland_ai_main_boer_army_above_cap = yes } add = -5 }
			modifier = { trigger = { sb_bechuanaland_ai_main_boer_army_twice_cap = yes } add = -10 }
			modifier = { trigger = { sb_bechuanaland_ai_main_boer_army_at_most_three_quarters_cap = yes } add = 10 }
		}
	}

	option = {
		name = sb_bechuanaland_corridor.010.b
		custom_tooltip = sb_bechuanaland_corridor_010_refuse_tt
		if = {
			limit = { sb_bechuanaland_warren_decision_window_is_valid = yes }

			base = 30
