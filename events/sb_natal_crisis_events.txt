# AI ZUL dice for blood river
sb_natal_crisis.019 = {
	type = country_event
	hidden = yes

	trigger = {
		country_definition = cd:ZUL
	}

	immediate = {
		# One-shot AI balance resolver for the Blood River diplomatic play.
		# It runs from the DP start hook, not from war start, so the support / no-
		# support branch is locked in before the fronts open.
		if = {
			limit = {
				NOT = { has_variable = sb_zul_blood_river_support_resolved_var }
			}
			set_variable = sb_zul_blood_river_support_resolved_var
			if = {
				limit = {
					is_ai = yes
					sb_frontier_player_challenge_enabled = yes
					OR = {
						c:ORA ?= {
							is_country_alive = yes
							is_player = yes
						}
						c:TRN ?= {
							is_country_alive = yes
							is_player = yes
						}
					}
				}
				sb_apply_zul_blood_river_readiness = yes
			}
			else_if = {
				limit = {
					is_ai = yes
					c:ORA ?= {
						is_country_alive = yes
						is_ai = yes
					}
					OR = {
						NOT = { exists = c:TRN }
						c:TRN ?= {
							is_ai = yes
						}
					}
					sb_frontier_ai_behavior_dynamic_historical = yes
				}
				random_list = { # Dynamic-Historical Blood River result
					30 = { # support ZUL
						set_variable = {
							name = sb_zul_blood_river_positive_roll_var
							months = 24
						}
						add_modifier = {
							name = sb_blood_river_zulu_limited_support
							months = 15
						}
						sb_zulu_add_dynastic_stability_20 = yes
						sb_boost_firearms_progress_10 = yes
					}
					70 = { # the relief force is destroyed
						set_variable = {
							name = sb_zul_blood_river_no_relief_roll_var
							months = 24
						}
						add_modifier = {
							name = sb_blood_river_zulu_no_relief
							months = 9
						}
					}
				}
			}
			else_if = {
				limit = {
					is_ai = yes
					c:ORA ?= {
						is_country_alive = yes
						is_ai = yes
					}
					OR = {
						NOT = { exists = c:TRN }
						c:TRN ?= {
							is_ai = yes
						}
					}
					sb_frontier_ai_behavior_strict_historical = yes
				}
				set_variable = {
					name = sb_zul_blood_river_no_relief_roll_var
					months = 24
				}
				add_modifier = {
					name = sb_blood_river_zulu_no_relief
					months = 15
				}
			}
		}
	}
}
