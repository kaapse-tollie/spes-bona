# The arbiter accepts outright.
sb_pink_map.020 = {
	type = country_event
	placement = root
	duration = 3
	dlc = dlc016

	title = portuguese_colonialism.5.t
	desc = portuguese_colonialism.5.d
	flavor = portuguese_colonialism.5.f

	event_image = { video = "unspecific_signed_contract" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	trigger = {
		is_country_alive = yes
		has_variable = pink_map_seeking
		has_variable = sb_pink_map_arbiter_scope
		var:sb_pink_map_arbiter_scope = { is_country_alive = yes }
	}

	immediate = {
		ROOT = { save_scope_as = por_scope }
		var:sb_pink_map_arbiter_scope = { save_scope_as = gbr_scope }
	}

	cancellation_trigger = {
		OR = {
			NOT = { has_variable = pink_map_seeking }
			NOT = { has_variable = sb_pink_map_arbiter_scope }
			AND = {
				has_variable = sb_pink_map_arbiter_scope
				var:sb_pink_map_arbiter_scope = { is_country_alive = no }
			}
		}
	}

	option = {
		name = portuguese_colonialism.5.a
		default_option = yes
		change_relations = { country = scope:gbr_scope value = 15 }
		sb_pink_map_accept_arbitration = yes
		sb_pink_map_finish_arbitration = yes
	}

	option = {
		name = portuguese_colonialism.5.b
		custom_tooltip = portuguese_colonialism.5.b_tt
		scope:gbr_scope = {
			post_notification = pink_map_britain_accepts_with_gift
			add_modifier = { name = gbr_receives_gift_from_portugal_for_approval_mod days = normal_modifier_time }
		}
		change_relations = { country = scope:gbr_scope value = 30 }
		create_diplomatic_catalyst = { type = catalyst_event_positive target = scope:gbr_scope }
		add_modifier = { name = portuguese_sends_gbr_gift_for_approval_mod days = short_modifier_time }
		sb_pink_map_accept_arbitration = yes
		sb_pink_map_finish_arbitration = yes
	}
}

# The arbiter accepts in exchange for a favour.
sb_pink_map.030 = {
	type = country_event
	placement = root
	duration = 3
	dlc = dlc016

	title = portuguese_colonialism.10.t
	desc = portuguese_colonialism.10.d
	flavor = portuguese_colonialism.10.f

	event_image = { video = "southamerica_aristocrats" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_default.dds"

	trigger = {
		is_country_alive = yes
		has_variable = pink_map_seeking
		has_variable = sb_pink_map_arbiter_scope
		var:sb_pink_map_arbiter_scope = { is_country_alive = yes }
	}

	immediate = {
		ROOT = { save_scope_as = por_scope }
		var:sb_pink_map_arbiter_scope = { save_scope_as = gbr_scope }
	}

	cancellation_trigger = {
		OR = {
			NOT = { has_variable = pink_map_seeking }
			NOT = { has_variable = sb_pink_map_arbiter_scope }
			AND = {
				has_variable = sb_pink_map_arbiter_scope
				var:sb_pink_map_arbiter_scope = { is_country_alive = no }
			}
		}
	}

	option = {
		name = portuguese_colonialism.10.a
		default_option = yes
		if = {
			limit = { NOT = { owes_obligation_to = scope:gbr_scope } }
			set_owes_obligation_to = { country = scope:gbr_scope setting = yes }
		}
		else = {
			custom_tooltip = {
				text = portuguese_colonialism.10.a_tt
				create_treaty = {
					name = treaty_name_the_pink_map_treaty
					first_country = ROOT
					second_country = scope:gbr_scope
					is_draft = no
					binding_period = { years = 2 }
					articles_to_create = {
						{
							article = money_transfer
							source_country = ROOT
							target_country = scope:gbr_scope
							inputs = { { quantity = 200 } }
						}
					}
				}
			}
		}
		scope:gbr_scope = { change_relations = { country = scope:por_scope value = 10 } }
		every_subject_or_below = {
			limit = { is_direct_subject_of = ROOT }
			add_liberty_desire = 5
		}
		sb_pink_map_accept_arbitration = yes
		sb_pink_map_finish_arbitration = yes
		scope:gbr_scope = { post_notification = pink_map_britain_accepts_with_favor }
	}

	option = {
		name = portuguese_colonialism.10.b
		scope:gbr_scope = {
			post_notification = pink_map_britain_portugal_backs_down
			change_relations = { country = scope:por_scope value = 25 }
		}
		add_modifier = { name = portuguese_backs_down_from_pursuing_the_pink_map_mod days = long_modifier_time }
		every_interest_group = {
			limit = {
				OR = {
					law_stance = { law = law_type:law_colonial_resettlement value > neutral }
					law_stance = { law = law_type:law_colonial_exploitation value > neutral }
					law_stance = { law = law_type:law_colonial_slavery value > neutral }
				}
			}
			add_modifier = { name = portuguese_refuses_to_back_down_from_pursuing_the_pink_map_ig_mod days = long_modifier_time }
		}
		sb_pink_map_remove_root_claim_package = yes
		sb_pink_map_finish_arbitration = yes
	}

	option = {
		name = portuguese_colonialism.10.c
		custom_tooltip = sb_pink_map_defiance_tt
		scope:gbr_scope = {
			post_notification = pink_map_britain_portugal_ignores
			add_modifier = {
				name = portuguese_refuses_to_back_down_from_pursuing_the_pink_map_mod
				days = short_modifier_time
			}
		}
		add_modifier = { name = portuguese_insists_on_pink_map_mod days = long_modifier_time }
		change_infamy = 25
		create_diplomatic_catalyst = { type = catalyst_event_negative target = scope:gbr_scope }
		sb_pink_map_apply_defiance = yes
	}
}
