sb_boer_republics.130 = {
	type = country_event
	fire_only_once = yes
	placement = root

	title = sb_boer_republics.130.t
	desc = sb_boer_republics.130.d
	flavor = sb_boer_republics.130.f

	event_image = {
		video = "gfx/event_pictures/africa_construction_colony.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_default.dds"
	duration = 3

	trigger = {
		OR = {
			country_definition = cd:TRN
			country_definition = cd:ORA
			country_definition = cd:ZPB
			country_definition = cd:LYD
			country_definition = cd:NAL
			country_definition = cd:SGO
		}
		country_has_primary_culture = cu:boer
	}

	option = {
		name = sb_boer_republics.130.a
		default_option = yes
		hidden_effect = { sb_record_trn_grondwet = yes }
		if = {
			limit = { has_modifier = sb_trek_frontier_republic }
			remove_modifier = sb_trek_frontier_republic
		}
		add_modifier = {
			name = sb_trek_centralizing_republic
			months = 120
		}
		if = {
			limit = { has_variable = sb_commandant_general_visible_var }
			remove_variable = sb_commandant_general_visible_var
		}
		activate_law = law_type:law_presidential_republic
		hidden_effect = {
			activate_law = law_type:law_agrarianism
		}
		activate_law = law_type:law_protectionism
		activate_law = law_type:law_landed_voting
		if = {
			limit = {
				NAND = {
					country_definition = cd:TRN
					OR = {
						ruler ?= { has_template = TRN_marthinus_wessel_pretorius }
						AND = {
							ruler.interest_group ?= ig:ig_landowners
							ig:ig_landowners.leader ?= {
								OR = { has_template = TRN_andries_pretorius has_template = TRN_marthinus_wessel_pretorius }
							}
						}
					}
				}
			}
			sb_transition_ruler_landowner_leadership = yes
		}
		if = {
			limit = {
				country_definition = cd:TRN
				NOT = { exists = c:LYD }
				NOT = { has_variable = sb_lydenburg_spawned_var }
			}
			set_variable = sb_lydenburg_spawn_var
			trigger_event = { id = sb_boer_republics.131 days = 60 popup = yes }
		}
		ai_chance = {
			base = 75
			modifier = {
				trigger = {
					is_ai = yes
					country_definition = cd:TRN
					sb_no_east_transvaal_claim_holder = yes
				}
				add = -75
			}
		}
	}

	option = {
		name = sb_boer_republics.130.b
		if = {
			limit = { has_modifier = sb_trek_centralizing_republic }
			remove_modifier = sb_trek_centralizing_republic
		}
		add_modifier = {
			name = sb_trek_frontier_republic
			months = 120
		}
		set_variable = sb_commandant_general_visible_var
		if = {
			limit = {
				country_definition = cd:TRN
				s:STATE_EAST_TRANSVAAL = {
					NOT = { is_homeland = cu:boer }
				}
			}
			if = {
				limit = { NOT = { has_claim = s:STATE_EAST_TRANSVAAL } }
				s:STATE_EAST_TRANSVAAL = { add_claim = root }
			}
			if = {
				limit = { NOT = { has_journal_entry = je_sb_pacifying_eastern_transvaal } }
				add_journal_entry = { type = je_sb_pacifying_eastern_transvaal }
			}
		}
		if = {
			limit = { has_variable = sb_lydenburg_spawn_var }
			remove_variable = sb_lydenburg_spawn_var
		}
		ai_chance = {
			base = 25
			modifier = {
				trigger = {
					is_ai = yes
					country_definition = cd:TRN
					sb_no_east_transvaal_claim_holder = yes
				}
				add = 75
			}
		}
	}
}

# One political return notice, including the household receipt only if TRN owns it.
sb_boer_republics.133 = {
	type = country_event
	placement = root
	title = sb_boer_republics.133.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					var:sb_trn_marthinus_appointment ?= flag:combined
					scope:sb_trn_marthinus ?= { sb_trn_appointment_candidate = yes }
					scope:sb_returning_andries ?= { is_character_alive = yes owner ?= c:TRN }
				}
				desc = sb_boer_republics.133.d.both
			}
			triggered_desc = {
				trigger = { var:sb_trn_marthinus_appointment ?= flag:combined scope:sb_trn_marthinus ?= { sb_trn_appointment_candidate = yes } }
				desc = sb_boer_republics.133.d.son
			}
			triggered_desc = {
				trigger = { scope:sb_returning_andries ?= { owner ?= c:TRN is_interest_group_leader = yes is_character_alive = yes } }
				desc = sb_boer_republics.133.d.father
			}
			triggered_desc = {
				trigger = { scope:sb_returning_andries ?= { owner ?= c:TRN is_character_alive = yes } }
				desc = sb_boer_republics.133.d.present
			}
			desc = sb_boer_republics.133.d
		}
	}
	flavor = {
		first_valid = {
			triggered_desc = { trigger = { has_variable = sb_trn_return_punitive_host } desc = sb_boer_republics.133.f.punitive }
			triggered_desc = { trigger = { has_variable = sb_trn_return_households_due } desc = sb_boer_republics.133.f.households }
			desc = sb_boer_republics.133.f
		}
	}
	event_image = { video = "gfx/event_pictures/africa_construction_colony.bk2" }
	icon = "gfx/interface/icons/event_icons/event_default.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	trigger = { country_definition = cd:TRN has_variable = sb_trn_return_notice_pending }
	option = {
		name = sb_boer_republics.133.a
		default_option = yes
		if = {
			limit = { has_variable = sb_trn_return_notice_pending }
			hidden_effect = { remove_variable = sb_trn_return_notice_pending }
			sb_appoint_trn_marthinus = { DELIVERY = combined }
			if = {
				limit = { has_variable = sb_trn_return_households_due }
				custom_tooltip = sb_natal_refugee_boer_pop_move_tt
				hidden_effect = {
					remove_variable = sb_trn_return_households_due
					sb_trigger_natal_refugee_migration_effect = yes
				}
			}
			hidden_effect = {
				if = { limit = { has_variable = sb_trn_return_punitive_host } remove_variable = sb_trn_return_punitive_host }
			}
		}
	}
}
