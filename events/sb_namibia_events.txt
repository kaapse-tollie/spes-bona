sb_nam.010 = {
	type = country_event
	placement = root

	title = sb_nam.010.t
	desc = sb_nam.010.d
	flavor = sb_nam.010.f

	event_image = {
		video = "gfx/event_pictures/africa_diplomats_negotiating.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3

	trigger = {
		country_definition = cd:GER
		game_date >= 1870.1.1
		NOT = { has_variable = sb_nam_colonial_prestige_resolved_var }
		OR = {
			ig:ig_landowners   ?= { is_in_government = yes }
			ig:ig_armed_forces ?= { is_in_government = yes }
		}
		OR = {
			sb_nam_has_qualifying_overseas_colonial_law = yes
			AND = {
				NOT = { has_journal_entry = je_government_petition_negotiation_version }
				NOT = { has_journal_entry = je_government_petition_negotiation_version_b }
			}
		}
	}

	option = {
		name = sb_nam.010.a
		default_option = yes
		custom_tooltip = sb_nam_010_a_tt
		hidden_effect = {
			set_variable = sb_nam_colonial_prestige_resolved_var
			set_variable = sb_nam_colonial_programme_backed_var
			if = {
				limit = {
					NOT = { sb_nam_has_qualifying_overseas_colonial_law = yes }
				}
				random_law = {
					limit = {
						type = law_type:law_colonial_exploitation
					}
					save_scope_as = sb_nam_promised_law
				}
				if = {
					limit = {
						ig:ig_landowners ?= { is_in_government = yes }
					}
					ig:ig_landowners ?= {
						save_scope_as = ig
						set_variable = {
							name = desired_law_var
							value = scope:sb_nam_promised_law
						}
						set_variable = {
							name = promise_quest_degree
							value = 1
						}
						set_variable = {
							name = promise_quest_type
							value = 2
						}
					}
				}
				else = {
					ig:ig_armed_forces ?= {
						save_scope_as = ig
						set_variable = {
							name = desired_law_var
							value = scope:sb_nam_promised_law
						}
						set_variable = {
							name = promise_quest_degree
							value = 1
						}
						set_variable = {
							name = promise_quest_type
							value = 2
						}
					}
				}
				promise_quest = yes
				if = {
					limit = { has_journal_entry = je_government_petition_negotiation_version }
					add_journal_entry = {
						type = je_government_petition_negotiation_version_b
					}
				}
				else = {
					add_journal_entry = {
						type = je_government_petition_negotiation_version
					}
				}
				if = {
					limit = {
						is_ai = yes
						NOT = { has_law = law_type:law_colonial_exploitation }
					}
					if = {
						limit = { enacting_any_law = yes }
						cancel_enactment = yes
					}
					start_enactment = law_type:law_colonial_exploitation
				}
			}
		}
	}

	option = {
		name = sb_nam.010.b
		custom_tooltip = sb_nam_010_b_tt
		hidden_effect = {
			set_variable = sb_nam_colonial_prestige_resolved_var
			if = {
				limit = {
					ig:ig_landowners ?= { is_in_government = yes }
				}
				ig:ig_landowners ?= {
					add_modifier = {
						name = sb_nam_german_colonial_rebuff
						months = 60
					}
				}
			}
			else = {
				ig:ig_armed_forces ?= {
					add_modifier = {
						name = sb_nam_german_colonial_rebuff
						months = 60
					}
				}
			}
		}
		ai_chance = { base = -1 }
	}
}

			save_scope_as = sb_nam_expedition_leader_3
		}

		name = sb_nam.013.a

			add_modifier = {
				name = modifier_large_expedition_cost
				multiplier = money_amount_multiplier_medium
			}
			add_journal_entry = { type = je_sb_map_namibian_coast }
		}
		ai_chance = { base = 60 }
	}

	option = {
		name = sb_nam.013.b
		trigger = { exists = scope:sb_nam_expedition_leader_2 }
		custom_tooltip = sb_nam_expedition_costs_tt
		hidden_effect = {
			scope:sb_nam_expedition_leader_2 = { set_as_expedition_leader = yes }
			add_modifier = {
				name = modifier_large_expedition_cost
				multiplier = money_amount_multiplier_medium
			}
			add_journal_entry = { type = je_sb_map_namibian_coast }
		}
		ai_chance = { base = 25 }
	}

	option = {
		name = sb_nam.013.c
		trigger = { exists = scope:sb_nam_expedition_leader_3 }
		custom_tooltip = sb_nam_expedition_costs_tt
		hidden_effect = {

				name = modifier_large_expedition_cost

			}
