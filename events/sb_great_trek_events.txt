sb_great_trek.001 = {
	type = country_event
	hidden = yes

	# Receipt is deliberately broad. Preflight belongs in the body so a busy
	# day-90 delivery consumes its reservation and the monthly watchdog can retry.
	trigger = {
		country_definition = cd:MTB
		is_country_alive = yes
		has_variable = sb_great_trek_opening_event_reserved_var
		NOT = { has_variable = sb_ndebele_opening_play_fired }
	}

	immediate = {
		remove_variable = sb_great_trek_opening_event_reserved_var
		if = {
			limit = {
				is_at_war = no
				is_active_in_diplomatic_play = no
				aggressive_diplomatic_plays_permitted = yes
				c:ORA ?= {
					is_country_alive = yes
					is_at_war = no
					is_active_in_diplomatic_play = no
					aggressive_diplomatic_plays_permitted = yes
				}
				NOT = {
					OR = {
						any_diplomatic_play = { initiator = root target = c:ORA is_diplomatic_play_type = dp_native_uprising }
						any_scope_war = { diplomatic_play = { initiator = root target = c:ORA is_diplomatic_play_type = dp_native_uprising } }
					}
				}
			}
			set_variable = sb_ndebele_opening_play_fired
			set_variable = sb_great_trek_opening_launch_lease_var
			create_diplomatic_play = {
				# Vanilla native uprising carries the colonization-rights article.
				type = dp_native_uprising
				target_country = c:ORA
			}
			if = {
				limit = { has_variable = sb_great_trek_opening_launch_lease_var }
				remove_variable = sb_great_trek_opening_launch_lease_var
				remove_variable = sb_ndebele_opening_play_fired
			}
		}
	}
}

sb_great_trek.002 = {
	type = country_event
	placement = ROOT

	title = sb_great_trek.002.t
	desc = sb_great_trek.002.d
	flavor = sb_great_trek.002.f

	duration = 3

	event_image = {
		video = "middleeast_jungle_expedition"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	trigger = {
		is_country_alive = yes
		has_variable = sb_great_trek_outcome_delivery_pending_var
		has_variable = sb_great_trek_outcome_delivery_queued_var
		NOT = { has_variable = sb_great_trek_outcome_delivery_received_var }
		c:MTB ?= {
			is_country_alive = yes
			has_diplomatic_pact = { who = root type = colonization_rights }
		}
	}

	immediate = {
		c:MTB ?= {
			save_scope_as = ndebele
			ruler = {
				save_scope_as = mzilikazi
			}
		}
		ROOT = {
			save_scope_as = transvaal
		}
		c:ZUL ?= {
			save_scope_as = zulu
		}
		s:STATE_TRANSVAAL = {
			save_scope_as = transvaal_region
		}
		ROOT = { set_variable = { name = sb_great_trek_outcome_popup_receipt_var months = 4 } }
	}

	option = {
		name = sb_great_trek.002.a
		default_option = yes
		trigger = {
			is_country_alive = yes
			has_variable = sb_great_trek_outcome_delivery_pending_var
			has_variable = sb_great_trek_outcome_popup_receipt_var
			NOT = { has_variable = sb_great_trek_outcome_delivery_received_var }
			c:MTB ?= {
				is_country_alive = yes
				has_diplomatic_pact = { who = root type = colonization_rights }
			}
		}

		hidden_effect = {
			remove_variable = sb_great_trek_outcome_delivery_pending_var
			remove_variable = sb_great_trek_outcome_delivery_queued_var
			if = { limit = { has_variable = sb_great_trek_outcome_popup_receipt_var } remove_variable = sb_great_trek_outcome_popup_receipt_var }
			set_variable = { name = sb_great_trek_outcome_delivery_received_var days = 30 }
			if = {
				limit = {
					exists = c:MSH
				}
				c:MTB ?= {
					annex = c:MSH
				}
			}
			c:MTB ?= {
				sb_move_mzilikazi_ndebele_to_zimbabwe = yes
			}
		}
			c:MTB ?= {
				s:STATE_VRYSTAAT = {
					set_owner_of_provinces = {
					country = root
					provinces = {
						x52B31E xF17D3D xF075D7 x373E53 x9D6A16 xCEE3D2
					}
				}
					set_owner_of_provinces = {
						country = c:PDI
						provinces = {
							x83E266 x6253E3 xED7851 x75FA4B
						}
					}
				}
			every_scope_state = {
				limit = {
					OR = {
						state_region = s:STATE_TRANSVAAL
						state_region = s:STATE_NORTHERN_TRANSVAAL
					}
				}
				set_state_owner = c:TSW
			}
			every_scope_state = {
				limit = {
					state_region = s:STATE_EAST_TRANSVAAL
				}
				set_state_owner = c:PDI
			}
		}
		if = {
			limit = { exists = c:PDI }
			s:STATE_TRANSVAAL = {
				set_owner_of_provinces = {
					country = c:PDI
					provinces = { x902190 x04FA29 xA84679 xAFDF6D x7F6508 x9A137B }
				}
			}
			s:STATE_NORTHERN_TRANSVAAL = {
				set_owner_of_provinces = {
					country = c:PDI
					provinces = { xD022FF x3E338E x5078AD xB19AFE x7E986B xDABE56 }
				}
			}
		}
		sb_move_mtb_defeat_cape_ec_trekkers_to_vrystaat = yes
		if = {
			limit = { exists = c:TSO }
			s:STATE_EAST_TRANSVAAL = {
				set_owner_of_provinces = {
					country = c:TSO
					provinces = { x0591EA x972FB2 xDF169D x927F5A }
				}
			}
		}
		if = {
			limit = { exists = c:PDI }
			s:STATE_EAST_TRANSVAAL = {
				set_owner_of_provinces = {
					country = c:PDI
					provinces = { x7470A2 xBD0AEB x9037B0 xDC7726 }
				}
			}
		}
		ai_chance = {
			base = 100
		}
	}


	option = {
		name = sb_great_trek.002.a
		trigger = {
			has_variable = sb_great_trek_outcome_delivery_pending_var
			has_variable = sb_great_trek_outcome_popup_receipt_var
			NOT = { AND = { is_country_alive = yes NOT = { has_variable = sb_great_trek_outcome_delivery_received_var } c:MTB ?= { is_country_alive = yes has_diplomatic_pact = { who = root type = colonization_rights } } } }
		}
		remove_variable = sb_great_trek_outcome_delivery_queued_var
		if = { limit = { has_variable = sb_great_trek_outcome_popup_receipt_var } remove_variable = sb_great_trek_outcome_popup_receipt_var }
	}
}

sb_great_trek.006 = {
	type = country_event
	placement = root

	title = sb_great_trek.006.t
	desc = sb_great_trek.006.d
	flavor = sb_great_trek.006.f

	event_image = {
		video = "gfx/event_pictures/africa_construction_colony.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3

	trigger = {
		# The Potchefstroom split can be surfaced either to ORA itself or to a
		# player-controlled Natalia, but the actual spawn work still runs from ORA
		# scope so the inland ownership handoff stays consistent.
		OR = {
			country_definition = cd:ORA
			country_definition = cd:NAL
		}
		NOT = { exists = c:TRN }
		c:ORA ?= {
			is_country_alive = yes
			has_variable = sb_trn_v2_spawn_pending_var
			NOT = { has_variable = sb_trn_v2_spawned_var }
			# Inline the west-Transvaal foothold audit directly here. This avoids a
			# boot-time dependency on a custom scripted trigger for the one place the
			# split event needs to know whether ORA has actually established itself
			# in the Potchefstroom corridor.
			OR = {
				p:xD0A010.state.owner = this
				p:x8BDC4D.state.owner = this
				p:xA22DC2.state.owner = this
				p:xD02190.state.owner = this
				p:x6C1786.state.owner = this
				p:x9A137B.state.owner = this
			}
		}
	}

	option = {
		name = sb_great_trek.006.a
		trigger = {
			is_player = yes
		}
		c:ORA ?= {
			sb_spawn_transvaal_republic_v2 = yes
		}
		play_as = c:TRN
		ai_chance = {
			base = 0
		}
	}

	option = {
		name = sb_great_trek.006.b
		default_option = yes
		c:ORA ?= {
			sb_spawn_transvaal_republic_v2 = yes
		}
		ai_chance = {
			base = 100
		}
	}
}
