# Hidden delay endpoint. Invalid borders fall back to the normal Boer flight;
# wars merely leave the question ready for the next peaceful monthly pulse.
sb_klip_river_county.001 = {
	type = country_event
	hidden = yes

	trigger = {
		country_definition = cd:NAL
		has_variable = sb_klip_river_county_pending_var
		NOT = { has_variable = sb_klip_river_county_resolved_var }
	}

	immediate = {
		if = {
			limit = { has_variable = sb_klip_river_county_scheduled_var }
			remove_variable = sb_klip_river_county_scheduled_var
		}
		if = {
			limit = { NOT = { sb_klip_river_county_setup_valid = yes } }
			sb_klip_river_prepare_standard_boer_flight = yes
		}
		else_if = {
			limit = { sb_klip_river_county_ready_for_commission = yes }
			set_variable = sb_klip_river_county_ready_var
			trigger_event = { id = sb_klip_river_county.010 days = 1 popup = yes }
		}
		else = {
			set_variable = sb_klip_river_county_ready_var
		}
	}
}
