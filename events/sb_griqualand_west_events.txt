sb_griqualand_west.021 = {
	type = country_event
	placement = root

	title = sb_griqualand_west.021.t
	desc = sb_griqualand_west.021.d
	flavor = sb_griqualand_west.021.f

	event_image = {
		video = "europenorthamerica_gold_prospectors"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	duration = 3

	immediate = {
		sb_griqualand_west_save_country_loc_scopes = yes
	}

	trigger = {
		has_global_variable = sb_kimberley_discovery_pending_var
		OR = {
			p:x68B5E8.state.owner = root
			AND = {
				country_definition = cd:CAP
				c:WBL ?= {
					is_country_alive = yes
					any_scope_state = { state_region = s:STATE_NORTHERN_CAPE }
				}
			}
		}
	}

	option = {
		name = sb_griqualand_west.021.a
		default_option = yes
		hidden_effect = {
			sb_kimberley_discover_diamonds = yes
			if = {
				limit = {
					c:CAP ?= {
						is_country_alive = yes
						has_claim = s:STATE_NORTHERN_CAPE
					}
					c:ORA ?= {
						is_country_alive = yes
					}
					OR = {
						c:ORA ?= { is_subject = no }
						c:ORA ?= {
							is_direct_subject_of = c:TRN
							OR = {
								is_subject_type = subject_type_sb_boer_presidential_union
								is_subject_type = subject_type_sb_boer_confederal_partner
							}
						}
					}
				}
				if = {
					limit = { c:ORA ?= { is_subject = no } }
					c:ORA ?= {
						trigger_event = { id = sb_griqualand_west.022 days = 3 popup = yes }
					}
				}
				else = {
					c:TRN ?= {
						set_variable = sb_kimberley_oranje_proxy_var
						trigger_event = { id = sb_griqualand_west.022 days = 3 popup = yes }
					}
				}
			}
			else_if = {
				limit = {
					c:CAP ?= {
						is_country_alive = yes
						has_claim = s:STATE_NORTHERN_CAPE
					}
					c:WBL ?= {
						is_country_alive = yes
						any_scope_state = { state_region = s:STATE_NORTHERN_CAPE }
					}
					NOT = {
						c:ORA ?= {
							is_country_alive = yes
							OR = {
								is_subject = no
								is_direct_subject_of = c:TRN
							}
						}
					}
				}
				c:CAP ?= {
					trigger_event = { id = sb_griqualand_west.244 days = 3 popup = yes }
				}
			}
		}
		ai_chance = { base = 100 }
	}
}
