	placement = root

				desc = sb_griqualand_west.021.waterboer_uncontested_d
				trigger = {
					NOT = { sb_cap_has_northern_cape_foothold = yes }

				}
			}

				desc = sb_griqualand_west.021.d

			}
		}
	}
	flavor = sb_griqualand_west.021.f

	event_image = {
		video = "europenorthamerica_gold_prospectors"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	duration = 3

	immediate = {
		sb_griqualand_west_save_country_loc_scopes = yes
		c:CAP ?= {
			if = {
				limit = {
					sb_cap_has_northern_cape_foothold = yes
					NOT = { has_claim = s:STATE_GRIQUALAND_WEST }
				}
				s:STATE_GRIQUALAND_WEST = { add_claim = c:CAP }
			}
		}
	}

	trigger = {
		has_global_variable = sb_kimberley_discovery_pending_var
		OR = {
			p:x68B5E8.state.owner = root
			AND = {
				country_definition = cd:CAP
				sb_cap_has_northern_cape_foothold = yes
				c:WBL ?= {
					is_country_alive = yes
					any_scope_state = { state_region = s:STATE_GRIQUALAND_WEST }
				}
			}
			AND = {
				NOT = { sb_cap_has_northern_cape_foothold = yes }
				OR = {
					country_definition = cd:ORA
						AND = {
							country_definition = cd:TRN
							OR = {
								has_variable = sb_kimberley_oranje_proxy_var
								has_variable = sb_oranje_federated_into_trn_var
							}
						}
				}
				c:WBL ?= {
					is_country_alive = yes
					any_scope_state = { state_region = s:STATE_GRIQUALAND_WEST }
				}
			}
		}
	}

	option = {
		name = sb_griqualand_west.021.a
		default_option = yes
		hidden_effect = {
			sb_kimberley_discover_diamonds = yes
			if = {
				limit = { NOT = { sb_cap_has_northern_cape_foothold = yes } }
				c:CAP ?= {
					if = {
						limit = { has_claim = s:STATE_GRIQUALAND_WEST }
						s:STATE_GRIQUALAND_WEST = { remove_claim = c:CAP }
					}
				}
				if = {
					limit = {
						OR = {
							country_definition = cd:ORA
							country_definition = cd:TRN
						}
					}
					if = {
						limit = { country_definition = cd:TRN }
						set_variable = sb_kimberley_oranje_proxy_var
					}
					set_variable = sb_kimberley_ora_pressed_claim_var
					sb_griqualand_west_grant_oranje_claim = yes
					trigger_event = { id = sb_griqualand_west.245 days = 3 popup = yes }
				}
				else = {
					sb_kimberley_finalize_wbl_diamond_mines_route = yes
				}
			}
			else = {
				if = {
					limit = {
						c:CAP ?= {
							is_country_alive = yes
							NOT = { has_claim = s:STATE_GRIQUALAND_WEST }
						}
					}
					s:STATE_GRIQUALAND_WEST = { add_claim = c:CAP }
				}
				if = {
					limit = {
						c:CAP ?= {
							is_country_alive = yes
							has_claim = s:STATE_GRIQUALAND_WEST
						}
						OR = {
							AND = {
								c:ORA ?= { is_country_alive = yes }
								OR = {
									c:ORA ?= { is_subject = no }
									c:ORA ?= {
										is_direct_subject_of = c:TRN
										OR = {
											is_subject_type = subject_type_sb_boer_presidential_union
											is_subject_type = subject_type_sb_boer_confederal_partner
										}
									}
								}
							}
							c:TRN ?= {
								is_country_alive = yes
								has_variable = sb_oranje_federated_into_trn_var
							}
						}
					}
					if = {
						limit = {
							c:ORA ?= {
								is_country_alive = yes
								is_subject = no
							}
						}
						c:ORA ?= {
							trigger_event = { id = sb_griqualand_west.022 days = 2 popup = yes }
						}
					}
					else = {
						c:TRN ?= {
							set_variable = sb_kimberley_oranje_proxy_var
							trigger_event = { id = sb_griqualand_west.022 days = 5 popup = yes }
						}
					}
				}
				else_if = {
					limit = {
						c:CAP ?= {
							is_country_alive = yes
							has_claim = s:STATE_GRIQUALAND_WEST
						}
						c:WBL ?= {
							is_country_alive = yes
							any_scope_state = { state_region = s:STATE_GRIQUALAND_WEST }
						}
						NOT = {
							c:ORA ?= {
								is_country_alive = yes
								OR = {
									is_subject = no
									AND = {
										is_direct_subject_of = c:TRN
										OR = {
											is_subject_type = subject_type_sb_boer_presidential_union
											is_subject_type = subject_type_sb_boer_confederal_partner
										}
									}
								}
							}
						}
					}
					c:CAP ?= {
						trigger_event = { id = sb_griqualand_west.246 days = sb_travel_cape_kimberley_message_days popup = yes }
					}
				}
			}
		}
		ai_chance = { base = 100 }
	}
}
