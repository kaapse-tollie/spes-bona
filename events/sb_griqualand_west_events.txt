sb_griqualand_west.021 = {
	type = country_event
	placement = root

	title = sb_griqualand_west.021.t
	desc = sb_griqualand_west.021.d
	flavor = sb_griqualand_west.021.f

	event_image = {
		video = "europenorthamerica_gold_prospectors"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	duration = 3

	trigger = {
		has_global_variable = sb_kimberley_discovery_pending_var
		OR = {
			p:x68B5E8.state.owner = root
			AND = {
				country_definition = cd:CAP
				c:WBL ?= {
					is_country_alive = yes
					p:x68B5E8.state.owner = this
				}
			}
		}
	}

	option = {
		name = sb_griqualand_west.021.a
		default_option = yes
		custom_tooltip = sb_griqualand_west_021_tt
		hidden_effect = {
			sb_phase3_discover_kimberley_diamonds = yes
			if = {
				limit = {
					c:CAP ?= {
						is_country_alive = yes
						has_claim = s:STATE_NORTHERN_CAPE
					}
					c:ORA ?= {
						is_country_alive = yes
						is_subject = no
					}
				}
				c:ORA ?= {
					trigger_event = { id = sb_griqualand_west.022 days = 3 popup = yes }
				}
			}
			else_if = {
				limit = {
					c:CAP ?= {
						is_country_alive = yes
						has_claim = s:STATE_NORTHERN_CAPE
					}
					c:WBL ?= {
						is_country_alive = yes
						p:x68B5E8.state.owner = this
					}
					NOT = {
						c:ORA ?= {
							is_country_alive = yes
							is_subject = no
						}
					}
				}
				c:CAP ?= {
					trigger_event = { id = sb_griqualand_west.244 days = 3 popup = yes }
				}
			}
		}
		ai_chance = { base = 100 }
	}
}

sb_griqualand_west.025 = {
	type = country_event
	placement = root

	title = sb_griqualand_west.025.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = sb_griqualand_west.025.cap_d
				trigger = { has_variable = sb_phase3_wbl_cap_protection_demand_var }
			}
			triggered_desc = {
				desc = sb_griqualand_west.025.ora_d
				trigger = { has_variable = sb_phase3_wbl_ora_protection_demand_var }
			}
			triggered_desc = {
				desc = sb_griqualand_west.025.d
				trigger = { always = yes }
			}
		}
	}
	flavor = sb_griqualand_west.025.f

	event_image = {
		video = "africa_diplomats_negotiating"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3

	trigger = {
		country_definition = cd:WBL
		p:x68B5E8.state.owner = root
		OR = {
			has_variable = sb_phase3_wbl_cap_protection_demand_var
			has_variable = sb_phase3_wbl_ora_protection_demand_var
		}
	}

	option = {
		name = sb_griqualand_west.025.a
		default_option = yes
		custom_tooltip = sb_griqualand_west_025_accept_tt
		if = {
			limit = { has_variable = sb_phase3_wbl_cap_protection_demand_var }
			c:CAP ?= {
				if = {
					limit = { root = { is_subject = yes } }
					transfer_subject = root
				}
				annex = root
			}
			sb_phase3_finalize_kimberley_cap_route = yes
		}
		if = {
			limit = { has_variable = sb_phase3_wbl_ora_protection_demand_var }
			c:ORA ?= {
				if = {
					limit = { root = { is_subject = yes } }
					transfer_subject = root
				}
				else = {
					create_diplomatic_pact = {
						country = root
						type = puppet
					}
				}
			}
			if = {
				limit = {
					is_subject_of = c:ORA
					NOT = { is_subject_type = subject_type_puppet }
				}
				change_subject_type = subject_type_puppet
			}
			sb_phase3_finalize_kimberley_ora_route = yes
		}
		ai_chance = { base = 60 }
	}

	option = {
		name = sb_griqualand_west.025.b
		custom_tooltip = sb_griqualand_west_025_refuse_tt
		if = {
			limit = { has_variable = sb_phase3_wbl_cap_protection_demand_var }
			set_variable = sb_phase3_wbl_cap_crisis_war_var
			c:CAP ?= {
				set_variable = sb_phase3_wbl_cap_crisis_war_var
				create_diplomatic_play = {
					type = dp_annex_war
					target_country = root
				}
			}
		}
		if = {
			limit = { has_variable = sb_phase3_wbl_ora_protection_demand_var }
			set_variable = sb_phase3_wbl_ora_crisis_war_var
			c:ORA ?= {
				set_variable = sb_phase3_wbl_ora_crisis_war_var
				create_diplomatic_play = {
					type = dp_make_protectorate
					target_country = root
				}
			}
		}
		ai_chance = { base = 40 }
	}
}
