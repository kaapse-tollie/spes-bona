sb_boer_conventions.142 = {
	type = country_event
	hidden = yes

	trigger = {
		country_definition = cd:GBR
	}

	immediate = {
		if = {
			limit = {
				c:TRN ?= {
					has_variable = sb_gbr_highveld_slavery_response_pending_var
				}
			}
			c:TRN ?= {
				save_scope_as = highveld_target
			}
		}
		else_if = {
			limit = {
				c:ORA ?= {
					has_variable = sb_gbr_highveld_slavery_response_pending_var
				}
			}
			c:ORA ?= {
				save_scope_as = highveld_target
			}
		}
		else_if = {
			limit = {
				c:ZPB ?= {
					has_variable = sb_gbr_highveld_zpb_response_pending_var
				}
			}
			c:ZPB ?= {
				save_scope_as = highveld_target
			}
		}

		if = {
			limit = {
				exists = scope:highveld_target
				scope:highveld_target ?= {
					is_country_alive = yes
					is_subject = no
					is_at_war = no
					is_active_in_diplomatic_play = no
					aggressive_diplomatic_plays_permitted = yes
					sb_has_any_slavery_law = yes
				}
				aggressive_diplomatic_plays_permitted = yes
				is_at_war = no
				is_active_in_diplomatic_play = no
			}
			scope:highveld_target ?= {
				if = {
					limit = { has_variable = sb_gbr_highveld_slavery_response_pending_var }
					remove_variable = sb_gbr_highveld_slavery_response_pending_var
					set_variable = sb_gbr_highveld_slavery_response_consumed_var
				}
				if = {
					limit = { has_variable = sb_gbr_highveld_zpb_response_pending_var }
					remove_variable = sb_gbr_highveld_zpb_response_pending_var
					set_variable = sb_gbr_highveld_zpb_response_consumed_var
				}
				if = {
					limit = { is_country_type = recognized }
					set_country_type = unrecognized
				}
				change_relations = {
					country = c:GBR
					value = -25
				}
				sb_grant_gbr_investment_rights_in_root = yes
			}
			create_diplomatic_play = {
				type = dp_ban_slavery
				target_country = scope:highveld_target
			}
			random_diplomatic_play = {
				limit = {
					initiator = root
					target = scope:highveld_target
					is_diplomatic_play_type = dp_ban_slavery
				}
				add_war_goal = {
					holder = root
					target_country = scope:highveld_target
					type = make_protectorate
				}
				add_war_goal = {
					holder = scope:highveld_target
					target_country = root
					type = humiliation
					primary_demand = yes
				}
				if = {
					limit = {
						c:CAP ?= {
							is_country_alive = yes
							is_direct_subject_of = root
						}
					}
					add_war_goal = {
						holder = scope:highveld_target
						target_country = c:CAP
						type = liberate_subject
					}
				}
				if = {
					limit = {
						scope:highveld_target ?= {
							sb_boer_convention_nal_is_british_colony = yes
						}
						c:NAL ?= {
							is_direct_subject_of = root
						}
					}
					add_war_goal = {
						holder = scope:highveld_target
						target_country = c:NAL
						type = liberate_subject
					}
				}
			}
		}
		else_if = {
			limit = {
				exists = scope:highveld_target
				scope:highveld_target ?= {
					OR = {
						NOT = { is_country_alive = yes }
						is_subject = yes
						has_law = law_type:law_slavery_banned
					}
				}
			}
			scope:highveld_target ?= {
					if = {
						limit = { has_variable = sb_gbr_highveld_slavery_response_pending_var }
						remove_variable = sb_gbr_highveld_slavery_response_pending_var
						set_variable = sb_gbr_highveld_slavery_response_consumed_var
					}
					if = {
						limit = { has_variable = sb_gbr_highveld_zpb_response_pending_var }
						remove_variable = sb_gbr_highveld_zpb_response_pending_var
						set_variable = sb_gbr_highveld_zpb_response_consumed_var
					}
				}
			}
			else_if = {
				limit = {
					exists = scope:highveld_target
				}
				trigger_event = { id = sb_boer_conventions.142 days = 90 popup = no }
			}
	}
}
