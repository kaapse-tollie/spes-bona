# Hidden 1.14 record goal. Peace acceptance is durable, but claim mutation waits
# for the route close resolver so territorial transfer and callback order are safe.
sb_griqualand_story_revoke_claim = {
	icon = "gfx/interface/icons/war_goals/revoke_claim.dds"
	kind = custom
	settings = {
		require_target_be_part_of_war
		can_add_for_other_country
		targets_enemy_claims
		assent_required
		skip_build_list
		validate_revoke_claims
	}
	execution_priority = 80
	contestion_type = control_own_state
	target_type = country
	possible = { always = no }
	valid = {
		scope:target_state ?= { state_region = s:STATE_GRIQUALAND_WEST }
		scope:target_country ?= { has_claim = s:STATE_GRIQUALAND_WEST }
	}
	maneuvers = {
		value = 0
		add = { desc = "MANEUVERS_BASE_VALUE" value = 5 }
	}
	infamy = { value = 0 }
	on_enforced = { sb_griqualand_record_claim_revocation = yes }
	ai = { is_significant_demand = no }
}

# TRN is the diplomatic party while live subject ORA remains the legal claim owner.
# After federation TRN itself may be the qualified claim owner.
sb_revoke_oranje_griqualand_claim = {
	icon = "gfx/interface/icons/war_goals/revoke_claim.dds"
	kind = custom
	settings = {
		require_target_be_part_of_war
		can_add_for_other_country
		assent_required
		skip_build_list
	}
	execution_priority = 80
	contestion_type = control_own_state
	target_type = country
	possible = { always = no }
	valid = {
		scope:target_country ?= {
			country_definition = cd:TRN
			is_diplomatic_play_enemy_of = root
		}
		scope:target_state ?= { state_region = s:STATE_GRIQUALAND_WEST }
		OR = {
			c:ORA ?= {
				is_country_alive = yes
				is_direct_subject_of = scope:target_country
				has_claim = s:STATE_GRIQUALAND_WEST
			}
			scope:target_country ?= {
				has_variable = sb_oranje_federated_into_trn_var
				has_claim = s:STATE_GRIQUALAND_WEST
			}
		}
	}
	maneuvers = {
		value = 0
		add = { desc = "MANEUVERS_BASE_VALUE" value = 5 }
	}
	infamy = { value = 0 }
	on_enforced = { sb_griqualand_record_claim_revocation = yes }
	ai = { is_significant_demand = no }
}

# TRN leads the play while enforcing its subject Oranje's legal claim. No safe
# third-party timer inverse exists, so this transfer requires assent.
sb_return_oranje_griqualand = {
	icon = "gfx/interface/icons/war_goals/return_state.dds"
	kind = custom
	settings = {
		require_target_be_part_of_war
		annexes_entire_state
		assent_required
		skip_build_list
		validate_conflicts_conquer_state
		validate_target_not_treaty_port
		conflicts_with_country_creation
		conflicts_with_annex_state
	}
	execution_priority = 70
	contestion_type = control_target_state
	target_type = state
	possible = { always = no }
	valid = {
		ROOT = {
			country_definition = cd:TRN
			c:ORA ?= {
				is_country_alive = yes
				is_direct_subject_of = ROOT
				has_claim = s:STATE_GRIQUALAND_WEST
			}
		}
		scope:target_country ?= { country_definition = cd:WBL }
		scope:target_state ?= {
			owner = scope:target_country
			state_region = s:STATE_GRIQUALAND_WEST
		}
	}
	maneuvers = { value = 0 }
	infamy = { value = 0 }
	on_enforced = { scope:target_state ?= { set_state_owner = c:ORA } }
	ai = { is_significant_demand = yes }
}
