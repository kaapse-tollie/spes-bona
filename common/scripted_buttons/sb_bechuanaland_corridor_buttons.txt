je_sb_bechuanaland_demand_caprivi_button = {
	name = "je_sb_bechuanaland_demand_caprivi_button"
	desc = "je_sb_bechuanaland_demand_caprivi_button_desc"

	visible = {
		has_journal_entry = je_sb_bechuanaland_corridor
		sb_bechuanaland_swa_overlord_candidate = yes
	}

	possible = {
		custom_tooltip = {
			text = je_sb_bechuanaland_demand_caprivi_possible_tt
			hidden_trigger = {
				c:SWA ?= { owns_entire_state_region = STATE_HEREROLAND }
				sb_bechuanaland_caprivi_not_held_by_swa = yes
				NOT = { has_variable = sb_bechuanaland_caprivi_demanded_var }
				OR = {
					is_player = yes
					AND = {
						is_ai = yes
						sb_frontier_ai_scripting_enabled = yes
						NOT = { has_variable = sb_bechuanaland_caprivi_ai_delay_var }
						is_at_war = no
						in_default = no
						c:SWA ?= { is_at_war = no }
						sb_bechuanaland_swa_controls_namibian_core = yes
						sb_bechuanaland_primary_boer_actor_at_peace = yes
					}
				}
			}
		}
	}

	ai_chance = {
		value = 1000
	}

	effect = {
		set_variable = sb_bechuanaland_caprivi_demanded_var
		c:GBR ?= { trigger_event = { id = sb_bechuanaland_corridor.030 days = 3 popup = yes } }
	}
}

je_sb_bechuanaland_send_crisis_support_button = {
	name = "je_sb_bechuanaland_send_crisis_support_button"
	desc = "je_sb_bechuanaland_send_crisis_support_button_desc"

	visible = {
		has_journal_entry = je_sb_bechuanaland_corridor
		OR = {
			country_definition = cd:GBR
			sb_bechuanaland_swa_overlord_candidate = yes
		}
	}

	possible = {
		custom_tooltip = {
			text = je_sb_bechuanaland_send_crisis_support_possible_tt
			hidden_trigger = {
				has_global_variable = sb_bechuanaland_crisis_war_active_var
				NOT = { has_variable = sb_bechuanaland_support_cooldown_var }
			}
		}
	}

	ai_chance = {
		value = 1000
	}

	effect = {
		set_variable = { name = sb_bechuanaland_support_cooldown_var months = 12 }
		if = {
			limit = { country_definition = cd:GBR }
			sb_bechuanaland_apply_support_to_british_side = yes
		}
		else_if = {
			limit = { sb_bechuanaland_swa_overlord_candidate = yes }
			sb_bechuanaland_apply_support_to_boer_swa_side = yes
		}
	}
}
