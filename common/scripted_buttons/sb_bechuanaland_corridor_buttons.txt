je_sb_bechuanaland_support_settler_government_button = {
	name = "je_sb_bechuanaland_support_settler_government_button"
	desc = "je_sb_bechuanaland_support_settler_government_button_desc"

	visible = {
		has_journal_entry = je_sb_bechuanaland_corridor
		sb_bechuanaland_is_primary_boer_actor = yes
		NOT = { country_definition = cd:SGO }
		c:SGO ?= { is_country_alive = yes }
	}

	possible = {
		custom_tooltip = {
			text = je_sb_bechuanaland_support_settler_government_possible_tt
			hidden_trigger = {
				in_default = no
				NOT = { has_variable = sb_bechuanaland_boer_subsidy_cooldown_var }
				c:SGO ?= { is_country_alive = yes }
			}
		}
	}

	ai_chance = {
		value = 100
		if = {
			limit = { has_global_variable = sb_bechuanaland_boer_actor_global_scope }
			global_var:sb_bechuanaland_boer_actor_global_scope = {
				if = {
					limit = {
						gold_reserves < 0
						net_fixed_income <= 250
					}
					add = -200
				}
			}
		}
		else = {
			add = -200
		}
	}

	effect = {
		set_variable = { name = sb_bechuanaland_boer_subsidy_active_var months = 12 }
		set_variable = { name = sb_bechuanaland_boer_subsidy_cooldown_var months = 12 }
		add_modifier = {
			name = sb_bechuanaland_finance_boer_settlers_cost
			months = 12
		}
		c:SGO ?= {
			add_modifier = {
				name = sb_bechuanaland_financed_boer_settlement
				months = 12
			}
		}
	}
}

je_sb_bechuanaland_demand_caprivi_button = {
	name = "je_sb_bechuanaland_demand_caprivi_button"
	desc = "je_sb_bechuanaland_demand_caprivi_button_desc"

	visible = {
		has_journal_entry = je_sb_bechuanaland_corridor
		sb_bechuanaland_swa_active_sponsor = yes
		sb_bechuanaland_crisis_escalation_available = yes
	}

	possible = {
		custom_tooltip = {
			text = je_sb_bechuanaland_demand_caprivi_possible_tt
			hidden_trigger = {
				c:SWA ?= { owns_entire_state_region = STATE_HEREROLAND }
				sb_bechuanaland_caprivi_not_held_by_swa = yes
				NOT = { has_variable = sb_bechuanaland_caprivi_demanded_var }
				OR = {
					is_player = yes
					AND = {
						is_ai = yes
						sb_frontier_ai_scripting_enabled = yes
						is_at_war = no
						in_default = no
						c:SWA ?= { is_at_war = no }
						sb_bechuanaland_primary_boer_actor_at_peace = yes
					}
				}
			}
		}
	}

	ai_chance = {
		value = 1000
	}

	effect = {
		if = {
			limit = {
				sb_bechuanaland_crisis_escalation_available = yes
				exists = c:GBR
				c:GBR ?= { is_country_alive = yes }
			}
			change_infamy = 5
			change_relations = { country = c:GBR value = -5 }
			sb_bechuanaland_shift_swa_influence_5 = yes
			sb_bechuanaland_reserve_caprivi_decision = yes
			set_variable = sb_bechuanaland_caprivi_demanded_var
			c:GBR ?= { trigger_event = { id = sb_bechuanaland_corridor.030 days = 3 popup = yes } }
		}
		custom_tooltip = sb_bechuanaland_influence_swa_gain_5_tt
	}
}
