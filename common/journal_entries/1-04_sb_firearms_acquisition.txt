# Generic firearms-acquisition JE for southern African kingdoms that still
# rely heavily on melee-centric warfare at game start.
#
# One supply-based counter: 260 qualifying weeks from an unseeded start.
# Imports and domestic production use the same market-supply test.

je_sb_adopt_firearms = {
	group = je_group_historical_content

	icon = "gfx/interface/icons/event_icons/event_military.dds"
	scripted_progress_bar = sb_firearms_adoption_bar
	widget = {
		gui = "gui/sb_firearms_penalties.gui"
		name = "sb_firearms_penalties"
		container = "custom_widget_container_2"
	}

	is_shown_in_lobby = {
		hidden_trigger = {
			sb_is_firearms_acquisition_country = yes
			NOT = { has_variable = sb_firearms_modernized_var }
		}
	}

	possible = {
		hidden_trigger = {
			sb_is_firearms_acquisition_country = yes
			NOT = { has_variable = sb_firearms_modernized_var }
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_sb_adopt_firearms_status_supplied
				trigger = { sb_has_firearms_market_supply = yes }
			}
			desc = je_sb_adopt_firearms_status_no_progress
		}
	}

	immediate = {
		if = {
			limit = { NOT = { has_variable = sb_firearms_je_seeded_var } }
			sb_seed_initial_firearms_progress = yes
			set_variable = sb_firearms_je_seeded_var
		}
		sb_update_firearms_progress = yes
		sb_repair_iron_age_weapons_penalty = yes
	}

	on_yearly_pulse = {
		effect = {
			sb_repair_iron_age_weapons_penalty = yes
		}
	}

	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = { has_variable = sb_firearms_modernized_var }
					sb_has_firearms_market_supply = yes
					var:sb_firearms_supply_weeks_var < 260
				}
				change_variable = { name = sb_firearms_supply_weeks_var add = 1 }
			}
			sb_update_firearms_progress = yes
			sb_sync_iron_age_weapons_penalty = yes
		}
	}

	complete = {
		custom_tooltip = {
			text = je_sb_adopt_firearms_completion_tt
			OR = {
				has_variable = sb_firearms_modernized_var
				var:sb_firearms_supply_weeks_var >= 260
			}
		}
	}

	on_complete = {
		custom_tooltip = je_sb_adopt_firearms_remove_penalties_tt
		show_as_tooltip = {
			sb_apply_firearms_completion_rewards = yes
		}
		hidden_effect = {
			set_variable = sb_firearms_modernized_var
			sb_remove_iron_age_weapons_modifiers = yes
			if = {
				limit = { has_variable = sb_firearms_applied_tier_var }
				remove_variable = sb_firearms_applied_tier_var
			}
			remove_variable = sb_firearms_supply_weeks_var
			remove_variable = sb_firearms_progress_months_var
			trigger_event = { id = sb_firearms.001 }
		}
	}

	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
