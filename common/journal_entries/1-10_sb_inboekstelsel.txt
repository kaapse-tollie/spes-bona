# Spes Bona - Inboekstelsel System

je_sb_inboekstelsel_system = {
	group = je_group_historical_content

	icon = "gfx/interface/icons/law_icons/inboekstelsel_system.dds"

	scripted_button = je_sb_inboekstelsel_enforce_button
	scripted_button = je_sb_inboekstelsel_relax_button

	is_shown_when_inactive = {
		sb_is_boer_inboekstelsel_country = yes
		sb_has_inboekstelsel_law = yes
		NOT = { has_variable = sb_inboekstelsel_reopen_required_var }
	}

	possible = {
		sb_is_boer_inboekstelsel_country = yes
		sb_has_inboekstelsel_law = yes
	}

	status_desc = {
		desc = je_sb_inboekstelsel_system_status
	}

	complete = {
		always = no
	}

	invalid = {
		OR = {
			NOT = { sb_is_boer_inboekstelsel_country = yes }
			NOT = { sb_has_inboekstelsel_law = yes }
		}
	}

	on_invalid = {
		if = {
			limit = {
				sb_is_boer_inboekstelsel_country = yes
				NOT = { sb_has_inboekstelsel_law = yes }
			}
			if = {
				limit = { has_variable = sb_inboekstelsel_reopen_pending_var }
				remove_variable = sb_inboekstelsel_reopen_pending_var
			}
			set_variable = sb_inboekstelsel_reopen_required_var
		}
	}
	should_be_pinned_by_default_uninvolved_or_context = yes

	weight = 1000
}
