# Spes Bona - BST frontier and Sotho-Tswana journal entries

je_sb_bst_ora_frontier = {
	group = je_group_global_international_situations

	icon = "gfx/interface/icons/event_icons/event_raiding.dds"

	should_be_involved = {
		OR = {
			c:BST ?= root
			sb_bst_oranje_frontier_actor = yes
		}
	}

	should_show_when_not_involved = {
		always = no
	}

	is_shown_in_lobby = {
		OR = {
			c:BST ?= this
			c:ORA ?= this
			c:TRN ?= this
			c:LYD ?= this
			c:ZPB ?= this
			c:NAL ?= this
		}
	}

	is_shown_when_inactive = {
		sb_bst_any_oranje_frontier_actor_exists = yes
		c:BST ?= {
			is_country_alive = yes
			is_subject = no
			sb_bst_holds_frontier_land = yes
		}
	}

	possible = {
		sb_bst_any_oranje_frontier_actor_exists = yes
		c:BST ?= {
			is_country_alive = yes
			is_subject = no
			sb_bst_holds_frontier_land = yes
		}
	}

	scripted_button = je_sb_bst_raid_oranje_button

	status_desc = {
		desc = je_sb_bst_ora_frontier_status
	}

	custom_completion_header = je_sb_bst_ora_frontier_oranje_victory

	complete = {
		custom_tooltip = {
			text = je_sb_bst_ora_frontier_oranje_victory_goal
			any_country = {
				sb_bst_oranje_frontier_actor = yes
				owns_entire_state_region = STATE_VRYSTAAT
				owns_entire_state_region = STATE_DRAKENSBERG
			}
		}
	}

	custom_on_completion_header = je_sb_bst_ora_frontier_on_oranje_victory

	on_complete = {
		random_country = {
			limit = {
				sb_bst_oranje_frontier_actor = yes
				owns_entire_state_region = STATE_VRYSTAAT
				owns_entire_state_region = STATE_DRAKENSBERG
			}
			if = {
				limit = {
					is_country_alive = yes
					owns_entire_state_region = STATE_DRAKENSBERG
					NOT = { has_variable = sb_settled_drakensberg_var }
					NOT = {
						c:BST ?= {
							is_country_alive = yes
						}
					}
				}
				if = {
					limit = {
						NOT = { has_variable = sb_ora_annexed_bst_var }
						NOT = { has_variable = sb_settled_drakensberg_var }
					}
					set_variable = sb_ora_annexed_bst_var
				}
				trigger_event = { id = sb_bst_frontier.300 days = 1 popup = yes }
			}
		}
	}

	custom_failure_header = je_sb_bst_ora_frontier_basotho_victory

	fail = {
		custom_tooltip = {
			text = je_sb_bst_ora_frontier_basotho_victory_goal
			c:BST ?= {
				is_country_alive = yes
				sb_bst_holds_frontier_land = yes
			}
			NOT = { sb_bst_any_oranje_frontier_actor_exists = yes }
		}
	}

	custom_on_failure_header = je_sb_bst_ora_frontier_on_basotho_victory

	on_fail = {
		c:BST ?= {
			if = {
				limit = {
					is_country_alive = yes
					is_subject = no
					owns_entire_state_region = STATE_VRYSTAAT
				}
				if = {
					limit = {
						NOT = { has_journal_entry = je_sb_sotho_tswana_nation }
						NOT = { has_variable = sb_sotho_tswana_nation_completed_var }
					}
					add_journal_entry = {
						type = je_sb_sotho_tswana_nation
					}
				}
				trigger_event = { id = sb_bst_frontier.310 days = 1 popup = yes }
			}
		}
	}

	on_yearly_pulse = {
		effect = {
			c:BST ?= {
				if = {
					limit = {
						is_ai = yes
						sb_bst_frontier_raid_valid = yes
						NOT = { has_variable = sb_bst_oranje_raid_cooldown_var }
					}
					sb_bst_execute_oranje_raid = yes
				}
			}
		}
	}

	invalid = {
		OR = {
			c:BST ?= {
				is_country_alive = yes
				NOT = { sb_bst_holds_frontier_land = yes }
			}
			c:BST ?= {
				is_subject = yes
			}
		}
	}

	weight = 1000
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_sb_settle_drakensberg = {
	group = je_group_historical_content

	icon = "gfx/interface/icons/je_icons/je_great_trek.dds"

	possible = {
			country_has_primary_culture = cu:boer
			has_variable = sb_ora_annexed_bst_var
			NOT = { has_journal_entry = je_sb_bst_ora_frontier }
			NOT = { has_variable = sb_settled_drakensberg_var }
			NOT = {
				c:BST ?= {
				is_country_alive = yes
			}
		}
	}

	status_desc = {
		desc = je_sb_settle_drakensberg_status
	}

	complete = {
		owns_entire_state_region = STATE_DRAKENSBERG
		any_scope_state = {
			state_region = s:STATE_DRAKENSBERG
			culture_percent_state = {
				target = cu:boer
				value >= 0.15
			}
		}
	}

	on_complete = {
		set_variable = sb_settled_drakensberg_var
		remove_variable = sb_ora_annexed_bst_var
		s:STATE_DRAKENSBERG = {
			add_homeland = cu:boer
		}
	}

	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_sb_sotho_tswana_nation = {
	group = je_group_historical_content

	icon = "gfx/interface/icons/invention_icons/nationalism.dds"

	possible = {
		country_definition = cd:BST
		is_subject = no
		owns_entire_state_region = STATE_VRYSTAAT
	}

	status_desc = {
		desc = je_sb_sotho_tswana_nation_status
	}

	complete = {
		owns_entire_state_region = STATE_VRYSTAAT
		owns_entire_state_region = STATE_DRAKENSBERG
		has_technology_researched = nationalism
		sb_bst_boer_free_sta_route = yes
	}

	on_complete = {
		set_variable = sb_sotho_tswana_nation_completed_var
		sb_bst_centralize_sotho_tswana_clients = yes
	}

	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
