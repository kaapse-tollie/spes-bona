# SB override: keep vanilla Scramble for Africa behavior, but attach the
# colonisation-speed compensation here instead of on Civilizing Mission.
TRY_REPLACE:je_scramble_for_africa = {

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_foreign_affairs

	is_shown_when_inactive = {
		NOT = { is_country_type = unrecognized }
		has_technology_researched = quinine
		any_scope_state = {
			is_incorporated = no
			is_in_geographic_region = geographic_region_africa
		}
	}

	possible = {
		has_technology_researched = malaria_prevention
		OR = {
			has_law_or_variant = law_type:law_colonial_resettlement
			has_law_or_variant = law_type:law_colonial_exploitation
		}
		OR = {
			any_scope_state = {
				is_in_geographic_region = geographic_region_africa
				is_incorporated = no
				is_treaty_port = no
			}
			any_subject_or_below = {
				is_in_geographic_region = geographic_region_africa
			}
		}
	}

	immediate = {
		trigger_event = {
			id = africa_colonial_events.1
			popup = yes
		}
		save_scope_as = scramble_for_africa_self

		add_modifier = {
			name = sb_scramble_for_africa_colonial_growth
			months = -1
		}

		set_variable = {
			name = controlled_colonies_in_africa_cached
			value = controlled_colonies_in_africa
		}
	}

	complete = {
		custom_tooltip = {
			text = scramble_for_africa_complete_tt
			var:controlled_colonies_in_africa_cached ?= {
				this >= scope:journal_entry.goal_value
			}
		}
		custom_tooltip = {
			text = scramble_for_africa_complete_2_tt
			any_state_region_in_africa = {
				any_scope_state = {
					owner = {
						OR = {
							is_country_type = recognized
							top_overlord ?= {
								is_country_type = recognized
							}
						}
					}
				}
				percent >= 0.95
			}
		}
	}

	on_complete = {
		trigger_event = {
			id = africa_colonial_events.2
			popup = yes
		}
		if = {
			limit = { has_modifier = sb_scramble_for_africa_colonial_growth }
			remove_modifier = sb_scramble_for_africa_colonial_growth
		}
		if = {
			limit = {
				has_variable = controlled_colonies_in_africa_cached
			}
			remove_variable = controlled_colonies_in_africa_cached
		}
	}

	current_value = {
		value = 7
	}

	fail = {
		custom_tooltip = {
			text = scramble_for_africa_fail_tt
			NOR = {
				any_scope_state = {
					is_incorporated = no
					is_in_geographic_region = geographic_region_africa
				}
				any_subject_or_below = {
					is_in_geographic_region = geographic_region_africa
				}
			}
		}
	}

	on_fail = {
		if = {
			limit = { has_modifier = sb_scramble_for_africa_colonial_growth }
			remove_modifier = sb_scramble_for_africa_colonial_growth
		}
		if = {
			limit = {
				has_variable = controlled_colonies_in_africa_cached
			}
			remove_variable = controlled_colonies_in_africa_cached
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = scramble_for_africa_6_states
				trigger = {
					any_scope_state = {
						count >= 6
						is_incorporated = no
						is_in_geographic_region = geographic_region_africa
					}
				}
			}
			triggered_desc = {
				desc = scramble_for_africa_4_states
				trigger = {
					any_scope_state = {
						count >= 4
						is_incorporated = no
						is_in_geographic_region = geographic_region_africa
					}
				}
			}
			triggered_desc = {
				desc = scramble_for_africa_3_states
				trigger = {
					any_scope_state = {
						count >= 1
						is_incorporated = no
						is_in_geographic_region = geographic_region_africa
					}
				}
			}
			triggered_desc = {
				desc = scramble_for_africa_0_states
			}
		}
	}

	on_timeout = {
	}

	on_weekly_pulse = {
		effect = {
			if = {
				limit = { NOT = { has_modifier = sb_scramble_for_africa_colonial_growth } }
				add_modifier = {
					name = sb_scramble_for_africa_colonial_growth
					months = -1
				}
			}
			set_variable = {
				name = controlled_colonies_in_africa_cached
				value = controlled_colonies_in_africa
			}
		}
	}

	on_yearly_pulse = {
		random_events = {
			100 = 0
			10 = africa_colonial_events.4
			10 = africa_colonial_events.5
			10 = africa_colonial_events.6
			5 = africa_colonial_events.7
			10 = patagonia.2
			10 = amazon.2

			10 = africa_colonial_events.40
			10 = africa_colonial_events.41
			10 = africa_colonial_events.51
			10 = africa_colonial_events.53
			10 = africa_colonial_events.54
		}
	}

	weight = 200

	should_be_pinned_by_default_uninvolved_or_context = yes
}
