# Spes Bona - Highveld journal entry override
# ==============================================================================
# Disables je_struggle_for_the_highveld entirely.
#
# WHY: The vanilla Struggle for the Highveld JE is a simplified, single-JE
#   approach to Southern African politics. Spes Bona replaces it with Cape
#   politics, Great Trek, Highveld settlement, and frontier-specific systems.
#
# HOW: The REPLACE: prefix overwrites the vanilla definition. We set:
#   - is_shown_in_lobby = always no  (hidden from country picker)
#   - possible = always no           (can never activate)
#   - completion outcome = always no (can never complete normally)
#   - failure outcome = always yes   (immediately fails if somehow active)
#   - failure effect = empty        (no effects on failure)
#   - weight = 0                     (lowest priority)
#
# VANILLA FILE: common/journal_entries/??_struggle_for_the_highveld.txt
# RELATED OVERRIDES: common/scripted_buttons/zz_sb_highveld_vanilla_overrides.txt
#   (disables the associated vanilla buttons)
# ==============================================================================

REPLACE:je_struggle_for_the_highveld = {
	activation_description = { always = yes }

	group = je_group_historical_content

	is_shown_in_lobby = {
		always = no
	}

	possible = {
		always = no
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			always = no
		}
		effect = {}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			always = yes
		}
		effect = {
		}
	}

	# Failure outcome immediately concludes any accidentally activated instance.

	# Failure has no side effects.

	weight = 0
}
