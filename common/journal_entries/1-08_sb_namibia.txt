je_sb_consolidation_of_south_west_africa = {
	group = je_group_historical_content

	icon = "gfx/interface/icons/notification_icons/colonial.dds"

	scripted_progress_bar = sb_nam_rule_harshness_bar

	possible = {
		sb_nam_can_run_consolidation_je = yes
	}

	scripted_button = je_sb_delegate_south_west_affairs_to_cape_button

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_sb_consolidation_of_south_west_africa_status_exterminatory
				trigger = {
					scope:journal_entry = {
						"scripted_bar_progress(sb_nam_rule_harshness_bar)" >= 10
					}
				}
			}
			triggered_desc = {
				desc = je_sb_consolidation_of_south_west_africa_status_brutal
				trigger = {
					scope:journal_entry = {
						"scripted_bar_progress(sb_nam_rule_harshness_bar)" >= 7
					}
				}
			}
			triggered_desc = {
				desc = je_sb_consolidation_of_south_west_africa_status_coercive
				trigger = {
					scope:journal_entry = {
						"scripted_bar_progress(sb_nam_rule_harshness_bar)" >= 5
					}
				}
			}
			triggered_desc = {
				desc = je_sb_consolidation_of_south_west_africa_status_supervised
				trigger = {
					scope:journal_entry = {
						"scripted_bar_progress(sb_nam_rule_harshness_bar)" >= 2
					}
				}
			}
			desc = je_sb_consolidation_of_south_west_africa_status_accommodating
		}
	}

	complete = {
		custom_tooltip = {
			text = je_sb_consolidation_of_south_west_africa_tt_ports
			p:xDEAA3C.state.owner = root
			p:xB04050.state.owner = root
		}
		custom_tooltip = {
			text = je_sb_consolidation_of_south_west_africa_tt_events
			has_variable = sb_nam_stage_100_done_var
			has_variable = sb_nam_stage_110_done_var
			has_variable = sb_nam_stage_120_done_var
			has_variable = sb_nam_stage_130_done_var
		}
		custom_tooltip = {
			text = je_sb_consolidation_of_south_west_africa_tt_namaqualand
			any_scope_state = {
				state_region = s:STATE_NAMAQUALAND
				owner = root
				num_provinces >= 35
			}
		}
		custom_tooltip = {
			text = je_sb_consolidation_of_south_west_africa_tt_hereroland
			any_scope_state = {
				state_region = s:STATE_HEREROLAND
				owner = root
				num_provinces >= 68
			}
		}
	}

	invalid = {
		OR = {
			NOT = { p:xDEAA3C.state.owner = root }
			NOT = { p:xB04050.state.owner = root }
		}
	}

	on_monthly_pulse = {
		effect = {
			sb_nam_repair_consolidation_seed = yes
			sb_nam_sync_consolidation_modifiers = yes
		}
	}

	on_complete = {
		set_variable = sb_nam_consolidation_resolved_var
		sb_nam_clear_final_harshness_flags = yes
		sb_nam_clear_administration_package_flags = yes
		if = {
			limit = {
				scope:journal_entry = {
					"scripted_bar_progress(sb_nam_rule_harshness_bar)" >= 10
				}
			}
			set_variable = sb_nam_final_harshness_10_var
		}
		else_if = {
			limit = {
				scope:journal_entry = {
					"scripted_bar_progress(sb_nam_rule_harshness_bar)" >= 7
				}
			}
			set_variable = sb_nam_final_harshness_7_var
		}
		else_if = {
			limit = {
				scope:journal_entry = {
					"scripted_bar_progress(sb_nam_rule_harshness_bar)" >= 5
				}
			}
			set_variable = sb_nam_final_harshness_5_var
		}
		else_if = {
			limit = {
				scope:journal_entry = {
					"scripted_bar_progress(sb_nam_rule_harshness_bar)" >= 2
				}
			}
			set_variable = sb_nam_final_harshness_2_var
		}
		else = {
			set_variable = sb_nam_final_harshness_0_var
		}
		if = {
			limit = {
				scope:journal_entry = {
					"scripted_bar_progress(sb_nam_rule_harshness_bar)" >= 9
				}
			}
			set_variable = sb_nam_administration_exterminatory_var
		}
		else_if = {
			limit = {
				scope:journal_entry = {
					"scripted_bar_progress(sb_nam_rule_harshness_bar)" >= 6
				}
			}
			set_variable = sb_nam_administration_extractive_var
		}
		else_if = {
			limit = {
				scope:journal_entry = {
					"scripted_bar_progress(sb_nam_rule_harshness_bar)" >= 3
				}
			}
			set_variable = sb_nam_administration_charter_var
		}
		else = {
			set_variable = sb_nam_administration_mission_var
		}
		if = {
			limit = {
				NOT = { country_definition = cd:GBR }
				NOT = { country_definition = cd:CAP }
				NOT = { country_definition = cd:SAF }
			}
			trigger_event = { id = sb_nam.140 days = 1 popup = yes }
		}
		else = {
			sb_nam_seed_minority_rights_from_harshness = yes
		}
		sb_nam_clear_consolidation_runtime = yes
	}

	on_invalid = {
		sb_nam_clear_consolidation_runtime = yes
	}

	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
}

		}
	}

	complete = {

			text = je_sb_map_namibian_coast_complete_tt
			scope:journal_entry = {
				"scripted_bar_progress(sb_namibian_coast_survey_progress_bar)" >= 18
			}
		}
	}

	on_complete = {
		trigger_event = { id = sb_nam.014 popup = yes }
	}

	fail = {
		custom_tooltip = {
			text = je_sb_map_namibian_coast_fail_tt
			scope:journal_entry = {
				"scripted_bar_progress(sb_namibian_coast_expedition_peril_bar)" >= 8
			}
		}
	}

	on_fail = {
		trigger_event = { id = sb_nam.015 popup = yes }
	}

	invalid = {
		OR = {
			in_default = yes
			is_subject = yes
			NOT = { is_country_type = recognized }
			NOT = { sb_nam_has_qualifying_overseas_colonial_law = yes }
			NOT = {
				has_strategic_region_interest_tier = {
					strategic_region = sr:region_southern_africa
					value >= 2
				}
			}
			NOT = {
				any_scope_character = {
					has_variable = expedition_leader_var
				}
			}
			has_global_variable = sb_namibian_coast_race_closed_var
		}
	}

	on_invalid = {
		if = {
			limit = { has_global_variable = sb_namibian_coast_race_closed_var }
			trigger_event = { id = sb_nam.016 popup = yes }
		}
		else = {
			trigger_event = { id = sb_nam.015 popup = yes }
		}
	}

	weight = 200
	transferable = no
	should_be_pinned_by_default_uninvolved_or_context = yes
}
