je_sb_consolidating_gaza = {
	group = je_group_historical_content

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	is_shown_in_lobby = {
		OR = {
			c:GZA ?= this
			c:NGN ?= this
		}
	}

	immediate = {
		if = {
			limit = { NOT = { has_variable = sb_gaza_consolidation_seeded_var } }
			set_variable = sb_gaza_consolidation_seeded_var
			sb_gaza_apply_startup_consolidation_effects = yes
		}
	}

	status_desc = {
		desc = je_sb_consolidating_gaza_status
	}

	complete = {
		hidden_trigger = {
			NOT = { has_variable = sb_gaza_consolidation_grace_var }
		}
		s:STATE_LOURENCO_MARQUES = {
			any_scope_state = {
				owner = root
				turmoil < 0.25
			}
		}
		s:STATE_ZAMBEZIA = {
			any_scope_state = {
				owner = root
				turmoil < 0.25
				is_incorporated = yes
			}
		}
	}

	fail = {
		custom_tooltip = {
			text = je_sb_consolidating_gaza_fail_soshangane_tt
			hidden_trigger = {
				ruler = {
					NOT = { has_template = GZA_soshangane }
				}
			}
		}
	}

	invalid = {
		OR = {
			NOT = { has_variable = sb_gaza_intro_sequence_completed_var }
			custom_tooltip = {
				text = je_sb_consolidating_gaza_invalid_subject_tt
				hidden_trigger = {
					is_subject = yes
				}
			}
			custom_tooltip = {
				text = je_sb_consolidating_gaza_invalid_holdings_tt
				hidden_trigger = {
					NOT = { sb_gaza_has_target_holdings = yes }
				}
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			sb_gaza_monthly_random_event = yes
		}
	}

	on_complete = {
		sb_gaza_completion_effect = yes
	}

	event_outcome_completed_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { always = yes }
				desc = event_effects_sb_gaza.090_tt
			}
		}
	}

	on_fail = {
		sb_gaza_failure_effect = yes
	}

	event_outcome_failed_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { always = yes }
				desc = event_effects_sb_gaza.095_tt
			}
		}
	}

	on_invalid = {
		sb_gaza_clear_runtime = yes
	}

	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
}

		}
		custom_tooltip = {
			text = sb_delagoa_east_transvaal_railway_complete_tt
			sb_has_railway_in_east_transvaal = yes
		}
		custom_tooltip = {
			text = sb_delagoa_northern_transvaal_railway_complete_tt
			sb_has_railway_in_northern_transvaal = yes
		}
		custom_tooltip = {
			text = sb_delagoa_market_leader_transit_complete_tt
			sb_delagoa_actor_has_trade_through = yes
		}
	}

	invalid = {
		OR = {
			exists = c:SAF
			has_variable = sb_delagoa_route_complete_var
			has_global_variable = sb_delagoa_route_complete_global_var
			NOT = { sb_delagoa_has_valid_gateway = yes }
			NOT = { sb_is_delagoa_route_actor = yes }

			random_scope_state = {
				limit = {
					owner = {
						owns_entire_state_region = STATE_LOURENCO_MARQUES
					}
				}
				owner = { trigger_event = { id = sb_delagoa.021 days = 1 popup = yes } }
			}
		}
	}

		first_valid = {
			triggered_desc = {
				trigger = { always = yes }
				desc = event_effects_sb_delagoa.020_tt
			}
		}
	}

	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_sb_imperial_confederation_scheme = {
	group = je_group_global_international_situations

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	scripted_progress_bar = sb_imperial_confederation_subject_control_bar
	scripted_progress_bar = sb_imperial_confederation_sea_access_bar

	widget = {
		gui = "gui/com_journal_injects/injects.gui"
		name = "com_journal_entry_progress_bars"
		container = "com_custom_widget_container_scripted_progress_bars"
	}

	immediate = {
		sb_imperial_confederation_validate_involvement = yes
		sb_imperial_confederation_sync_subject_control_counts = yes
		sb_imperial_confederation_sync_subject_control_bar = yes
		sb_imperial_confederation_sync_sea_access_bar = yes
	}

	should_be_involved = {
		sb_should_be_involved_in_imperial_confederation = yes
	}

	should_show_when_not_involved = {
		sb_should_be_involved_in_imperial_confederation = yes
	}

	is_shown_in_lobby = {
		always = no
	}

	is_shown_when_inactive = {
		always = no
	}

	possible = {
		c:GBR ?= {
			is_country_alive = yes
		}
		has_global_variable = sb_imperial_confederation_scheme_unlocked_var
		NOT = { has_global_variable = sb_imperial_confederation_scheme_failed_var }
		NOT = { exists = c:SAF }
	}

	scripted_button = je_sb_imperial_confederation_bind_trade_button
	scripted_button = je_sb_imperial_confederation_form_saf_button

	status_desc = {
		desc = je_sb_imperial_confederation_scheme_status
	}

	custom_completion_header = je_sb_imperial_confederation_scheme_imperial_victory

	complete = {
		custom_tooltip = {
			text = je_sb_imperial_confederation_success_confederate_tt
			exists = c:SAF
		}
	}

	custom_failure_header = je_sb_imperial_confederation_scheme_failure

	fail = {
		OR = {
			custom_tooltip = {
				text = je_sb_imperial_confederation_failure_sea_access_tt
				sb_imperial_confederation_sea_access_failure_ready = yes
			}
			custom_tooltip = {
				text = je_sb_imperial_confederation_failure_terminal_tt
				OR = {
					has_global_variable = sb_imperial_confederation_scheme_failed_var
					AND = {
						has_global_variable = sb_imperial_confederation_scheme_resolved_var
						NOT = { exists = c:SAF }
					}
				}
			}
			custom_tooltip = {
				text = je_sb_imperial_confederation_failure_no_boers_tt
				NOT = { sb_imperial_confederation_has_live_boer_participant = yes }
			}
			custom_tooltip = {
				text = je_sb_imperial_confederation_failure_british_war_tt
				sb_imperial_confederation_britain_at_war_with_participant = yes
			}
			custom_tooltip = {
				text = je_sb_imperial_confederation_failure_subject_escape_tt
				sb_imperial_confederation_bound_subject_escaped = yes
			}
		}
	}

	on_complete = {
		set_global_variable = sb_imperial_confederation_terminal_je_settled_var
		sb_imperial_confederation_clear_runtime = yes
	}

	on_fail = {
		if = {
			limit = { NOT = { has_global_variable = sb_imperial_confederation_scheme_failed_var } }
			sb_imperial_confederation_fail_scheme = yes
		}
		set_global_variable = sb_imperial_confederation_terminal_je_settled_var
		sb_imperial_confederation_clear_runtime = yes
	}

	weight = 1000
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
}
