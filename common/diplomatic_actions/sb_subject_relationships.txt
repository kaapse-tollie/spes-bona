sb_responsible_colony = {
	groups= {
		general
	}
	show_in_lens = no
	texture = "gfx/interface/icons/diplomatic_action_icons/responsible_government.dds"

	potential = {
		always = no
	}

	pact = {
		cost = 10
		has_junior_participant = yes
		maintenance_paid_by = first_country
		income_transfer = 0.3
		second_country_gets_income_transfer = no
		income_transfer_based_on_second_country = yes

		max_target_involvement = 8000
		target_involvement_applies_to = source_country

		relations_progress_per_day = 1
		relations_improvement_max = 50

		subject_type = subject_type_sb_responsible_colony

			second_modifier = {
				country_law_enactment_imposition_success_add = 0.1
				country_liberty_desire_add = -0.05
			}

		actor_can_break = {
			is_diplomatic_play_committed_participant = no
		}

		target_can_break = {
			always = no
		}

		manual_break_effect = {
			create_bidirectional_truce = { country = scope:target_country months = 60 }
		}

		auto_break_effect = {
			create_bidirectional_truce = { country = scope:target_country months = 60 }
		}
	}

	ai = {
		evaluation_chance = {
			value = 0
		}

		will_break = {
			always = no
		}
	}
}

sb_boer_confederal_partner = {
	groups= {
		general
	}
	show_in_lens = no
	texture = "gfx/interface/icons/diplomatic_action_icons/responsible_government.dds"

	potential = {
		always = no
	}

	pact = {
		cost = 10
		has_junior_participant = yes
		maintenance_paid_by = first_country
		income_transfer = 0.05
		second_country_gets_income_transfer = no
		income_transfer_based_on_second_country = yes

		max_target_involvement = 8000
		target_involvement_applies_to = source_country

		relations_progress_per_day = 1
		relations_improvement_max = 50

		subject_type = subject_type_sb_boer_confederal_partner

		second_modifier = {
			country_law_enactment_imposition_success_add = 0.1
		}

		actor_can_break = {
			is_diplomatic_play_committed_participant = no
		}

		target_can_break = {
			always = no
		}

		manual_break_effect = {
			if = {
				limit = {
					has_variable = sb_martinus_confederation_active_var
					scope:target_country ?= {
						has_variable = sb_martinus_confederal_partner_var
					}
				}
				remove_variable = sb_martinus_confederation_active_var
				scope:target_country ?= {
					remove_variable = sb_martinus_confederal_partner_var
					if = {
						limit = { has_variable = sb_martinus_confederal_relationship_v3_var }
						remove_variable = sb_martinus_confederal_relationship_v3_var
					}
				}
			}
			create_bidirectional_truce = { country = scope:target_country months = 60 }
		}

		auto_break_effect = {
			create_bidirectional_truce = { country = scope:target_country months = 60 }
		}
	}

	ai = {
		evaluation_chance = {
			value = 0
		}

		will_break = {
			always = no
		}
	}
}
