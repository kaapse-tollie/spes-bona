# Swaziland (SWZ) — vanilla override.
# Historically: The Swazi kingdom under Sobhuza I (d. c. 1839) and then Mswati II (r. 1840-1868).
# A Nguni kingdom that survived the Mfecane by diplomacy and strategic mountain retreats.
# Smaller and less militarised than Zulu but maintained independence until British protectorate (1903).
# This override removes the vanilla Highveld JE and adjusts the eastern
# highveld setup for the mod's regional content.
# Setup stays conservative relative to ZUL while retaining the documented
# Delagoa Bay trade capacity and the mod's narrower Swazi military profile.
COUNTRIES = {
	c:SWZ ?= {
		# Tier 6 tech: pre-industrial Nguni society, similar to ZUL.
		effect_starting_technology_tier_6_tech = yes
		# Delagoa Bay exchange was established before Dingane imposed an external
		# blockade in 1834; imported access does not imply domestic manufacture.
		add_technology_researched = international_trade

		# Traditional politics: monarchy under the Ngwenyama (king) and Ndlovukazi (queen mother).
		# For unrecognized countries this effect supplies Mercantilism, representing
		# controlled exchange without turning a Zulu blockade into a Swazi policy.
		effect_starting_politics_traditional = yes
		activate_law = law_type:law_sb_amabutho_system
		# Swazi power rested on age-regiment musters and rapid local defense more
		# than on a large permanent standing army. The Amabutho System and starting
		# formation keep that capacity visible and bounded.
		# Subjecthood: subjects of the king, not citizens.
		activate_law = law_type:law_subjecthood
		# Traditionalism: economy based on cattle-keeping and agriculture under royal control.
		activate_law = law_type:law_traditionalism
		# Land-Based Taxation: tribute from chiefs and homesteads.
		activate_law = law_type:law_land_based_taxation
		# Slave Trade: captives from Mfecane-era conflicts, not full trade otherwise messess up demographics later
		activate_law = law_type:law_legacy_slavery

		# Armed Forces ruling: the king's regiments (emabutfo) were central to state power.
		# Keep the vanilla-style plain scope here rather than switching to `?=`, so
		# this override does not change startup behavior beyond the intended edits.
		ig:ig_armed_forces = {
			add_ruling_interest_group = yes
		}

		# Swazi authority historically extended farther west into the highveld
		# than the starting owned slice suggests. This also helps keep early
		# TRN colonization pressure out of the eastern escarpment.
		s:STATE_EAST_TRANSVAAL = {
			add_claim = c:SWZ
		}

		# Make firearms modernization visible in the initial journal UI.
		add_journal_entry = { type = je_sb_adopt_firearms }
		}
	}
