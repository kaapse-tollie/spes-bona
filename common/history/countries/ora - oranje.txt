# Orange Free State (ORA) — Boer republic between the Orange and Vaal rivers.
# Historically, the Orange River Sovereignty was a British protectorate
# (1848-1854) before London withdrew and the Orange Free State
# (Oranje-Vrystaat) was proclaimed. At the 1836 start, Voortrekker settlement
# in the region is only beginning.
# ORA shares most of TRN's setup as a Boer frontier republic. The key
# difference is that ORA now carries the whole inland Voortrekker movement at
# start, so it does begin with frontier colonization active in the v2 setup.
COUNTRIES = {
	c:ORA ?= {
		set_variable = sb_commandant_general_visible_var
		set_variable = sb_early_republic_var
		set_variable = sb_voortrekker_start_var

		# Election date: 1839, mirroring the early Volksraad rhythm used for TRN.
		set_next_election_date = 1839.10.12

		# Selective frontier package: Tier 6 supplies the agrarian, organized-army,
		# and settlement baseline. The additions preserve ORA's carried commando,
		# republican, administrative, trade, diplomatic, and colonization capacity
		# without importing Tier 3's industry, artillery, professions, or broader
		# naval package.
		effect_starting_technology_tier_6_tech = yes
		add_technology_researched = mandatory_service
		add_technology_researched = military_drill
		add_technology_researched = line_infantry
		add_technology_researched = rationalism
		add_technology_researched = tech_bureaucracy
		add_technology_researched = democracy
		# Agrarianism requires romanticism; academia is deliberately not granted.
		add_technology_researched = romanticism
		add_technology_researched = international_trade
		# Natalia inherits ORA's capabilities; navigation lets it build Port Natal
		# and reach the harbour ultimatum narrative without granting Admiralty.
		add_technology_researched = navigation
		add_technology_researched = international_relations
		add_technology_researched = colonization
		add_technology_researched = urban_planning
		# Conservative politics: same Calvinist Boer social structure as TRN.
		effect_starting_politics_conservative = yes
		# Commandant-General Republic: Boer frontier variant of Presidential
		# Republic that keeps the elected Volksraad structure while allowing the
		# ruler to serve as field commandant.
		activate_law = law_type:law_sb_commandant_general_republic
		# Oligarchy: the early trekker Volksraad is still a narrow burgher elite
		# rather than a settled constitutional electorate.
		activate_law = law_type:law_oligarchy
		# National Supremacy: racially exclusive polity, same as TRN.
		activate_law = law_type:law_national_supremacy
		# Elected Bureaucrats: local officials elected by burghers, same as TRN.
		activate_law = law_type:law_elected_bureaucrats
		# Agrarianism: pastoral economy on the highveld grasslands.
		activate_law = law_type:law_agrarianism
		# Land-Based Taxation: farm quitrents as primary revenue.
		activate_law = law_type:law_land_based_taxation
			# Homesteading: settler land claims, same pattern as TRN.
			activate_law = law_type:law_homesteading
			# National Militia: Boer commando system.
			activate_law = law_type:law_national_militia
			active_law:lawgroup_army_model ?= {
				if = {
					limit = {
						NOT = { has_amendment = amendment_type:amendment_sb_commando_corps }
					}
					add_amendment = {
						type = amendment_sb_commando_corps
						sponsor = PREV.ig:ig_rural_folk
					}
				}
			}
			# Discrete Inboekstelsel: same indentured-labour practices as TRN.
			activate_law = law_type:law_discrete_inboekstelsel
			# V2 single-tag trek start: ORA carries the inland trek movement before
		# Transvaal splinters out later, so it needs colonial affairs from day one.
		activate_law = law_type:law_frontier_colonization
		active_law:lawgroup_colonization ?= {
			if = {
				limit = {
					NOT = { has_amendment = amendment_type:amendment_homestead_acts }
				}
				add_amendment = {
					type = amendment_homestead_acts
					sponsor = PREV.ig:ig_rural_folk
				}
			}
		}

		# V2 player-start coalition: keep the opening republic anchored on the
		# Groot Boere, Kommandante, and NG Kerk.
		ig:ig_landowners ?= {
			add_ruling_interest_group = yes
		}
		ig:ig_armed_forces ?= {
			add_ruling_interest_group = yes
		}
		ig:ig_devout ?= {
			add_ruling_interest_group = yes
		}

		# Replace the vanilla Highveld JE with the mod's Great Trek content.
		add_journal_entry = { type = je_sb_great_trek }
		# Intro popup scheduled by sb_on_game_started_after_lobby; see ZUL note.
	}
}
