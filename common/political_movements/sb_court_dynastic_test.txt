# Created only by a reserved court crisis; never by ordinary movement formation.
movement_sb_dynastic_test = {
	category = movement_category_ideological
	
	ideology = ideology_sb_contested_crown
	character_ideologies = { } # Allegiance is separate from an IG leader's wider ideology.
	# Sponsorship is an IG-level crisis position, not a source of permanent leader ideologies.
	can_pressure_interest_group = { always = yes }
	
	disband_trigger = { owner = { NOT = { has_variable = sb_court_dynastic_crisis_var } } }
	
	
	creation_trigger = { always = no }
	creation_weight = { value = 0 }

	
	# The native character-support interface is for agitators, not battalion allocation.
	character_support_trigger = {
		var:sb_court_home_scope ?= {
			has_variable = sb_court_dynastic_crisis_var
			var:sb_court_dynastic_rival_scope ?= prev
		}
	}
	# Native agitator selection chooses the highest valid weight. Only the named prince qualifies.
	character_support_weight = { value = 100000 }
	
	pop_support_trigger = {
		NOT = {
			is_pop_type = slaves
		}
		mog_plp_movement_mainland_general_pop_support_trigger = yes    # Grey_PLP
	}
	
	pop_support_factors = {
		movement_support_low_urbanization
		movement_support_high_literacy
		movement_support_aristocrats
		movement_support_farmers
		movement_support_peasants
		movement_support_clergymen
		movement_support_officers
	}
	
	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
					OR = {
						is_being_suppressed = yes
						is_being_bolstered = yes
					}
				}
			}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}
		add = {
			desc = "URBAN_STATE" # Increases support in rural states, decreases in urban
			value = 0.5
			subtract = state.state_urbanization_rate
			multiply = 10
		}
		
		add = {
			value = 5
			desc = "POP_BASE_SUPPORT"
		}
		
		if = {
			limit = {
				is_pop_type = aristocrats
			}
			add = {
				value = 50
				desc = "POP_ARISTOCRATS"
			}
		}
		else_if = {
			limit = {
				is_pop_type = farmers
			}
			add = {
				value = 20
				desc = "POP_FARMERS"
			}
		}
		else_if = {
			limit = {
				is_pop_type = peasants
			}
			add = {
				value = 20
				desc = "POP_PEASANTS"
			}
		}
		else_if = {
			limit = {
				is_pop_type = clergymen
			}
			add = {
				value = 20
				desc = "POP_CLERGYMEN"
			}
		}
		else_if = {
			limit = {
				is_pop_type = officers
			}
			add = {
				value = 25
				desc = "POP_OFFICERS"
			}
		}
		else_if = {
			limit = {
				is_pop_type = soldiers
			}
			add = {
				value = 10
				desc = "POP_SOLDIERS"
			}
		}
		
		# This is the prince's cause: a disliked incumbent must not suppress his rival's recruitment.
		multiply = { value = owner.sb_court_claimant_popularity_attraction }
		
		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.2
			max = 0.7 # Starts more powerful but caps out early
		}
		
		# Grey_PLP
		multiply = {
			value = mog_plp_movement_mainland_general_pop_support_weight_incorporation_percent_mult
		}
	}
	
	revolution = {
		# Any IG may sponsor, subject to native approval, stance and 5% backing gates.
		interest_group_can_join = { always = yes }
  possible = { owner = { sb_court_dynastic_enabled = yes has_variable = sb_court_dynastic_test_pending_var } }
		weight = {
			value = 100
		}
		
		target_fraction_of_states = { value = owner.var:sb_court_dynastic_target_var }
		state_weight = {
			value = sb_court_dynastic_state_weight
			# Coalition geography is scoped to this crisis; retain CMF's common hook.
			multiply = { value = mog_plp_movement_all_general_state_weight_mult }
		}
	}
	
	
	
 active_law_radicalism_multiplier = 0

}
