REPLACE:movement_utilitarian = {
	category = movement_category_ideological

	active_law_radicalism_multiplier = 0.5

	ideology = ideology_utilitarian_movement
	character_ideologies = {
		ideology_utilitarian_leader
	}

    disband_trigger = {
        has_council_republic_law = yes
    }
    on_disbanded = {
        fold_movement_into_anti_communist_movement_effect = yes
    }

	creation_trigger = {
		NOT = {
			AND = {
				country_definition = cd:CAP
				has_journal_entry = je_sb_cape_politics
			}
		}
		OR = {
			country_has_primary_culture = cu:british
			country_has_primary_culture = cu:anglo_african
			country_has_primary_culture = cu:anglo_canadian
			country_has_primary_culture = cu:australian
			anz_country_is_pakeha = yes #anzfp
		}
	}
	creation_weight = {
		value = 100
		if = {
			limit = {
				owner ?= { # If a movement of this type exists in a neighbouring country, make it more likely to spawn here as well
					any_neighbouring_state = {
						owner = {
							prestige > ROOT.owner.prestige
							any_political_movement = {
								is_political_movement_type = movement_utilitarian
							}
						}
					}
				}
			}
			multiply = {
				value = 2
			}
		}
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}

	character_support_trigger = {
		OR = {
			AND = {
				culture = {
					has_discrimination_trait_group = heritage_group_european
				}
				has_reactionary_ideology = no
				has_conservative_ideology = no
				has_socialist_ideology = no
			}
			has_ideology = ideology:ideology_utilitarian_leader
		}
	}
	character_support_weight = {
		value = 200
		if = {
			limit = {
				has_ideology = ideology:ideology_utilitarian_leader
			}
			multiply = {
				value = 5
			}
		}
	}

	pop_support_trigger = {
		OR = {
			AND = {
				owner = {
					country_is_colonial_or_company = yes
				}
				culture = {
					has_discrimination_trait_group = heritage_group_european
				}
			}
			is_pop_type = bureaucrats
			is_pop_type = capitalists
			is_pop_type = engineers
			is_pop_type = clerks
			is_pop_type = officers
			is_pop_type = academics
		}
		NOR = {
			is_pop_type = laborers
			is_pop_type = peasants
			is_pop_type = slaves
		}
		mog_plp_movement_mainland_general_pop_support_trigger = yes    # Grey_PLP
	}

	pop_support_factors = {
		movement_support_colonial_rule
		movement_support_high_acceptance
		movement_support_bureaucrats
		movement_support_academics
		movement_support_officers
		movement_support_clerks
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
					OR = {
						is_being_suppressed = yes
						is_being_bolstered = yes
					}
				}
			}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}

		add = {
			value = 1
			desc = "POP_BASE_SUPPORT"
		}

		if = {
			limit = {
				owner = {
					country_is_colonial_or_company = yes
				}
				culture = {
					has_discrimination_trait_group = heritage_group_european
				}
			}
			add = {
				desc = "colonial_rule"
				value = 15
				if = {
					limit = {
						is_pop_type = bureaucrats
					}
					multiply = 1.5
				}
			}
		}

		if = {
			limit = {
				is_pop_type = bureaucrats
			}
			add = {
				desc = "POP_BUREAUCRATS"
				value = 9
			}
		}
		else_if = {
			limit = {
				is_pop_type = academics
			}
			add = {
				desc = "POP_ACADEMICS"
				value = 6
			}
		}
		else_if = {
			limit = {
				is_pop_type = capitalists
			}
			add = {
				desc = "POP_CAPITALISTS"
				value = 3
			}
		}
		else_if = {
			limit = {
				is_pop_type = officers
			}
			add = {
				desc = "POP_OFFICERS"
				value = 3
			}
		}
		else_if = {
			limit = {
				is_pop_type = clerks
			}
			add = {
				desc = "POP_CLERKS"
				value = 3
			}
		}

		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.10
		}

		multiply = {
			desc = "POP_ACCEPTANCE"
			value = pop_acceptance
			divide = 75
		}

		# Grey_PLP - always put last
		multiply = {
			value = mog_plp_movement_mainland_general_pop_support_weight_incorporation_percent_mult
		}
	}

	revolution = {
		possible = {
			sb_chiefs_other_uprising_allowed = yes
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = 1
			if = {
				limit = {
					OR = {
						any_scope_building = {
							is_building_group = bg_bureaucracy
							level >= 10
						}
					}
				}
				multiply = 10
			}
			else_if = {
				limit = {
					OR = {
						any_scope_building = {
							is_building_group = bg_bureaucracy
							level >= 5
						}
					}
				}
				multiply = 5
			}
			else = {
				multiply = 0
			}
			# Grey_PLP - always put last
			multiply = {
				value = mog_plp_movement_all_general_state_weight_mult
			}
		}
	}

	secession = {
		possible = {
			sb_chiefs_other_uprising_allowed = yes
			political_movement_support >= 0.05
			scope:clout <= 0
		}

		weight = {
			value = 1000
		}

		state_weight = {
			value = 1
			if = {
				limit = {
					OR = {
						any_scope_building = {
							is_building_group = bg_bureaucracy
							level >= 10
						}
					}
				}
				multiply = 10
			}
			else_if = {
				limit = {
					OR = {
						any_scope_building = {
							is_building_group = bg_bureaucracy
							level >= 5
						}
					}
				}
				multiply = 5
			}
			else = {
				multiply = 0
			}
			# Grey_PLP - always put last
			multiply = {
				value = mog_plp_movement_all_general_state_weight_mult
			}
		}
	}

	additional_radicalism_factors = {
		add = movement_radicalism_schools_funding
		if = {
			limit = {
				neighboring_country_has_insurrectionary_movement_trigger = { TYPE = movement_utilitarian }
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}
		add = movement_radicalism_suppressed_after_civil_war
	}
}

			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_republican_leader
					scope:interest_group ?= {
						has_ideology = ideology:ideology_republican
					}
				}
			}
			multiply = {
				value = 2
			}
		}
	}

	pop_support_trigger = {
		NOR = {
			is_pop_type = aristocrats
			is_pop_type = slaves
			AND = {
				is_pop_type = clergymen
				religion = owner.religion
			}
		}
		mog_plp_movement_mainland_general_pop_support_trigger = yes    # Grey_PLP
	}

	pop_support_factors = {
		movement_support_country_is_autocratic
		movement_support_high_urbanization
		movement_support_high_literacy
		movement_support_radicalization
		movement_support_capitalists
		movement_support_shopkeepers
		movement_support_academics
		movement_support_bureaucrats
		movement_support_clerks
		movement_support_machinists
		movement_support_peasants
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
					OR = {
						is_being_suppressed = yes
						is_being_bolstered = yes
					}
				}
			}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}

		add = {
			desc = "URBAN_STATE"
			value = 10
			multiply = state.state_urbanization_rate
		}

		if = {
			limit = {
				strata = upper
			}
			if = {
				limit = {
					is_pop_type = capitalists
				}
				add = {
					value = 6
					desc = "POP_CAPITALISTS"
				}
			}
			else = {
				add = {
					value = 3
					desc = "UPPER_NO_ICON"
				}
			}
		}
		else_if = {
			limit = {
				strata = middle
			}
			if = {
				limit = {
					is_pop_type = shopkeepers
				}
				add = {
					value = 25
					desc = "POP_SHOPKEEPERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = academics
				}
				add = {
					value = 80
					desc = "POP_ACADEMICS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = bureaucrats
				}
				add = {
					value = 12
					desc = "POP_BUREAUCRATS"
				}
			}
			else = {
				add = {
					value = 6
					desc = "MIDDLE_NO_ICON"
				}
			}
		}
		else_if = {
			limit = {
				strata = lower
			}
			if = {
				limit = {
					is_pop_type = clerks
				}
				add = {
					value = 18
					desc = "POP_CLERKS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = machinists
				}
				add = {
					value = 21
					desc = "POP_MACHINISTS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = laborers
				}
				add = {
					value = 15
					desc = "POP_LABORERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = peasants
				}
				add = {
					value = 2
					desc = "POP_PEASANTS"
				}
			}
			else = {
				add = {
					value = 9
					desc = "LOWER_NO_ICON"
				}
			}
		}

		if = {
			limit = {
				owner ?= {
					country_has_voting_franchise = no
				}
			}
			multiply = {
				value = 1.25
				desc = "COUNTRY_IS_AUTOCRATIC"
			}
		}
		else_if = {
			limit = {
				owner ?= {
					OR = {
						has_law_or_variant = law_type:law_census_voting
						has_law_or_variant = law_type:law_universal_suffrage
					}
				}
			}
			multiply = {
				value = 0.75
				desc = "COUNTRY_IS_DEMOCRATIC"
			}
		}

		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.10
		}

		multiply = { # Multiplied by 2x max
			desc = "POP_RADICALS"
			value = pop_radical_fraction
			add = 1
		}

		# Grey_PLP - always put last
		multiply = {
			value = mog_plp_movement_mainland_general_pop_support_weight_incorporation_percent_mult
		}
	}

	revolution = {
		possible = {
			sb_chiefs_other_uprising_allowed = yes
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = { # State urbanisation rate + clout + 0.5*turmoil - loyalists
            value = state_urbanization_rate
            add = scope:clout
            add = { value = this.turmoil divide = 2 }
            subtract = this.loyalists

			# Grey_PLP - always put last
			multiply = {
				value = mog_plp_movement_all_general_state_weight_mult
			}
		}
	}

	secession = {
		possible = {
			sb_chiefs_other_uprising_allowed = yes
			political_movement_support >= 0.05
			scope:clout <= 0
		}

		weight = {
			value = 1000
		}

		state_weight = {
			value = state_urbanization_rate
            add = { value = this.turmoil divide = 2 }
            subtract = this.loyalists

			# Grey_PLP - always put last
			multiply = {
				value = mog_plp_movement_all_general_state_weight_mult
			}
		}
	}

	additional_radicalism_factors = {
		add = movement_radicalism_schools_funding
		owner = {
			if = {
				limit = { has_institution = institution_police }

				add = {
					value = "institution_investment_level(institution_police)"
					subtract = 2
					multiply = 0.05
					desc = "INSTITUTION_FUNDING_LEVEL_police"
				}
			}
			else = {
				add = {
					value = -0.1
					desc = "NO_INSTITUTION_police"
				}
			}
			if = {
				limit = {
					NOT = {
						has_law_or_variant = law_type:law_universal_suffrage
					}
				}
				add = {
					desc = "NO_UNIVERSAL_SUFFRAGE"
					value = 0.135
					multiply = {
						value = owner.modifier:state_expected_sol_from_literacy
						multiply = 0.10
						add = 0.5
					}
					min = 0.15
				}
			}
			if = {
				limit = {
					has_journal_entry = je_russian_revolution
				}
				add = {
					desc = "russian_revolution_radicalism"
					value = 0.15
				}
			}
		}

		if = {
			limit = {
				owner ?= {
					country_has_voting_franchise = yes
					electoral_confidence < 0.9
				}
			}
			add = {
				value = 1
				subtract = owner.electoral_confidence
				divide = 5
				desc = "low_electoral_confidence"
			}
		}

		if = {
			limit = {
				neighboring_country_has_insurrectionary_movement_trigger = { TYPE = movement_radical }
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}
		add = movement_radicalism_suppressed_after_civil_war
	}
