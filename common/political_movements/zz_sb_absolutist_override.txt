# CMF 1.66.0: preserve all hooks outside the authored Zulu constituency exclusion.
# Equivalent creation/disband gates put cheap exclusions before movement scans.
REPLACE:movement_royalist_absolutist = {
	category = movement_category_ideological
	
	ideology = ideology_absolutist_movement
	character_ideologies = {
		ideology_royalist
		ideology_authoritarian
		ideology_gbbf_bavarocrat #GOTB
	}
	
	disband_trigger = {
		OR = {
			AND = {
				NOT = {
					has_variable = king_in_parliament_possible
				}
				country_has_voting_franchise = yes
			}
			has_variable = monarchy_stamped_out_var
			sb_uses_authored_zulu_movements = yes
			any_political_movement = {
				OR = {
					is_political_movement_type = movement_carlist
					is_political_movement_type = movement_miguelist
				}
			}
			has_law = law_type:law_bakufu
			has_council_republic_law = yes
		}
	}
	on_disbanded = {
		if = {
			limit = {
				owner ?= {
					sb_uses_authored_zulu_movements = no
					country_has_voting_franchise = yes
					NOT = {
						has_variable = monarchy_stamped_out_var
					}
					NOT = { has_council_republic_law = yes }
					NOT = {
						any_political_movement = {
							is_political_movement_type = movement_royalist_constitutional
						}
					}
				}
			}
			owner ?= {
				create_political_movement = { type = movement_royalist_constitutional }
			}
		}
		fold_movement_into_anti_communist_movement_effect = yes
	}
	
	creation_trigger = {
		has_law = law_type:law_monarchy
		country_has_voting_franchise = no
		NOT = { c:FRA ?= this }
		NOR = {
			has_law = law_type:law_bakufu
			AND = {
				c:JAP ?= this
				NOT = { has_variable = japan_restoration_complete }
			}
		}
		sb_uses_authored_zulu_movements = no
		# One pass over movements, after the inexpensive country exclusions.
		NOT = {
			any_political_movement = {
				OR = {
					is_political_movement_type = movement_royalist_constitutional
					is_political_movement_type = movement_carlist
					is_political_movement_type = movement_miguelist
				}
			}
		}
	}
	creation_weight = {
		value = 100
		if = {
			limit = {
				owner ?= { # If a movement of this type exists in a neighbouring country, make it more likely to spawn here as well
					any_neighbouring_state = {
						owner = {
							prestige > ROOT.owner.prestige
							any_political_movement = {
								is_political_movement_type = movement_royalist_absolutist
							}
						}
					}
				}
			}
			multiply = {
				value = 2
			}
		}
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
		if = {
			limit = {
				owner ?= {
					country_has_primary_culture = cu:russian
				}
			}
			set_core_ideology = ideology_tsarist_movement
		}
	}
	
	character_support_trigger = {
		OR = {
			has_ideology = ideology:ideology_royalist
			AND = {
				has_ideology = ideology:ideology_authoritarian
				scope:interest_group ?= {
					law_stance = {
						law = law_type:law_monarchy
						value > neutral
					}
				}
			}
			AND = {
				scope:interest_group ?= {
					law_stance = {
						law = law_type:law_monarchy
						value > neutral
					}
				}
				NOT = {
					law_stance = {
						law = law_type:law_monarchy
						value < approve
					}
					law_stance = {
						law = law_type:law_autocracy
						value < approve
					}
				}
			}
		}
	}
	character_support_weight = {
		value = 200
		if = {
			limit = {
				has_ideology = ideology:ideology_royalist
			}
			multiply = {
				value = 5
			}
		}
	}
	
	pop_support_trigger = {
		NOT = {
			is_pop_type = slaves
		}
		mog_plp_movement_mainland_general_pop_support_trigger = yes    # Grey_PLP
	}
	
	pop_support_factors = {
		movement_support_ruler_popularity
		movement_support_low_urbanization
		movement_support_high_literacy
		movement_support_aristocrats
		movement_support_farmers
		movement_support_peasants
		movement_support_clergymen
		movement_support_officers
	}
	
	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
					OR = {
						is_being_suppressed = yes
						is_being_bolstered = yes
					}
				}
			}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}
		add = {
			desc = "URBAN_STATE" # Increases support in rural states, decreases in urban
			value = 0.5
			subtract = state.state_urbanization_rate
			multiply = 10
		}
		
		add = {
			value = 5
			desc = "POP_BASE_SUPPORT"
		}
		
		if = {
			limit = {
				is_pop_type = aristocrats
			}
			add = {
				value = 50
				desc = "POP_ARISTOCRATS"
			}
		}
		else_if = {
			limit = {
				is_pop_type = farmers
			}
			add = {
				value = 20
				desc = "POP_FARMERS"
			}
		}
		else_if = {
			limit = {
				is_pop_type = peasants
			}
			add = {
				value = 20
				desc = "POP_PEASANTS"
			}
		}
		else_if = {
			limit = {
				is_pop_type = clergymen
			}
			add = {
				value = 20
				desc = "POP_CLERGYMEN"
			}
		}
		else_if = {
			limit = {
				is_pop_type = officers
			}
			add = {
				value = 25
				desc = "POP_OFFICERS"
			}
		}
		else_if = {
			limit = {
				is_pop_type = soldiers
			}
			add = {
				value = 10
				desc = "POP_SOLDIERS"
			}
		}
		
		if = {
			limit = {
				owner ?= {
					has_law = law_type:law_monarchy
				}
				owner.ruler ?= {
					NOT = { has_trait = trait_child }
					law_stance = {
						law = law_type:law_autocracy
						value > neutral
					}
				}
			}
			multiply = { # Absolutist are more reliant on the monarchs popularity than constitutionalists
				add = owner.ruler.popularity
				multiply = 0.01
				add = 1
				min = 0.2
				max = 1.8
				desc = "RULER_POPULARITY"
			}
		}
		else_if = {
			limit = {
				owner ?= {
					has_law = law_type:law_monarchy
				}
				owner.ruler ?= {
					has_trait = trait_child
					law_stance = {
						law = law_type:law_autocracy
						value > neutral
					}
				}
			}
			multiply = {
				value = 0.75
				desc = "RULER_IS_CHILD"
			}
		}
		else_if = {
			limit = {
				owner ?= {
					has_law = law_type:law_monarchy
				}
				owner.ruler ?= {
					law_stance = {
						law = law_type:law_autocracy
						value < neutral
					}
				}
			}
			multiply = {
				value = 0.75
				desc = "RULER_IS_LIBERAL"
			}
		}
		
		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.2
			max = 0.7 # Starts more powerful but caps out early
		}
		
		# Grey_PLP - always put last
		multiply = {
			value = mog_plp_movement_mainland_general_pop_support_weight_incorporation_percent_mult
		}
	}
	
	revolution = {
		possible = { sb_chiefs_other_uprising_allowed = yes }
		weight = {
			value = 100
		}
		
		state_weight = {
			value = 1
			add = scope:clout

			subtract = state_urbanization_rate
			
			if = {
				limit = { is_capital = yes }
				if = {
					limit = { owner ?= { has_law = law_type:law_monarchy } }
					multiply = 10.0
				}
				else = {
					multiply = 0.1
				}
			}
			# Grey_PLP - always put last
			multiply = {
				value = mog_plp_movement_all_general_state_weight_mult
			}
		}
	}
	
	secession = {
		possible = {
			sb_chiefs_other_uprising_allowed = yes
			political_movement_support >= 0.05
			scope:clout <= 0
		}
		
		weight = {
			value = 1000
		}
		
		state_weight = {
			value = 1
			
			subtract = state_urbanization_rate
			
			if = {
				limit = { is_capital = yes }
				if = {
					limit = { owner ?= { has_law = law_type:law_monarchy } }
					multiply = 10.0
				}
				else = {
					multiply = 0.1
				}
			}
			# Grey_PLP - always put last
			multiply = {
				value = mog_plp_movement_all_general_state_weight_mult
			}
		}
		
		forced_tags = {
			BHT = {
				trigger = {
					owner ?= { c:BIC ?= this }
				}
				weight = 1000
			}
		}
	}
	
	additional_radicalism_factors = {
        add = movement_radicalism_royalist_monarchy_abolished

        if = {
            limit = {
                neighboring_country_has_insurrectionary_movement_trigger = { TYPE = movement_royalist_absolutist }
            }
            add = {
                desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
                value = 0.15
            }
        }
        add = movement_radicalism_suppressed_after_civil_war
	}
}
