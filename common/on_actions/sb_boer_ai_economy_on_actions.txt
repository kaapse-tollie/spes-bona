	on_actions = {
		sb_on_boer_ai_economy_yearly_pulse

			limit = {

				game_date < 1856.1.1
			}
			sb_boer_ai_economy_1855 = yes
		}

			limit = {
				game_date >= 1860.1.1

			}
			sb_boer_ai_economy_1860 = yes
		}

			limit = {
				game_date >= 1865.1.1

			}
			sb_boer_ai_economy_1865 = yes
		}
	}
