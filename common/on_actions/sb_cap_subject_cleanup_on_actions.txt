# CAP/ABY subject-state transitions and low-frequency repair.

on_become_subject = {
	on_actions = { sb_on_cap_aby_subject_state_changed }
}

on_become_independent = {
	on_actions = { sb_on_cap_aby_subject_state_changed sb_on_bechuanaland_post_open_independence }
}

on_country_formed = {
	on_actions = { sb_on_bechuanaland_post_open_country_formed }
}

on_yearly_pulse_country = {
	on_actions = { sb_on_cap_aby_subject_state_changed }
}

sb_on_bechuanaland_post_open_independence = {
	effect = {
		sb_bechuanaland_handle_post_open_lifecycle = yes
		if = {
			limit = { sb_bechuanaland_corridor_crisis_is_live = yes }
			sb_bechuanaland_refresh_influence_scopes = yes
		}
	}
}

sb_on_bechuanaland_post_open_country_formed = {
	effect = { sb_bechuanaland_handle_post_open_lifecycle = yes }
}

sb_on_cap_aby_subject_state_changed = {
	trigger = {
		OR = {
			country_definition = cd:CAP
			country_definition = cd:ABY
		}
	}
	effect = { sb_refresh_cap_aby_subject_state = yes }
}
