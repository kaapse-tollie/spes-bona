# Spes Bona - Cape law-checkpoint hooks
#
# The Cape Qualified Franchise is intentionally attached while Cultural
# Exclusion is being enacted, not as an automatic JE outcome. Scheduling it from
# enactment start avoids missing the event when the law passes without an
# advance checkpoint.

on_law_enactment_started = {
	on_actions = { sb_on_cape_law_enactment_started }
}

on_law_enactment_ended = {
	on_actions = { sb_on_cape_law_enactment_ended }
}

sb_on_cape_law_enactment_started = {
	effect = {
		if = {
			limit = {
				country_definition = cd:CAP
				is_enacting_law = law_type:law_cultural_exclusion
				NOT = { has_variable = sb_cqf_first_checkpoint_pending_var }
				currently_enacting_law ?= {
					NOT = { has_amendment = amendment_type:amendment_sb_cape_qualified_franchise }
				}
			}
			set_variable = {
				name = sb_cqf_first_checkpoint_pending_var
				days = 30
			}
			trigger_event = { id = sb_cape.130 days = 21 popup = yes }
		}
	}
}

sb_on_cape_law_enactment_ended = {
	effect = {
		if = {
			limit = {
				country_definition = cd:CAP
				has_variable = sb_cqf_first_checkpoint_pending_var
			}
			remove_variable = sb_cqf_first_checkpoint_pending_var
		}
	}
}
