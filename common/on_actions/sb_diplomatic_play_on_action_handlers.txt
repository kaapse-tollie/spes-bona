# Diplomatic-play, war, and subject-transition handlers.

sb_on_spes_bona_diplomatic_play_started = {
	effect = {
		sb_withdraw_ai_frontier_goods_exports_to_boers = yes
		sb_restore_ai_frontier_force_floors_for_play = yes
		sb_refresh_british_cape_boer_restraint = yes
		if = {
			limit = { is_diplomatic_play_type = dp_sb_xhosa_war_7 }
			scope:target ?= {
				if = {
					limit = {
						OR = {
							country_definition = cd:CAP
							country_definition = cd:ABY
						}
					}
					set_variable = sb_xhosa_frontier_war_7_resolved_var
					if = { limit = { has_variable = sb_xhosa_frontier_war_7_scheduled_var } remove_variable = sb_xhosa_frontier_war_7_scheduled_var }
				}
			}
		}
		if = {
			limit = { is_diplomatic_play_type = dp_sb_xhosa_war_8 }
			scope:target ?= {
				if = {
					limit = {
						OR = {
							country_definition = cd:CAP
							country_definition = cd:ABY
						}
					}
					set_variable = sb_xhosa_frontier_war_8_resolved_var
					if = { limit = { has_variable = sb_xhosa_frontier_war_8_scheduled_var } remove_variable = sb_xhosa_frontier_war_8_scheduled_var }
				}
			}
		}
		if = {
			limit = { is_diplomatic_play_type = dp_sb_xhosa_war_9 }
			scope:target ?= {
				if = {
					limit = {
						OR = {
							country_definition = cd:CAP
							country_definition = cd:ABY
						}
					}
					set_variable = sb_xhosa_frontier_war_9_resolved_var
					if = { limit = { has_variable = sb_xhosa_frontier_war_9_scheduled_var } remove_variable = sb_xhosa_frontier_war_9_scheduled_var }
				}
			}
		}
		if = {
			limit = {
				is_diplomatic_play_type = dp_sb_natal_crisis
			}
			if = {
				limit = {
					c:ZUL ?= {
						is_country_alive = yes
						is_player = yes
					}
					c:ORA ?= {
						is_country_alive = yes
						is_ai = yes
					}
					OR = {
						NOT = { exists = c:TRN }
						c:TRN ?= {
							is_ai = yes
						}
					}
				}
				c:ORA ?= {
					set_variable = {
						name = sb_ora_blood_river_player_zul_extra_force_var
						months = 24
					}
				}
			}
			# Resolve Blood River balance at DP start so the support / no-support
			# outcome is deterministic before war and cannot reroll on later joins.
			c:ZUL ?= {
				if = {
					limit = { is_country_alive = yes }
						trigger_event = { id = sb_natal_crisis.019 }
				}
			}
			sb_raise_blood_river_ai_boer_relief_forces = yes
			sb_join_trn_to_active_natal_war = yes
		}
		sb_handle_natalia_player_support_join = yes
			if = {
				limit = {
					OR = {
						scope:target ?= { country_definition = cd:SWZ }
						root = {
							any_scope_play_involved = {
								c:SWZ ?= THIS
								is_diplomatic_play_target = yes
							}
						}
					}
					c:SWZ ?= {
						is_country_alive = yes
					}
				}
				c:SWZ ?= {
					if = {
						limit = {
							root = {
								any_scope_play_involved = {
									is_player = yes
									is_diplomatic_play_enemy_of = c:SWZ
								}
							}
						}
						sb_swazi_raise_defensive_muster_vs_player = yes
					}
					else_if = {
						limit = {
							NOT = { has_variable = sb_zulu_swazi_war_active_var }
						}
						sb_swazi_raise_defensive_muster = yes
					}
				}
			}
	}
}

sb_on_spes_bona_diplo_play_back_down = {
	effect = {
		sb_cleanup_trn_frontier_deployment_train_for_play = yes
		sb_cleanup_ai_frontier_force_support_for_play = yes
		sb_klip_river_handle_backdown = yes
		sb_bechuanaland_handle_crisis_backdown = yes
		# If Oranje backs down from Pretorius's coercive claim, treat it as a
		# Transvaal victory rather than waiting for a war-goal enforcement hook.
		if = {
			limit = {
				root = {
					is_diplomatic_play_type = dp_sb_martinus_humiliation
				}
				scope:actor ?= {
					country_definition = cd:ORA
				}
				scope:target ?= {
					country_definition = cd:TRN
					is_country_alive = yes
				}
			}
			scope:target = {
				sb_martinus_create_presidential_union_with_ora = yes
			}
		}
		# Pretorius cannot subsequently seek the Free State presidency if
		# Transvaal backs down from the claim it attempted to enforce.
		if = {
			limit = {
				root = {
					is_diplomatic_play_type = dp_sb_martinus_humiliation
				}
				scope:actor ?= {
					country_definition = cd:TRN
				}
				scope:target ?= {
					country_definition = cd:ORA
					is_country_alive = yes
				}
			}
			scope:actor = {
				set_variable = sb_martinus_ambition_resolved_var
				set_variable = sb_martinus_legal_union_resolved_var
				if = {
					limit = { has_variable = sb_martinus_legal_union_pending_var }
					remove_variable = sb_martinus_legal_union_pending_var
				}
			}
			scope:target = {
				if = {
					limit = { has_variable = sb_martinus_legal_union_pending_var }
					remove_variable = sb_martinus_legal_union_pending_var
				}
				sb_martinus_restore_ora_historical_groot_boere_leader = yes
			}
		}

		# Dingane backing down from the Swazi play should still count as a dynastic
		# humiliation rather than a free reset.
		if = {
			limit = {
				root = {
					is_diplomatic_play_type = dp_conquer_state
					scope:target ?= { country_definition = cd:SWZ }
				}
				scope:actor ?= {
					country_definition = cd:ZUL
					has_variable = sb_zulu_swazi_war_active_var
				}
			}
			scope:actor = {
				remove_variable = sb_zulu_swazi_war_active_var
				sb_zulu_apply_swazi_campaign_defeat = yes
			}
			c:SWZ ?= {
				if = {
					limit = { has_variable = sb_zulu_swazi_war_active_var }
					remove_variable = sb_zulu_swazi_war_active_var
				}
			}
		}

		# Kimberley claim dispute: the backing-down side determines whether the
		# Cape proceeds against Waterboersland or Oranje takes over the claim.
		if = {
			limit = {
				root = { is_diplomatic_play_type = dp_sb_griqualand_revoke_claim }
				scope:actor ?= {
					country_definition = cd:ORA
					has_variable = sb_kimberley_cap_ora_dispute_war_var
				}
				scope:target ?= {
					country_definition = cd:CAP
					has_variable = sb_kimberley_cap_ora_dispute_war_var
				}
			}
			scope:actor = { remove_variable = sb_kimberley_cap_ora_dispute_war_var }
			scope:target = {
				remove_variable = sb_kimberley_cap_ora_dispute_war_var
				trigger_event = { id = sb_griqualand_west.249 days = 3 popup = yes }
			}
		}
		if = {
			limit = {
				root = { is_diplomatic_play_type = dp_humiliation }
				scope:actor ?= {
					country_definition = cd:CAP
					has_variable = sb_kimberley_cap_ora_dispute_war_var
				}
				scope:target ?= {
					OR = {
						country_definition = cd:ORA
						country_definition = cd:TRN
					}
					has_variable = sb_kimberley_cap_ora_dispute_war_var
				}
			}
				if = {
						limit = { c:CAP ?= { has_claim = s:STATE_GRIQUALAND_WEST } }
						s:STATE_GRIQUALAND_WEST = { remove_claim = c:CAP }
					}
					sb_griqualand_west_grant_oranje_claim = yes
			scope:actor = { remove_variable = sb_kimberley_cap_ora_dispute_war_var }
			scope:target = {
				remove_variable = sb_kimberley_cap_ora_dispute_war_var
				trigger_event = { id = sb_griqualand_west.245 days = 3 popup = yes }
			}
		}

		if = {
			limit = {
				root = {
					OR = {
						is_diplomatic_play_type = dp_sb_griqualand_cap_revoke_claim_locked
						is_diplomatic_play_type = dp_sb_griqualand_cap_revoke_oranje_proxy_locked
						is_diplomatic_play_type = dp_sb_griqualand_cap_return_wbl_locked
						is_diplomatic_play_type = dp_sb_griqualand_oranje_return_wbl_locked
						is_diplomatic_play_type = dp_sb_griqualand_oranje_return_wbl_proxy_locked
					}
				}
			}
			sb_griqualand_west_try_finalize_refusal_contest = yes
		}

		# Waterboersland crisis wars only seed the remaining fields after the
		# annexation/protectorate question is actually resolved.
		if = {
			limit = {
				scope:actor ?= {
					country_definition = cd:WBL
					has_variable = sb_griqualand_west_cap_annexation_war_var
				}
				scope:target ?= {
					country_definition = cd:CAP
					has_variable = sb_griqualand_west_cap_annexation_war_var
				}
			}
			sb_kimberley_finalize_cap_annexation_route = yes
		}
		if = {
			limit = {
				scope:actor ?= {
					country_definition = cd:CAP
					has_variable = sb_griqualand_west_cap_annexation_war_var
				}
				scope:target ?= {
					country_definition = cd:WBL
					has_variable = sb_griqualand_west_cap_annexation_war_var
				}
			}
			sb_kimberley_finalize_wbl_diamond_mines_route = yes
		}
		if = {
			limit = {
				scope:actor ?= {
					country_definition = cd:WBL
					has_variable = sb_griqualand_west_ora_submission_war_var
				}
				scope:target ?= {
					OR = {
						country_definition = cd:ORA
						country_definition = cd:TRN
					}
					has_variable = sb_griqualand_west_ora_submission_war_var
				}
			}
			if = {
				limit = { scope:target ?= { country_definition = cd:TRN } }
				sb_kimberley_finalize_trn_puppet_route = yes
			}
			else = {
				sb_kimberley_finalize_ora_puppet_route = yes
			}
		}
		if = {
			limit = {
				scope:actor ?= {
					OR = {
						country_definition = cd:ORA
						country_definition = cd:TRN
					}
					has_variable = sb_griqualand_west_ora_submission_war_var
				}
				scope:target ?= {
					country_definition = cd:WBL
					has_variable = sb_griqualand_west_ora_submission_war_var
				}
			}
			sb_kimberley_finalize_wbl_diamond_mines_route = yes
		}


	}
}

					country_definition = cd:CAP
					has_variable = sb_kimberley_cap_ora_dispute_war_var
					scope:diplomatic_play ?= {
						is_diplomatic_play_type = dp_humiliation
					}
				scope:target ?= {
					OR = {
						country_definition = cd:ORA
						country_definition = cd:TRN
					}
					has_variable = sb_kimberley_cap_ora_dispute_war_var
				}
			}
			remove_variable = sb_kimberley_cap_ora_dispute_war_var
			scope:target = { remove_variable = sb_kimberley_cap_ora_dispute_war_var }
			trigger_event = { id = sb_griqualand_west.250 days = 3 popup = yes }
		}

		if = {
			limit = {
				country_definition = cd:GZA
				has_variable = sb_gaza_eastern_transvaal_claim_if_zulu_defeated_var
				scope:target ?= {
					country_definition = cd:ZUL
				}
			}
			if = {
				limit = { NOT = { has_claim = s:STATE_EAST_TRANSVAAL } }
				s:STATE_EAST_TRANSVAAL = { add_claim = root }
			}
			remove_variable = sb_gaza_eastern_transvaal_claim_if_zulu_defeated_var
		}

		if = {
			limit = {
				country_definition = cd:ZUL
				scope:target ?= {
					country_definition = cd:GZA
					has_variable = sb_gaza_eastern_transvaal_claim_if_zulu_defeated_var
				}
			}
			scope:target = {
				remove_variable = sb_gaza_eastern_transvaal_claim_if_zulu_defeated_var
			}
		}

		if = {
			limit = {
				OR = {
					country_definition = cd:ORA
					country_definition = cd:TRN
				}
				has_variable = sb_kimberley_cap_ora_dispute_war_var
				scope:diplomatic_play ?= {
					is_diplomatic_play_type = dp_sb_griqualand_revoke_claim
				}
				scope:target ?= {
					country_definition = cd:CAP
					has_variable = sb_kimberley_cap_ora_dispute_war_var
				}
			}
				if = {
						limit = { c:CAP ?= { has_claim = s:STATE_GRIQUALAND_WEST } }
						s:STATE_GRIQUALAND_WEST = { remove_claim = c:CAP }
					}
					sb_griqualand_west_grant_oranje_claim = yes
			remove_variable = sb_kimberley_cap_ora_dispute_war_var
			scope:target = { remove_variable = sb_kimberley_cap_ora_dispute_war_var }
			trigger_event = { id = sb_griqualand_west.245 days = 3 popup = yes }
		}

		if = {
			limit = {
				country_definition = cd:CAP
				has_variable = sb_griqualand_west_cap_annexation_war_var
				scope:target ?= {
					country_definition = cd:WBL
					has_variable = sb_griqualand_west_cap_annexation_war_var
				}
			}
			sb_kimberley_finalize_cap_annexation_route = yes
		}

		if = {
			limit = {
				country_definition = cd:WBL
				has_variable = sb_griqualand_west_cap_annexation_war_var
				scope:target ?= {
					country_definition = cd:CAP
					has_variable = sb_griqualand_west_cap_annexation_war_var
				}
			}
			sb_kimberley_finalize_wbl_diamond_mines_route = yes
		}

		if = {
			limit = {
				OR = {
					country_definition = cd:ORA
					country_definition = cd:TRN
				}
				has_variable = sb_griqualand_west_ora_submission_war_var
				scope:target ?= {
					country_definition = cd:WBL
					has_variable = sb_griqualand_west_ora_submission_war_var
				}
			}
			if = {
				limit = { country_definition = cd:TRN }
				sb_kimberley_finalize_trn_puppet_route = yes
			}
			else = {
				sb_kimberley_finalize_ora_puppet_route = yes
			}
		}

		if = {
			limit = {
				country_definition = cd:WBL
				has_variable = sb_griqualand_west_ora_submission_war_var
				scope:target ?= {
					OR = {
						country_definition = cd:ORA
						country_definition = cd:TRN
					}
					has_variable = sb_griqualand_west_ora_submission_war_var
				}
			}
			sb_kimberley_finalize_wbl_diamond_mines_route = yes
		}


		if = {
			limit = {
				country_definition = cd:ZUL
				scope:diplomatic_play ?= {
					is_diplomatic_play_type = dp_sb_anglo_zulu_return_state_locked
				}
			}
			trigger_event = { id = sb_anglo_zulu.020 days = 3 popup = yes }
		}

		if = {
			limit = {
				country_definition = cd:NAL
				scope:diplomatic_play ?= {
					is_diplomatic_play_type = dp_sb_anglo_zulu_return_state_locked
				}
			}
			trigger_event = { id = sb_anglo_zulu.030 days = 3 popup = yes }
		}

		if = {
			limit = {
				country_definition = cd:CAP
				scope:diplomatic_play ?= {
					OR = {
						is_diplomatic_play_type = dp_sb_xhosa_war_7
						is_diplomatic_play_type = dp_sb_xhosa_war_8
						is_diplomatic_play_type = dp_sb_xhosa_war_9
					}
				}
			}
			add_modifier = {
				name = sb_cape_victory_against_xhosa
				months = 24
			}
		}

		if = {
			limit = {
				country_definition = cd:ZUL
				NOT = {
					scope:target ?= {
						country_definition = cd:SWZ
					}
				}
				scope:diplomatic_play ?= {
					NOT = { is_diplomatic_play_type = dp_sb_anglo_zulu_return_state_locked }
				}
				NOT = { has_variable = sb_zulu_generic_war_resolution_recent_var }
				NOT = { has_variable = sb_natal_war_active_var }
				NOT = { has_variable = sb_port_natal_british_war_var }
				NOT = { has_variable = sb_zulu_swazi_war_active_var }
			}
			sb_zulu_apply_generic_war_victory = yes
		}

		if = {
			limit = {
				NOT = {
					country_definition = cd:SWZ
				}
				scope:target ?= {
					country_definition = cd:ZUL
					NOT = { has_variable = sb_zulu_generic_war_resolution_recent_var }
					NOT = { has_variable = sb_natal_war_active_var }
					NOT = { has_variable = sb_port_natal_british_war_var }
					NOT = { has_variable = sb_zulu_swazi_war_active_var }
				}
				scope:diplomatic_play ?= {
					NOT = { is_diplomatic_play_type = dp_sb_anglo_zulu_return_state_locked }
				}
			}
			scope:target = {
				sb_zulu_apply_generic_war_defeat = yes
			}
		}

		if = {
			limit = {
				country_definition = cd:ZUL
				has_variable = sb_zulu_swazi_war_active_var
				scope:target ?= {
					country_definition = cd:SWZ
					has_variable = sb_zulu_swazi_war_active_var
				}
			}
			remove_variable = sb_zulu_swazi_war_active_var
			scope:target = {
				remove_variable = sb_zulu_swazi_war_active_var
			}
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_EAST_TRANSVAAL
					}
				}
				sb_zulu_apply_swazi_campaign_victory = yes
				if = {
					limit = {
						is_ai = yes
						sb_frontier_ai_scripting_enabled = yes
						sb_frontier_ai_behavior_strict_historical = no
						has_variable = sb_zulu_blood_river_victory_var
						ruler = {
							has_template = ZUL_dingane
						}
						c:GZA ?= {
							is_country_alive = yes
							is_ai = yes
							is_subject = no
						}
						NOT = { has_variable = sb_zulu_gaza_followup_resolved_var }
					}
					set_variable = sb_zulu_gaza_followup_resolved_var
					random_list = {
						50 = {
							set_variable = sb_zulu_gaza_followup_pending_var
						}
						50 = { }
					}
				}
			}
			else = {
				sb_zulu_apply_swazi_campaign_stalemate = yes
			}
		}

		if = {
			limit = {
				country_definition = cd:SWZ
				has_variable = sb_zulu_swazi_war_active_var
				scope:target ?= {
					country_definition = cd:ZUL
					has_variable = sb_zulu_swazi_war_active_var
				}
			}
			remove_variable = sb_zulu_swazi_war_active_var
			scope:target = {
				remove_variable = sb_zulu_swazi_war_active_var
				sb_zulu_apply_swazi_campaign_defeat = yes
			}
		}

		# ORA wins the Natal crisis war.
		if = {
			limit = {
				country_definition = cd:ORA
				scope:target ?= {
					country_definition = cd:ZUL
					has_variable = sb_natal_war_active_var
				}
				has_variable = sb_natal_war_active_var
				scope:diplomatic_play ?= {
					is_diplomatic_play_type = dp_sb_natal_crisis
				}
			}
			remove_variable = sb_natal_war_active_var
			scope:target = {
				remove_variable = sb_natal_war_active_var
			}
				trigger_event = { id = sb_natal_crisis.070 days = 3 popup = yes }
		}

		# ZUL wins the Natal crisis war.
		if = {
			limit = {
				country_definition = cd:ZUL
				scope:target ?= {
					country_definition = cd:ORA
					has_variable = sb_natal_war_active_var
				}
				has_variable = sb_natal_war_active_var
				scope:diplomatic_play ?= {
					is_diplomatic_play_type = dp_sb_natal_crisis
				}
			}
			remove_variable = sb_natal_war_active_var
			scope:target = {
				remove_variable = sb_natal_war_active_var
			}
			trigger_event = { id = sb_natal_crisis.080 days = 3 popup = yes }
		}

		# If no Natal colony exists, a completed Blood River settlement lets
		# Britain reorganize annexed Zulu territory as a new colonial shell.
		if = {
			limit = {
				country_definition = cd:GBR
				NOT = { exists = c:NAL }
				scope:target ?= {
					country_definition = cd:ZUL
				}
				scope:diplomatic_play ?= {
					is_diplomatic_play_type = dp_annex_war
				}
				OR = {
					scope:target ?= {
						has_variable = sb_zulu_blood_river_resolved_var
					}
					c:ORA ?= {
						has_variable = sb_natal_path_resolved_var
					}
				}
			}
			set_variable = {
				name = sb_natal_proclamation_pending_var
				days = 30
			}
			scope:target = {
				if = {
					limit = { has_variable = sb_port_natal_british_war_var }
					remove_variable = sb_port_natal_british_war_var
				}
				if = {
					limit = { has_variable = sb_port_natal_raid_var }
					remove_variable = sb_port_natal_raid_var
				}
			}
			trigger_event = { id = sb_zulu_dynasty.091 days = 7 popup = yes }
		}

		# If British Natal already exists, any Zululand acquired directly by
		# Britain in a victorious war is handed to the established colony.
		if = {
			limit = {
				country_definition = cd:GBR
				scope:target ?= {
					country_definition = cd:ZUL
				}
				scope:diplomatic_play ?= {
					OR = {
						is_diplomatic_play_type = dp_annex_war
						is_diplomatic_play_type = dp_conquer_state
					}
				}
				c:NAL ?= {
					is_country_alive = yes
					is_subject_of = c:GBR
					has_variable = sb_natalia_british_colony_resolved_var
				}
				any_scope_state = {
					state_region = s:STATE_ZULULAND
				}
			}
			scope:target = {
				if = {
					limit = { has_variable = sb_port_natal_british_war_var }
					remove_variable = sb_port_natal_british_war_var
				}
				if = {
					limit = { has_variable = sb_port_natal_raid_var }
					remove_variable = sb_port_natal_raid_var
				}
			}
			trigger_event = { id = sb_anglo_zulu.040 days = 3 popup = yes }
		}

		# A surviving victory over the British should feel exceptional.
		if = {
			limit = {
				country_definition = cd:ZUL
				scope:target ?= {
					country_definition = cd:GBR
				}
				has_variable = sb_port_natal_british_war_var
				scope:diplomatic_play ?= {
					is_diplomatic_play_type = dp_annex_war
				}
			}
			remove_variable = sb_port_natal_british_war_var
			if = {
				limit = { has_variable = sb_port_natal_raid_var }
				remove_variable = sb_port_natal_raid_var
			}
			trigger_event = { id = sb_zulu_dynasty.092 days = 7 popup = yes }
		}

		# ORA wins the opening Ndebele war and the inland trek consolidates.
		if = {
			limit = {
				country_definition = cd:ORA
				scope:target ?= {
					country_definition = cd:MTB
				}
				scope:diplomatic_play ?= {
					is_diplomatic_play_type = dp_native_uprising
				}
			}
			trigger_event = { id = sb_great_trek.002 days = 3 popup = yes }
		}
	}
