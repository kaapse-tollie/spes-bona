				# Exactly one non-timer accepted side awards that side's dynastic result.
				if = {
					limit = { is_country_alive = yes has_variable = sb_zulu_generic_goal_accepted_pending_var NOT = { has_variable = sb_zulu_generic_defeat_goal_accepted_pending_var } }
					sb_zulu_apply_generic_war_victory = yes
				}
				else_if = {
					limit = { is_country_alive = yes has_variable = sb_zulu_generic_defeat_goal_accepted_pending_var NOT = { has_variable = sb_zulu_generic_goal_accepted_pending_var } }
					sb_zulu_apply_generic_war_defeat = yes
				}
				if = { limit = { has_variable = sb_zulu_generic_goal_accepted_pending_var } remove_variable = sb_zulu_generic_goal_accepted_pending_var }
				if = { limit = { has_variable = sb_zulu_generic_defeat_goal_accepted_pending_var } remove_variable = sb_zulu_generic_defeat_goal_accepted_pending_var }
				remove_variable = sb_zulu_generic_war_play_scope
				remove_variable = sb_zulu_generic_war_active_var
			}
			remove_global_variable = sb_zulu_generic_war_coordinator_scope_global_var
		}
		if = {
			limit = {
				root = { is_diplomatic_play_type = dp_conquer_state }
				scope:actor ?= { country_definition = cd:ZUL }
				scope:target ?= { country_definition = cd:SWZ }
				c:ZUL ?= {
					has_variable = sb_zulu_swazi_war_active_var
					has_variable = sb_zulu_swazi_play_scope
					var:sb_zulu_swazi_play_scope = root
				}
			}
			c:ZUL ?= {
				if = {
					limit = {
						is_country_alive = yes
						has_variable = sb_zulu_swazi_goal_seen_pending_var
						has_variable = sb_zulu_swazi_target_state_scope
						var:sb_zulu_swazi_target_state_scope ?= { owner = c:ZUL }
						NOT = {
							OR = {
								has_variable = sb_swazi_goal_accepted_pending_var
								AND = {
									has_variable = sb_swazi_goal_seen_pending_var
									var:sb_zulu_swazi_target_state_scope ?= { owner = c:SWZ }
								}
							}
						}
					}
					sb_zulu_apply_swazi_campaign_victory = yes
				}
				else_if = {
					limit = {
						is_country_alive = yes
						has_variable = sb_zulu_swazi_target_state_scope
						OR = {
							has_variable = sb_swazi_goal_accepted_pending_var
							AND = {
								has_variable = sb_swazi_goal_seen_pending_var
								var:sb_zulu_swazi_target_state_scope ?= { owner = c:SWZ }
							}
						}
						NOT = {
							AND = {
								has_variable = sb_zulu_swazi_goal_seen_pending_var
								var:sb_zulu_swazi_target_state_scope ?= { owner = c:ZUL }
							}
						}
					}
					sb_zulu_apply_swazi_campaign_defeat = yes
				}
				else_if = {
					limit = { is_country_alive = yes }
					# Zero or two surviving goal-side predicates are non-exclusive.
					sb_zulu_apply_swazi_campaign_stalemate = yes
				}
			}
			sb_cleanup_swazi_defensive_muster_for_play = yes
			sb_zulu_clear_swazi_campaign_runtime = yes
		}
		if = {
			limit = {
				root = { is_diplomatic_play_type = dp_native_uprising }
				scope:actor ?= { country_definition = cd:MTB }
				scope:target ?= { country_definition = cd:ORA }
				c:ORA ?= { has_variable = sb_great_trek_opening_play_scope var:sb_great_trek_opening_play_scope = root }
			}
			if = {
				limit = {
					c:ORA ?= { is_country_alive = yes }
					c:MTB ?= {
						is_country_alive = yes
						has_diplomatic_pact = { who = c:ORA type = colonization_rights }
					}
				}
				c:ORA = { sb_great_trek_queue_outcome_delivery = yes }
			}
			c:ORA ?= {
				if = { limit = { has_variable = sb_great_trek_goal_accepted_pending_var } remove_variable = sb_great_trek_goal_accepted_pending_var }
				remove_variable = sb_great_trek_opening_play_scope
			}
			c:MTB ?= { if = { limit = { has_variable = sb_great_trek_opening_launch_lease_var } remove_variable = sb_great_trek_opening_launch_lease_var } }
		}
		sb_klip_river_handle_war_end = yes
		sb_bechuanaland_resolve_crisis_war_at_end = yes

		# Route-local close cleanup. Leases are consumed on start, but all terminal
		# paths clear them again so failed/backed-down plays cannot poison a retry.
		if = {
			limit = { root = { is_diplomatic_play_type = dp_annex_war initiator = c:ORA target = c:BST } c:ORA ?= { has_variable = sb_ora_bst_1856_play_scope var:sb_ora_bst_1856_play_scope = root } }
			sb_cleanup_ora_bst_1856_play_support = yes
			c:ORA = { remove_variable = sb_ora_bst_1856_play_scope }
		}
		if = {
			limit = { root = { is_diplomatic_play_type = dp_sb_nrp_boundary_confrontation } c:NRP ?= { has_variable = sb_nrp_boundary_play_scope var:sb_nrp_boundary_play_scope = root } }
			c:NRP = { sb_nrp_commit_deferred_defence_pact = yes sb_nrp_clear_boundary_confrontation_runtime = yes }
		}
		c:NAL ?= { if = { limit = { has_variable = sb_natal_zulu_secession_launch_lease_var } remove_variable = sb_natal_zulu_secession_launch_lease_var } }
		sb_refresh_british_cape_boer_restraint = yes
		sb_cleanup_trn_frontier_deployment_train_for_play = yes
		sb_cleanup_blood_river_assistance_for_play = yes
		sb_cleanup_swazi_defensive_muster_for_play = yes
		sb_cleanup_bst_gun_war_defence_for_play = yes
		# A failed or negotiated British annexation leaves live ZUL untouched and
		# discards only the snapshot created for this annexation play.
		if = {
			limit = {
				root = { is_diplomatic_play_type = dp_annex_war }
				scope:actor ?= { sb_zululand_british_postwar_owner = yes }
				c:NAL ?= { has_variable = sb_zululand_preannexation_royal_house_archived_var }
				scope:target ?= {
					country_definition = cd:ZUL
					is_country_alive = yes
				}
			}
			scope:target = { sb_zululand_discard_preannexation_royal_house_archive = yes }
		}
		# War end uses the identical exact-root terminal resolver as back-down.
		sb_zpb_crackdown_resolve_bound_terminal = yes

		if = {
			limit = {
				root = { is_diplomatic_play_type = dp_annex_war }
				scope:actor ?= {
					country_definition = cd:GBR
					has_variable = sb_anglo_zulu_pressure_active_var
				}
				scope:target ?= {
					country_definition = cd:ZUL
					has_variable = sb_anglo_zulu_pressure_active_var
				}
			}
			scope:actor = {
				remove_variable = sb_anglo_zulu_pressure_active_var
			}
			scope:target = {
				remove_variable = sb_anglo_zulu_pressure_active_var
			}
			c:NAL ?= {
				if = {
					limit = { has_variable = sb_anglo_zulu_pressure_active_var }
					remove_variable = sb_anglo_zulu_pressure_active_var
				}
			}
		}
	}
}
