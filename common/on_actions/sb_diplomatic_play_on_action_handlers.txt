# Diplomatic-play, war, and subject-transition handlers.

sb_on_spes_bona_diplomatic_play_started = {
	effect = {
		sb_withdraw_ai_frontier_goods_exports_to_boers = yes
		sb_restore_ai_frontier_force_floors_for_play = yes
		sb_refresh_british_cape_boer_restraint = yes
		if = {
			limit = {
				is_diplomatic_play_type = dp_annex_war
				scope:initiator ?= { sb_zululand_british_postwar_owner = yes }
				scope:target ?= {
					country_definition = cd:ZUL
					is_country_alive = yes
				}
				c:NAL ?= {
					is_country_alive = yes
					has_variable = sb_natalia_british_colony_resolved_var
				}
			}
			scope:target = { sb_zululand_archive_preannexation_royal_house = yes }
		}
		if = {
			limit = { is_diplomatic_play_type = dp_sb_xhosa_war_7 }
			scope:target ?= {
				if = {
					limit = {
						OR = {
							country_definition = cd:CAP
							country_definition = cd:ABY
						}
					}
					set_variable = sb_xhosa_frontier_war_7_resolved_var
					if = { limit = { has_variable = sb_xhosa_frontier_war_7_scheduled_var } remove_variable = sb_xhosa_frontier_war_7_scheduled_var }
				}
			}
		}
		if = {
			limit = { is_diplomatic_play_type = dp_sb_xhosa_war_8 }
			scope:target ?= {
				if = {
					limit = {
						OR = {
							country_definition = cd:CAP
							country_definition = cd:ABY
						}
					}
					set_variable = sb_xhosa_frontier_war_8_resolved_var
					if = { limit = { has_variable = sb_xhosa_frontier_war_8_scheduled_var } remove_variable = sb_xhosa_frontier_war_8_scheduled_var }
				}
			}
		}
		if = {
			limit = { is_diplomatic_play_type = dp_sb_xhosa_war_9 }
			scope:target ?= {
				if = {
					limit = {
						OR = {
							country_definition = cd:CAP
							country_definition = cd:ABY
						}
					}
					set_variable = sb_xhosa_frontier_war_9_resolved_var
					if = { limit = { has_variable = sb_xhosa_frontier_war_9_scheduled_var } remove_variable = sb_xhosa_frontier_war_9_scheduled_var }
				}
			}
		}
		if = {
			limit = {
				is_diplomatic_play_type = dp_sb_natal_crisis
			}
			if = {
				limit = {
					OR = {
						AND = {
							scope:initiator ?= { country_definition = cd:ORA has_variable = sb_natal_war_active_var }
							scope:target ?= { country_definition = cd:ZUL has_variable = sb_natal_war_active_var }
						}
						AND = {
							scope:initiator ?= { country_definition = cd:ZUL has_variable = sb_natal_war_active_var }
							scope:target ?= { country_definition = cd:ORA has_variable = sb_natal_war_active_var }
						}
					}
				}
				if = { limit = { has_global_variable = sb_natal_story_initiator_demand_accepted_global_var } remove_global_variable = sb_natal_story_initiator_demand_accepted_global_var }
				if = { limit = { has_global_variable = sb_natal_story_target_demand_accepted_global_var } remove_global_variable = sb_natal_story_target_demand_accepted_global_var }
				if = { limit = { has_global_variable = sb_natal_story_war_resolved_global_var } remove_global_variable = sb_natal_story_war_resolved_global_var }
				set_global_variable = sb_natal_story_war_active_global_var
			}
			# Resolve Blood River balance at DP start so the support / no-support
			# outcome is deterministic before war and cannot reroll on later joins.
			c:ZUL ?= {
				if = {
					limit = { is_country_alive = yes }
						trigger_event = { id = sb_natal_crisis.019 }
				}
			}
			sb_join_trn_to_active_natal_war = yes
			sb_reconcile_blood_river_assistance_for_play = yes
		}
		if = {
			limit = {
				is_diplomatic_play_type = dp_sb_martinus_humiliation
				OR = {
					AND = {
						scope:initiator ?= { country_definition = cd:TRN has_variable = sb_martinus_ambition_resolved_var has_variable = sb_martinus_story_humiliation_route_var }
						scope:target ?= { country_definition = cd:ORA }
					}
					AND = {
						scope:initiator ?= { country_definition = cd:ORA }
						scope:target ?= { country_definition = cd:TRN has_variable = sb_martinus_ambition_resolved_var has_variable = sb_martinus_story_humiliation_route_var }
					}
				}
			}
			if = { limit = { has_global_variable = sb_martinus_story_initiator_demand_accepted_global_var } remove_global_variable = sb_martinus_story_initiator_demand_accepted_global_var }
			if = { limit = { has_global_variable = sb_martinus_story_target_demand_accepted_global_var } remove_global_variable = sb_martinus_story_target_demand_accepted_global_var }
			if = { limit = { has_global_variable = sb_martinus_story_war_resolved_global_var } remove_global_variable = sb_martinus_story_war_resolved_global_var }
			set_global_variable = sb_martinus_story_war_active_global_var
		}
		# Positive identity for an ordinary Zulu war. Known story routes keep their
		# own markers and are excluded rather than inferred later by absence alone.
		if = {
			limit = {
				OR = { scope:initiator ?= { country_definition = cd:ZUL } scope:target ?= { country_definition = cd:ZUL } }
				c:ZUL ?= {
					is_country_alive = yes
					NOT = { has_variable = sb_anglo_zulu_pressure_active_var }
					NOT = { has_variable = sb_natal_war_active_var }
					NOT = { has_variable = sb_port_natal_british_war_var }
					NOT = { has_variable = sb_zulu_swazi_war_active_var }
					NOT = { has_variable = sb_klip_river_secession_active_var }
					NOT = { has_variable = sb_klip_river_punitive_active_var }
					NOT = { has_variable = sb_natal_zulu_restoration_active_var }
				}
				NOT = { c:GZA ?= { has_variable = sb_gaza_eastern_transvaal_claim_if_zulu_defeated_var } }
			}
			c:ZUL = {
				set_variable = sb_zulu_generic_war_active_var
				set_variable = { name = sb_zulu_generic_war_play_scope value = root }
			}
		}
		# Safe post-create configuration is authorized only by an exact initiator lease.
		sb_griqualand_configure_started_sequence_play = yes
		sb_griqualand_configure_started_aligned_proxy_play = yes
		sb_nrp_configure_started_boundary_play = yes
		sb_natal_configure_started_zulu_secession_play = yes
		sb_bechuanaland_configure_started_story_play = yes
		sb_handle_natalia_player_support_join = yes
		sb_route_swazi_defensive_muster_for_play = yes
		sb_reconcile_bst_gun_war_defence_for_play = yes
	}
}

		sb_restore_ai_frontier_force_floors_for_play = yes
		sb_refresh_british_cape_boer_restraint = yes

	}
}

sb_on_spes_bona_diplo_play_abandon_side = {
	effect = {
		# Membership has changed but the play still exists. Re-evaluate only the
		# packages whose owner rule depends on the current committed sides.
		sb_reconcile_blood_river_assistance_for_play = yes
		sb_route_swazi_defensive_muster_for_play = yes
		sb_reconcile_bst_gun_war_defence_for_play = yes
		sb_cleanup_trn_frontier_deployment_train_for_play = yes
	}
}

sb_on_spes_bona_diplo_play_war_start = {
	effect = {
		sb_restore_ai_frontier_force_floors_for_play = yes
		sb_mobilize_ai_frontier_historical_formations_for_play = yes
		sb_schedule_trn_frontier_deployment_for_play = yes
		sb_reconcile_blood_river_assistance_for_play = yes
		sb_route_swazi_defensive_muster_for_play = yes
		sb_reconcile_bst_gun_war_defence_for_play = yes
		if = {
			limit = {
				is_diplomatic_play_type = dp_sb_natal_crisis
				sb_frontier_play_artificial_assistance_enabled = { RECIPIENT = c:ORA }
				c:ORA ?= {
					is_country_alive = yes
					is_ai = yes
					has_variable = sb_natal_war_active_var
				}
			}
			# ORA is the only active Boer war participant in fresh V2 Blood River
			# starts, so this is the live front-commitment hook.
			c:ORA ?= {
				sb_ensure_ora_ai_natal_front_commitment = yes
			}
		}
	}
}

		}
	}
}

sb_on_spes_bona_diplo_play_back_down = {
	effect = {
		# Resolve/record route state before generic participant cleanup.
		# A backed-down play never reaches on_war_end. Record the demand won by
		# the non-backing side and run the same idempotent 0/1/2 resolver now.
		if = {
			limit = {
				root = {
					OR = {
						is_diplomatic_play_type = dp_sb_natal_crisis
						is_diplomatic_play_type = dp_sb_martinus_humiliation
					}
				}
			}
			sb_story_humiliation_record_back_down = yes
			sb_story_humiliation_resolve = yes
		}

		if = {
			limit = {
				has_global_variable = sb_griqualand_west_refusal_contest_active_var
				root = {
					OR = {
						is_diplomatic_play_type = dp_sb_griqualand_cap_revoke_claim_locked
						is_diplomatic_play_type = dp_sb_griqualand_cap_revoke_oranje_proxy_locked
						is_diplomatic_play_type = dp_sb_griqualand_oranje_revoke_cap_claim_locked
						is_diplomatic_play_type = dp_sb_griqualand_cap_return_wbl_locked
						is_diplomatic_play_type = dp_sb_griqualand_oranje_return_wbl_locked
						is_diplomatic_play_type = dp_sb_griqualand_oranje_independent_return_wbl_proxy_locked
						is_diplomatic_play_type = dp_sb_griqualand_oranje_return_wbl_proxy_locked
					}
				}
			}
			sb_griqualand_sequence_record_back_down = yes
			sb_griqualand_west_close_routed_play = yes
			sb_griqualand_clear_launch_leases = yes
		}
		sb_klip_river_handle_backdown = yes
		sb_bechuanaland_handle_crisis_backdown = yes

		if = {
			limit = { root = { is_diplomatic_play_type = dp_sb_nrp_boundary_confrontation } }
			scope:initiator ?= { if = { limit = { has_variable = sb_nrp_boundary_launch_lease_var } remove_variable = sb_nrp_boundary_launch_lease_var } }
			c:NRP ?= {
				if = { limit = { has_variable = sb_nrp_boundary_confrontation_active_var } remove_variable = sb_nrp_boundary_confrontation_active_var }
				if = { limit = { has_variable = sb_nrp_boundary_initiator_scope } remove_variable = sb_nrp_boundary_initiator_scope }
			}
		}
		c:NAL ?= { if = { limit = { has_variable = sb_natal_zulu_secession_launch_lease_var } remove_variable = sb_natal_zulu_secession_launch_lease_var } }
		scope:initiator ?= {
			if = { limit = { has_variable = sb_bechuanaland_direct_launch_lease_var } remove_variable = sb_bechuanaland_direct_launch_lease_var }
			if = { limit = { has_variable = sb_bechuanaland_proxy_launch_lease_var } remove_variable = sb_bechuanaland_proxy_launch_lease_var }
			if = { limit = { has_variable = sb_bechuanaland_warren_direct_launch_lease_var } remove_variable = sb_bechuanaland_warren_direct_launch_lease_var }
			if = { limit = { has_variable = sb_bechuanaland_warren_proxy_launch_lease_var } remove_variable = sb_bechuanaland_warren_proxy_launch_lease_var }
			if = { limit = { has_variable = sb_bechuanaland_total_launch_lease_var } remove_variable = sb_bechuanaland_total_launch_lease_var }
		}
		sb_refresh_british_cape_boer_restraint = yes
		sb_cleanup_trn_frontier_deployment_train_for_play = yes
		sb_cleanup_blood_river_assistance_for_play = yes
		sb_cleanup_swazi_defensive_muster_for_play = yes
		sb_cleanup_bst_gun_war_defence_for_play = yes
		if = {
			limit = {
				root = { is_diplomatic_play_type = dp_sb_trn_zpb_crackdown_locked }
				scope:actor ?= { country_definition = cd:TRN }
				scope:target ?= {
					country_definition = cd:ZPB
					is_country_alive = yes
					has_variable = sb_zpb_crackdown_total_conquest_route_var
				}
			}
			scope:target = { set_variable = sb_zpb_crackdown_successor_pending_var }
		}
		if = {
			limit = {
				root = { is_diplomatic_play_type = dp_sb_trn_zpb_crackdown_locked }
				scope:actor ?= {
					country_definition = cd:ZPB
					has_variable = sb_zpb_crackdown_total_conquest_route_var
				}
			}
			scope:actor = {
				remove_variable = sb_zpb_crackdown_total_conquest_route_var
				if = {
					limit = { has_variable = sb_zpb_crackdown_successor_pending_var }
					remove_variable = sb_zpb_crackdown_successor_pending_var
				}
			}
		}
		# Dingane backing down from the Swazi play should still count as a dynastic
		# humiliation rather than a free reset.
		if = {
			limit = {
				root = {
					is_diplomatic_play_type = dp_conquer_state
					scope:target ?= { country_definition = cd:SWZ }
				}
				scope:actor ?= {
					country_definition = cd:ZUL
					has_variable = sb_zulu_swazi_war_active_var
				}
			}
			scope:actor = {
				remove_variable = sb_zulu_swazi_war_active_var
				sb_zulu_apply_swazi_campaign_defeat = yes
			}
			c:SWZ ?= {
				if = {
					limit = { has_variable = sb_zulu_swazi_war_active_var }
					remove_variable = sb_zulu_swazi_war_active_var
				}
			}
		}

		# Ordinary Zulu play: the non-backing side receives the single terminal
		# outcome, keyed to the exact play saved at start.
		if = {
			limit = { c:ZUL ?= { is_country_alive = yes has_variable = sb_zulu_generic_war_active_var has_variable = sb_zulu_generic_war_play_scope var:sb_zulu_generic_war_play_scope = root } }
			if = { limit = { scope:actor ?= { country_definition = cd:ZUL } } c:ZUL = { sb_zulu_apply_generic_war_defeat = yes } }
			else = { c:ZUL = { sb_zulu_apply_generic_war_victory = yes } }
			c:ZUL = {
				if = { limit = { has_variable = sb_zulu_generic_goal_accepted_pending_var } remove_variable = sb_zulu_generic_goal_accepted_pending_var }
				if = { limit = { has_variable = sb_zulu_generic_defeat_goal_accepted_pending_var } remove_variable = sb_zulu_generic_defeat_goal_accepted_pending_var }
				remove_variable = sb_zulu_generic_war_play_scope
				remove_variable = sb_zulu_generic_war_active_var
			}
		}

		# Kimberley claim dispute: the backing-down side determines whether the
		# Cape proceeds against Waterboersland or Oranje takes over the claim.
		if = {
			limit = {
				root = { is_diplomatic_play_type = dp_humiliation }
				scope:actor ?= {
					country_definition = cd:CAP
					has_variable = sb_kimberley_cap_ora_dispute_war_var
				}
				scope:target ?= {
					OR = {
						country_definition = cd:ORA
						country_definition = cd:TRN
					}
					has_variable = sb_kimberley_cap_ora_dispute_war_var
				}
			}
				if = {
						limit = { c:CAP ?= { has_claim = s:STATE_GRIQUALAND_WEST } }
						s:STATE_GRIQUALAND_WEST = { remove_claim = c:CAP }
					}
					sb_griqualand_west_grant_oranje_claim = yes
			scope:actor = { remove_variable = sb_kimberley_cap_ora_dispute_war_var }
			scope:target = {
				remove_variable = sb_kimberley_cap_ora_dispute_war_var
				trigger_event = { id = sb_griqualand_west.245 days = 3 popup = yes }
			}
		}


		# Waterboersland crisis wars only seed the remaining fields after the
		# annexation/protectorate question is actually resolved.
		if = {
			limit = {
				scope:actor ?= {
					country_definition = cd:WBL
					has_variable = sb_griqualand_west_cap_annexation_war_var
				}
				scope:target ?= {
					country_definition = cd:CAP
					has_variable = sb_griqualand_west_cap_annexation_war_var
				}
			}
			sb_kimberley_finalize_cap_annexation_route = yes
		}
		if = {
			limit = {
				scope:actor ?= {
					country_definition = cd:CAP
					has_variable = sb_griqualand_west_cap_annexation_war_var
				}
				scope:target ?= {
					country_definition = cd:WBL
					has_variable = sb_griqualand_west_cap_annexation_war_var
				}
			}
			sb_kimberley_finalize_wbl_diamond_mines_route = yes
		}
		if = {
			limit = {
				scope:actor ?= {
					country_definition = cd:WBL
					has_variable = sb_griqualand_west_ora_submission_war_var
				}
				scope:target ?= {
					OR = {
						country_definition = cd:ORA
						country_definition = cd:TRN
					}
					has_variable = sb_griqualand_west_ora_submission_war_var
				}
			}
			if = {
				limit = { scope:target ?= { country_definition = cd:TRN } }
				sb_kimberley_finalize_trn_puppet_route = yes
			}
			else = {
				sb_kimberley_finalize_ora_puppet_route = yes
			}
		}
		if = {
			limit = {
				scope:actor ?= {
					OR = {
						country_definition = cd:ORA
						country_definition = cd:TRN
					}
					has_variable = sb_griqualand_west_ora_submission_war_var
				}
				scope:target ?= {
					country_definition = cd:WBL
					has_variable = sb_griqualand_west_ora_submission_war_var
				}
			}
			sb_kimberley_finalize_wbl_diamond_mines_route = yes
		}

		# Britain backing down from the scheduled annexation counts as a Zulu
		# victory; Zulu backing down is resolved by the enforced annexation goal.
		if = {
			limit = {
				root = { is_diplomatic_play_type = dp_annex_war }
				scope:actor ?= {
					country_definition = cd:GBR
					has_variable = sb_anglo_zulu_pressure_active_var
				}
				scope:target ?= {
					country_definition = cd:ZUL
					has_variable = sb_anglo_zulu_pressure_active_var
				}
			}
			scope:target = {
				sb_zululand_discard_preannexation_royal_house_archive = yes
				trigger_event = { id = sb_anglo_zulu.020 days = 3 popup = yes }
			}
		}

	}
}

sb_on_spes_bona_war_end = {
	effect = {
		# Final predicates and accepted records must resolve before route cleanup.
		sb_story_humiliation_resolve = yes
		if = {
			limit = {
				has_global_variable = sb_griqualand_west_refusal_contest_active_var
				root = {
					OR = {
						is_diplomatic_play_type = dp_sb_griqualand_cap_revoke_claim_locked
						is_diplomatic_play_type = dp_sb_griqualand_cap_revoke_oranje_proxy_locked
						is_diplomatic_play_type = dp_sb_griqualand_oranje_revoke_cap_claim_locked
						is_diplomatic_play_type = dp_sb_griqualand_cap_return_wbl_locked
						is_diplomatic_play_type = dp_sb_griqualand_oranje_return_wbl_locked
						is_diplomatic_play_type = dp_sb_griqualand_oranje_independent_return_wbl_proxy_locked
						is_diplomatic_play_type = dp_sb_griqualand_oranje_return_wbl_proxy_locked
					}
				}
			}
			sb_griqualand_west_close_routed_play = yes
			sb_griqualand_clear_launch_leases = yes
		}

		# Reconcile accepted frontier records once, after the final war state exists.
		if = {
			limit = { root = { is_diplomatic_play_type = dp_humiliation } c:CAP ?= { has_variable = sb_kimberley_cap_claim_goal_accepted_pending_var has_variable = sb_kimberley_cap_ora_dispute_war_var } }
			c:CAP = { remove_variable = sb_kimberley_cap_claim_goal_accepted_pending_var trigger_event = { id = sb_griqualand_west.250 days = 3 popup = yes } }
		}
		if = {
			limit = {
				root = { is_diplomatic_play_type = dp_annex_war }
				scope:target ?= {
					country_definition = cd:ZUL
					OR = { has_variable = sb_anglo_zulu_pressure_active_var has_variable = sb_port_natal_british_war_var }
				}
