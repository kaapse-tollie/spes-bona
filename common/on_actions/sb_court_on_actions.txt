on_ruler_selected = { on_actions = { sb_court_on_ruler_selected } }
on_character_death = { on_actions = { sb_court_on_character_death } }
on_revolution_start = { on_actions = { sb_court_on_revolution_start } }
on_revolution_end = { on_actions = { sb_court_on_revolution_end } }
on_civil_war_won = { on_actions = { sb_court_on_civil_war_won } }
on_law_enactment_started = { on_actions = { sb_court_block_ordinary_contested_enactment } }
# can_enact remains true for the movement's demand evaluation; normal reform is cancelled here.
sb_court_block_ordinary_contested_enactment = {
 trigger = { OR = { is_enacting_law = law_type:law_sb_contested_crown is_enacting_law = law_type:law_sb_contested_social_crown } }
 effect = { cancel_enactment = yes }
}
sb_court_on_ruler_selected = {
 effect = {
  owner ?= {
   if = {
    limit = { sb_institutions_profile_country = yes NOT = { has_variable = sb_court_selecting_head_var } }
    save_scope_as = sb_court_country
    if = {
     limit = { has_variable = sb_court_dynastic_war_scope var:sb_court_dynastic_candidate_scope ?= { is_character_alive = yes } }
     set_variable = sb_court_selecting_head_var
     sb_court_set_dynastic_head = yes
     remove_variable = sb_court_selecting_head_var
    }
    if = {
     limit = { has_variable = sb_institutions_active_var sb_court_has_royal_law = yes scope:previous ?= { is_character_alive = no } NOT = { has_variable = sb_court_regent_scope } NOT = { has_variable = sb_court_dynastic_crisis_var } }
     scope:previous = { save_scope_as = sb_court_outgoing }
     sb_court_capture_accession = yes
    }
    if = {
     limit = { NOT = { has_variable = sb_court_accession_open_var } NOT = { has_variable = sb_court_dynastic_war_scope } NOT = { has_variable = sb_court_regent_scope } NOT = { has_variable = sb_court_dynastic_crisis_var } }
     sb_court_ensure_roster = yes
     ruler = { save_scope_as = sb_court_new_ruler }
     every_in_list = {
      variable = sb_court_members
      limit = { sb_court_member_of_country = yes var:sb_court_endorsement_var = 1 NOT = { this = scope:sb_court_new_ruler } }
      set_variable = { name = sb_court_endorsement_var value = 0 }
     }
     every_in_list = {
      variable = sb_court_members
      limit = { sb_court_member_of_country = yes var:sb_court_claimant_scope ?= scope:sb_court_new_ruler }
      set_variable = { name = sb_court_endorsement_var value = 1 }
      remove_variable = sb_court_claimant_scope
     }
     ruler = { set_variable = { name = sb_court_endorsement_var value = 1 } }
     set_variable = { name = sb_court_incumbent_scope value = ruler }
     remove_variable = sb_court_privileges_this_reign_var
     if = { limit = { ruler = { has_template = ZUL_mbuyazi_heir } } set_global_variable = sb_zulu_mbuyazi_succession_global_var }
     else_if = { limit = { ruler = { has_template = ZUL_cetshwayo_heir } } remove_global_variable = sb_zulu_mbuyazi_succession_global_var }
     sb_court_select_chiefly_head = yes
    }
    sb_institutions_recalculate = yes
   }
  }
 }
}
sb_court_on_character_death = {
 effect = {
  if = {
   limit = { has_variable = sb_court_registered_var }
   save_scope_as = sb_court_deceased
   var:sb_court_home_scope ?= {
    if = {
     limit = { has_variable = sb_institutions_active_var sb_court_has_royal_law = yes var:sb_court_incumbent_scope ?= scope:sb_court_deceased NOT = { has_variable = sb_court_regent_scope } NOT = { has_variable = sb_court_dynastic_crisis_var } NOT = { has_variable = sb_court_selecting_head_var } }
     scope:sb_court_deceased = { save_scope_as = sb_court_outgoing }
     sb_court_capture_accession = yes
    }
    every_in_list = {
     variable = sb_court_members limit = { var:sb_court_claimant_scope ?= scope:sb_court_deceased }
     set_variable = { name = sb_court_endorsement_var value = 0 }
     remove_variable = sb_court_claimant_scope
    }
    every_in_list = {
     variable = sb_court_members limit = { var:sb_court_successor_pledge_scope ?= scope:sb_court_deceased }
     remove_variable = sb_court_successor_pledge_scope
    }
    every_in_list = {
     variable = sb_court_members limit = { var:sb_court_council_promise_scope ?= scope:sb_court_deceased }
     remove_variable = sb_court_council_promise_scope
    }
    # Defer until the native death removal completes; one pending refresh per country.
    if = { limit = { NOT = { has_variable = sb_court_death_refresh_var } } set_variable = sb_court_death_refresh_var trigger_event = { id = sb_court_politics.900 days = 1 } }
    # The identity references are cleared before that refresh.
    if = { limit = { var:sb_court_chiefly_head_scope ?= scope:sb_court_deceased } remove_variable = sb_court_chiefly_head_scope }
    if = { limit = { var:sb_court_restoration_claimant_scope ?= scope:sb_court_deceased } remove_variable = sb_court_restoration_claimant_scope }
   }
  }
 }
}
sb_court_on_revolution_start = {
 effect = {
  if = {
   limit = { sb_institutions_profile_country = yes }
   sb_court_track_dynastic_war = yes
   # If another movement split instead, release our pending episode before its own handler.
   if = { limit = { has_variable = sb_court_dynastic_test_pending_var NOT = { has_variable = sb_court_dynastic_war_scope } } sb_court_cancel_dynastic_crisis = yes }
   sb_court_track_revolution = yes
  }
 }
}
sb_court_on_revolution_end = { effect = { sb_court_record_dynastic_end = yes if = { limit = { has_variable = sb_court_civil_war_var } sb_court_record_war_end = yes } } }
sb_court_on_civil_war_won = { effect = { sb_court_resolve_dynastic_win = yes if = { limit = { has_variable = sb_court_war_ended_var } sb_court_resolve_won_war = yes } } }
