# A terminal chiefdom receipt must not survive destruction or retagging of NAL.
# The first live country pulse clears the global/container state; later pulses
# are inert because cleanup removes the global single-flight marker.
# Recover generation containers even when their fixed-tag event recipients died.
# The first live country pulse performs idempotent cleanup; valid held receipts
# are left to their saved recipient and timed popup watchdogs.
# Klip River generations carry their original actor scopes. This global pulse
# removes only orphaned saved actors; it never adopts a replacement c:NAL.
sb_on_klip_river_orphan_monthly = {
	trigger = {
		is_country_alive = yes
		OR = {
			AND = {
				container_exists = sb_klip_river_secession_generation_state
				NOT = { container:sb_klip_river_secession_generation_state ?= { has_variable = sb_klip_river_secession_generation_actor_scope var:sb_klip_river_secession_generation_actor_scope ?= { is_country_alive = yes country_definition = cd:NAL } } }
			}
			AND = {
				container_exists = sb_klip_river_punitive_generation_state
				NOT = { container:sb_klip_river_punitive_generation_state ?= { has_variable = sb_klip_river_punitive_generation_actor_scope var:sb_klip_river_punitive_generation_actor_scope ?= { is_country_alive = yes country_definition = cd:NAL } } }
			}
			any_country = {
				is_country_alive = yes
				NOT = { country_definition = cd:NAL }
				OR = { has_variable = sb_klip_river_county_creation_lease_var has_variable = sb_klip_river_county_created_this_attempt_var }
			}
		}
	}
	effect = { sb_klip_river_recover_orphaned_generations = yes }
}

# Close the one-day completion race synchronously. The completed marker is durable;
# only unfinished route leases are cleared so a later NAL reacquisition can retry.
sb_on_zululand_incorporation_state_owner_change = {
	trigger = { state_region = s:STATE_ZULULAND }
	effect = {
		if = {
			limit = { NOT = { owner ?= { country_definition = cd:NAL } } }
			c:NAL ?= {
				if = { limit = { has_variable = sb_zululand_incorporation_requested_var } remove_variable = sb_zululand_incorporation_requested_var }
				if = { limit = { has_variable = sb_zululand_incorporation_started_var } remove_variable = sb_zululand_incorporation_started_var }
				if = { limit = { has_variable = sb_zululand_post_annex_incorporation_pending_var } remove_variable = sb_zululand_post_annex_incorporation_pending_var }
				if = { limit = { has_variable = sb_zululand_incorporation_complete_pending_var } remove_variable = sb_zululand_incorporation_complete_pending_var }
			}
		}
	}
}

					has_principle = principle:principle_colonial_offices_3
				}
				NOT = { has_variable = sb_gbr_cap_frontier_colonization_demand_cooldown_var }
				c:CAP ?= {
					is_country_alive = yes
					is_subject_of = c:GBR

					NOT = { has_law = law_type:law_colonial_resettlement }
					NOT = { has_law = law_type:law_colonial_exploitation }

						OR = {
							is_enacting_law = law_type:law_frontier_colonization
							is_enacting_law = law_type:law_colonial_resettlement
							is_enacting_law = law_type:law_colonial_exploitation
						}
					}
					enacting_any_law = no
				}
			}
			set_variable = {
				name = sb_gbr_cap_frontier_colonization_demand_cooldown_var

			}
			c:CAP ?= {
				trigger_event = { id = sb_cape.204 days = sb_travel_london_southern_africa_message_days popup = yes }
			}
		}
	}
}
