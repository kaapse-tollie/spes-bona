sb_on_mineral_discoveries_acquired_technology = {
	effect = {
		if = {
			limit = {
				has_technology_researched = pneumatic_tools
				NOT = { has_global_variable = sb_vrystaat_gold_potential_seeded_var }
			}
			set_global_variable = sb_vrystaat_gold_potential_seeded_var
			s:STATE_VRYSTAAT = {
				change_resource_potential = {
					type = building_gold_field
					amount = 4
				}
			}
		}
		if = {
			limit = {
				has_technology_researched = dynamite
				# Gauteng is the localized name for STATE_TRANSVAAL.
				any_scope_state = { state_region = s:STATE_TRANSVAAL }
				NOT = { has_global_variable = sb_transvaal_gold_potential_seeded_var }
			}
			set_global_variable = sb_transvaal_gold_potential_seeded_var
			s:STATE_TRANSVAAL = {
				change_resource_potential = {
					type = building_gold_field
					amount = 94
				}
			}
			s:STATE_EAST_TRANSVAAL = {
				change_resource_potential = {
					type = building_gold_field
					amount = 4
				}
			}
		}
		if = {
			limit = {
				has_technology_researched = nitroglycerin
				any_scope_state = { state_region = s:STATE_GRIQUALAND_WEST }
				NOT = { has_global_variable = sb_griqualand_west_iron_nitroglycerin_seeded_var }
			}
			set_global_variable = sb_griqualand_west_iron_nitroglycerin_seeded_var
			s:STATE_GRIQUALAND_WEST = {
				change_resource_potential = {
					type = building_iron_mine
					amount = 4
				}
			}
		}
		if = {
			limit = {
				has_technology_researched = dynamite
				any_scope_state = { state_region = s:STATE_GRIQUALAND_WEST }
				NOT = { has_global_variable = sb_griqualand_west_iron_dynamite_seeded_var }
			}
			set_global_variable = sb_griqualand_west_iron_dynamite_seeded_var
			s:STATE_GRIQUALAND_WEST = {
				change_resource_potential = {
					type = building_iron_mine
					amount = 5
				}
			}
		}
		if = {
			limit = {
				has_technology_researched = pneumatic_tools
				any_scope_state = { state_region = s:STATE_GRIQUALAND_WEST }
				NOT = { has_global_variable = sb_griqualand_west_iron_pneumatic_tools_seeded_var }
			}
			set_global_variable = sb_griqualand_west_iron_pneumatic_tools_seeded_var
			s:STATE_GRIQUALAND_WEST = {
				change_resource_potential = {
					type = building_iron_mine
					amount = 10
				}
			}
		}
	}
}

sb_on_kimberley_yearly_pulse = {
	effect = {
		if = {
			limit = {
				sb_can_discover_kimberley_diamonds = yes
			}
			set_global_variable = sb_kimberley_discovery_pending_var
			if = {
				limit = {
					p:x68B5E8.state.owner = {
						NOT = { country_definition = cd:WBL }
					}
				}
				p:x68B5E8.state.owner = {
					trigger_event = { id = sb_griqualand_west.027 days = 1 popup = yes }
				}
			}
			else_if = {
				limit = {
					sb_cap_has_northern_cape_foothold = yes
					c:WBL ?= {
						is_country_alive = yes
						any_scope_state = { state_region = s:STATE_GRIQUALAND_WEST }
					}
				}
				c:CAP ?= {
					trigger_event = { id = sb_griqualand_west.021 days = 1 popup = yes }
				}
			}
			else_if = {
				limit = {
					c:TRN ?= {
						is_country_alive = yes
						OR = {
							has_variable = sb_oranje_federated_into_trn_var
							c:ORA ?= {
								is_country_alive = yes
								is_direct_subject_of = c:TRN
								OR = {
									is_subject_type = subject_type_sb_boer_presidential_union
									is_subject_type = subject_type_sb_boer_confederal_partner
								}
							}
						}
					}
				}
				c:TRN ?= {
					set_variable = sb_kimberley_oranje_proxy_var
					trigger_event = { id = sb_griqualand_west.021 days = 1 popup = yes }
				}
			}
			else_if = {
				limit = {
					c:ORA ?= {
						is_country_alive = yes
					}
				}
				c:ORA ?= {
					trigger_event = { id = sb_griqualand_west.021 days = 1 popup = yes }
				}
			}
			else = {
				p:x68B5E8.state.owner = {
					trigger_event = { id = sb_griqualand_west.021 days = 1 popup = yes }
				}
			}
		}
	}
}

sb_on_bechuanaland_corridor_monthly_pulse = {
	effect = {
		if = {
			limit = { sb_bechuanaland_corridor_can_open = yes }
			sb_bechuanaland_open_corridor_is = yes
		}
		if = {
			limit = {
				sb_bechuanaland_corridor_ready_after_kimberley = yes
				NOT = { has_global_variable = sb_bechuanaland_corridor_open_global_var }
				NOT = { has_global_variable = sb_bechuanaland_corridor_resolved_global_var }
				OR = {
					exists = c:SAF
					NOT = {
						c:CAP ?= {
							is_country_alive = yes
							is_subject_of = c:GBR
						}
					}
				}
			}
			sb_bechuanaland_skip_corridor_crisis = yes
		}
		if = {
			limit = {
				has_global_variable = sb_bechuanaland_corridor_open_global_var
				NOT = { has_global_variable = sb_bechuanaland_corridor_resolved_global_var }
			}
			sb_bechuanaland_handle_post_open_lifecycle = yes
			if = {
				limit = { sb_bechuanaland_corridor_crisis_is_live = yes }
				sb_bechuanaland_repair_abandoned_crisis_attempts = yes
				sb_bechuanaland_repair_sgo_subject_if_needed = yes
				sb_bechuanaland_repair_stalled_cap_warren_dispatch = yes
				sb_bechuanaland_try_launch_pending_crisis = yes
				sb_bechuanaland_maintain_crisis_support = yes
				sb_bechuanaland_advance_influence_monthly = yes
				if = {
					limit = {
						container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_war_active_var }
						NOT = { sb_bechuanaland_has_active_corridor_diplomatic_play = yes }
					}
					sb_bechuanaland_queue_unresolved_settlement_event = yes
				}
			}
		}
	}
}

sb_on_griqualand_west_refusal_contest_monthly_pulse = {
	effect = {
		if = {
			limit = {
				c:TRN ?= {
					is_country_alive = yes
					has_claim = s:STATE_GRIQUALAND_WEST
					OR = {
						has_variable = sb_kimberley_oranje_proxy_var
						has_variable = sb_kimberley_ora_pressed_claim_var
						has_variable = sb_kimberley_trn_claimant_var
					}
				}
				c:ORA ?= {
					is_country_alive = yes
					is_direct_subject_of = c:TRN
					OR = {
						is_subject_type = subject_type_sb_boer_presidential_union
						is_subject_type = subject_type_sb_boer_confederal_partner
					}
				}
			}
			sb_griqualand_west_grant_oranje_claim = yes
		}
		sb_griqualand_west_try_finalize_refusal_contest = yes
	}
}

sb_on_griqualand_east_monthly_pulse_country = {
	effect = {
		if = {
			limit = {
				country_definition = cd:CAP
				has_variable = sb_griqualand_east_pending_var
				NOT = { has_variable = sb_griqualand_east_resolved_var }
				NOT = { exists = c:GLE }
				OR = {
					exists = c:ABY
					NOT = { sb_eastern_cape_frontier_is_unsettled = yes }
				}
			}
			remove_variable = sb_griqualand_east_pending_var
			set_variable = sb_griqualand_east_resolved_var
		}
		if = {
			limit = {
				country_definition = cd:CAP
				sb_cap_can_form_gle = yes
			}
			set_variable = sb_griqualand_east_pending_var
			trigger_event = { id = sb_griqualand_east.220 days = 14 popup = yes }
		}
		if = {
			limit = {
				country_definition = cd:CAP
				has_variable = sb_gle_willing_annexation_pending_var
				NOT = { sb_cap_can_integrate_gle = yes }
			}
			remove_variable = sb_gle_willing_annexation_pending_var
		}
		if = {
			limit = {
				country_definition = cd:CAP
				sb_cap_can_integrate_gle = yes
				NOT = { has_variable = sb_gle_willing_annexation_pending_var }
			}
			set_variable = sb_gle_willing_annexation_pending_var
			trigger_event = { id = sb_griqualand_east.221 days = 1 popup = yes }
		}
	}
}
