sb_on_namibia_monthly_pulse_country = {
	trigger = {
		OR = {
			country_definition = cd:GER
			country_definition = cd:CAP
			country_definition = cd:SWA
			has_variable = sb_nam_selected_policy_var
			has_variable = sb_nam_policy_report_pending_var
			has_variable = sb_nam_company_payments_left_var
			has_modifier = sb_nam_company_slot
			p:xDEAA3C.state.owner = root
			p:xB04050.state.owner = root
			AND = {
				NOT = { is_country_type = decentralized }
				NOT = { has_technology_researched = sb_namibian_coastal_access }
				sb_nam_has_coastal_engineering_technologies = yes
			}
		}
	}
	effect = {
		sb_nam_grant_coastal_access_if_eligible = yes

		if = {
			limit = { sb_nam_can_offer_berlin_programme = yes }
			trigger_event = { id = sb_nam.010 popup = yes }
		}

		if = {
			limit = {
				sb_nam_german_survey_can_receive_maps = yes
				NOT = { has_variable = sb_nam_german_missionary_support_offered_var }
				je:je_sb_map_namibian_coast = { NOT = { has_variable = sb_nam_maps_offer_var } }
			}
			je:je_sb_map_namibian_coast = {
				save_scope_as = sb_nam_maps_survey
				set_variable = { name = sb_nam_maps_offer_var value = 1 }
			}
			trigger_event = { id = sb_nam.011 days = 390 popup = yes }
		}

		if = {
				limit = {
					country_definition = cd:CAP
					game_date >= 1867.1.1
					game_date <= 1871.12.31
					NOT = { has_variable = sb_nam_rehoboth_branch_resolved_var }
					OR = {
						sb_nam_has_any_colonial_law = yes
					has_modifier = sb_cape_favour_settlers
				}
			}
			trigger_event = { id = sb_nam.020 popup = yes }
		}

		if = {
			limit = {
				p:xDEAA3C.state.owner = root
				NOT = { is_country_type = decentralized }
				NOT = { has_claim = s:STATE_NAMAQUALAND }
				sb_nam_receives_direct_coastal_claims = yes
				c:CAP ?= {
					NOT = {
						any_scope_state = {
							state_region = s:STATE_NAMAQUALAND
							owner = prev
							num_provinces >= 2
						}
					}
				}
			}
			s:STATE_NAMAQUALAND = {
				add_claim = root
			}
		}

		sb_nam_maintain_swa_company = yes

		if = {
			limit = { country_definition = cd:SWA is_country_alive = yes }
			if = {
				limit = { has_variable = sb_nam_selected_policy_var }
				sb_nam_sync_frontier_generation = yes
			}
			if = {
				limit = { NOT = { has_variable = sb_nam_sponsor_claims_relinquished_var } }
				if = {
					limit = { var:sb_nam_policy_founder_var ?= { has_variable = sb_nam_policy_colony_founded_var } }
					var:sb_nam_policy_founder_var = { sb_nam_relinquish_swa_coastal_claims = yes }
				}
				else_if = {
					limit = { overlord ?= { has_variable = sb_nam_mapping_permanent_engagement_var } }
					overlord = { sb_nam_relinquish_swa_coastal_claims = yes }
				}
			}
			if = {
				limit = {
					has_modifier = sb_nam_administrative_capacity
					NOT = { has_variable = sb_nam_capacity_term_applied_var }
					sb_nam_legacy_charter_is_funded = no
				}
				sb_nam_apply_administrative_capacity = yes
			}
			sb_nam_offer_extermination_orders = yes
		}
		if = {
			limit = {
				has_variable = sb_nam_policy_expiry_var
				var:sb_nam_policy_expiry_var > game_date
				NOT = { var:sb_nam_selected_policy_var = 2 }
			}
			sb_nam_sync_regional_policy = yes
		}
		if = {
			limit = { has_variable = sb_nam_policy_report_pending_var NOT = { sb_nam_has_survey_station = yes } }
			remove_variable = sb_nam_policy_report_pending_var
		}
		if = {
			limit = { sb_nam_seeded_policy_colony_is_pending = yes }
			sb_nam_reconcile_seeded_policy_colony = yes
		}
		else_if = {
			limit = { sb_nam_can_found_policy_colony = yes sb_nam_controls_both_consolidation_ports = yes }
			if = {
				limit = { NOT = { has_journal_entry = je_sb_map_namibian_coast } }
				sb_nam_offer_policy_report = yes
			}
			sb_nam_try_found_policy_colony = yes
		}
	}
}

sb_on_namibia_global_monthly_pulse = {
	effect = {
		sb_nam_close_coastal_race_if_fully_occupied = yes
	}
}

sb_on_swa_frontier_state_owner_change = {
	trigger = {
		OR = {
			has_modifier = sb_nam_frontier_generation
			AND = {
				c:SWA ?= { has_variable = sb_nam_selected_policy_var }
				OR = {
					sb_nam_is_policy_region = yes
					owner = c:SWA
					c:SWA = { is_adjacent_to_state = PREV }
				}
			}
		}
	}
	effect = {
		if = {
			limit = { has_modifier = sb_nam_frontier_generation NOT = { owner ?= c:SWA } }
			remove_modifier = sb_nam_frontier_generation
		}
		c:SWA ?= {
			if = {
				limit = { has_variable = sb_nam_selected_policy_var }
				sb_nam_sync_frontier_generation = yes
			}
		}
	}
}

sb_on_swa_company_disbanded = {
	trigger = {
		scope:company ?= { is_company_type = company_type:company_sb_south_west_africa_company }
	}
	effect = {
		if = { limit = { has_modifier = sb_nam_company_slot } remove_modifier = sb_nam_company_slot }
		if = {
			limit = {
				c:SWA ?= {
					is_direct_subject_of = PREV
					is_subject_type = subject_type_chartered_company
				}
			}
			c:SWA = { set_country_type = colonial }
			remove_diplomatic_pact = { country = c:SWA type = chartered_company }
			create_diplomatic_pact = { country = c:SWA type = colony }
		}
	}
}
