sb_check_colonial_administration = {
	effect = {
		if = {
			limit = {
				sb_is_ordinary_colonial_subject = yes
				overlord ?= scope:sb_colonial_administration_overlord
				NOT = { has_law = law_type:law_colonial_administration }
			}
			activate_law = law_type:law_colonial_administration
		}
	}
}
