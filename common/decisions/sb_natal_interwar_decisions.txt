decision_sb_integrate_restored_zululand = {
	is_shown = {
		country_definition = cd:NAL
		OR = {
			has_variable = sb_natal_gradual_zulu_restoration_var
			has_variable = sb_zululand_zibhebhu_victory_var
			has_variable = sb_zululand_crown_restored_victory_var
		}
		c:ZUL ?= {
			is_country_alive = yes
			is_direct_subject_of = root
			OR = {
				is_subject_type = subject_type_puppet
				is_subject_type = subject_type_vassal
			}
		}
	}

	possible = { sb_zululand_nal_can_annex_subject = yes }

	when_taken = {
		set_variable = sb_zululand_post_annex_incorporation_pending_var
		annex = c:ZUL
		if = {
			limit = { has_variable = sb_natal_gradual_zulu_restoration_var }
			remove_variable = sb_natal_gradual_zulu_restoration_var
		}
		set_variable = sb_natal_zulu_restoration_resolved_var
		set_variable = sb_natal_zulu_settlement_resolved_var
		sb_natal_clear_zulu_restoration_runtime = yes
		sb_zululand_apply_post_annex_incorporation = yes
	}

	ai_chance = { value = 100 }
}
