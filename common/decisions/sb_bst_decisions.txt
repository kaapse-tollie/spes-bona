sb_seek_british_protection = {
	is_shown = {
		country_definition = cd:BST
		is_subject = no
		has_journal_entry = je_sb_bst_ora_frontier
		sb_bst_controls_only_drakensberg = yes
		NOT = { has_variable = sb_bst_gbr_protection_request_resolved_var }
		sb_has_adjacent_british_protection_route = yes
		c:GBR ?= {
			is_country_alive = yes
		}
	}

	possible = {
		always = yes
	}

	when_taken = {
		set_variable = sb_bst_gbr_protection_request_resolved_var
		c:GBR ?= {
			trigger_event = { id = sb_bst_frontier.100 days = sb_travel_london_southern_africa_message_days popup = yes }
		}
	}

	ai_chance = {
		value = 0
	}
}

decision_sb_transfer_basutoland_to_cape = {
	is_shown = {
		c:BST ?= {
			is_country_alive = yes
			is_direct_subject_of = root
		}
		c:CAP ?= {
			is_country_alive = yes
			is_direct_subject_of = root
		}
	}

	possible = {
		c:BST ?= {
			is_country_alive = yes
			is_direct_subject_of = root
			is_subject_type = subject_type_puppet
		}
		c:CAP ?= {
			is_country_alive = yes
			is_direct_subject_of = root
			OR = {
				is_subject_type = subject_type_dominion
				is_subject_type = subject_type_sb_dominion
				is_subject_type = subject_type_sb_responsible_colony
				is_subject_type = subject_type_sb_responsible_colony_monarchy
			}
			sb_cap_controls_bst_frontier = yes
		}
	}

	when_taken = {
		c:CAP ?= {
			transfer_subject = c:BST
		}
		c:BST ?= {
			if = {
				limit = {
					is_direct_subject_of = c:CAP
					NOT = { is_subject_type = subject_type_puppet }
				}
				change_subject_type = subject_type_puppet
			}
		}
	}

	ai_chance = {
		value = 100
	}
}
