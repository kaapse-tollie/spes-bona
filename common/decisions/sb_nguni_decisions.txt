# Spes Bona - Nguni decisions

sb_proclaim_nguni_nation = {
	is_shown = {
		OR = {
			country_definition = cd:ZUL
			country_definition = cd:SWZ
			country_definition = cd:GZA
		}
		is_subject = no
	}

	possible = {
		is_at_war = no
		is_active_in_diplomatic_play = no
		NOT = { sb_southern_formation_has_blocking_story_transaction = yes }
		NOT = {
			OR = {
				has_variable = sb_natal_story_launch_pending_var
				has_variable = sb_natal_story_launch_retry_queued_var
				has_variable = sb_natal_story_launch_lease_var
				has_variable = sb_natal_story_play_scope
				has_variable = sb_natal_guns_bargain_generation_scope
				has_variable = sb_natal_refusal_generation_scope
			}
		}
		NOT = {
			c:ZUL ?= {
				OR = {
					has_variable = sb_natal_story_launch_pending_var
					has_variable = sb_natal_story_launch_retry_queued_var
					has_variable = sb_natal_story_launch_lease_var
					has_variable = sb_natal_story_play_scope
					has_variable = sb_natal_guns_bargain_generation_scope
					has_variable = sb_natal_refusal_generation_scope
				}
			}
		}
		NOT = { has_global_variable = sb_zululand_chiefdoms_open_global_var }
		NOT = { container_exists = sb_natal_guns_bargain_generation_state }
		NOT = { container_exists = sb_natal_refusal_generation_state }
		NOT = { container_exists = sb_natal_terminal_outcome_state }
		NOT = { has_variable = sb_natal_terminal_outcome_scope }
		NOT = { has_variable = sb_natal_terminal_outcome_pending_var }
		NOT = { has_variable = sb_natal_terminal_outcome_popup_receipt_var }
		NOT = { has_variable = sb_zulu_swazi_war_active_var }
		NOT = { has_variable = sb_zulu_gaza_followup_pending_var }
		NOT = { has_variable = sb_zulu_swazi_outcome_delivery_pending_var }
		NOT = {
			c:ZUL ?= {
				is_country_alive = yes
				OR = {
					has_variable = sb_zulu_swazi_war_active_var
					has_variable = sb_zulu_gaza_followup_pending_var
					has_variable = sb_zulu_swazi_outcome_delivery_pending_var
					has_variable = sb_zulu_swazi_outcome_delivery_queued_var
				}
			}
		}
		has_technology_researched = nationalism
		sb_nguni_conqueror_path_open = yes
	}

	when_taken = {
		if = {
			limit = { country_definition = cd:ZUL }
			set_variable = sb_ngu_inherited_zul_content_var
		}
		else_if = {
			limit = { country_definition = cd:GZA }
			set_variable = sb_ngu_inherited_gza_content_var
		}
		change_tag = NGN
		c:NGN ?= {
			sb_nguni_apply_formation_inheritance = yes
		}
	}

	ai_chance = {
		value = 100
	}
}
