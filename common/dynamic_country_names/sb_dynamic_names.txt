# Spes Bona - Dynamic Country Names
# V3 dynamic names override the base country tag name when trigger conditions are met.
# Higher priority values take precedence when multiple names match.
# Cross-ref: localization/ for the actual display strings (dyn_c_sb_* keys).

# TRN: Potchefstroom -> Transvaal Republic -> Suid-Afrikaanse Republiek
# REPLACE: overwrites any vanilla dynamic names for TRN.
# Historically, the Boers initially called their polity after Potchefstroom (est. 1838),
# the first major Voortrekker settlement in the Transvaal. Once the early trek
# phase ends, the polity is better read as the Transvaal Republic. The fuller
# Suid-Afrikaanse Republiek title is reserved for control of the whole
# Transvaal macro-region, excluding the Swazi-held slice.
REPLACE:TRN = {
	# Potchefstroom — early small Boer republic name.
	# priority = 0: lowest priority, only used when sb_early_republic_var is set.
	# This variable is set at game start and cleared by the trek/consolidation JE.
	dynamic_country_name = {
		name = dyn_c_sb_potchefstroom
		adjective = dyn_c_sb_potchefstroom_adj

		is_main_tag_only = yes
		priority = 0

		trigger = {
			scope:actor ?= {
				has_variable = sb_early_republic_var
			}
		}
	}

	# Suid-Afrikaanse Republiek — full Transvaal consolidation short of Swaziland.
	# priority = 20: highest TRN naming tier once the whole macro-region is held.
	dynamic_country_name = {
		name = dyn_c_sb_suid_afrikaanse_republiek
		adjective = dyn_c_sb_suid_afrikaanse_republiek_adj

		is_main_tag_only = yes
		priority = 20

		trigger = {
			scope:actor ?= {
				NOT = { has_variable = sb_early_republic_var }
				sb_controls_transvaal_except_swazi = yes
			}
		}
	}

	# Transvaal Republic — mature western republic once the trek phase has ended.
	dynamic_country_name = {
		name = dyn_c_sb_transvaal_republic
		adjective = dyn_c_sb_transvaal_republic_adj

		is_main_tag_only = yes
		priority = 10

		trigger = {
			scope:actor ?= {
				NOT = { has_variable = sb_early_republic_var }
			}
		}
	}
}

# BIC/BHT — vanilla's Dominion of India trigger catches any non-company subject.
# SB responsible colonies are still autonomy-level 1, so they keep the colonial
# name tier instead of being displayed as dominions.
REPLACE:BIC = {
	dynamic_country_name = {
		name = dyn_c_british_raj
		adjective = dyn_c_british_raj_adj
		use_overlord_prefix = yes

		is_main_tag_only = yes
		priority = 60

		trigger = {
			scope:actor ?= {
				any_overlord_or_above = {
					c:GBR ?= this
				}
				OR = {
					is_subject_type = subject_type_sb_responsible_colony
					is_subject_type = subject_type_sb_responsible_colony_monarchy
				}
			}
		}
	}

	dynamic_country_name = {
		name = dyn_c_british_raj
		adjective = dyn_c_british_raj_adj
		use_overlord_prefix = yes

		is_main_tag_only = yes
		priority = 50

		trigger = {
			exists = c:GBR
			c:GBR ?= {
				has_diplomatic_pact = {
					who = scope:actor
					type = colony
					is_initiator = yes
				}
			}
		}
	}

	dynamic_country_name = {
		name = dyn_c_east_india_company
		adjective = dyn_c_east_india_company_adj
		use_overlord_prefix = yes

		is_main_tag_only = yes
		priority = 10

		trigger = {
			scope:actor ?= {
				is_subject = yes
				has_government_type = gov_chartered_company
			}
		}
	}

	dynamic_country_name = {
		name = dyn_c_dominion_of_india
		adjective = BIC_ADJ

		is_main_tag_only = yes
		priority = 25

		trigger = {
			scope:actor ?= {
				is_subject = yes
				NOT = { has_government_type = gov_chartered_company }
				NOT = { is_subject_type = subject_type_sb_responsible_colony }
				NOT = { is_subject_type = subject_type_sb_responsible_colony_monarchy }
			}
		}
	}

	dynamic_country_name = {
		name = dyn_c_madras_presidency
		adjective = dyn_c_east_india_company_adj

		is_main_tag_only = yes
		priority = 25

		trigger = {
			scope:actor ?= {
				is_subject = yes
				capital ?= {
					is_in_geographic_region = geographic_region_madras_old
				}
				NOR = {
					any_scope_state = {
						is_in_geographic_region = geographic_region_bengal_old
					}
					any_scope_state = {
						is_in_geographic_region = geographic_region_bombay_old
					}
				}
			}
		}
	}

	dynamic_country_name = {
		name = dyn_c_bombay_presidency
		adjective = dyn_c_east_india_company_adj

		is_main_tag_only = yes
		priority = 25

		trigger = {
			scope:actor ?= {
				is_subject = yes
				capital ?= {
					is_in_geographic_region = geographic_region_bombay_old
				}
				NOR = {
					any_scope_state = {
						is_in_geographic_region = geographic_region_madras_old
					}
					any_scope_state = {
						is_in_geographic_region = geographic_region_bengal_old
					}
				}
			}
		}
	}

	dynamic_country_name = {
		name = dyn_c_bengal_presidency

		is_main_tag_only = yes
		priority = 25
