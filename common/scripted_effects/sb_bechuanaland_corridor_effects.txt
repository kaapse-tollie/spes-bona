# Spes Bona - Bechuanaland Corridor / Caprivi Crisis effects

sb_bechuanaland_save_loc_scopes = {
	c:GBR ?= { if = { limit = { is_country_alive = yes } save_scope_as = sb_gbr_scope } }
	c:CAP ?= { if = { limit = { is_country_alive = yes } save_scope_as = sb_cap_scope } }
	c:SWA ?= {
		if = {
			limit = { is_country_alive = yes }
			save_scope_as = sb_swa_scope
		}
	}
	c:SGO ?= { if = { limit = { is_country_alive = yes } save_scope_as = sb_sgo_scope } }
	c:WBL ?= { if = { limit = { is_country_alive = yes } save_scope_as = sb_wbl_scope } }

	if = {
		limit = { any_country = { sb_bechuanaland_swa_overlord_candidate = yes } }
		random_country = {
			limit = { sb_bechuanaland_swa_overlord_candidate = yes }
			save_scope_as = sb_swa_overlord_scope
		}
	}

	if = {
		limit = { c:TRN ?= { sb_bechuanaland_boer_actor = yes } }
		c:TRN ?= { save_scope_as = sb_boer_actor_scope }
	}
	else_if = {
		limit = { c:ORA ?= { sb_bechuanaland_boer_actor = yes } }
		c:ORA ?= { save_scope_as = sb_boer_actor_scope }
	}
}

sb_bechuanaland_refresh_influence_scopes = {
	if = { limit = { has_global_variable = sb_bechuanaland_boer_actor_global_scope } remove_global_variable = sb_bechuanaland_boer_actor_global_scope }
	if = {
		limit = { c:TRN ?= { sb_bechuanaland_boer_actor = yes } }
		set_global_variable = { name = sb_bechuanaland_boer_actor_global_scope value = c:TRN }
	}
	else_if = {
		limit = { c:ORA ?= { sb_bechuanaland_boer_actor = yes } }
		set_global_variable = { name = sb_bechuanaland_boer_actor_global_scope value = c:ORA }
	}

	if = {
		limit = { any_country = { sb_bechuanaland_swa_overlord_candidate = yes } }
		random_country = {
			limit = { sb_bechuanaland_swa_overlord_candidate = yes }
			save_temporary_scope_as = sb_bechuanaland_current_swa_overlord
		}
		set_global_variable = {
			name = sb_bechuanaland_swa_overlord_global_scope
			value = scope:sb_bechuanaland_current_swa_overlord
		}
	}
	else_if = {
		limit = {
			has_global_variable = sb_bechuanaland_swa_overlord_global_scope
			global_var:sb_bechuanaland_swa_overlord_global_scope = { is_country_alive = no }
		}
		remove_global_variable = sb_bechuanaland_swa_overlord_global_scope
	}

	if = { limit = { has_global_variable = sb_bechuanaland_swa_claimant_global_scope } remove_global_variable = sb_bechuanaland_swa_claimant_global_scope }
	if = {
		limit = { c:SWA ?= { is_country_alive = yes } }
		set_global_variable = { name = sb_bechuanaland_swa_claimant_global_scope value = c:SWA }
	}
	else_if = {
		limit = {
			has_global_variable = sb_bechuanaland_swa_overlord_global_scope
			global_var:sb_bechuanaland_swa_overlord_global_scope = { is_country_alive = yes }
		}
		global_var:sb_bechuanaland_swa_overlord_global_scope = {
			save_temporary_scope_as = sb_bechuanaland_surviving_swa_overlord
		}
		set_global_variable = {
			name = sb_bechuanaland_swa_claimant_global_scope
			value = scope:sb_bechuanaland_surviving_swa_overlord
		}
	}
}

sb_bechuanaland_initialize_influence_score = {
	if = {
		limit = { NOT = { has_global_variable = sb_bechuanaland_influence_score_global_var } }
		set_global_variable = { name = sb_bechuanaland_influence_score_global_var value = 0 }
		if = {
			limit = {
				OR = {
					has_global_variable = sb_bechuanaland_caprivi_escalated_var
					has_global_variable = sb_bechuanaland_swa_influence_positive_var
				}
			}
			set_global_variable = { name = sb_bechuanaland_influence_score_global_var value = -10 }
		}
		else_if = {
			limit = { has_global_variable = sb_bechuanaland_boer_influence_positive_var }
			set_global_variable = { name = sb_bechuanaland_influence_score_global_var value = 10 }
		}
		set_global_variable = sb_bechuanaland_influence_just_initialized_global_var
	}
	if = { limit = { has_global_variable = sb_bechuanaland_boer_influence_positive_var } remove_global_variable = sb_bechuanaland_boer_influence_positive_var }
	if = { limit = { has_global_variable = sb_bechuanaland_swa_influence_positive_var } remove_global_variable = sb_bechuanaland_swa_influence_positive_var }
	if = { limit = { has_global_variable = sb_bechuanaland_caprivi_escalated_var } remove_global_variable = sb_bechuanaland_caprivi_escalated_var }
}

sb_bechuanaland_apply_influence_score_to_all = {
	every_country = {
		limit = { has_journal_entry = je_sb_bechuanaland_corridor }
		je:je_sb_bechuanaland_corridor ?= {
			set_bar_progress = {
				name = sb_bechuanaland_boer_swa_influence_bar
				value = global_var:sb_bechuanaland_influence_score_global_var
			}
		}
	}
}

sb_bechuanaland_sync_influence_bar = {
	sb_bechuanaland_refresh_influence_scopes = yes
	sb_bechuanaland_initialize_influence_score = yes
	if = {
		limit = { has_global_variable = sb_bechuanaland_influence_just_initialized_global_var }
		remove_global_variable = sb_bechuanaland_influence_just_initialized_global_var
	}
	else_if = {
		limit = { c:GBR ?= { has_journal_entry = je_sb_bechuanaland_corridor } }
		c:GBR ?= {
			je:je_sb_bechuanaland_corridor ?= {
				set_global_variable = {
					name = sb_bechuanaland_influence_score_global_var
					value = "scripted_bar_progress(sb_bechuanaland_boer_swa_influence_bar)"
				}
			}
		}
	}
	set_global_variable = {
		name = sb_bechuanaland_influence_score_global_var
		value = {
			value = global_var:sb_bechuanaland_influence_score_global_var
			min = -40
			max = 40
		}
	}
	sb_bechuanaland_apply_influence_score_to_all = yes
}

sb_bechuanaland_skip_corridor_crisis = {
	if = { limit = { has_global_variable = sb_bechuanaland_corridor_watch_global_var } remove_global_variable = sb_bechuanaland_corridor_watch_global_var }

	set_global_variable = sb_bechuanaland_corridor_resolved_global_var

sb_bechuanaland_open_corridor_is = {
	remove_global_variable = sb_bechuanaland_corridor_watch_global_var
	set_global_variable = sb_bechuanaland_corridor_open_global_var
	set_global_variable = { name = sb_bechuanaland_resolution_grace_global_var months = 12 }
	sb_bechuanaland_clear_arid_corridor_modifier = yes
	s:STATE_BOTSWANA = { add_claim = c:GBR }
	s:STATE_BECHUANALAND = {
		if = { limit = { exists = c:CAP } add_claim = c:CAP }
	}

	every_country = {
		limit = {
			sb_bechuanaland_is_involved_actor = yes
			NOT = { has_journal_entry = je_sb_bechuanaland_corridor }
		}
		add_contextless_journal_entry = je_sb_bechuanaland_corridor
	}
	every_country = {
		limit = { sb_bechuanaland_swa_overlord_candidate = yes }
		set_variable = { name = sb_bechuanaland_caprivi_ai_delay_var months = 12 }
	}
	sb_bechuanaland_sync_influence_bar = yes
	every_country = {
		limit = {
			has_journal_entry = je_sb_bechuanaland_corridor
			is_player = yes
		}
		trigger_event = { id = sb_bechuanaland_corridor.001 days = 1 popup = yes }
	}
}

sb_bechuanaland_ensure_sgo_subject_of_root = {
	save_temporary_scope_as = sb_sgo_sponsor_scope
	c:SGO ?= {
		if = {
			limit = {
				has_global_variable = sb_bechuanaland_corridor_open_global_var
				NOT = { has_journal_entry = je_sb_bechuanaland_corridor }
			}
			add_contextless_journal_entry = je_sb_bechuanaland_corridor
		}
	}
	if = {
		limit = {
			OR = {
				is_country_type = recognized
				country_is_colonial_or_company = yes
			}
		}
		if = {
			limit = { c:SGO ?= { is_country_alive = yes is_subject = no } }
			create_diplomatic_pact = { country = c:SGO type = protectorate }
		}
		else_if = {
			limit = {
				c:SGO ?= {
					is_direct_subject_of = scope:sb_sgo_sponsor_scope
					NOT = { is_subject_type = subject_type_protectorate }
				}
			}
			c:SGO ?= { change_subject_type = subject_type_protectorate }
		}
	}
	else_if = {
		limit = { is_country_type = unrecognized }
		if = {
			limit = { c:SGO ?= { is_country_alive = yes is_subject = no } }
			create_diplomatic_pact = { country = c:SGO type = tributary }
		}
		else_if = {
			limit = {
				c:SGO ?= {
					is_direct_subject_of = scope:sb_sgo_sponsor_scope
					NOT = { is_subject_type = subject_type_tributary }
				}
			}
			c:SGO ?= { change_subject_type = subject_type_tributary }
		}
	}
	c:SGO ?= {
		if = {
			limit = {
				is_country_alive = yes
				is_direct_subject_of = scope:sb_sgo_sponsor_scope
				OR = {
					is_subject_type = subject_type_protectorate
					is_subject_type = subject_type_tributary
				}
			}
			set_variable = sb_sgo_scripted_subject_established_var
		}
	}
	if = {
		limit = { has_global_variable = sb_bechuanaland_corridor_open_global_var }
		sb_bechuanaland_sync_influence_bar = yes
	}
}

sb_bechuanaland_repair_sgo_subject_if_needed = {
	if = {
		limit = {
			has_global_variable = sb_bechuanaland_corridor_open_global_var
			NOT = { has_global_variable = sb_bechuanaland_corridor_resolved_global_var }
			c:SGO ?= {
				is_country_alive = yes
				NOT = { has_variable = sb_sgo_scripted_subject_established_var }
				OR = {
					has_variable = sb_sgo_scripted_frontier_setup_var
					has_variable = sb_early_republic_var
				}
			}
		}
		if = {
			limit = {
				c:TRN ?= { sb_bechuanaland_boer_actor = yes }
				c:SGO ?= {
					OR = {
						is_subject = no
						is_direct_subject_of = c:TRN
					}
				}
			}
			c:TRN ?= { sb_bechuanaland_ensure_sgo_subject_of_root = yes }
		}
		else_if = {
			limit = {
				c:ORA ?= { sb_bechuanaland_boer_actor = yes }
				c:SGO ?= {
					OR = {
						is_subject = no
						is_direct_subject_of = c:ORA
					}
				}
			}
			c:ORA ?= { sb_bechuanaland_ensure_sgo_subject_of_root = yes }
		}
	}
}

sb_bechuanaland_ensure_sgo_population_floor = {
	if = {
		limit = {
			exists = c:SGO
			exists = s:STATE_BECHUANALAND.region_state:SGO
		}
		s:STATE_BECHUANALAND.region_state:SGO = {
			save_scope_as = sb_sgo_population_target
			set_variable = {
				name = sb_sgo_tswana_population_temp_var
				value = {
					every_scope_pop = {
						limit = { culture = cu:tswana }
						add = total_size
					}
				}
			}
		}
		if = {
			limit = {
				scope:sb_sgo_population_target.var:sb_sgo_tswana_population_temp_var < 8000
				exists = s:STATE_BECHUANALAND.region_state:TSW
				s:STATE_BECHUANALAND.region_state:TSW = {
					any_scope_pop = { culture = cu:tswana }
				}
			}
			s:STATE_BECHUANALAND.region_state:TSW = {
				ordered_scope_pop = {
					limit = { culture = cu:tswana }
					order_by = total_size
					position = 0
					check_range_bounds = no
					move_partial_pop = {
						state = scope:sb_sgo_population_target
						population = {
							value = 8000
							subtract = scope:sb_sgo_population_target.var:sb_sgo_tswana_population_temp_var
							max = total_size
						}
					}
				}
			}
		}
		scope:sb_sgo_population_target = {
			remove_variable = sb_sgo_tswana_population_temp_var
		}
		if = {
			limit = {
				scope:sb_sgo_population_target = { state_population < 10000 }
				exists = s:STATE_BECHUANALAND.region_state:TSW
				s:STATE_BECHUANALAND.region_state:TSW = {
					any_scope_pop = { culture = cu:tswana }
				}
			}
			s:STATE_BECHUANALAND.region_state:TSW = {
				ordered_scope_pop = {
					limit = { culture = cu:tswana }
					order_by = total_size
					position = 0
					check_range_bounds = no
					move_partial_pop = {
						state = scope:sb_sgo_population_target
						population = {
							value = 10000
							subtract = scope:sb_sgo_population_target.state_population
							max = total_size
						}
					}
				}
			}
		}
	}
}
