# Spes Bona - Bechuanaland Corridor / Caprivi Crisis effects
#
# Lifecycle contract
# - One named script container, parented to GBR, owns every shared corridor scope,
#   phase marker, score, and participant list.
# - The one contextless corridor JE receives only the scopes and values needed by
#   its presentation; it is addressed through GBR's standard je: link.
# - Country variables are reserved for country-local cooldowns and modifiers.
# - Queue effects consume one decision lease before recording one pending route.
# - Launch effects consume that route only after all participants validate.
# - Settlement effects save their event scopes before destroying the container.
# Callers must not retain PREV across delayed events.

sb_bechuanaland_create_corridor_container = {
	if = {
		limit = {
			NOT = { container_exists = sb_bechuanaland_corridor_state }
			c:GBR ?= { is_country_alive = yes }
		}
		create_container = {
			name = sb_bechuanaland_corridor_state
			tags = { sb_story sb_bechuanaland_corridor }
			parent = c:GBR
			on_created = {
				set_variable = { name = sb_bechuanaland_influence_score_var value = 0 }
			}
		}
	}
}

sb_bechuanaland_save_loc_scopes = {
	c:GBR ?= { if = { limit = { is_country_alive = yes } save_scope_as = sb_gbr_scope } }
	c:CAP ?= { if = { limit = { is_country_alive = yes } save_scope_as = sb_cap_scope } }
	c:SWA ?= {
		if = {
			limit = { is_country_alive = yes }
			save_scope_as = sb_swa_scope
		}
	}
	c:SGO ?= { if = { limit = { is_country_alive = yes } save_scope_as = sb_sgo_scope } }
	c:WBL ?= { if = { limit = { is_country_alive = yes } save_scope_as = sb_wbl_scope } }

	if = {
		limit = { any_country = { sb_bechuanaland_swa_active_sponsor = yes } }
		random_country = {
			limit = { sb_bechuanaland_swa_active_sponsor = yes }
			save_scope_as = sb_swa_overlord_scope
		}
	}
	if = {
		limit = {
			container_exists = sb_bechuanaland_corridor_state
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_warren_defender_scope }
		}
		container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_warren_defender_scope = {
			save_scope_as = sb_warren_defender_scope
		}
	}

	if = {
		limit = { c:SGO ?= { sb_bechuanaland_boer_actor = yes } }
		c:SGO ?= { save_scope_as = sb_boer_actor_scope }
	}
	else_if = {
		limit = { c:TRN ?= { sb_bechuanaland_boer_actor = yes } }
		c:TRN ?= { save_scope_as = sb_boer_actor_scope }
	}
	else_if = {
		limit = { c:ORA ?= { sb_bechuanaland_boer_actor = yes } }
		c:ORA ?= { save_scope_as = sb_boer_actor_scope }
	}
}

sb_bechuanaland_refresh_influence_scopes = {
	if = { limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_gbr_scope } } container:sb_bechuanaland_corridor_state = { remove_variable = sb_bechuanaland_gbr_scope } }
	if = {
		limit = { c:GBR ?= { is_country_alive = yes } }
		container:sb_bechuanaland_corridor_state = {
			set_variable = { name = sb_bechuanaland_gbr_scope value = c:GBR }
		}
	}
	if = { limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_swa_scope } } container:sb_bechuanaland_corridor_state = { remove_variable = sb_bechuanaland_swa_scope } }
	if = {
		limit = { c:SWA ?= { is_country_alive = yes } }
		container:sb_bechuanaland_corridor_state = {
			set_variable = { name = sb_bechuanaland_swa_scope value = c:SWA }
		}
	}

	if = { limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_boer_actor_scope } } container:sb_bechuanaland_corridor_state = { remove_variable = sb_bechuanaland_boer_actor_scope } }
	if = {
		limit = { c:SGO ?= { sb_bechuanaland_boer_actor = yes } }
		container:sb_bechuanaland_corridor_state = {
			set_variable = { name = sb_bechuanaland_boer_actor_scope value = c:SGO }
		}
	}
	else_if = {
		limit = { c:TRN ?= { sb_bechuanaland_boer_actor = yes } }
		container:sb_bechuanaland_corridor_state = {
			set_variable = { name = sb_bechuanaland_boer_actor_scope value = c:TRN }
		}
	}
	else_if = {
		limit = { c:ORA ?= { sb_bechuanaland_boer_actor = yes } }
		container:sb_bechuanaland_corridor_state = {
			set_variable = { name = sb_bechuanaland_boer_actor_scope value = c:ORA }
		}
	}

	# Retain a rank-demoted incumbent, but replace a dead, detached, or
	# British-aligned sponsor with the current qualifying GP overlord.
	if = {
		limit = {
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_swa_overlord_scope }
			container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_swa_overlord_scope = { sb_bechuanaland_swa_active_sponsor = yes }
		}
		container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_swa_overlord_scope = {
			save_temporary_scope_as = sb_bechuanaland_current_swa_overlord
		}
	}
	else = {
		if = { limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_swa_overlord_scope } } container:sb_bechuanaland_corridor_state = { remove_variable = sb_bechuanaland_swa_overlord_scope } }
		if = {
			limit = { any_country = { sb_bechuanaland_swa_overlord_candidate = yes } }
			random_country = {
				limit = { sb_bechuanaland_swa_overlord_candidate = yes }
				save_temporary_scope_as = sb_bechuanaland_current_swa_overlord
			}
			container:sb_bechuanaland_corridor_state = {
				set_variable = {
					name = sb_bechuanaland_swa_overlord_scope
					value = scope:sb_bechuanaland_current_swa_overlord
				}
			}
		}
	}
	if = {
		limit = {
			exists = scope:sb_bechuanaland_current_swa_overlord
			has_global_variable = sb_bechuanaland_corridor_open_global_var
			NOT = { has_global_variable = sb_bechuanaland_corridor_resolved_global_var }
		}
		scope:sb_bechuanaland_current_swa_overlord = {
			if = {
				limit = { NOT = { has_journal_entry = je_sb_bechuanaland_corridor } }
				add_contextless_journal_entry = je_sb_bechuanaland_corridor
			}
		}
	}

	if = { limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_swa_claimant_scope } } container:sb_bechuanaland_corridor_state = { remove_variable = sb_bechuanaland_swa_claimant_scope } }
	if = {
		limit = { c:SWA ?= { is_country_alive = yes } }
		container:sb_bechuanaland_corridor_state = {
			set_variable = { name = sb_bechuanaland_swa_claimant_scope value = c:SWA }
		}
	}
	else_if = {
		limit = {
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_swa_overlord_scope }
			container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_swa_overlord_scope = { is_country_alive = yes }
		}
		container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_swa_overlord_scope = {
			save_temporary_scope_as = sb_bechuanaland_surviving_swa_overlord
		}
		container:sb_bechuanaland_corridor_state = {
			set_variable = {
				name = sb_bechuanaland_swa_claimant_scope
				value = scope:sb_bechuanaland_surviving_swa_overlord
			}
		}
	}
}

	if = {

# ROOT = the singleton contextless corridor JE.
sb_bechuanaland_project_corridor_journal = {
	if = {
		limit = { NOT = { has_variable = com_hide_scripted_buttons } }
		set_variable = com_hide_scripted_buttons
	}
	if = {
		limit = {
			container_exists = sb_bechuanaland_corridor_state
			container:sb_bechuanaland_corridor_state = {
				has_variable = sb_bechuanaland_influence_score_var
			}
		}
		container:sb_bechuanaland_corridor_state = {
			save_temporary_scope_as = sb_bechuanaland_container
		}
		if = { limit = { has_variable = sb_bechuanaland_gbr_scope } remove_variable = sb_bechuanaland_gbr_scope }
		if = { limit = { has_variable = sb_bechuanaland_swa_scope } remove_variable = sb_bechuanaland_swa_scope }
		if = { limit = { has_variable = sb_bechuanaland_swa_overlord_scope } remove_variable = sb_bechuanaland_swa_overlord_scope }
		if = { limit = { has_variable = sb_bechuanaland_boer_actor_scope } remove_variable = sb_bechuanaland_boer_actor_scope }
		if = { limit = { scope:sb_bechuanaland_container = { has_variable = sb_bechuanaland_gbr_scope } } set_variable = { name = sb_bechuanaland_gbr_scope value = scope:sb_bechuanaland_container.var:sb_bechuanaland_gbr_scope } }
		if = { limit = { scope:sb_bechuanaland_container = { has_variable = sb_bechuanaland_swa_scope } } set_variable = { name = sb_bechuanaland_swa_scope value = scope:sb_bechuanaland_container.var:sb_bechuanaland_swa_scope } }
		if = { limit = { scope:sb_bechuanaland_container = { has_variable = sb_bechuanaland_swa_overlord_scope } } set_variable = { name = sb_bechuanaland_swa_overlord_scope value = scope:sb_bechuanaland_container.var:sb_bechuanaland_swa_overlord_scope } }
		if = { limit = { scope:sb_bechuanaland_container = { has_variable = sb_bechuanaland_boer_actor_scope } } set_variable = { name = sb_bechuanaland_boer_actor_scope value = scope:sb_bechuanaland_container.var:sb_bechuanaland_boer_actor_scope } }

		set_variable = {
			name = sb_bechuanaland_influence_score_var
			value = scope:sb_bechuanaland_container.var:sb_bechuanaland_influence_score_var
		}
		set_bar_progress = {
			name = sb_bechuanaland_boer_swa_influence_bar
			value = scope:sb_bechuanaland_container.var:sb_bechuanaland_influence_score_var
		}
	}
}

sb_bechuanaland_apply_influence_score_to_all = {
	if = {
		limit = {
			container_exists = sb_bechuanaland_corridor_state
			c:GBR ?= { has_journal_entry = je_sb_bechuanaland_corridor }
		}
		c:GBR = {
			je:je_sb_bechuanaland_corridor ?= {
				sb_bechuanaland_project_corridor_journal = yes
			}
		}
	}
}

# Evaluate expensive actor, treaty, territory, and relation checks once. Every
# The contextless JE reads these flags for its progress breakdown.
sb_bechuanaland_cache_influence_sources = {
	sb_bechuanaland_clear_influence_source_cache = yes
	if = {
		limit = {
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_boer_actor_scope }
			OR = {
				c:SGO ?= {
					is_country_alive = yes
					is_direct_subject_of = container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_boer_actor_scope
					OR = {
						is_subject_type = subject_type_protectorate
						is_subject_type = subject_type_tributary
					}
					sb_bechuanaland_owns_corridor_land = yes
				}
				AND = {
					NOT = { c:SGO ?= { is_country_alive = yes } }
					container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_boer_actor_scope = {
						sb_bechuanaland_owns_corridor_land = yes
					}
				}
			}
		}
		container:sb_bechuanaland_corridor_state = { set_variable = sb_bechuanaland_influence_source_boer_settlers_var }
	}
	if = {
		limit = {
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_boer_actor_scope }
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_swa_overlord_scope }
			container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_boer_actor_scope = {
				is_in_same_power_bloc = container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_swa_overlord_scope
			}
		}
		container:sb_bechuanaland_corridor_state = { set_variable = sb_bechuanaland_influence_source_boer_in_swa_bloc_var }
	}
	if = {
		limit = {
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_boer_actor_scope }
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_swa_overlord_scope }
			container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_boer_actor_scope = {
				any_scope_treaty = {
					binds = container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_swa_overlord_scope
					any_scope_article = {
						has_type = foreign_investment_rights
						source_country = container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_boer_actor_scope
						target_country = container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_swa_overlord_scope
					}
				}
			}
		}
		container:sb_bechuanaland_corridor_state = { set_variable = sb_bechuanaland_influence_source_swa_investment_var }
	}
	if = {
		limit = {
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_boer_actor_scope }
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_swa_overlord_scope }
			container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_boer_actor_scope = {
				any_scope_treaty = {
					binds = container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_swa_overlord_scope
					any_scope_article = {
						has_type = trade_privilege
						source_country = container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_boer_actor_scope
						target_country = container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_swa_overlord_scope
					}
					any_scope_article = {
						has_type = trade_privilege
						source_country = container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_swa_overlord_scope
						target_country = container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_boer_actor_scope
					}
				}

# Root-agnostic terminal routing for prerequisites lost after the corridor opens.
sb_bechuanaland_handle_post_open_lifecycle = {
	if = {
		limit = {
			has_global_variable = sb_bechuanaland_corridor_open_global_var
			NOT = { has_global_variable = sb_bechuanaland_corridor_resolved_global_var }
			NOT = { container_exists = sb_bechuanaland_corridor_state }
		}
		remove_global_variable = sb_bechuanaland_corridor_open_global_var
		sb_bechuanaland_record_invalid_terminal_outcome = yes
		set_global_variable = sb_bechuanaland_corridor_resolved_global_var
	}
	else_if = {
		limit = {
			has_global_variable = sb_bechuanaland_corridor_open_global_var
			NOT = { has_global_variable = sb_bechuanaland_corridor_resolved_global_var }
			container_exists = sb_bechuanaland_corridor_state
		}
		sb_bechuanaland_refresh_influence_scopes = yes
		if = {
			limit = { c:SAF ?= { is_country_alive = yes } }
			if = { limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_british_victory_var } } container:sb_bechuanaland_corridor_state = { remove_variable = sb_bechuanaland_british_victory_var } }
			if = { limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_boer_swa_victory_var } } container:sb_bechuanaland_corridor_state = { remove_variable = sb_bechuanaland_boer_swa_victory_var } }
			if = {
				limit = { c:GBR ?= { has_claim = s:STATE_BOTSWANA } }
					s:STATE_BOTSWANA = { add_claim = c:SAF }
				}
				sb_bechuanaland_record_invalid_terminal_outcome = yes
				set_global_variable = sb_bechuanaland_corridor_resolved_global_var
				sb_bechuanaland_cleanup_corridor_crisis = yes
			}
			else_if = {
				limit = { sb_bechuanaland_cap_is_swa_overlord_subject = yes }
				sb_bechuanaland_set_boer_swa_crisis_victory = yes
			}
		else_if = {
			limit = {
				NOT = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_war_active_var } }
				NOT = { sb_bechuanaland_cap_is_swa_overlord_subject = yes }
				NOT = {
					c:CAP ?= {
						is_country_alive = yes
						is_subject_of = c:GBR
					}
				}
			}
			if = { limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_british_victory_var } } container:sb_bechuanaland_corridor_state = { remove_variable = sb_bechuanaland_british_victory_var } }
			if = { limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_boer_swa_victory_var } } container:sb_bechuanaland_corridor_state = { remove_variable = sb_bechuanaland_boer_swa_victory_var } }
			if = { limit = { c:GBR ?= { has_claim = s:STATE_BOTSWANA } } s:STATE_BOTSWANA = { remove_claim = c:GBR } }
			sb_bechuanaland_record_invalid_terminal_outcome = yes
			set_global_variable = sb_bechuanaland_corridor_resolved_global_var
			sb_bechuanaland_cleanup_corridor_crisis = yes
		}
		else_if = {
			limit = {

# British alignment removes the SWA appeal but preserves Warren's local choice.
sb_bechuanaland_queue_british_alignment_warren = {
	if = {
		limit = {
			sb_bechuanaland_corridor_crisis_is_live = yes
			sb_bechuanaland_swa_overlord_is_british_aligned = yes
			sb_bechuanaland_crisis_escalation_available = yes
		}
		container:sb_bechuanaland_corridor_state = {
			set_variable = { name = sb_bechuanaland_british_alignment_warren_pending_var days = 30 }
		}
		sb_bechuanaland_reserve_warren_decision = yes
		s:STATE_BECHUANALAND = { add_claim = c:CAP }
		s:STATE_GRIQUALAND_WEST = { add_claim = c:CAP }
		if = {
			limit = { c:SGO ?= { is_country_alive = yes is_subject = yes } }
			every_country = {
				limit = { c:SGO ?= { is_direct_subject_of = PREV } }
				sb_bechuanaland_record_gbr_warren_dispatch = yes
				trigger_event = { id = sb_bechuanaland_corridor.034 days = 1 popup = yes }
			}
		}
		else_if = {
			limit = { c:WBL ?= { is_country_alive = yes is_subject = yes } }
			every_country = {
				limit = { c:WBL ?= { is_direct_subject_of = PREV } }
				sb_bechuanaland_record_gbr_warren_dispatch = yes
				trigger_event = { id = sb_bechuanaland_corridor.034 days = 1 popup = yes }
			}
		}
		else_if = {
			limit = {
				c:TRN ?= {
					is_country_alive = yes
					OR = {
						owns_entire_state_region = STATE_GRIQUALAND_WEST
						owns_entire_state_region = STATE_BECHUANALAND
					}
				}
			}
			c:TRN ?= {
				sb_bechuanaland_record_gbr_warren_dispatch = yes
				trigger_event = { id = sb_bechuanaland_corridor.034 days = 1 popup = yes }
			}
		}
		else_if = {
			limit = {
				c:ORA ?= {
					is_country_alive = yes
					is_subject = no
					OR = {
						owns_entire_state_region = STATE_GRIQUALAND_WEST
						owns_entire_state_region = STATE_BECHUANALAND
					}
				}
			}
			c:ORA ?= {
				sb_bechuanaland_record_gbr_warren_dispatch = yes
				trigger_event = { id = sb_bechuanaland_corridor.034 days = 1 popup = yes }
			}
		}
		else = {
			sb_bechuanaland_finish_warren_attempt = yes
			sb_bechuanaland_clear_escalation_decision = yes
			if = { limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_british_alignment_warren_pending_var } } container:sb_bechuanaland_corridor_state = { remove_variable = sb_bechuanaland_british_alignment_warren_pending_var } }
			sb_bechuanaland_set_british_crisis_victory = yes
		}
	}
}

	}

	# cultural changes make it fail the broad republic trigger, unless it now
	# belongs to the opposing British subject system.

			container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_boer_actor_scope = {

			sb_bechuanaland_add_to_boer_network = yes
		}
	}
	# Stage subjects before appending them. Clausewitz list iteration is not
	# guaranteed to remain stable when the traversed list is modified in place.
	container:sb_bechuanaland_corridor_state = {
		save_temporary_scope_as = sb_bechuanaland_container
		if = { limit = { has_variable_list = sb_bechuanaland_boer_network_staging } clear_variable_list = sb_bechuanaland_boer_network_staging }

			variable = sb_bechuanaland_boer_network

							NOT = {
								is_target_in_variable_list = {

				variable = sb_bechuanaland_boer_network_staging
				save_temporary_scope_as = sb_bechuanaland_network_subject
				if = {
					limit = {
					NOT = {
						is_target_in_variable_list = {
							name = sb_bechuanaland_boer_network
							target = scope:sb_bechuanaland_network_subject
						}
						}
					}
					add_to_variable_list = {
						name = sb_bechuanaland_boer_network
						target = scope:sb_bechuanaland_network_subject
					}

	}
	else = { sb_bechuanaland_cancel_pending_crisis = yes }

sb_bechuanaland_try_launch_pending_crisis = {
	if = {
		limit = {
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_escalation_pending_var }
			NOT = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_war_active_var } }
		}
		sb_bechuanaland_refresh_influence_scopes = yes
		if = {
			limit = {
				container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_boer_support_var }
				NOT = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_boer_actor_scope } }
			}
			container:sb_bechuanaland_corridor_state = { remove_variable = sb_bechuanaland_crisis_boer_support_var }
		}
		if = {
			limit = { NOT = { sb_bechuanaland_crisis_core_is_valid = yes } }
			sb_bechuanaland_cancel_pending_crisis = yes
		}
		else = {
			if = {
				limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_boer_choice_pending_var } }
				if = {
					limit = {
						container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_boer_actor_scope }
						NOT = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_boer_choice_dispatched_var } }
					}
					# Outlive the three-month event so the monthly pulse cannot dispatch duplicates.
					container:sb_bechuanaland_corridor_state = {
						set_variable = { name = sb_bechuanaland_crisis_boer_choice_dispatched_var months = 4 }
					}
					if = {
						limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_direct_route_var } }
						container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_boer_actor_scope = {
							trigger_event = { id = sb_bechuanaland_corridor.032 days = 3 popup = yes }
						}
					}
					else_if = {
						limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_proxy_route_var } }
						container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_boer_actor_scope = {
							trigger_event = { id = sb_bechuanaland_corridor.033 days = 3 popup = yes }
						}
					}
				}
				else_if = {
					limit = { NOT = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_boer_actor_scope } } }
					sb_bechuanaland_choose_boer_crisis_neutrality = yes
				}
			}
			if = {
				limit = {
					container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_escalation_pending_var }
					NOT = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_boer_choice_pending_var } }
				}
				if = {
					limit = { NOT = { sb_bechuanaland_crisis_participant_snapshot_is_valid = yes } }
					sb_bechuanaland_refresh_crisis_launch_participants = yes
				}
				if = {
					limit = { sb_bechuanaland_crisis_launch_participants_ready = yes }
					if = {
						limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_caprivi_route_var } }
						if = {
							limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_direct_route_var } }
							sb_bechuanaland_launch_direct_crisis = yes
						}
						else_if = {
							limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_proxy_route_var } }
							sb_bechuanaland_launch_proxy_crisis = yes
						}
					}
					else_if = {
						limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_direct_route_var } }
						sb_bechuanaland_launch_warren_direct_crisis = yes
					}
					else_if = {

					}
				}

			}
		}
		if = {

			}
			s:STATE_ZAMBIA = { add_claim = c:GBR }
			s:STATE_ZAMBEZI = { add_claim = c:GBR }
			s:STATE_KAZEMBE = { add_claim = c:GBR }
		}

						set_state_type = incorporated
					}
				}
			}
			sb_bechuanaland_record_british_terminal_outcome = yes
			set_global_variable = sb_bechuanaland_corridor_resolved_global_var
			container:sb_bechuanaland_corridor_state = { remove_variable = sb_bechuanaland_british_settlement_ready_var }

sb_bechuanaland_refresh_crisis_support_recipients = {
	every_country = {
		limit = { has_modifier = sb_bechuanaland_crisis_support }
		remove_modifier = sb_bechuanaland_crisis_support
	}
	c:SWA ?= {
		if = {
			limit = { sb_bechuanaland_fighting_british_side = yes }
			add_modifier = { name = sb_bechuanaland_crisis_support }
		}
	}
	if = {
		limit = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_boer_actor_scope } }
		every_country = {
			limit = {
				sb_bechuanaland_fighting_british_side = yes
				OR = {
					this = container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_boer_actor_scope
					AND = {
						country_has_primary_culture = cu:boer
						is_subject_of = container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_boer_actor_scope
					}
				}
			}
			add_modifier = { name = sb_bechuanaland_crisis_support }
		}
	}
}

sb_bechuanaland_maintain_crisis_support = {
	if = {
		limit = {
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_swa_overlord_scope }
			container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_swa_overlord_scope = {
				has_variable = sb_bechuanaland_swa_crisis_support_active_var
			}
			sb_bechuanaland_boer_swa_network_fighting_british_side = yes
		}
		container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_swa_overlord_scope = {
			if = {
				limit = { NOT = { has_modifier = sb_bechuanaland_crisis_support_cost } }
				add_modifier = { name = sb_bechuanaland_crisis_support_cost }
			}
		}
		sb_bechuanaland_refresh_crisis_support_recipients = yes
	}
	else = {
		every_country = {
			limit = {
				OR = {
					has_variable = sb_bechuanaland_swa_crisis_support_active_var
					has_modifier = sb_bechuanaland_crisis_support_cost
					has_modifier = sb_bechuanaland_crisis_support
				}
			}
			if = {
				limit = { has_variable = sb_bechuanaland_swa_crisis_support_active_var }
				remove_variable = sb_bechuanaland_swa_crisis_support_active_var
			}
			if = {
				limit = { has_modifier = sb_bechuanaland_crisis_support_cost }
				remove_modifier = sb_bechuanaland_crisis_support_cost
			}
			if = {
				limit = { has_modifier = sb_bechuanaland_crisis_support }
				remove_modifier = sb_bechuanaland_crisis_support
			}
		}
	}
}

		sb_bechuanaland_clear_influence_source_cache = yes

				has_variable = sb_bechuanaland_swa_subsidy_active_var

				has_modifier = sb_bechuanaland_crisis_support

		}
		if = { limit = { has_variable = sb_bechuanaland_boer_subsidy_active_var } remove_variable = sb_bechuanaland_boer_subsidy_active_var }
		if = { limit = { has_variable = sb_bechuanaland_boer_subsidy_cooldown_var } remove_variable = sb_bechuanaland_boer_subsidy_cooldown_var }
		if = { limit = { has_variable = sb_bechuanaland_swa_subsidy_active_var } remove_variable = sb_bechuanaland_swa_subsidy_active_var }
		if = { limit = { has_variable = sb_bechuanaland_swa_subsidy_cooldown_var } remove_variable = sb_bechuanaland_swa_subsidy_cooldown_var }
		if = { limit = { has_variable = sb_bechuanaland_swa_garrison_cooldown_var } remove_variable = sb_bechuanaland_swa_garrison_cooldown_var }

		if = { limit = { has_modifier = sb_bechuanaland_crisis_support } remove_modifier = sb_bechuanaland_crisis_support }
		if = { limit = { has_modifier = sb_bechuanaland_crisis_support_cost } remove_modifier = sb_bechuanaland_crisis_support_cost }
	}
	if = {
		limit = { container_exists = sb_bechuanaland_corridor_state }

	}
}
