# Spes Bona - Bechuanaland Corridor / Caprivi Crisis effects

sb_bechuanaland_save_loc_scopes = {
	c:GBR ?= { if = { limit = { is_country_alive = yes } save_scope_as = sb_gbr_scope } }
	c:CAP ?= { if = { limit = { is_country_alive = yes } save_scope_as = sb_cap_scope } }
	c:SWA ?= {
		if = {
			limit = { is_country_alive = yes }
			save_scope_as = sb_swa_scope
		}
	}
	c:SGO ?= { if = { limit = { is_country_alive = yes } save_scope_as = sb_sgo_scope } }
	c:WBL ?= { if = { limit = { is_country_alive = yes } save_scope_as = sb_wbl_scope } }

	if = {
		limit = { any_country = { sb_bechuanaland_swa_overlord_candidate = yes } }
		random_country = {
			limit = { sb_bechuanaland_swa_overlord_candidate = yes }
			save_scope_as = sb_swa_overlord_scope
		}
	}

	if = {
		limit = { c:TRN ?= { sb_bechuanaland_boer_actor = yes } }
		c:TRN ?= { save_scope_as = sb_boer_actor_scope }
	}
	else_if = {
		limit = { c:ORA ?= { sb_bechuanaland_boer_actor = yes } }
		c:ORA ?= { save_scope_as = sb_boer_actor_scope }
	}
}

sb_bechuanaland_sync_influence_bar = {
	every_country = {
		limit = { has_journal_entry = je_sb_bechuanaland_corridor }
		je:je_sb_bechuanaland_corridor ?= {
			if = {
				limit = { has_global_variable = sb_bechuanaland_boer_influence_positive_var }
				set_bar_progress = {
					name = sb_bechuanaland_boer_swa_influence_bar
					value = 45
				}
			}
			else_if = {
				limit = { has_global_variable = sb_bechuanaland_swa_influence_positive_var }
				set_bar_progress = {
					name = sb_bechuanaland_boer_swa_influence_bar
					value = -45
				}
			}
			else = {
				set_bar_progress = {
					name = sb_bechuanaland_boer_swa_influence_bar
					value = 0
				}
			}
		}
	}
}

sb_bechuanaland_add_boer_influence = {
	if = { limit = { has_global_variable = sb_bechuanaland_swa_influence_positive_var } remove_global_variable = sb_bechuanaland_swa_influence_positive_var }
	set_global_variable = sb_bechuanaland_boer_influence_positive_var
	sb_bechuanaland_sync_influence_bar = yes
}

sb_bechuanaland_add_swa_influence = {
	if = { limit = { has_global_variable = sb_bechuanaland_boer_influence_positive_var } remove_global_variable = sb_bechuanaland_boer_influence_positive_var }
	set_global_variable = sb_bechuanaland_swa_influence_positive_var
	sb_bechuanaland_sync_influence_bar = yes
}

sb_bechuanaland_skip_corridor_crisis = {
	if = { limit = { has_global_variable = sb_bechuanaland_corridor_watch_global_var } remove_global_variable = sb_bechuanaland_corridor_watch_global_var }
	if = { limit = { has_global_variable = sb_bechuanaland_corridor_open_global_var } remove_global_variable = sb_bechuanaland_corridor_open_global_var }
	set_global_variable = sb_bechuanaland_corridor_resolved_global_var
	sb_bechuanaland_clear_arid_corridor_modifier = yes
}

sb_bechuanaland_open_corridor_is = {
	remove_global_variable = sb_bechuanaland_corridor_watch_global_var
	set_global_variable = sb_bechuanaland_corridor_open_global_var
	set_global_variable = { name = sb_bechuanaland_resolution_grace_global_var months = 12 }
	sb_bechuanaland_clear_arid_corridor_modifier = yes
	s:STATE_BOTSWANA = { add_claim = c:GBR }
	s:STATE_BECHUANALAND = {
		if = { limit = { exists = c:CAP } add_claim = c:CAP }
	}

	every_country = {
		limit = {
			sb_bechuanaland_is_involved_actor = yes
			NOT = { has_journal_entry = je_sb_bechuanaland_corridor }
		}
		add_contextless_journal_entry = je_sb_bechuanaland_corridor
	}
	every_country = {
		limit = { sb_bechuanaland_swa_overlord_candidate = yes }
		set_variable = { name = sb_bechuanaland_caprivi_ai_delay_var months = 12 }
	}
	sb_bechuanaland_sync_influence_bar = yes
	every_country = {
		limit = {
			has_journal_entry = je_sb_bechuanaland_corridor
			is_player = yes
		}
		trigger_event = { id = sb_bechuanaland_corridor.001 days = 1 popup = yes }
	}
}

sb_bechuanaland_ensure_sgo_protectorate_of_root = {
	save_scope_as = sb_sgo_protector_scope
	c:SGO ?= {
		if = {
			limit = {
				has_global_variable = sb_bechuanaland_corridor_open_global_var
				NOT = { has_journal_entry = je_sb_bechuanaland_corridor }
			}
			add_contextless_journal_entry = je_sb_bechuanaland_corridor
			sb_bechuanaland_sync_influence_bar = yes
		}
		if = {
			limit = {
				is_country_alive = yes
				NOT = { is_subject_of = scope:sb_sgo_protector_scope }
			}
			scope:sb_sgo_protector_scope = {
				create_diplomatic_pact = {
					country = c:SGO
					type = protectorate
				}
			}
		}
		if = {
			limit = {
				is_country_alive = yes
				is_subject_of = root
				NOT = { is_subject_type = subject_type_protectorate }
			}
			change_subject_type = subject_type_protectorate
		}
	}
}

}

sb_bechuanaland_ensure_sgo_population_floor = {
	if = {
		limit = {
			exists = c:SGO
			exists = s:STATE_BECHUANALAND.region_state:SGO
		}
		s:STATE_BECHUANALAND.region_state:SGO = {
			save_scope_as = sb_sgo_population_target
			set_variable = {
				name = sb_sgo_tswana_population_temp_var
				value = {
					every_scope_pop = {
						limit = { culture = cu:tswana }
						add = total_size
					}
				}
			}
		}
		if = {
			limit = {
				scope:sb_sgo_population_target.var:sb_sgo_tswana_population_temp_var < 8000
				exists = s:STATE_BECHUANALAND.region_state:TSW
				s:STATE_BECHUANALAND.region_state:TSW = {
					any_scope_pop = { culture = cu:tswana }
				}
			}
			s:STATE_BECHUANALAND.region_state:TSW = {
				ordered_scope_pop = {
					limit = { culture = cu:tswana }
					order_by = total_size
					position = 0
					check_range_bounds = no
					move_partial_pop = {
						state = scope:sb_sgo_population_target
						population = {
							value = 8000
							subtract = scope:sb_sgo_population_target.var:sb_sgo_tswana_population_temp_var
							max = total_size
						}
					}
				}
			}
		}
		scope:sb_sgo_population_target = {
			remove_variable = sb_sgo_tswana_population_temp_var
		}
		if = {
			limit = {
				scope:sb_sgo_population_target = { state_population < 10000 }
				exists = s:STATE_BECHUANALAND.region_state:TSW
				s:STATE_BECHUANALAND.region_state:TSW = {
					any_scope_pop = { culture = cu:tswana }
				}
			}
			s:STATE_BECHUANALAND.region_state:TSW = {
				ordered_scope_pop = {
					limit = { culture = cu:tswana }
					order_by = total_size
					position = 0
					check_range_bounds = no
					move_partial_pop = {
						state = scope:sb_sgo_population_target
						population = {
							value = 10000
							subtract = scope:sb_sgo_population_target.state_population
							max = total_size
						}
					}
				}
			}
		}
	}
}

sb_bechuanaland_save_dual_return_scopes = {
	if = {
		limit = {
			c:WBL ?= {
				is_country_alive = yes
				owns_entire_state_region = STATE_GRIQUALAND_WEST
				OR = {
					is_subject_of = c:ORA
					is_subject_of = c:TRN
				}
			}
		}
		c:WBL ?= { save_temporary_scope_as = sb_bechuanaland_glw_return_target }
		s:STATE_GRIQUALAND_WEST.region_state:WBL ?= { save_temporary_scope_as = sb_bechuanaland_glw_return_state }
	}
	else_if = {
		limit = { c:TRN ?= { owns_entire_state_region = STATE_GRIQUALAND_WEST } }
		c:TRN ?= { save_temporary_scope_as = sb_bechuanaland_glw_return_target }
		s:STATE_GRIQUALAND_WEST.region_state:TRN ?= { save_temporary_scope_as = sb_bechuanaland_glw_return_state }
	}
	else_if = {
		limit = { c:ORA ?= { owns_entire_state_region = STATE_GRIQUALAND_WEST } }
		c:ORA ?= { save_temporary_scope_as = sb_bechuanaland_glw_return_target }
		s:STATE_GRIQUALAND_WEST.region_state:ORA ?= { save_temporary_scope_as = sb_bechuanaland_glw_return_state }
	}

	if = {
		limit = {
			c:SGO ?= {
				is_country_alive = yes
				owns_entire_state_region = STATE_BECHUANALAND
				OR = {
					is_subject_of = c:ORA
					is_subject_of = c:TRN
				}
			}
		}
		c:SGO ?= { save_temporary_scope_as = sb_bechuanaland_corridor_return_target }
		s:STATE_BECHUANALAND.region_state:SGO ?= { save_temporary_scope_as = sb_bechuanaland_corridor_return_state }
	}
	else_if = {
		limit = { c:TRN ?= { owns_entire_state_region = STATE_BECHUANALAND } }
		c:TRN ?= { save_temporary_scope_as = sb_bechuanaland_corridor_return_target }
		s:STATE_BECHUANALAND.region_state:TRN ?= { save_temporary_scope_as = sb_bechuanaland_corridor_return_state }
	}
	else_if = {
		limit = { c:ORA ?= { owns_entire_state_region = STATE_BECHUANALAND } }
		c:ORA ?= { save_temporary_scope_as = sb_bechuanaland_corridor_return_target }
		s:STATE_BECHUANALAND.region_state:ORA ?= { save_temporary_scope_as = sb_bechuanaland_corridor_return_state }
	}
}

sb_bechuanaland_start_cap_dual_return_war = {
	sb_bechuanaland_save_dual_return_scopes = yes
	if = {
		limit = {
			exists = scope:sb_bechuanaland_glw_return_target
			exists = scope:sb_bechuanaland_glw_return_state
			exists = scope:sb_bechuanaland_corridor_return_target
			exists = scope:sb_bechuanaland_corridor_return_state
		}
		c:CAP ?= {
			sb_bechuanaland_mark_british_war_side = yes
			create_diplomatic_play = {
				type = dp_sb_bechuanaland_cap_dual_return_locked
				target_country = scope:sb_bechuanaland_glw_return_target
				target_state = scope:sb_bechuanaland_glw_return_state
			}
		}
		random_diplomatic_play = {
			limit = {
				initiator = c:CAP
				is_diplomatic_play_type = dp_sb_bechuanaland_cap_dual_return_locked
			}
			if = {
				limit = {
					scope:sb_bechuanaland_corridor_return_target = {
						NOT = { this = scope:sb_bechuanaland_glw_return_target }
					}
				}
				add_target_backers = { scope:sb_bechuanaland_corridor_return_target }
			}
			add_war_goal = {
				holder = c:CAP
				type = return_state
				target_country = scope:sb_bechuanaland_corridor_return_target
				target_state = scope:sb_bechuanaland_corridor_return_state
			}
		}
		c:GBR ?= { sb_bechuanaland_mark_british_war_side = yes }
		scope:sb_bechuanaland_glw_return_target = { sb_bechuanaland_mark_boer_swa_war_side = yes }
		scope:sb_bechuanaland_corridor_return_target = { sb_bechuanaland_mark_boer_swa_war_side = yes }
		c:ORA ?= { if = { limit = { is_country_alive = yes } sb_bechuanaland_mark_boer_swa_war_side = yes } }
		c:TRN ?= { if = { limit = { is_country_alive = yes } sb_bechuanaland_mark_boer_swa_war_side = yes } }
		c:WBL ?= { if = { limit = { is_country_alive = yes } sb_bechuanaland_mark_boer_swa_war_side = yes } }
		c:SGO ?= { if = { limit = { is_country_alive = yes } sb_bechuanaland_mark_boer_swa_war_side = yes } }
	}
}

sb_bechuanaland_transfer_dual_corridor_to_cap = {
	every_country = {
		limit = {
			OR = {
				country_definition = cd:WBL
				country_definition = cd:ORA
				country_definition = cd:TRN
			}
			owns_entire_state_region = STATE_GRIQUALAND_WEST
		}
		save_temporary_scope_as = sb_bechuanaland_state_giver
		transfer_state = {
			STATE = s:STATE_GRIQUALAND_WEST
			GIVER = scope:sb_bechuanaland_state_giver
			TAKER = c:CAP
		}
	}
	every_country = {
		limit = {
			OR = {
				country_definition = cd:SGO
				country_definition = cd:ORA
				country_definition = cd:TRN
			}
			owns_entire_state_region = STATE_BECHUANALAND
		}
		save_temporary_scope_as = sb_bechuanaland_state_giver
		transfer_state = {
			STATE = s:STATE_BECHUANALAND
			GIVER = scope:sb_bechuanaland_state_giver
			TAKER = c:CAP
		}
	}
}

sb_bechuanaland_apply_british_settlement = {
	if = {
		limit = { NOT = { has_global_variable = sb_bechuanaland_corridor_resolved_global_var } }
		set_global_variable = sb_bechuanaland_british_victory_global_var
		set_global_variable = sb_bechuanaland_corridor_resolved_global_var
	}
	if = {
		limit = { c:CAP ?= { is_country_alive = yes } }
		if = {
			limit = {
				c:CAP ?= {
					OR = {
						NOT = { sb_cap_is_responsible_or_dominion = yes }
						sb_bechuanaland_cap_hardline_for_botswana = yes
					}
				}
			}
			sb_bechuanaland_make_tsw_subject_of_gbr = yes
		}
		else_if = {
			limit = { c:CAP ?= { sb_bechuanaland_cap_liberal_botswana_route = yes } }
			sb_bechuanaland_make_tsw_subject_of_cap = yes
		}
		else = {
			if = { limit = { c:GBR ?= { has_claim = s:STATE_BOTSWANA } } s:STATE_BOTSWANA = { remove_claim = c:GBR } }
			s:STATE_BECHUANALAND = { add_claim = c:CAP }
			c:CAP ?= {
				if = {
					limit = {
						NOT = { has_technology_researched = malaria_prevention }
						can_research = malaria_prevention
					}
					add_technology_progress = { progress = 7500 technology = malaria_prevention }
				}
			}
		}
	}
	sb_bechuanaland_cleanup_corridor_crisis = yes
}

sb_bechuanaland_apply_boer_swa_settlement = {
	if = {
		limit = { NOT = { has_global_variable = sb_bechuanaland_corridor_resolved_global_var } }
		set_global_variable = sb_bechuanaland_boer_swa_victory_global_var
		set_global_variable = sb_bechuanaland_corridor_resolved_global_var
	}
	if = { limit = { c:GBR ?= { has_claim = s:STATE_BOTSWANA } } s:STATE_BOTSWANA = { remove_claim = c:GBR } }
	if = { limit = { c:CAP ?= { has_claim = s:STATE_BECHUANALAND } } s:STATE_BECHUANALAND = { remove_claim = c:CAP } }
	if = { limit = { exists = c:SGO } s:STATE_BECHUANALAND = { add_claim = c:SGO } }
	if = {
		limit = { has_global_variable = sb_bechuanaland_caprivi_escalated_var }
		s:STATE_BOTSWANA = { add_claim = c:SWA }
	}
	else_if = {
		limit = {
			has_global_variable = sb_bechuanaland_boer_influence_positive_var
			exists = c:SGO
		}
		s:STATE_BOTSWANA = { add_claim = c:SGO }
	}
	else = {
		s:STATE_BOTSWANA = { add_claim = c:SWA }
	}
	if = {
		limit = {
			has_global_variable = sb_kimberley_cap_victory_global_var
			has_global_variable = sb_bechuanaland_boer_influence_positive_var
			exists = c:SGO
		}
		sb_bechuanaland_transfer_tsw_corridor_to_sgo = yes
	}
	sb_bechuanaland_cleanup_corridor_crisis = yes
}

sb_bechuanaland_cleanup_corridor_crisis = {
	if = { limit = { has_global_variable = sb_bechuanaland_corridor_watch_global_var } remove_global_variable = sb_bechuanaland_corridor_watch_global_var }
	if = { limit = { has_global_variable = sb_bechuanaland_corridor_open_global_var } remove_global_variable = sb_bechuanaland_corridor_open_global_var }
	if = { limit = { has_global_variable = sb_bechuanaland_resolution_grace_global_var } remove_global_variable = sb_bechuanaland_resolution_grace_global_var }
	if = { limit = { has_global_variable = sb_bechuanaland_crisis_war_active_var } remove_global_variable = sb_bechuanaland_crisis_war_active_var }
	if = { limit = { has_global_variable = sb_bechuanaland_caprivi_escalated_var } remove_global_variable = sb_bechuanaland_caprivi_escalated_var }
	c:GBR ?= { if = { limit = { has_variable = sb_bechuanaland_british_war_side_var } remove_variable = sb_bechuanaland_british_war_side_var } }
	c:CAP ?= {
		if = { limit = { has_variable = sb_bechuanaland_british_war_side_var } remove_variable = sb_bechuanaland_british_war_side_var }
		if = { limit = { has_variable = sb_bechuanaland_cap_warren_sent_var } remove_variable = sb_bechuanaland_cap_warren_sent_var }
		if = { limit = { has_modifier = sb_bechuanaland_crisis_support } remove_modifier = sb_bechuanaland_crisis_support }
	}
	c:SWA ?= {
		if = { limit = { has_variable = sb_bechuanaland_boer_swa_war_side_var } remove_variable = sb_bechuanaland_boer_swa_war_side_var }
		if = { limit = { has_modifier = sb_bechuanaland_crisis_support } remove_modifier = sb_bechuanaland_crisis_support }
	}
	c:SGO ?= {
		if = { limit = { has_variable = sb_bechuanaland_boer_swa_war_side_var } remove_variable = sb_bechuanaland_boer_swa_war_side_var }
		if = { limit = { has_modifier = sb_bechuanaland_crisis_support } remove_modifier = sb_bechuanaland_crisis_support }
	}
	c:WBL ?= { if = { limit = { has_variable = sb_bechuanaland_boer_swa_war_side_var } remove_variable = sb_bechuanaland_boer_swa_war_side_var } }
	c:ORA ?= { if = { limit = { has_variable = sb_bechuanaland_boer_swa_war_side_var } remove_variable = sb_bechuanaland_boer_swa_war_side_var } }
	c:TRN ?= { if = { limit = { has_variable = sb_bechuanaland_boer_swa_war_side_var } remove_variable = sb_bechuanaland_boer_swa_war_side_var } }
	every_country = {
		limit = {
			has_journal_entry = je_sb_bechuanaland_corridor
		}
		if = { limit = { has_variable = sb_bechuanaland_support_cooldown_var } remove_variable = sb_bechuanaland_support_cooldown_var }
	}
	every_country = {
		limit = { has_variable = sb_bechuanaland_caprivi_ai_delay_var }
		remove_variable = sb_bechuanaland_caprivi_ai_delay_var
	}
}
