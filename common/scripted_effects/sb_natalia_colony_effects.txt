# Spes Bona - Natalia British Colony and Refugee Effects
#
# This file owns the later Natalia branch where the Boer republic is either
# forced into submission or reorganized by Britain as a crown colony. Keeping
# the colony/refugee work separate from `sb_natalia_effects.txt` leaves the
# Blood River and republican-foundation path much easier to navigate.

sb_assign_british_natal_colony_territory = {
	s:STATE_NATAL = {
		every_scope_state = {
			limit = { owner = ROOT }
			set_state_owner = c:NAL
		}
		add_claim = c:NAL
	}
	s:STATE_ZULULAND = {
		every_scope_state = {
			limit = { owner = ROOT }
			set_state_owner = c:NAL
		}
	}
}

# Britain can respond to Dingane's coastal raid by annexing the Zulu-held Natal
# littoral outright and reorganizing it as a crown colony.
sb_place_natalia_under_british_protectorate = {
	change_relations = {
		country = c:GBR
		value = 50
	}
	if = {
		limit = {
			NOT = { is_subject_of = c:GBR }
		}
		c:GBR ?= {
			create_diplomatic_pact = {
				country = root
				type = protectorate
			}
		}
	}
	else_if = {
		limit = {
			is_subject_of = c:GBR
			NOT = { is_subject_type = subject_type_protectorate }
			NOT = { is_subject_type = subject_type_colony }
		}
		change_subject_type = subject_type_protectorate
	}
}

sb_disable_natalia_great_trek_for_british_colony = {
	set_variable = sb_natalia_great_trek_disabled_var
	if = {
		limit = { has_variable = sb_trek_stage2_pending_var }
		remove_variable = sb_trek_stage2_pending_var
	}
	every_scope_state = {
		remove_modifier = sb_trek_migration_pull
	}
}

sb_convert_natalia_to_british_crown_colony = {
	if = {
		limit = {
			NOT = { has_variable = sb_natalia_british_colony_resolved_var }
		}
		set_variable = sb_natalia_british_colony_resolved_var
	}
	change_relations = {
		country = c:GBR
		value = 50
	}
	# Vanilla colony subject types only remain valid on countries with the
	# colonial country type. Without this, Natalia can briefly become a colony
	# and then immediately drop back out of subjecthood on the next tick.
	if = {
		limit = {
			NOT = { is_country_type = colonial }
		}
		set_country_type = colonial
	}
	if = {
		limit = {
			is_subject = no
		}
		c:GBR ?= {
			create_diplomatic_pact = {
				country = prev
				type = colony
			}
		}
	}
	else_if = {
		limit = { NOT = { is_direct_subject_of = c:GBR } }
		c:GBR ?= { transfer_subject = prev }
	}
	if = {
		limit = {
			is_direct_subject_of = c:GBR
			NOT = { is_subject_type = subject_type_colony }
		}
		change_subject_type = subject_type_colony
	}
}

sb_ensure_british_natal_loyalist_lobby = {
	every_political_lobby = {
		limit = {
			target = c:GBR
			OR = {
				is_political_lobby_type = lobby_anti_country
				is_political_lobby_type = lobby_anti_overlord
			}
		}
		disband_political_lobby = yes
	}
	if = {
		limit = {
			NOT = {
				any_political_lobby = {
					target = c:GBR
					OR = {
						is_political_lobby_type = lobby_pro_country
						is_political_lobby_type = lobby_pro_overlord
					}
				}
			}
		}
		create_political_lobby = {
			type = lobby_pro_overlord
			target = c:GBR
			add_interest_group = ig:ig_armed_forces
			add_interest_group = ig:ig_landowners
			add_interest_group = ig:ig_industrialists
		}
	}
}

sb_apply_british_natal_colony_setup = {
	sb_disable_natalia_great_trek_for_british_colony = yes
	# The native Crown Colony post-change callback must keep the governor
	# appointed below. Let its protection expire after the government refresh.
	set_variable = { name = keep_current_ruler_var days = 7 }
	sb_convert_natalia_to_british_crown_colony = yes
	if = {
		limit = { NOT = { has_variable = sb_natal_zulu_movement_seeded_var } }
		set_variable = sb_natal_zulu_movement_seeded_var
		sb_natal_create_zulu_national_movement = yes
	}
	if = {
		limit = {
			has_variable = sb_commandant_general_visible_var
		}
		remove_variable = sb_commandant_general_visible_var
	}
	if = {
		limit = {
			has_variable = sb_early_republic_var
		}
		remove_variable = sb_early_republic_var
	}
	if = {
		limit = {
			NOT = { has_variable = southern_africa_subject_var }
		}
		set_variable = southern_africa_subject_var
	}
	activate_law = law_type:law_colonial_administration
	activate_law = law_type:law_professional_army
	activate_law = law_type:law_slavery_banned
	activate_law = law_type:law_freedom_of_conscience
	activate_law = law_type:law_subjecthood
	active_law:lawgroup_citizenship ?= {
		if = {
			limit = {
				NOT = { has_amendment = amendment_type:amendment_racialized_subjecthood }
			}
			add_amendment = {
				type = amendment_racialized_subjecthood
				sponsor = PREV.ig:ig_armed_forces
			}
		}
	}
	activate_law = law_type:law_extraction_economy
	if = {
		limit = {
			c:ZUL ?= { is_country_alive = yes }
			NOT = { has_truce_with = c:ZUL }
		}
		create_bidirectional_truce = {
			country = c:ZUL
			months = 120
		}
	}
	active_law:lawgroup_army_model ?= {
		if = {
			limit = {
				has_amendment = amendment_type:amendment_sb_commando_corps
			}
			every_scope_amendment = {
				limit = {
					type = amendment_type:amendment_sb_commando_corps
				}
				remove_amendment = yes
			}
		}
	}
	if = {
		limit = {
			NOT = { country_has_primary_culture = cu:anglo_african }
		}
		add_primary_culture = cu:anglo_african
	}
	if = {
		limit = {
			country_has_primary_culture = cu:british
		}
		remove_primary_culture = cu:british
	}
	if = {
		limit = {
			country_has_primary_culture = cu:boer
		}
		remove_primary_culture = cu:boer
	}
	if = {
		limit = { NOT = { has_variable = sb_crown_administration_initialized_var } }
		sb_apply_generic_british_colony_interest_group_identity = yes
		sb_establish_crown_administration = yes
	}
	sb_ensure_british_natal_loyalist_lobby = yes
	every_interest_group = {
		remove_ruling_interest_group = yes
	}
	ig:ig_armed_forces ?= { add_ruling_interest_group = yes }
	s:STATE_NATAL = {
		set_owner_of_provinces = {
			country = root
			provinces = { x279045 }
		}
		region_state:NAL ?= {
			set_state_type = incorporated
			every_scope_pop = {
				limit = {
					culture = cu:british
				}
				change_pop_culture = {
					target = cu:anglo_african
					value = 1
				}
			}
			evaluate_and_assign_state_hub_dynamic_names = yes
		}
	}
	sb_install_british_natal_governor = yes
	set_global_variable = sb_british_natal_ever_formed_global_var
	s:STATE_NATAL = {
		every_scope_state = {
			if = {
				limit = { has_modifier = sb_port_natal_migration_deterrent }
				remove_modifier = sb_port_natal_migration_deterrent
			}
		}
	}
	sb_natal_schedule_shepstone_system = yes
}

sb_mark_natalia_boer_leadership_for_deferred_exile = {
	random_scope_character = {
		limit = {
			has_template = TRN_andries_pretorius
		}
		set_variable = sb_klip_river_deferred_exile_var
	}
	if = {
		limit = {
			NOT = {
				any_scope_character = {
					has_variable = sb_klip_river_deferred_exile_var
				}
			}
			exists = ruler
			ruler = { culture = cu:boer }
		}
		ruler = {
			set_variable = sb_klip_river_deferred_exile_var
		}
	}
	if = {
		limit = {
			NOT = {
				any_scope_character = {
					has_variable = sb_klip_river_deferred_exile_var
				}
			}
			any_scope_character = {
				culture = cu:boer
				is_character_alive = yes
			}
		}
		random_scope_character = {
			limit = {
				culture = cu:boer
				is_character_alive = yes
			}
			set_variable = sb_klip_river_deferred_exile_var
		}
	}
}

sb_exile_natalia_boer_leadership = {
	if = {
		limit = {
			NOT = { exists = scope:natal_refugee_receiver }
		}
		sb_select_natalia_refugee_receiver = yes
	}
	random_scope_character = {
		limit = {
			has_variable = sb_klip_river_deferred_exile_var
		}
		save_scope_as = natal_exiled_leader
	}
	if = {
		limit = {
			NOT = { exists = scope:natal_exiled_leader }
			any_scope_character = {
				has_template = TRN_andries_pretorius
			}
		}
		random_scope_character = {
			limit = {
				has_template = TRN_andries_pretorius
			}
			save_scope_as = natal_exiled_leader
		}
	}
	if = {
		limit = {
			NOT = { exists = scope:natal_exiled_leader }
			exists = ruler
			ruler = {
				culture = cu:boer
			}
		}
		ruler = {
			save_scope_as = natal_exiled_leader
		}
	}
	if = {
		limit = {
			exists = scope:natal_refugee_receiver
			scope:natal_exiled_leader ?= {
				OR = { NOT = { has_template = TRN_andries_pretorius } NOT = { c:TRN ?= { is_country_alive = yes } } }
			}
		}
		scope:natal_exiled_leader ?= {
			if = {
				limit = {
					NOT = { has_role_of_type = politician }
				}
				add_character_role = politician
			}
			transfer_character = scope:natal_refugee_receiver
			set_interest_group = scope:natal_refugee_receiver.ig:ig_landowners
			if = {
				limit = { has_variable = sb_klip_river_deferred_exile_var }
				remove_variable = sb_klip_river_deferred_exile_var
			}
		}
	}
}

sb_install_british_natal_governor = {
	random_scope_character = {
		limit = {
			has_variable = sb_martin_west_var
		}
		save_scope_as = natal_colonial_governor
	}
	if = {
		limit = {
			NOT = { exists = scope:natal_colonial_governor }
		}
		create_character = {
			template = NAL_martin_west
			on_created = {
				save_scope_as = natal_colonial_governor
			}
		}
	}
	c:NAL = {
		every_interest_group = {
			remove_ruling_interest_group = yes
		}
		ig:ig_armed_forces ?= { add_ruling_interest_group = yes }
	}
	scope:natal_colonial_governor ?= {
		set_interest_group = c:NAL.ig:ig_armed_forces
		if = {
			limit = {
				NOT = { has_role_of_type = politician }
			}
			add_character_role = politician
		}
		set_as_interest_group_leader = yes
		set_character_as_ruler = yes
	}
}

sb_finalize_british_natal_colony = {
	if = {
		limit = { has_variable = sb_natalia_ultimatum_truce_var }
		if = {
			limit = {
				c:GBR ?= {
					is_country_alive = yes
				}
				has_truce_with = c:GBR
			}
			end_truce = c:GBR
		}
		remove_variable = sb_natalia_ultimatum_truce_var
	}
	sb_mark_natalia_boer_leadership_for_deferred_exile = yes
	if = {
		limit = { NOT = { any_scope_character = { has_template = TRN_andries_pretorius has_variable = sb_klip_river_deferred_exile_var } } }
		sb_exile_natalia_boer_leadership = yes
	}
	sb_apply_british_natal_colony_setup = yes
	sb_annul_natalia_boer_treaties = yes
	if = {
		limit = { sb_klip_river_county_setup_valid = yes }
		sb_schedule_klip_river_county_question = yes
	}
	else = {
		sb_klip_river_prepare_standard_boer_flight = yes
	}
}

sb_found_natal_colony_after_british_annexation = {
	sb_create_british_natal_colony_if_missing = yes
	sb_assign_british_natal_colony_territory = yes
	c:NAL ?= {
		sb_seed_british_natal_colony_shell = yes
		sb_apply_british_natal_colony_setup = yes
		# This colony was carved out only after ZUL ceased to exist; it never
		# inherited the Boer Natalia boundary dispute.
		set_variable = sb_klip_river_direct_colony_excluded_var
	}
	c:GBR ?= { sb_british_transfer_archived_zulu_firearms_to_natal = yes sb_british_transfer_preannex_zulu_royal_house_to_natal = yes }
}

sb_trigger_selected_natal_refugee_receiver_followup = {
	if = {
		limit = {
			exists = scope:natal_refugee_receiver
			scope:natal_refugee_receiver ?= {
				country_definition = cd:ORA
			}
		}
		set_variable = { name = sb_klip_household_notice_host value = flag:ora }
		scope:natal_refugee_receiver = {
			trigger_event = { id = sb_natal_crisis.116 days = 1 popup = yes }
		}
	}
	else_if = {
		limit = {
			exists = scope:natal_refugee_receiver
			scope:natal_refugee_receiver ?= {
				country_definition = cd:TRN
			}
		}
		set_variable = { name = sb_klip_household_notice_host value = flag:trn }
		scope:natal_refugee_receiver = {
			if = {
				limit = { has_variable = sb_trn_return_notice_pending }
				set_variable = sb_trn_return_households_due
			}
			else = { trigger_event = { id = sb_natal_crisis.117 days = 1 popup = yes } }
		}
	}
	else_if = {
		limit = {
			exists = scope:natal_refugee_receiver
		}
		scope:natal_refugee_receiver ?= {
			sb_trigger_natal_refugee_migration_effect = yes
		}
	}
}

# Add a Boer ally directly to Natalia's side once Britain turns the ultimatum
# into a live diplomatic play.
sb_join_natalia_against_britain = {
	save_scope_as = natalia_supporter
	random_diplomatic_play = {
		limit = {
			is_diplomatic_play_type = dp_make_protectorate
			initiator = c:GBR
			target = c:NAL
			NOT = {
				any_scope_play_involved = {
					scope:natalia_supporter ?= THIS
					is_diplomatic_play_committed_participant = yes
				}
			}
		}
		add_target_backers = { scope:natalia_supporter }
		add_war_goal = {
			holder = c:GBR
			target_country = scope:natalia_supporter
			type = make_protectorate
		}
		if = {
			limit = {
				scope:natalia_supporter ?= {
					country_definition = cd:ORA
				}
			}
			sb_add_natalia_liberate_cape_goal_if_oranje_supports = yes
		}
	}
}

sb_add_natalia_liberate_cape_goal_if_oranje_supports = {
	if = {
		limit = {
			any_scope_play_involved = {
				c:ORA ?= THIS
				is_diplomatic_play_committed_participant = yes
				is_diplomatic_play_enemy_of = c:GBR
			}
			c:NAL ?= {
				NOT = { has_variable = sb_natalia_liberate_cape_goal_added_var }
			}
		}
		add_war_goal = {
			holder = c:NAL
			target_country = c:CAP
			type = liberate_subject
		}
		c:NAL ?= {
			set_variable = {
				name = sb_natalia_liberate_cape_goal_added_var
				months = 24
			}
		}
	}
}

sb_handle_natalia_player_support_join = {
	if = {
		limit = {
			is_diplomatic_play_type = dp_make_protectorate
			initiator = c:GBR
			target = c:NAL
			any_scope_play_involved = {
				is_player = yes
				is_diplomatic_play_committed_participant = yes
				is_diplomatic_play_enemy_of = c:GBR
				OR = {
					country_definition = cd:ORA
					country_definition = cd:TRN
					country_definition = cd:ZPB
					country_definition = cd:LYD
				}
			}
		}
		c:ORA ?= {
			if = { limit = { is_country_alive = yes is_ai = yes } sb_support_natalia_against_britain = yes }
		}
		c:TRN ?= {
			if = { limit = { is_country_alive = yes is_ai = yes } sb_support_natalia_against_britain = yes }
		}
		c:ZPB ?= {
			if = { limit = { is_country_alive = yes is_ai = yes } sb_support_natalia_against_britain = yes }
		}
		c:LYD ?= {
			if = { limit = { is_country_alive = yes is_ai = yes } sb_support_natalia_against_britain = yes }
		}
		sb_add_natalia_liberate_cape_goal_if_oranje_supports = yes
	}
}

# Support Natalia against Britain through the standard Boer compact helper; do
# not create separate short defensive-pact-only treaties here.
sb_support_natalia_against_britain = {
	if = {
		limit = {
			country_definition = cd:TRN
			c:NAL ?= { has_variable = sb_british_natal_ultimatum_var }
		}
		c:NAL ?= {
			set_variable = sb_natalia_trn_backed_ultimatum_var
		}
	}
	sb_ensure_boer_republic_compacts_for_root = yes
	sb_join_natalia_against_britain = yes
}

# The receiver's event always draws the standard departure from Natal.
sb_trigger_natal_refugee_migration_effect = {
	c:NAL ?= {
		if = { limit = { has_variable = sb_klip_household_notice_host } remove_variable = sb_klip_household_notice_host }
		sb_klip_river_distribute_boer_emigrants = yes
	}
}

sb_klip_river_capture_refugee_capital = {
	if = {
		limit = { is_country_alive = yes }
		capital ?= { save_temporary_scope_as = $DESTINATION$ }
	}
}

# Pop scope. Only whole, positive quotas have a native move.
sb_klip_river_move_refugee_quota = {
	if = {
		limit = {
			exists = scope:sb_natal_refugee_$TAG$_capital
			exists = scope:sb_natal_refugee_$TAG$_quota
		}
		if = {
			limit = { scope:sb_natal_refugee_$TAG$_quota > 0 }
			move_partial_pop = {
				state = scope:sb_natal_refugee_$TAG$_capital
				population = scope:sb_natal_refugee_$TAG$_quota
			}
		}
	}
}

# Country scope is the source of the emigrants. A held receiver event can have a foreign ROOT.
sb_klip_river_distribute_boer_emigrants = {
	if = {
		limit = {
			is_country_alive = yes
			NOT = { has_variable = sb_klip_river_boer_population_moved_var }
		}
		set_variable = sb_klip_river_boer_population_moved_var
		c:TRN ?= { sb_klip_river_capture_refugee_capital = { DESTINATION = sb_natal_refugee_trn_capital } }
		c:ORA ?= { sb_klip_river_capture_refugee_capital = { DESTINATION = sb_natal_refugee_ora_capital } }
		c:ZPB ?= { sb_klip_river_capture_refugee_capital = { DESTINATION = sb_natal_refugee_zpb_capital } }
		c:LYD ?= { sb_klip_river_capture_refugee_capital = { DESTINATION = sb_natal_refugee_lyd_capital } }
		save_temporary_scope_value_as = {
			name = sb_natal_refugee_non_ora_count
			value = {
				value = 0
				if = { limit = { exists = scope:sb_natal_refugee_trn_capital } add = 1 }
				if = { limit = { exists = scope:sb_natal_refugee_zpb_capital } add = 1 }
				if = { limit = { exists = scope:sb_natal_refugee_lyd_capital } add = 1 }
			}
		}
		# Tooltips can traverse the effect before the destinations and count are set.
		if = {
			limit = {
				exists = scope:sb_natal_refugee_non_ora_count
				OR = {
					exists = scope:sb_natal_refugee_trn_capital
					exists = scope:sb_natal_refugee_ora_capital
					exists = scope:sb_natal_refugee_zpb_capital
					exists = scope:sb_natal_refugee_lyd_capital
				}
			}
			every_scope_state = {
				every_scope_pop = {
					limit = { culture = cu:boer }
					save_temporary_scope_value_as = {
						name = sb_natal_refugee_departing
						value = { value = total_size multiply = 0.95 floor = yes }
					}
					save_temporary_scope_value_as = {
						name = sb_natal_refugee_ora_quota
						value = {
							value = 0
							if = {
								limit = { exists = scope:sb_natal_refugee_ora_capital }
								value = scope:sb_natal_refugee_departing
								if = { limit = { scope:sb_natal_refugee_non_ora_count > 0 } divide = 10 floor = yes }
							}
						}
					}
					save_temporary_scope_value_as = {
						name = sb_natal_refugee_shared_pool
						value = {
							value = scope:sb_natal_refugee_departing
							subtract = scope:sb_natal_refugee_ora_quota
							if = {
								limit = { exists = scope:sb_natal_refugee_trn_capital }
								subtract = { value = scope:sb_natal_refugee_departing divide = 2 floor = yes }
							}
						}
					}
					# Divide from LYD back to TRN. Flooring the last share first gives
					# spare people to TRN, then ZPB, then LYD.
					save_temporary_scope_value_as = {
						name = sb_natal_refugee_lyd_quota
						value = {
							value = 0
							if = {
								limit = { exists = scope:sb_natal_refugee_lyd_capital }
								value = scope:sb_natal_refugee_shared_pool
								divide = scope:sb_natal_refugee_non_ora_count
								floor = yes
							}
						}
					}
					save_temporary_scope_value_as = {
						name = sb_natal_refugee_zpb_quota
						value = {
							value = 0
							if = {
								limit = { exists = scope:sb_natal_refugee_zpb_capital }
								value = scope:sb_natal_refugee_shared_pool
								subtract = scope:sb_natal_refugee_lyd_quota
								if = { limit = { exists = scope:sb_natal_refugee_trn_capital } divide = 2 floor = yes }
							}
						}
					}
					# TRN receives its reserved half and its share of the pool.
					save_temporary_scope_value_as = {
						name = sb_natal_refugee_trn_quota
						value = {
							value = scope:sb_natal_refugee_departing
							subtract = scope:sb_natal_refugee_ora_quota
							subtract = scope:sb_natal_refugee_lyd_quota
							subtract = scope:sb_natal_refugee_zpb_quota
						}
					}
					sb_klip_river_move_refugee_quota = { TAG = trn }
					sb_klip_river_move_refugee_quota = { TAG = ora }
					sb_klip_river_move_refugee_quota = { TAG = zpb }
					sb_klip_river_move_refugee_quota = { TAG = lyd }
				}
			}
		}
