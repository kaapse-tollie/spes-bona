# Spes Bona - Natalia British Colony and Refugee Effects
#
# This file owns the later Natalia branch where the Boer republic is either
# forced into submission or reorganized by Britain as a crown colony. Keeping
# the colony/refugee work separate from `sb_natalia_effects.txt` leaves the
# Blood River and republican-foundation path much easier to navigate.

sb_assign_british_natal_colony_territory = {
	s:STATE_ZULULAND = {
		set_owner_of_provinces = {
			country = c:NAL
			provinces = {
				x279045
				xBE6FEE x1A084B xBFA16B x9E9742 x88FAD4 x904EBE x41C070 xE882CE xE1E455 xBBCA32
				x5B124F xFF0EF1 x552449 xE0EB02 x85695F xDE0EDE x7ACC38 xB1F868 x3CED3D x11A090 xCD31DB
			}
		}
		add_claim = c:NAL
	}
}

# Britain can respond to Dingane's coastal raid by annexing the Zulu-held Natal
# littoral outright and reorganizing it as a crown colony.
sb_place_natalia_under_british_protectorate = {
	change_relations = {
		country = c:GBR
		value = 50
	}
	if = {
		limit = {
			NOT = { is_subject_of = c:GBR }
		}
		c:GBR ?= {
			create_diplomatic_pact = {
				country = root
				type = protectorate
			}
		}
	}
	else_if = {
		limit = {
			is_subject_of = c:GBR
			NOT = { is_subject_type = subject_type_protectorate }
			NOT = { is_subject_type = subject_type_colony }
		}
		change_subject_type = subject_type_protectorate
	}
}

sb_convert_natalia_to_british_crown_colony = {
	if = {
		limit = {
			NOT = { has_variable = sb_natalia_british_colony_resolved_var }
		}
		set_variable = sb_natalia_british_colony_resolved_var
	}
	change_relations = {
		country = c:GBR
		value = 50
	}
	# Vanilla colony subject types only remain valid on countries with the
	# colonial country type. Without this, Natalia can briefly become a colony
	# and then immediately drop back out of subjecthood on the next tick.
	if = {
		limit = {
			NOT = { is_country_type = colonial }
		}
		set_country_type = colonial
	}
	if = {
		limit = {
			NOT = { is_subject_of = c:GBR }
		}
		c:GBR ?= {
			create_diplomatic_pact = {
				country = root
				type = colony
			}
		}
	}
	else_if = {
		limit = {
			is_subject_of = c:GBR
			NOT = { is_subject_type = subject_type_colony }
		}
		change_subject_type = subject_type_colony
	}
}

sb_apply_british_natal_colony_setup = {
	sb_convert_natalia_to_british_crown_colony = yes
	if = {
		limit = {
			has_variable = sb_commandant_general_visible_var
		}
		remove_variable = sb_commandant_general_visible_var
	}
	if = {
		limit = {
			has_variable = sb_early_republic_var
		}
		remove_variable = sb_early_republic_var
	}
	if = {
		limit = {
			NOT = { has_variable = southern_africa_subject_var }
		}
		set_variable = southern_africa_subject_var
	}
	activate_law = law_type:law_colonial_administration
	activate_law = law_type:law_professional_army
	activate_law = law_type:law_slavery_banned
	activate_law = law_type:law_racial_segregation
	if = {
		limit = {
			c:ZUL ?= { is_country_alive = yes }
			NOT = { has_truce_with = c:ZUL }
		}
		create_bidirectional_truce = {
			country = c:ZUL
			months = 120
		}
	}
	active_law:lawgroup_army_model ?= {
		if = {
			limit = {
				has_amendment = amendment_type:amendment_sb_commando_corps
			}
			every_scope_amendment = {
				limit = {
					type = amendment_type:amendment_sb_commando_corps
				}
				remove_amendment = yes
			}
		}
	}
	if = {
		limit = {
			NOT = { country_has_primary_culture = cu:anglo_african }
		}
		add_primary_culture = cu:anglo_african
	}
	if = {
		limit = {
			country_has_primary_culture = cu:british
		}
		remove_primary_culture = cu:british
	}
	if = {
		limit = {
			country_has_primary_culture = cu:boer
		}
		remove_primary_culture = cu:boer
	}
	sb_apply_generic_british_colony_interest_group_identity = yes
	every_interest_group = {
		remove_ruling_interest_group = yes
	}
	ig:ig_armed_forces ?= { add_ruling_interest_group = yes }
	s:STATE_ZULULAND = {
		set_owner_of_provinces = {
			country = root
			provinces = { x279045 }
		}
		region_state:NAL ?= {
			set_state_type = incorporated
			every_scope_pop = {
				limit = {
					culture = cu:british
				}
				change_pop_culture = {
					target = cu:anglo_african
					value = 1
				}
			}
		}
	}
	sb_install_british_natal_governor = yes
}

sb_select_natalia_refugee_receiver = {
	random_country = {
		limit = {
			is_country_alive = yes
			OR = {
				country_definition = cd:ORA
				country_definition = cd:TRN
				country_definition = cd:ZPB
				country_definition = cd:LYD
			}
		}
		save_scope_as = natal_refugee_receiver
	}
}

			}
		}
		every_scope_treaty = {
			limit = {
				binds = c:ORA
			}
			withdraw = {
				country = root
			}
		}
	}
	if = {
		limit = {
			any_scope_treaty = {
				binds = c:TRN
			}
		}
		every_scope_treaty = {
			limit = {
				binds = c:TRN
			}
			withdraw = {
				country = root
			}
		}
	}
	if = {
		limit = {
			exists = c:ZPB
			any_scope_treaty = {
				binds = c:ZPB
			}
		}
		every_scope_treaty = {
			limit = {
				binds = c:ZPB
			}
			withdraw = {
				country = root
			}
		}
	}
	if = {
		limit = {
			exists = c:LYD
			any_scope_treaty = {
				binds = c:LYD
			}
		}
		every_scope_treaty = {
			limit = {
				binds = c:LYD
			}
			withdraw = {
				country = root
			}
		}
	}
}

sb_mark_natalia_boer_leadership_for_deferred_exile = {
	random_scope_character = {
		limit = {
			has_template = TRN_andries_pretorius
		}
		set_variable = sb_klip_river_deferred_exile_var
	}
	if = {
		limit = {
			NOT = {
				any_scope_character = {
					has_variable = sb_klip_river_deferred_exile_var
				}
			}
			exists = ruler
			ruler = { culture = cu:boer }
		}
		ruler = {
			set_variable = sb_klip_river_deferred_exile_var
		}
	}
	if = {
		limit = {
			NOT = {
				any_scope_character = {
					has_variable = sb_klip_river_deferred_exile_var
				}
			}
			any_scope_character = {
				culture = cu:boer
				is_character_alive = yes
			}
		}
		random_scope_character = {
			limit = {
				culture = cu:boer
				is_character_alive = yes
			}
			set_variable = sb_klip_river_deferred_exile_var
		}
	}
}

sb_exile_natalia_boer_leadership = {
	if = {
		limit = {
			NOT = { exists = scope:natal_refugee_receiver }
		}
		sb_select_natalia_refugee_receiver = yes
	}
	random_scope_character = {
		limit = {
			has_variable = sb_klip_river_deferred_exile_var
		}
		save_scope_as = natal_exiled_leader
	}
	if = {
		limit = {
			NOT = { exists = scope:natal_exiled_leader }
			any_scope_character = {
				has_template = TRN_andries_pretorius
			}
		}
		random_scope_character = {
			limit = {
				has_template = TRN_andries_pretorius
			}
			save_scope_as = natal_exiled_leader
		}
	}
	if = {
		limit = {
			NOT = { exists = scope:natal_exiled_leader }
			exists = ruler
			ruler = {
				culture = cu:boer
			}
		}
		ruler = {
			save_scope_as = natal_exiled_leader
		}
	}
	scope:natal_exiled_leader ?= {
		if = {
			limit = {
				NOT = { has_role_of_type = politician }
			}
			add_character_role = politician
		}
		transfer_character = scope:natal_refugee_receiver
		set_interest_group = scope:natal_refugee_receiver.ig:ig_landowners
		if = {
			limit = { has_variable = sb_klip_river_deferred_exile_var }
			remove_variable = sb_klip_river_deferred_exile_var
		}
	}
}

sb_install_british_natal_governor = {
	random_scope_character = {
		limit = {
			has_variable = sb_martin_west_var
		}
		save_scope_as = natal_colonial_governor
	}
	if = {
		limit = {
			NOT = { exists = scope:natal_colonial_governor }
		}
		create_character = {
			template = NAL_martin_west
			on_created = {
				save_scope_as = natal_colonial_governor
			}
		}
	}
	c:NAL = {
		every_interest_group = {
			remove_ruling_interest_group = yes
		}
		ig:ig_armed_forces ?= { add_ruling_interest_group = yes }
	}
	scope:natal_colonial_governor ?= {
		set_interest_group = c:NAL.ig:ig_armed_forces
		if = {
			limit = {
				NOT = { has_role_of_type = politician }
			}
			add_character_role = politician
		}
		set_as_interest_group_leader = yes
		set_character_as_ruler = yes
	}
}

sb_ensure_british_natal_colony_governor = {
	if = {
		limit = {
			country_definition = cd:NAL
			has_variable = sb_natalia_british_colony_resolved_var
			OR = {
				NOT = { exists = ruler }
				ruler = {
					NOR = {
						culture = cu:british
						culture = cu:anglo_african
					}
				}
			}
		}
		sb_install_british_natal_governor = yes
	}
}

sb_finalize_british_natal_colony = {
	if = {
		limit = { has_variable = sb_natalia_ultimatum_truce_var }
		if = {
			limit = {
				c:GBR ?= {
					is_country_alive = yes
				}
				has_truce_with = c:GBR
			}
			end_truce = c:GBR
		}
		remove_variable = sb_natalia_ultimatum_truce_var
	}
	sb_mark_natalia_boer_leadership_for_deferred_exile = yes
	sb_apply_british_natal_colony_setup = yes
	sb_annul_natalia_boer_treaties = yes
	if = {
		limit = { sb_klip_river_county_setup_valid = yes }
		sb_schedule_klip_river_county_question = yes
	}
	else = {
		sb_klip_river_prepare_standard_boer_flight = yes
	}
}

sb_found_natal_colony_after_british_annexation = {
	sb_create_british_natal_colony_if_missing = yes
	sb_assign_british_natal_colony_territory = yes
	c:NAL ?= {
		sb_seed_british_natal_colony_shell = yes
		sb_apply_british_natal_colony_setup = yes
		# This colony was carved out only after ZUL ceased to exist; it never
		# inherited the Boer Natalia boundary dispute.
		set_variable = sb_klip_river_direct_colony_excluded_var
	}
}

sb_trigger_random_natal_refugee_receiver_event = {
	sb_select_natalia_refugee_receiver = yes
	sb_trigger_selected_natal_refugee_receiver_followup = yes
}

sb_trigger_selected_natal_refugee_receiver_followup = {
	if = {
		limit = {
			exists = scope:natal_refugee_receiver
			scope:natal_refugee_receiver ?= {
				country_definition = cd:ORA
			}
		}
		scope:natal_refugee_receiver = {
			trigger_event = { id = sb_natal_crisis.116 days = 1 popup = yes }
		}
	}
	else_if = {
		limit = {
			exists = scope:natal_refugee_receiver
			scope:natal_refugee_receiver ?= {
				country_definition = cd:TRN
			}
		}
		scope:natal_refugee_receiver = {
			trigger_event = { id = sb_natal_crisis.117 days = 1 popup = yes }
		}
	}
	else_if = {
		limit = {
			exists = scope:natal_refugee_receiver
		}
		scope:natal_refugee_receiver ?= {
			sb_trigger_natal_refugee_migration_effect = yes
		}
	}
}

# Add a Boer ally directly to Natalia's side once Britain turns the ultimatum
# into a live diplomatic play.
sb_join_natalia_against_britain = {
	save_scope_as = natalia_supporter
	random_diplomatic_play = {
		limit = {
			is_diplomatic_play_type = dp_make_protectorate
			initiator = c:GBR
			target = c:NAL
			NOT = {
				any_scope_play_involved = {
					scope:natalia_supporter ?= THIS
				}
			}
		}
		add_target_backers = { scope:natalia_supporter }
		add_war_goal = {
			holder = c:GBR
			target_country = scope:natalia_supporter
			type = make_protectorate
		}
		if = {
			limit = {
				scope:natalia_supporter ?= {
					country_definition = cd:ORA
				}
			}
			sb_add_natalia_liberate_cape_goal_if_oranje_supports = yes
		}
	}
}

sb_add_natalia_liberate_cape_goal_if_oranje_supports = {
	if = {
		limit = {
			any_scope_play_involved = {
				c:ORA ?= THIS
				is_diplomatic_play_enemy_of = c:GBR
			}
			c:NAL ?= {
				NOT = { has_variable = sb_natalia_liberate_cape_goal_added_var }
			}
		}
		add_war_goal = {
			holder = c:NAL
			target_country = c:CAP
			type = liberate_subject
		}
		c:NAL ?= {
			set_variable = {
				name = sb_natalia_liberate_cape_goal_added_var
				months = 24
			}
		}
	}
}

sb_handle_natalia_player_support_join = {
	if = {
		limit = {
			is_diplomatic_play_type = dp_make_protectorate
			any_scope_play_involved = { c:GBR ?= THIS }
			any_scope_play_involved = { c:NAL ?= THIS }
			any_scope_play_involved = {
				is_player = yes
				is_diplomatic_play_enemy_of = c:GBR
				OR = {
					country_definition = cd:ORA
					country_definition = cd:TRN
					country_definition = cd:ZPB
					country_definition = cd:LYD
				}
			}
		}
		c:ORA ?= {
			if = { limit = { is_country_alive = yes is_ai = yes } sb_support_natalia_against_britain = yes }
		}
		c:TRN ?= {
			if = { limit = { is_country_alive = yes is_ai = yes } sb_support_natalia_against_britain = yes }
		}
		c:ZPB ?= {
			if = { limit = { is_country_alive = yes is_ai = yes } sb_support_natalia_against_britain = yes }
		}
		c:LYD ?= {
			if = { limit = { is_country_alive = yes is_ai = yes } sb_support_natalia_against_britain = yes }
		}
		sb_add_natalia_liberate_cape_goal_if_oranje_supports = yes
	}
}

# Support Natalia against Britain through the standard Boer compact helper; do
# not create separate short defensive-pact-only treaties here.
sb_support_natalia_against_britain = {
	if = {
		limit = {
			country_definition = cd:TRN
			c:NAL ?= { has_variable = sb_british_natal_ultimatum_var }
		}
		c:NAL ?= {
			set_variable = sb_natalia_trn_backed_ultimatum_var
		}
	}
	sb_ensure_boer_republic_compacts_for_root = yes
	sb_join_natalia_against_britain = yes
}

# Boer refugees from Natal move inland toward whichever republic receives them.
sb_trigger_natal_refugee_migration_effect = {
	save_temporary_scope_as = natal_refugee_coordinator
	c:NAL ?= {
		save_scope_as = natalia_source
	}
	set_variable = {
		name = sb_natal_refugee_receiver_count_var
		value = 0
	}
	if = {
		limit = { c:ORA ?= { is_country_alive = yes } }
		c:ORA.capital ?= {
			save_scope_as = natal_refugee_receiver_a
		}
		change_variable = {
			name = sb_natal_refugee_receiver_count_var
			add = 1
		}
	}
	if = {
		limit = { c:TRN ?= { is_country_alive = yes } }
		c:TRN.capital ?= {
			if = {
				limit = { NOT = { exists = scope:natal_refugee_receiver_a } }
				save_scope_as = natal_refugee_receiver_a
			}
			else_if = {
				limit = { NOT = { exists = scope:natal_refugee_receiver_b } }
				save_scope_as = natal_refugee_receiver_b
			}
		}
		change_variable = {
			name = sb_natal_refugee_receiver_count_var
			add = 1
		}
	}
	if = {
		limit = { c:ZPB ?= { is_country_alive = yes } }
		c:ZPB.capital ?= {
			if = {
				limit = { NOT = { exists = scope:natal_refugee_receiver_a } }
				save_scope_as = natal_refugee_receiver_a
			}
			else_if = {
				limit = { NOT = { exists = scope:natal_refugee_receiver_b } }
				save_scope_as = natal_refugee_receiver_b
			}
			else_if = {
				limit = { NOT = { exists = scope:natal_refugee_receiver_c } }
				save_scope_as = natal_refugee_receiver_c
			}
		}
		change_variable = {
			name = sb_natal_refugee_receiver_count_var
			add = 1
		}
	}
	if = {
		limit = { c:LYD ?= { is_country_alive = yes } }
		c:LYD.capital ?= {
			if = {
				limit = { NOT = { exists = scope:natal_refugee_receiver_a } }
				save_scope_as = natal_refugee_receiver_a
			}
			else_if = {
				limit = { NOT = { exists = scope:natal_refugee_receiver_b } }
				save_scope_as = natal_refugee_receiver_b
			}
			else_if = {
				limit = { NOT = { exists = scope:natal_refugee_receiver_c } }
				save_scope_as = natal_refugee_receiver_c
			}
			else_if = {
				limit = { NOT = { exists = scope:natal_refugee_receiver_d } }
				save_scope_as = natal_refugee_receiver_d
			}
		}
		change_variable = {
			name = sb_natal_refugee_receiver_count_var
			add = 1
		}
	}
	if = {
		limit = {
			exists = scope:natalia_source
			scope:natal_refugee_coordinator.var:sb_natal_refugee_receiver_count_var > 0
		}
		c:NAL ?= {
			every_scope_state = {
				every_scope_pop = {
					limit = {
						culture = cu:boer
					}
					if = {
						limit = {
							scope:natal_refugee_coordinator.var:sb_natal_refugee_receiver_count_var == 1
						}
						move_partial_pop = {
							state = scope:natal_refugee_receiver_a
							population_ratio = 0.95
						}
					}
					else_if = {
						limit = {
							scope:natal_refugee_coordinator.var:sb_natal_refugee_receiver_count_var == 2
						}
						move_partial_pop = {
							state = scope:natal_refugee_receiver_a
							population_ratio = 0.475
						}
						move_partial_pop = {
							state = scope:natal_refugee_receiver_b
							population_ratio = 0.9048
						}
					}
					else_if = {
						limit = {
							scope:natal_refugee_coordinator.var:sb_natal_refugee_receiver_count_var == 3
						}
						move_partial_pop = {
							state = scope:natal_refugee_receiver_a
							population_ratio = 0.3167
						}
						move_partial_pop = {
							state = scope:natal_refugee_receiver_b
							population_ratio = 0.4634
						}
						move_partial_pop = {
							state = scope:natal_refugee_receiver_c
							population_ratio = 0.8636
						}
					}
					else_if = {
						limit = {
							scope:natal_refugee_coordinator.var:sb_natal_refugee_receiver_count_var >= 4
						}
						move_partial_pop = {
							state = scope:natal_refugee_receiver_a
							population_ratio = 0.2375
						}
						move_partial_pop = {
							state = scope:natal_refugee_receiver_b
							population_ratio = 0.3115
						}
						move_partial_pop = {
							state = scope:natal_refugee_receiver_c
							population_ratio = 0.4524
						}
						move_partial_pop = {
							state = scope:natal_refugee_receiver_d
							population_ratio = 0.8261
						}
					}
				}
			}
		}
	}
	remove_variable = sb_natal_refugee_receiver_count_var
}
