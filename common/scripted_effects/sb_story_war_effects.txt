# Shared 1.14 story-Humiliation recording and terminal resolution.
# The assent-required goal can only record a peace choice. The diplomatic play
# itself owns the one 0/1/2-side truth table and resolves after the final peace.

# Root = ZUL. The delayed guns-bargain reply carries this exact object as a
# saved event scope. Never use a timed country bit as its authentication token.
sb_natal_story_begin_guns_bargain_generation = {
	# A prior expired or invalid receipt must not block a new exact generation.
	if = { limit = { container_exists = sb_natal_guns_bargain_generation_state } sb_natal_story_recover_generation = yes }
	if = {
		limit = {
			country_definition = cd:ZUL
			is_country_alive = yes
			is_subject = no
			NOT = { container_exists = sb_natal_guns_bargain_generation_state }
			NOT = { has_variable = sb_natal_guns_bargain_generation_scope }
			c:ORA ?= { country_definition = cd:ORA is_country_alive = yes is_subject = no has_variable = sb_natal_guns_bargain_war_var }
		}
		ROOT = { save_scope_as = sb_natal_guns_bargain_generation_actor }
		c:ORA = { save_scope_as = sb_natal_guns_bargain_generation_counterparty }
		create_container = {
			name = sb_natal_guns_bargain_generation_state
			tags = { sb_story sb_natal_guns_bargain_generation }
			parent = root
			on_created = { set_variable = { name = sb_natal_guns_bargain_generation_pending_var days = 730 } }
		}
		container:sb_natal_guns_bargain_generation_state = {
			set_variable = { name = sb_natal_guns_bargain_generation_actor_scope value = scope:sb_natal_guns_bargain_generation_actor }
			set_variable = { name = sb_natal_guns_bargain_generation_counterparty_scope value = scope:sb_natal_guns_bargain_generation_counterparty }
			save_scope_as = sb_natal_guns_bargain_receipt_scope
		}
		set_variable = { name = sb_natal_guns_bargain_generation_scope value = scope:sb_natal_guns_bargain_receipt_scope }
		set_variable = { name = sb_natal_guns_bargain_delivery_queued_var days = 20 }
		c:ORA = {
			set_variable = { name = sb_natal_guns_bargain_generation_scope value = scope:sb_natal_guns_bargain_receipt_scope }
			trigger_event = { id = sb_natal_crisis.098 days = 31 }
		}
		# `.025.b` owns the single `.026` dispatch. These are only watchdogs.
		trigger_event = { id = sb_natal_crisis.098 days = 30 }
		trigger_event = { id = sb_natal_crisis.098 days = 125 }
		c:ORA ?= { if = { limit = { is_country_alive = yes } trigger_event = { id = sb_natal_crisis.098 days = 126 } } }
		trigger_event = { id = sb_natal_crisis.098 days = 731 }
		c:ORA ?= { if = { limit = { is_country_alive = yes } trigger_event = { id = sb_natal_crisis.098 days = 732 } } }
	}
	else_if = {
		limit = { NOT = { container_exists = sb_natal_guns_bargain_generation_state } }
		sb_natal_story_clear_guns_bargain_generation = yes
	}
}

sb_natal_story_clear_guns_bargain_receipt = {
	# The generation container remembers the exact pre-retag actors. Never use
	# current c:ZUL/c:ORA here: either tag may now name an unrelated replacement.
	if = {
		limit = { container_exists = sb_natal_guns_bargain_generation_state }
		container:sb_natal_guns_bargain_generation_state = {
			var:sb_natal_guns_bargain_generation_actor_scope ?= {
				if = { limit = { has_variable = sb_natal_guns_bargain_generation_scope } remove_variable = sb_natal_guns_bargain_generation_scope }
				if = { limit = { has_variable = sb_natal_guns_bargain_delivery_queued_var } remove_variable = sb_natal_guns_bargain_delivery_queued_var }
				if = { limit = { has_variable = sb_natal_story_launch_pending_var } remove_variable = sb_natal_story_launch_pending_var }
				if = { limit = { has_variable = sb_natal_story_launch_retry_queued_var } remove_variable = sb_natal_story_launch_retry_queued_var }
				if = { limit = { has_variable = sb_natal_story_launch_lease_var } remove_variable = sb_natal_story_launch_lease_var }
				if = { limit = { has_variable = sb_natal_story_absorption_commit_var } remove_variable = sb_natal_story_absorption_commit_var }
				if = { limit = { has_variable = sb_natal_guns_bargain_war_var } remove_variable = sb_natal_guns_bargain_war_var }
			}
			var:sb_natal_guns_bargain_generation_counterparty_scope ?= {
				if = { limit = { has_variable = sb_natal_guns_bargain_generation_scope } remove_variable = sb_natal_guns_bargain_generation_scope }
				if = { limit = { has_variable = sb_natal_story_launch_pending_var } remove_variable = sb_natal_story_launch_pending_var }
				if = { limit = { has_variable = sb_natal_guns_bargain_war_var } remove_variable = sb_natal_guns_bargain_war_var }
				if = { limit = { has_variable = sb_natal_diplomacy_started_var } remove_variable = sb_natal_diplomacy_started_var }
			}
			if = { limit = { has_variable = sb_natal_guns_bargain_popup_receipt_var } remove_variable = sb_natal_guns_bargain_popup_receipt_var }
			destroy_container = yes
		}
	}
}

sb_natal_story_clear_guns_bargain_generation = {
	# Receipt cleanup owns every exact actor link, including pre-launch leases.
	sb_natal_story_clear_guns_bargain_receipt = yes
}

# A held delivery that is only temporarily busy is reissued with the same
# unique receipt. Loss of a participant, ruler, route, or legal authority
# clears the receipt instead of reviving a stale choice.

			sb_natal_story_guns_bargain_delivery_can_retry = yes

		}
		set_variable = { name = sb_natal_guns_bargain_delivery_queued_var days = 45 }
		trigger_event = { id = sb_natal_crisis.026 days = 30 popup = yes }
		trigger_event = { id = sb_natal_crisis.098 days = 45 }
		c:ORA ?= { if = { limit = { is_country_alive = yes } trigger_event = { id = sb_natal_crisis.098 days = 46 } } }
		trigger_event = { id = sb_natal_crisis.098 days = 155 }
		c:ORA ?= { if = { limit = { is_country_alive = yes } trigger_event = { id = sb_natal_crisis.098 days = 156 } } }

		limit = { NOT = { sb_natal_story_guns_bargain_delivery_can_retry = yes } }

# Root may be either participant (or its retagged successor). A live admitted
# play is terminal-owned. Otherwise the bounded receipt either requeues the
# exact popup/launch or clears every permanent-invalid lock without reward.
sb_natal_story_recover_generation = {
	if = {
		limit = {
			NOT = {
				OR = {
					c:ZUL ?= { has_variable = sb_natal_story_play_scope }
					c:ORA ?= { has_variable = sb_natal_story_play_scope }
				}
			}
		}
		if = {
			limit = { container_exists = sb_natal_guns_bargain_generation_state }
			if = {
				limit = { sb_natal_story_guns_bargain_generation_is_valid = yes }
				if = {
					limit = { c:ZUL ?= { has_variable = sb_natal_story_launch_pending_var } }
					c:ZUL = { sb_natal_story_queue_launch_retry = yes }
				}
				else_if = {
					limit = {
						c:ZUL ?= { NOT = { has_variable = sb_natal_guns_bargain_delivery_queued_var } }
						container:sb_natal_guns_bargain_generation_state ?= { NOT = { has_variable = sb_natal_guns_bargain_popup_receipt_var } }
					}
					container:sb_natal_guns_bargain_generation_state = { save_scope_as = sb_natal_guns_bargain_receipt_scope }
					c:ZUL = { sb_natal_story_requeue_guns_bargain_delivery = yes }
				}
			}
			else = {
				sb_natal_story_clear_launch_reservation = yes
				sb_natal_story_clear_guns_bargain_generation = yes
			}
		}
		else_if = {
			limit = {
				OR = {
					c:ZUL ?= { has_variable = sb_natal_guns_bargain_generation_scope }
					c:ORA ?= { has_variable = sb_natal_guns_bargain_generation_scope }
				}
			}
			sb_natal_story_clear_launch_reservation = yes
			sb_natal_story_clear_guns_bargain_generation = yes
		}

		if = {
			limit = { container_exists = sb_natal_refusal_generation_state }
			if = {
				limit = { sb_natal_story_refusal_generation_is_valid = yes }
				if = {
					limit = { c:ZUL ?= { has_variable = sb_natal_story_launch_pending_var } }
					c:ZUL = { sb_natal_story_queue_launch_retry = yes }
				}
				else_if = {
					limit = {
						c:ORA ?= { NOT = { has_variable = sb_natal_refusal_delivery_queued_var } }
						container:sb_natal_refusal_generation_state ?= { NOT = { has_variable = sb_natal_refusal_popup_receipt_var } }
					}
					container:sb_natal_refusal_generation_state = { save_scope_as = sb_natal_refusal_receipt_scope }
					c:ORA = { sb_natal_story_requeue_refusal_delivery = yes }
				}
			}
			else = {
				sb_natal_story_clear_launch_reservation = yes
				sb_natal_story_clear_refusal_generation = yes
			}
		}
		else_if = {
			limit = {
				OR = {
					c:ORA ?= { has_variable = sb_natal_refusal_generation_scope }
					c:ZUL ?= { has_variable = sb_natal_refusal_generation_scope }
				}
			}
			sb_natal_story_clear_launch_reservation = yes
			sb_natal_story_clear_refusal_generation = yes
		}
	}
}

# Root = ZUL. Keep the chosen Blood River escalation alive across a transient
# diplomatic-play creation refusal without acknowledging an unbound play.
sb_natal_story_queue_launch_retry = {
	if = {
		limit = {
			country_definition = cd:ZUL
			is_country_alive = yes
			has_variable = sb_natal_story_launch_pending_var
			NOT = { has_variable = sb_natal_story_launch_retry_queued_var }
			c:ORA ?= { is_country_alive = yes has_variable = sb_natal_story_launch_pending_var }
			OR = {
				sb_natal_story_guns_bargain_generation_is_valid = yes
				sb_natal_story_refusal_generation_is_valid = yes
			}
		}
		set_variable = { name = sb_natal_story_launch_retry_queued_var days = 45 }
		trigger_event = { id = sb_natal_crisis.099 days = 30 }
		# The other participant survives a ZUL retag/death and clears the exact
		# generation on the next bounded watchdog.
		c:ORA = { trigger_event = { id = sb_natal_crisis.098 days = 46 } }
	}
	else_if = {
		limit = {
			NOT = {
				OR = {
					sb_natal_story_guns_bargain_generation_is_valid = yes
					sb_natal_story_refusal_generation_is_valid = yes
				}
			}
		}
		sb_natal_story_clear_launch_reservation = yes
		sb_natal_story_clear_guns_bargain_generation = yes
		sb_natal_story_clear_refusal_generation = yes
	}
}

# Root = ZUL. This is both the initial launcher and the hidden retry path.
sb_natal_story_attempt_exact_launch = {
	if = {
		limit = {
			country_definition = cd:ZUL
			is_country_alive = yes
			is_subject = no
			is_at_war = no
			is_active_in_diplomatic_play = no
			aggressive_diplomatic_plays_permitted = yes
			ruler = { is_character_alive = yes }
			has_variable = sb_natal_story_launch_pending_var
			NOT = { has_variable = sb_natal_story_play_scope }
			NOT = { has_variable = sb_natal_war_active_var }
			NOT = { has_global_variable = sb_natal_story_war_active_global_var }
			NOT = { has_truce_with = c:ORA }
			c:ORA ?= {
				is_country_alive = yes
				is_subject = no
				is_at_war = no
				is_active_in_diplomatic_play = no
				ruler = { is_character_alive = yes }
				has_variable = sb_natal_story_launch_pending_var
				NOT = { has_variable = sb_natal_story_play_scope }
			}
			OR = {
				sb_natal_story_guns_bargain_generation_is_valid = yes
				sb_natal_story_refusal_generation_is_valid = yes
			}
			NOT = {
				any_diplomatic_play = {
					is_diplomatic_play_type = dp_sb_natal_crisis
					initiator = root
					target = c:ORA
				}
			}
		}
		set_variable = sb_natal_story_launch_lease_var
		create_diplomatic_play = {
			type = dp_sb_natal_crisis
			target_country = c:ORA
			add_war_goal = {
				holder = c:ORA
				type = sb_story_humiliation
				target_country = root
			}
		}
	}
	# The admitted on-start path consumes the lease synchronously. A residual
	# lease means no exact root was acknowledged, so keep the mandate and retry.
	if = {
		limit = { has_variable = sb_natal_story_launch_lease_var }
		remove_variable = sb_natal_story_launch_lease_var
		sb_natal_story_queue_launch_retry = yes
	}
	else_if = {
		limit = {
			has_variable = sb_natal_story_launch_pending_var
			NOT = { has_variable = sb_natal_story_play_scope }
		}
		sb_natal_story_queue_launch_retry = yes
	}
}

			limit = { scope:sb_natal_terminal_receipt_scope.var:sb_natal_terminal_recipient_scope ?= { is_country_alive = yes } }

# Root = diplomatic play. Each route uses the same mutually exclusive truth table:
# both accepted -> mutual stand-down; initiator only; target only; neither -> status quo.
sb_story_humiliation_resolve = {
	if = {
		limit = {
			is_diplomatic_play_type = dp_sb_natal_crisis
			initiator ?= { has_variable = sb_natal_story_play_scope var:sb_natal_story_play_scope = root }
			target ?= { has_variable = sb_natal_story_play_scope var:sb_natal_story_play_scope = root }
			has_global_variable = sb_natal_story_war_active_global_var
			NOT = { has_global_variable = sb_natal_story_war_resolved_global_var }
		}
		set_global_variable = sb_natal_story_war_resolved_global_var
		# The saved play actors are the only recipient authority. This remains true
		# after a retag, while a dead recipient simply leaves an inert receipt.
		# 2: a negotiated mutual stand-down grants neither exclusive victory.
		if = {
			limit = {
				has_global_variable = sb_natal_story_initiator_demand_accepted_global_var
				has_global_variable = sb_natal_story_target_demand_accepted_global_var
			}
			initiator = { save_scope_as = sb_natal_terminal_recipient }
			target = { save_scope_as = sb_natal_terminal_counterparty }
			scope:sb_natal_terminal_recipient = { sb_natal_story_begin_terminal_zulu_neutral = yes }
			sb_story_humiliation_clear_natal_state = yes
		}
		# 1: the initiator demand means the saved Zulu-side actor won.
		else_if = {
			limit = { has_global_variable = sb_natal_story_initiator_demand_accepted_global_var }
			initiator = { save_scope_as = sb_natal_terminal_recipient }
			target = { save_scope_as = sb_natal_terminal_counterparty }
			scope:sb_natal_terminal_recipient = { sb_natal_story_begin_terminal_zulu_victory = yes }
			sb_story_humiliation_clear_natal_state = yes
		}
		# 1: the target demand means the saved Oranje-side actor won.
		else_if = {
			limit = { has_global_variable = sb_natal_story_target_demand_accepted_global_var }
			target = { save_scope_as = sb_natal_terminal_recipient }
			initiator = { save_scope_as = sb_natal_terminal_counterparty }
			scope:sb_natal_terminal_recipient = { sb_natal_story_begin_terminal_ora_victory = yes }
			sb_story_humiliation_clear_natal_state = yes
		}
		# 0: white peace uses the neutral follow-up, not Zulu triumph.
		else = {
			initiator = { save_scope_as = sb_natal_terminal_recipient }
			target = { save_scope_as = sb_natal_terminal_counterparty }
			scope:sb_natal_terminal_recipient = { sb_natal_story_begin_terminal_zulu_neutral = yes }
			sb_story_humiliation_clear_natal_state = yes
		}
	}
	else_if = {
		limit = {
			is_diplomatic_play_type = dp_sb_martinus_humiliation
			has_global_variable = sb_martinus_story_war_active_global_var
			NOT = { has_global_variable = sb_martinus_story_war_resolved_global_var }
			c:TRN ?= {
				has_variable = sb_martinus_story_play_scope
				var:sb_martinus_story_play_scope = root
			}
		}
		set_global_variable = sb_martinus_story_war_resolved_global_var
		# 2: both demands accepted; neither republic gets an exclusive result.
		if = {
			limit = {
				has_global_variable = sb_martinus_story_initiator_demand_accepted_global_var
				has_global_variable = sb_martinus_story_target_demand_accepted_global_var
			}
			sb_story_humiliation_martinus_stand_down = yes
		}
		# 1: initiator demand only. TRN alone can create the union.
		else_if = {
			limit = { has_global_variable = sb_martinus_story_initiator_demand_accepted_global_var }
			if = {
				limit = {
					initiator = c:TRN
					c:TRN ?= { is_country_alive = yes }
					c:ORA ?= { is_country_alive = yes is_subject = no }
				}
				c:TRN = { sb_martinus_begin_story_union_install = yes if = { limit = { NOT = { has_variable = sb_martinus_union_install_pending_var } NOT = { has_variable = sb_martinus_presidential_union_active_var } } sb_story_humiliation_clear_martinus_state = yes } }
			}
			else = { sb_story_humiliation_martinus_stand_down = yes }
		}
		# 1: target demand only. TRN alone can create the union.
		else_if = {
			limit = { has_global_variable = sb_martinus_story_target_demand_accepted_global_var }
			if = {
				limit = {
					target = c:TRN
					c:TRN ?= { is_country_alive = yes }
					c:ORA ?= { is_country_alive = yes is_subject = no }
				}
				c:TRN = { sb_martinus_begin_story_union_install = yes if = { limit = { NOT = { has_variable = sb_martinus_union_install_pending_var } NOT = { has_variable = sb_martinus_presidential_union_active_var } } sb_story_humiliation_clear_martinus_state = yes } }
			}
			else = { sb_story_humiliation_martinus_stand_down = yes }
		}
		# 0: white peace leaves both republics independent.
		else = { sb_story_humiliation_martinus_stand_down = yes }
		# Exact TRN success transfers authority to a durable generation-bound
		# install receipt. Every other terminal row cleans immediately.
		if = {
			limit = { NOT = { c:TRN ?= { sb_martinus_current_story_union_install_generation = yes } } }
			sb_story_humiliation_clear_martinus_state = yes
		}
	}
