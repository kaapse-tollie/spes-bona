# Startup treaty helpers for Southern Africa. We keep treaty history additive
# and normalize the relevant vanilla South African bundles at runtime so other
# mods can still extend 00_historical_treaties.txt safely.


# Post-trek republican recognition and Zoutpansberg crisis helpers.
sb_ensure_gbr_transit_rights_with_root = {
	save_scope_as = convention_target

	if = {
		limit = {
			NOT = {
				c:GBR ?= {
					any_scope_treaty = {
						binds = scope:convention_target
						any_scope_article = {
							has_type = transit_rights
							source_country = c:GBR
							target_country = scope:convention_target
						}
					}
				}
			}
		}
		c:GBR ?= {
			create_treaty = {
				first_country = c:GBR
				second_country = scope:convention_target
				is_draft = no
					binding_period = { years = 20 }
				articles_to_create = {
					{
						article = transit_rights
						source_country = c:GBR
						target_country = scope:convention_target
					}
					{
						article = goods_transfer
						source_country = c:GBR
						target_country = scope:convention_target
						inputs = {
							{ goods = g:merchant_marine }
							{ quantity = 10 }
						}
					}
					{
						article = host_power_bloc_embassy
						source_country = scope:convention_target
						target_country = c:GBR
					}
				}
			}
		}
	}
}

sb_ensure_convention_trade_through_with_root = {
	sb_ensure_gbr_transit_rights_with_root = yes
}



sb_withdraw_ai_frontier_goods_exports_to_boers = {
	c:ORA ?= {
		sb_withdraw_ai_ora_frontier_goods_exports_to_boers = yes
	}
	c:TRN ?= {
		sb_withdraw_ai_trn_frontier_goods_exports_to_boers = yes
	}
}

sb_withdraw_ai_ora_frontier_goods_exports_to_boers = {
	if = {
		limit = {
			is_country_alive = yes
			is_ai = yes
			is_active_in_diplomatic_play = yes
			OR = {
				has_journal_entry = je_sb_great_trek
				has_journal_entry = je_sb_bst_ora_frontier
			}
		}
		every_scope_treaty = {
			limit = {
				OR = {
					AND = {
						c:TRN ?= {
							is_country_alive = yes
							country_has_primary_culture = cu:boer
						}
						any_scope_article = {
							has_type = goods_transfer
							source_country = c:ORA
							target_country = c:TRN
						}
					}
					AND = {
						c:ZPB ?= {
							is_country_alive = yes
							country_has_primary_culture = cu:boer
						}
						any_scope_article = {
							has_type = goods_transfer
							source_country = c:ORA
							target_country = c:ZPB
						}
					}
					AND = {
						c:LYD ?= {
							is_country_alive = yes
							country_has_primary_culture = cu:boer
						}
						any_scope_article = {
							has_type = goods_transfer
							source_country = c:ORA
							target_country = c:LYD
						}
					}
					AND = {
						c:NAL ?= {
							is_country_alive = yes
							country_has_primary_culture = cu:boer
						}
						any_scope_article = {
							has_type = goods_transfer
							source_country = c:ORA
							target_country = c:NAL
						}
					}
				}
			}
			withdraw = { country = c:ORA }
		}
	}
}

sb_ensure_boer_republic_compacts_for_root = {
	if = {
		limit = {
			is_country_alive = yes
			country_has_primary_culture = cu:boer
			NOT = { country_has_primary_culture = cu:anglo_african }
		}
		sb_ensure_boer_republic_compact_with_ora = yes
		sb_ensure_boer_republic_compact_with_trn = yes
		sb_ensure_boer_republic_compact_with_zpb = yes
		sb_ensure_boer_republic_compact_with_lyd = yes
		sb_ensure_boer_republic_compact_with_nal = yes
		# Some compact pairs fail validation when transit-rights checks are not
		# available during setup. Fall back to direct creation so the Boer compact
		# network is always present.
		sb_ensure_boer_republic_defensive_compact_fallback_with_ora = yes
		sb_ensure_boer_republic_defensive_compact_fallback_with_trn = yes
		sb_ensure_boer_republic_defensive_compact_fallback_with_zpb = yes
		sb_ensure_boer_republic_defensive_compact_fallback_with_lyd = yes
		sb_ensure_boer_republic_defensive_compact_fallback_with_nal = yes
	}
}

# Runtime compact helpers are create-only. Do not withdraw live treaties here:
# explicit crisis paths handle real breaks, while generic ensure calls must not
# create visible annulment penalties during republic formation.
sb_ensure_boer_republic_compact_with_ora = {
	save_scope_as = boer_compact_source

	if = {
		limit = {
			is_country_alive = yes
			country_has_primary_culture = cu:boer
			NOT = { country_has_primary_culture = cu:anglo_african }
			NOT = { country_definition = cd:ORA }
			c:ORA ?= {
				is_country_alive = yes
				country_has_primary_culture = cu:boer
			}
			NOT = {
				any_scope_treaty = {
					binds = c:ORA
					any_scope_article = { has_type = defensive_pact }
				}
			}
			can_create_treaty = {
				first_country = scope:boer_compact_source
				second_country = c:ORA
				is_draft = no
				binding_period = { years = 25 }
				articles_to_create = {
					{ article = defensive_pact }
					{
						article = transit_rights
						source_country = scope:boer_compact_source
						target_country = c:ORA
					}
					{
						article = transit_rights
						source_country = c:ORA
						target_country = scope:boer_compact_source
					}
					{
						article = trade_privilege
						source_country = scope:boer_compact_source
						target_country = c:ORA
					}
					{
						article = trade_privilege
						source_country = c:ORA
						target_country = scope:boer_compact_source
					}
				}
			}
		}
		create_treaty = {
			first_country = scope:boer_compact_source
			second_country = c:ORA
			is_draft = no
			binding_period = { years = 25 }
			articles_to_create = {
				{ article = defensive_pact }
				{
					article = transit_rights
					source_country = scope:boer_compact_source
					target_country = c:ORA
				}
				{
					article = transit_rights
					source_country = c:ORA
					target_country = scope:boer_compact_source
				}
				{
					article = trade_privilege
					source_country = scope:boer_compact_source
					target_country = c:ORA
				}
				{
					article = trade_privilege
					source_country = c:ORA
					target_country = scope:boer_compact_source
				}
			}
		}
	}
}

# The generation receipt owns one held .150 delivery. It prevents an old popup
# from acting after a retry has issued a newer delivery (ABA protection).
sb_zpb_crackdown_clear_generation_receipt = {
	# The receipt may be retired after either actor retags. These are volatile
	# receipt markers, so sweep every live country rather than reselecting tags.
	every_country = {
		if = { limit = { has_variable = sb_zpb_crackdown_generation_scope } remove_variable = sb_zpb_crackdown_generation_scope }
		if = { limit = { has_variable = sb_zpb_crackdown_launch_lease_var } remove_variable = sb_zpb_crackdown_launch_lease_var }
		if = { limit = { has_variable = sb_zpb_crackdown_launch_ack_var } remove_variable = sb_zpb_crackdown_launch_ack_var }
	}
	# Frozen markers are state-local and ownership may have changed before an
	# ambiguous terminal callback. Sweep every live owner before retiring receipt.
	every_country = {
		every_scope_state = {
			if = { limit = { has_variable = sb_zpb_crackdown_frozen_target_state_var } remove_variable = sb_zpb_crackdown_frozen_target_state_var }
			if = { limit = { has_variable = sb_zpb_crackdown_frozen_contested_state_var } remove_variable = sb_zpb_crackdown_frozen_contested_state_var }
		}
	}
	if = { limit = { container_exists = sb_zpb_crackdown_generation_state } container:sb_zpb_crackdown_generation_state = { destroy_container = yes } }
}

# Route a current receipt to a held popup when launchable, or to one short
# retry lease when merely busy. Permanent loss clears only this route receipt.
sb_zpb_crackdown_prepare_held_delivery = {
	if = {
		limit = { NOT = { sb_trn_zpb_crackdown_route_live_authority = yes } }
		sb_zpb_crackdown_retire_permanently_invalid_delivery = yes
	}
	else = {
		if = {
			limit = {
				NOT = { has_variable = sb_zpb_crackdown_generation_scope }
				NOT = { container_exists = sb_zpb_crackdown_generation_state }
			}
			create_container = {
				name = sb_zpb_crackdown_generation_state
				tags = { sb_story sb_zpb_crackdown_generation }
				parent = root
				on_created = {
					set_variable = sb_zpb_crackdown_generation_active_var
					set_variable = sb_zpb_crackdown_delivery_pending_var
				}
			}
		}
		if = {
			limit = { container_exists = sb_zpb_crackdown_generation_state }
			container:sb_zpb_crackdown_generation_state = { save_temporary_scope_as = sb_zpb_crackdown_receipt }
			set_variable = { name = sb_zpb_crackdown_generation_scope value = scope:sb_zpb_crackdown_receipt }
		}
		if = {
			limit = { sb_trn_zpb_crackdown_current_receipt_authority = yes }
			var:sb_zpb_crackdown_generation_scope = {
				# Arrival consumes the short scheduling lease; an already-held popup
				# remains authoritative until its separate four-month authority expires.
				if = { limit = { has_variable = sb_zpb_crackdown_dispatch_lease_var } remove_variable = sb_zpb_crackdown_dispatch_lease_var }
			}
			if = {
				limit = {
					var:sb_zpb_crackdown_generation_scope ?= {
						NOT = { has_variable = sb_zpb_crackdown_popup_authority_var }
						NOT = { has_variable = sb_zpb_crackdown_dispatch_lease_var }
					}
					sb_trn_zpb_crackdown_preflight = yes
				}
				var:sb_zpb_crackdown_generation_scope = { set_variable = { name = sb_zpb_crackdown_popup_authority_var months = 4 } }
			}
			else_if = {
				limit = {
					var:sb_zpb_crackdown_generation_scope ?= {
						NOT = { has_variable = sb_zpb_crackdown_popup_authority_var }
						NOT = { has_variable = sb_zpb_crackdown_dispatch_lease_var }
					}
				}
				var:sb_zpb_crackdown_generation_scope = { set_variable = { name = sb_zpb_crackdown_dispatch_lease_var days = 30 } }
				trigger_event = { id = sb_boer_conventions.150 days = 30 popup = yes }
			}
		}
	}
}

sb_zpb_crackdown_handle_invalid_held_delivery = {
	if = {
		limit = { sb_trn_zpb_crackdown_route_live_authority = yes }
		# Retire the old receipt before retrying. A stale popup can therefore never
		# regain authority when a later delivery opens (ABA-safe rotation).
		sb_zpb_crackdown_clear_generation_receipt = yes
		sb_zpb_crackdown_prepare_held_delivery = yes
	}
	else = { sb_zpb_crackdown_retire_permanently_invalid_delivery = yes }
}

		scope:initiator = {
			set_variable = { name = sb_zpb_crackdown_play_scope value = root }
			set_variable = { name = sb_zpb_crackdown_target_scope value = scope:target }
			set_variable = { name = sb_zpb_crackdown_contested_state_scope value = scope:sb_zpb_crackdown_contested_state }

			var:sb_zpb_crackdown_generation_scope = {

			scope:target = { sb_apply_zpb_player_trn_civil_war_support = yes }

		# the reciprocal goals, so joins/transfers cannot expand this attempt.
		c:TRN = {
			every_scope_state = {
				limit = { has_variable = sb_zpb_crackdown_frozen_target_state_var }

# Both terminal callbacks call this exact root resolver. Terminal table:
# 0 = no exact receipt/root (no mutation); 1 = exact root without final proof
# (cleanup only); 2 = exact root with frozen physical proof (succession, cleanup).
# A live TRN subject is explicitly not success.
sb_zpb_crackdown_resolve_bound_terminal = {
	# on_diplo_play_back_down exposes only the DP root and scope:actor. Use the
	# DP's documented initiator/target links, never callback-only aliases.
	if = {
		limit = {
			is_diplomatic_play_type = dp_sb_trn_zpb_crackdown_locked
			initiator = { country_definition = cd:TRN }
			target = { country_definition = cd:ZPB }
		}
		initiator = { save_temporary_scope_as = sb_zpb_crackdown_terminal_initiator }
		target = { save_temporary_scope_as = sb_zpb_crackdown_terminal_target }
		if = {
			limit = {
				scope:sb_zpb_crackdown_terminal_initiator ?= {
					has_variable = sb_zpb_crackdown_play_scope
					var:sb_zpb_crackdown_play_scope = root
					has_variable = sb_zpb_crackdown_generation_scope
				}
				scope:sb_zpb_crackdown_terminal_target ?= {
					has_variable = sb_zpb_crackdown_play_scope
					var:sb_zpb_crackdown_play_scope = root
					has_variable = sb_zpb_crackdown_trn_scope
					var:sb_zpb_crackdown_trn_scope = scope:sb_zpb_crackdown_terminal_initiator
					has_variable = sb_zpb_crackdown_generation_scope
					var:sb_zpb_crackdown_generation_scope = scope:sb_zpb_crackdown_terminal_initiator.var:sb_zpb_crackdown_generation_scope
				}
				scope:sb_zpb_crackdown_terminal_initiator.var:sb_zpb_crackdown_generation_scope ?= {
					has_variable = sb_zpb_crackdown_generation_active_var
					has_variable = sb_zpb_crackdown_play_scope
					var:sb_zpb_crackdown_play_scope = root
					has_variable = sb_zpb_crackdown_contested_state_scope
				}
			}
			scope:sb_zpb_crackdown_terminal_initiator.var:sb_zpb_crackdown_generation_scope = { save_temporary_scope_as = sb_zpb_crackdown_terminal_receipt }
			if = {
				limit = {
					scope:sb_zpb_crackdown_terminal_receipt ?= {
						has_variable = sb_zpb_crackdown_contested_state_scope
						var:sb_zpb_crackdown_contested_state_scope ?= { has_variable = sb_zpb_crackdown_frozen_contested_state_var owner = scope:sb_zpb_crackdown_terminal_target }
					}
					NOT = { scope:sb_zpb_crackdown_terminal_target.var:sb_zpb_crackdown_trn_scope ?= { is_country_alive = yes } }
				}
				scope:sb_zpb_crackdown_terminal_target = { sb_zpb_assume_transvaal_after_crackdown_victory = yes }
			}
			scope:sb_zpb_crackdown_terminal_target = { sb_zpb_crackdown_clear_bound_target_runtime = yes }
			scope:sb_zpb_crackdown_terminal_initiator = { sb_zpb_crackdown_clear_bound_initiator_runtime = yes }
			# Exact terminal close clears every route-local active/appeal/backer/receipt
			# marker together, without touching external treaty or story state.
			sb_zpb_crackdown_clear_route_runtime = yes
		}
	}
}
# The only create path. It freezes the full TRN footprint before create, then
# uses the actual-root callback as its acknowledgement. Failure rolls back only
# the frozen attempt and requeues the same generation.
sb_open_trn_zpb_crackdown_play = {
	if = {
		limit = { sb_trn_zpb_crackdown_popup_authority = yes }
		var:sb_zpb_crackdown_generation_scope = { remove_variable = sb_zpb_crackdown_popup_authority_var set_variable = sb_zpb_crackdown_receipt_launch_lease_var }
		s:STATE_NORTHERN_TRANSVAAL.region_state:ZPB = { save_temporary_scope_as = sb_zpb_crackdown_launch_contested_state set_variable = sb_zpb_crackdown_frozen_contested_state_var }
		var:sb_zpb_crackdown_generation_scope = { set_variable = { name = sb_zpb_crackdown_contested_state_scope value = scope:sb_zpb_crackdown_launch_contested_state } }
		# This marker is only the pre-create TRN goal footprint, never the ZPB
		# contested-state evidence kept separately on the receipt above.
		every_scope_state = { set_variable = sb_zpb_crackdown_frozen_target_state_var }
		set_variable = sb_zpb_crackdown_launch_lease_var
		create_diplomatic_play = { type = dp_sb_trn_zpb_crackdown_locked target_country = c:ZPB target_state = s:STATE_NORTHERN_TRANSVAAL.region_state:ZPB }
		if = {
			limit = { has_variable = sb_zpb_crackdown_launch_ack_var }
			remove_variable = sb_zpb_crackdown_launch_ack_var
			var:sb_zpb_crackdown_generation_scope = { if = { limit = { has_variable = sb_zpb_crackdown_delivery_pending_var } remove_variable = sb_zpb_crackdown_delivery_pending_var } }
		}
		else = {
			if = { limit = { has_variable = sb_zpb_crackdown_launch_lease_var } remove_variable = sb_zpb_crackdown_launch_lease_var }
			var:sb_zpb_crackdown_generation_scope = { if = { limit = { has_variable = sb_zpb_crackdown_receipt_launch_lease_var } remove_variable = sb_zpb_crackdown_receipt_launch_lease_var } }
			every_scope_state = { limit = { has_variable = sb_zpb_crackdown_frozen_target_state_var } remove_variable = sb_zpb_crackdown_frozen_target_state_var }
			s:STATE_NORTHERN_TRANSVAAL.region_state:ZPB ?= { if = { limit = { has_variable = sb_zpb_crackdown_frozen_contested_state_var } remove_variable = sb_zpb_crackdown_frozen_contested_state_var } }
			var:sb_zpb_crackdown_generation_scope = { if = { limit = { has_variable = sb_zpb_crackdown_contested_state_scope } remove_variable = sb_zpb_crackdown_contested_state_scope } }
			sb_zpb_crackdown_prepare_held_delivery = yes
		}
	}
}
