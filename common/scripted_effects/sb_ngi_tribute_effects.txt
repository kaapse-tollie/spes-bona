# Spes Bona - Nguni Chiefdoms (NGI) tribute to the Zulu kingdom
# The southern-Natal chiefdom network paid tribute up the chain: smaller refugee groups to
# the Nhlangwini, and the Nhlangwini to Dingane (Wright, in Laband & Haswell 1988, 18-21).
# The transfer approximates a small share of the tag's income, scaled by population.
# Cross-ref: common/on_actions/sb_ngi_tribute_on_action_handlers.txt (pulse definition).

sb_ngi_tribute_payment = {
	value = 0
	every_scope_state = {
		add = state_population
	}
	multiply = 0.004
	min = 50
	max = 1500
}

sb_ngi_tribute_payment_neg = {
	value = sb_ngi_tribute_payment
	multiply = -1
}
