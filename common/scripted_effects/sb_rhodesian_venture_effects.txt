sb_de_beers_schedule_or_assign_cecil_rhodes = {
	# The establishment callback runs before the new company is safe for executive replacement.
	if = {
		limit = { NOT = { has_global_variable = sb_cecil_rhodes_entered_politics_global_var } }
		set_variable = sb_de_beers_rhodes_appointment_pending_var
	}
}

sb_de_beers_complete_cecil_rhodes_appointment = {
	if = {
		limit = { has_global_variable = sb_cecil_rhodes_entered_politics_global_var }
		remove_variable = sb_de_beers_rhodes_appointment_pending_var
	}
	else_if = {
		limit = { has_company = company_type:company_de_beers }
		company:company_de_beers = { save_scope_as = sb_de_beers_company_scope }
		sb_de_beers_assign_cecil_rhodes_to_saved_company = yes
	}
	else = {
		remove_variable = sb_de_beers_rhodes_appointment_pending_var
	}
}

sb_de_beers_make_cecil_rhodes_executive = {
	set_is_noble = no
	if = {
		limit = { has_role = character_role_ig_leader }
		remove_character_role = character_role_ig_leader
	}
	if = {
		limit = { has_role = character_role_politician }
		remove_character_role = character_role_politician
	}
	if = {
		limit = { NOT = { has_role = character_role_executive } }
		add_character_role = character_role_executive
	}
	set_character_as_executive = scope:sb_de_beers_company_scope
}

sb_de_beers_assign_cecil_rhodes_to_saved_company = {

	if = {
		limit = {
			is_template_used = SAF_cecil_rhodes
			NOT = { has_global_variable = sb_cecil_rhodes_ever_created_global_var }
		}
		set_global_variable = sb_cecil_rhodes_ever_created_global_var
	}

	random_character = {
		limit = {
			has_template = SAF_cecil_rhodes
			is_character_alive = yes
		}
		save_scope_as = sb_de_beers_rhodes_scope
	}
	if = {
		limit = { exists = scope:sb_de_beers_rhodes_scope }
		every_country = {
			limit = {
				has_company = company_type:company_de_beers
				company:company_de_beers = { executive = scope:sb_de_beers_rhodes_scope }
			}
			save_scope_as = sb_existing_rhodes_de_beers_owner_scope
		}
	}

	if = {
		limit = {
			exists = scope:sb_de_beers_rhodes_scope
			NOT = { exists = scope:sb_existing_rhodes_de_beers_owner_scope }
			NOT = {
				c:GBR ?= {
					company:company_sb_british_south_africa_company ?= {
						executive = scope:sb_de_beers_rhodes_scope
					}
				}
			}
		}
		scope:sb_de_beers_rhodes_scope = {
			transfer_character = root
			sb_de_beers_make_cecil_rhodes_executive = yes
		}
	}
	else_if = {
		limit = {
			NOT = { is_template_used = SAF_cecil_rhodes }
			NOT = { has_global_variable = sb_cecil_rhodes_ever_created_global_var }
		}
		set_global_variable = sb_cecil_rhodes_ever_created_global_var
		create_character = {
			template = SAF_cecil_rhodes
			role = character_role_executive
			on_created = {
				set_variable = sb_de_beers_new_rhodes_executive_var
			}
		}
		random_character = {
			limit = { has_variable = sb_de_beers_new_rhodes_executive_var }
			remove_variable = sb_de_beers_new_rhodes_executive_var
			save_scope_as = sb_de_beers_rhodes_scope
			sb_de_beers_make_cecil_rhodes_executive = yes
		}
	}

	if = {
		limit = {
			exists = scope:sb_de_beers_rhodes_scope
			scope:sb_de_beers_company_scope = {
				executive = scope:sb_de_beers_rhodes_scope
			}
		}
		remove_variable = sb_de_beers_rhodes_appointment_pending_var
	}
	else_if = {
		limit = {
			exists = scope:sb_de_beers_rhodes_scope
			OR = {
				exists = scope:sb_existing_rhodes_de_beers_owner_scope
				c:GBR ?= {
					company:company_sb_british_south_africa_company ?= {
						executive = scope:sb_de_beers_rhodes_scope
					}
				}
			}
		}
		remove_variable = sb_de_beers_rhodes_appointment_pending_var
	}
	else_if = {
		limit = {
			NOT = { exists = scope:sb_de_beers_rhodes_scope }
			has_global_variable = sb_cecil_rhodes_ever_created_global_var
		}
		remove_variable = sb_de_beers_rhodes_appointment_pending_var
	}
}

sb_de_beers_transition_cecil_rhodes_to_politics = {
	create_character = {
		role = character_role_executive
		interest_group = ig_industrialists
		on_created = {
			set_variable = sb_de_beers_political_replacement_executive_var
		}
	}
	random_character = {
		limit = { has_variable = sb_de_beers_political_replacement_executive_var }
		remove_variable = sb_de_beers_political_replacement_executive_var
		set_character_as_executive = scope:sb_de_beers_company_scope
	}
	scope:sb_de_beers_rhodes_scope = {
		if = {
			limit = { has_role = character_role_executive }
			remove_character_role = character_role_executive
		}
		if = {
			limit = { NOT = { has_role = character_role_politician } }
			add_character_role = character_role_politician
		}
		if = {
			limit = { NOT = { has_trait = experienced_political_operator } }
			add_trait = experienced_political_operator
		}
		if = {
			limit = { has_trait = basic_political_operator }
			remove_trait = basic_political_operator
		}
		if = {
			limit = { NOT = { has_trait = expert_entrepreneur } }
			add_trait = expert_entrepreneur
		}
		if = {
			limit = { has_trait = experienced_entrepreneur }
			remove_trait = experienced_entrepreneur
		}
		set_as_interest_group_leader = yes
		add_modifier = {
			name = sb_rhodes_diamond_magnate
			months = 120
		}
	}
	set_global_variable = sb_cecil_rhodes_entered_politics_global_var
	remove_variable = sb_de_beers_rhodes_appointment_pending_var
	remove_variable = sb_de_beers_rhodes_political_transition_pending_var
}

sb_rhodesian_venture_install_bsac_executive = {
	if = {
		limit = {
			is_template_used = SAF_cecil_rhodes
			NOT = { has_global_variable = sb_cecil_rhodes_ever_created_global_var }
		}
		set_global_variable = sb_cecil_rhodes_ever_created_global_var
	}
	random_character = {
		limit = {
			has_template = SAF_cecil_rhodes
			is_character_alive = yes
		}
		save_scope_as = sb_bsac_rhodes_scope
	}

	if = {
		limit = { exists = scope:sb_bsac_rhodes_scope }
		every_country = {
			limit = {
				has_company = company_type:company_de_beers
				company:company_de_beers = { executive = scope:sb_bsac_rhodes_scope }
			}
			save_scope_as = sb_rhodes_de_beers_owner_scope
			company:company_de_beers = { save_scope_as = sb_rhodes_de_beers_company_scope }
		}
		if = {
			limit = {
				exists = scope:sb_rhodes_de_beers_owner_scope
				exists = scope:sb_rhodes_de_beers_company_scope
			}
			scope:sb_rhodes_de_beers_owner_scope = {
				create_character = {
					role = character_role_executive
					interest_group = ig_industrialists
					on_created = {
						set_variable = sb_de_beers_replacement_executive_var
					}
				}
				random_character = {
					limit = { has_variable = sb_de_beers_replacement_executive_var }
					remove_variable = sb_de_beers_replacement_executive_var
					set_character_as_executive = scope:sb_rhodes_de_beers_company_scope
				}
			}
		}
		# Detach Rhodes from his old company before moving him into Britain's cast.
		scope:sb_bsac_rhodes_scope = {
			if = {
				limit = { has_role = character_role_executive }
				remove_character_role = character_role_executive
			}
			transfer_character = c:GBR
			sb_rhodesian_venture_prepare_rhodes_for_bsac = yes
			set_character_as_executive = scope:sb_bsac_company_scope
			sb_rhodesian_venture_set_rhodes_colonialist_if_installed = yes
			set_is_noble = no
		}
	}
	else_if = {
		limit = {
			NOT = { is_template_used = SAF_cecil_rhodes }
			NOT = { has_global_variable = sb_cecil_rhodes_ever_created_global_var }
		}
		set_global_variable = sb_cecil_rhodes_ever_created_global_var
		create_character = {
			template = SAF_cecil_rhodes
		}
		random_character = {
			limit = {
				has_template = SAF_cecil_rhodes
				is_character_alive = yes
			}
			sb_rhodesian_venture_prepare_rhodes_for_bsac = yes
			set_character_as_executive = scope:sb_bsac_company_scope
			sb_rhodesian_venture_set_rhodes_colonialist_if_installed = yes
			set_is_noble = no
		}
	}
	else = {
		create_character = {
			role = character_role_executive
			interest_group = ig_industrialists
			on_created = {
				set_variable = sb_bsac_generated_executive_var
			}
		}
		random_character = {
			limit = { has_variable = sb_bsac_generated_executive_var }
			remove_variable = sb_bsac_generated_executive_var
			set_character_as_executive = scope:sb_bsac_company_scope
		}
	}
}

									is_pop_type = bureaucrats

			capital = {

					building = building_government_administration

sb_update_british_south_africa_company_slot = {
	if = {
		limit = {
			sb_british_south_africa_company_is_linked_to_bsa = yes
			NOT = { has_modifier = sb_british_south_africa_company_charter_slot }
		}
		add_modifier = {
			name = sb_british_south_africa_company_charter_slot
			months = -1
		}
	}
	else_if = {
		limit = {
			has_modifier = sb_british_south_africa_company_charter_slot
			NOT = { sb_british_south_africa_company_is_linked_to_bsa = yes }
		}
		remove_modifier = sb_british_south_africa_company_charter_slot
	}
}

sb_disband_british_south_africa_company_country = {
	remove_variable = sb_bsac_starting_asset_created_var
	if = {
		limit = { has_modifier = sb_british_south_africa_company_charter_slot }
		remove_modifier = sb_british_south_africa_company_charter_slot
	}
	if = {
		limit = {
			exists = c:BSA
			c:BSA = {
				is_country_alive = yes
				is_subject_of = root
				is_subject_type = subject_type_chartered_company
			}
		}
		c:BSA = {
			if = {
				limit = {
					exists = owning_company
					owning_company = {
						is_company_type = company_type:company_sb_british_south_africa_company
						owner = ROOT
					}
				}
				owning_company = { remove_owned_country = c:BSA }
			}
			set_country_type = colonial
		}
		remove_diplomatic_pact = {
			country = c:BSA
			type = chartered_company
		}
		create_diplomatic_pact = {
			country = c:BSA
			type = colony
		}
	}
}
