# Persistent identities, assigned at registration. Kingship never reassigns a house.
# IDs 2–14 retain their existing memberships; 15–19 are authored royal branches.
sb_court_assign_authored_house = {
 if = {
  limit = { OR = { has_template = ZUL_dingane has_template = ZUL_uthumbu_claimed_son has_template = ZUL_uthumbu_successor } }
  set_variable = { name = sb_court_following_var value = 15 }
 }
 else_if = {
  limit = { OR = { has_template = ZUL_mpande_zulu } }
  set_variable = { name = sb_court_following_var value = 16 }
 }
 else_if = {
  limit = { OR = { has_template = ZUL_gqugqu_claimant } }
  set_variable = { name = sb_court_following_var value = 17 }
 }
 else_if = {
  limit = { OR = { has_template = ZUL_cetshwayo_heir has_template = ZUL_dinuzulu_heir has_template = ZUL_solomon_heir } }
  set_variable = { name = sb_court_following_var value = 18 }
 }
 else_if = {
  limit = { OR = { has_template = ZUL_mbuyazi_heir has_template = ZUL_mbuyazi_successor } }
  set_variable = { name = sb_court_following_var value = 19 }
 }
}
