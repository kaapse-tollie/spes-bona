# Spes Bona - BST frontier, protection, and STA effects

sb_bst_kill_non_moshoeshoe_ai_generals = {
	if = {
		limit = {
			country_definition = cd:BST
			is_ai = yes
		}
		every_scope_character = {
			limit = {
				is_character_alive = yes
				has_role_of_type = general
				NOT = { has_template = BST_Moshoeshoe }
				NOT = { has_trait = sb_moshoeshoe_lifeline }
			}
			kill_character = { hidden = yes }
		}
	}
}

sb_bst_apply_gun_war_ai_defence = {
	if = {
		limit = {
			is_ai = yes
			sb_frontier_ai_scripting_enabled = yes
			c:CAP ?= { is_player = yes }
		}
		add_modifier = {
			name = sb_bst_gun_war_defensive_muster_vs_player
			months = 24
		}
	}
	else_if = {
		limit = {
			is_ai = yes
			sb_frontier_ai_scripting_enabled = yes
			c:CAP ?= { is_ai = yes }
		}
		random_list = {
			66 = {
				add_modifier = {
					name = sb_bst_gun_war_defensive_muster
					months = 24
				}
			}
			33 = {}
		}
	}
}

sb_bst_execute_oranje_raid = {
	hidden_effect = {
		set_variable = {
			name = sb_bst_oranje_raid_cooldown_var
			days = 365
		}
	}

	random_country = {
		limit = {
			sb_bst_oranje_frontier_actor = yes
			NOT = { has_war_with = root }
			NOT = { has_truce_with = root }
		}
		custom_tooltip = sb_bst_raid_ora_tt
		hidden_effect = {
			trigger_event = { id = sb_bst_frontier.010 days = 1 popup = yes }
		}
	}

	add_treasury = 1000
	random_scope_state = {
		limit = {
			OR = {
				state_region = s:STATE_VRYSTAAT
				state_region = s:STATE_DRAKENSBERG
			}
		}
		kill_population_in_state = {
			value = 25
		}
	}
	if = {
		limit = {
			has_journal_entry = je_sb_adopt_firearms
			NOT = { has_variable = sb_firearms_modernized_var }
		}
		custom_tooltip = sb_bst_raid_firearms_progress_tt
		hidden_effect = {
			sb_boost_firearms_progress_raid = yes
		}
	}
}

sb_bst_restore_gbr_protectorate = {
	if = {
		limit = { has_variable = sb_bst_cap_annexation_crisis_var }
		remove_variable = sb_bst_cap_annexation_crisis_var
	}

	c:CAP ?= {
		if = {
			limit = { has_variable = sb_bst_cap_annexation_crisis_var }
			remove_variable = sb_bst_cap_annexation_crisis_var
		}
	}

	if = {
		limit = {
			is_subject_of = c:CAP
		}
		c:GBR ?= {
			transfer_subject = c:BST
		}
	}

	if = {
		limit = {
			NOT = { is_subject_of = c:GBR }
		}
		c:GBR ?= {
			create_diplomatic_pact = {
				country = root
				type = protectorate
			}
		}
	}
	else_if = {
		limit = {
			is_subject_of = c:GBR
			NOT = { is_subject_type = subject_type_protectorate }
		}
		change_subject_type = subject_type_protectorate
	}
}

# A white peace defeats Cape disarmament without enforcing BST's autonomy demand.
sb_bst_return_to_gbr_as_puppet = {
	if = {
		limit = { has_variable = sb_bst_cap_annexation_crisis_var }
		remove_variable = sb_bst_cap_annexation_crisis_var
	}

	c:CAP ?= {
		if = {
			limit = { has_variable = sb_bst_cap_annexation_crisis_var }
			remove_variable = sb_bst_cap_annexation_crisis_var
		}
	}

	if = {
		limit = { is_subject_of = c:CAP }
		c:GBR ?= {
			transfer_subject = c:BST
		}
	}

	if = {
		limit = {
			is_subject_of = c:GBR
			NOT = { is_subject_type = subject_type_puppet }
		}
		change_subject_type = subject_type_puppet
	}
}

sb_bst_centralize_sotho_tswana_clients = {
	c:TSW ?= {
		if = {
			limit = { is_country_alive = yes }
			if = {
				limit = {
					is_country_type = decentralized
				}
				set_country_type = unrecognized
			}
			if = {
				limit = {
					NOT = { is_subject_of = root }
				}
				root = {
					create_diplomatic_pact = {
						country = c:TSW
						type = tributary
					}
				}
			}
			else_if = {
				limit = {
					is_subject_of = root
					NOT = { is_subject_type = subject_type_tributary }
				}
				change_subject_type = subject_type_tributary
			}
			change_relations = {
				country = root
				value = 50
			}
			add_liberty_desire = -50
		}
	}
	c:PDI ?= {
		if = {
			limit = { is_country_alive = yes }
			if = {
				limit = {
					is_country_type = decentralized
				}
				set_country_type = unrecognized
			}
			if = {
				limit = {
					NOT = { is_subject_of = root }
				}
				root = {
					create_diplomatic_pact = {
						country = c:PDI
						type = tributary
					}
				}
			}
			else_if = {
				limit = {
					is_subject_of = root
					NOT = { is_subject_type = subject_type_tributary }
				}
				change_subject_type = subject_type_tributary
			}
			change_relations = {
				country = root
				value = 50
			}
			add_liberty_desire = -50
		}
	}
}
