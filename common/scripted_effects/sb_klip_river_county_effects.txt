# Klip River County boundary dispute, refugee, and war-resolution effects.

sb_schedule_klip_river_county_question = {
	if = {
		limit = {
			country_definition = cd:NAL
			NOT = { has_variable = sb_klip_river_county_pending_var }
			NOT = { has_variable = sb_klip_river_county_scheduled_var }
			NOT = { has_variable = sb_klip_river_county_resolved_var }
			NOT = { has_variable = sb_klip_river_boer_flight_started_var }
		}
		set_variable = sb_klip_river_county_pending_var
		set_variable = sb_klip_river_county_scheduled_var
		random_list = {
			1 = { trigger_event = { id = sb_klip_river_county.001 days = 120 popup = no } }
			1 = { trigger_event = { id = sb_klip_river_county.001 days = 210 popup = no } }
			1 = { trigger_event = { id = sb_klip_river_county.001 days = 300 popup = no } }
			1 = { trigger_event = { id = sb_klip_river_county.001 days = 390 popup = no } }
		}
	}
}

sb_klip_river_create_county = {
	if = {
		limit = { NOT = { exists = c:KLR } }
		# Stage the rebel muster before partitioning Natal so the requested share
		# is calculated from the colony's complete Boer population.
		s:STATE_NATAL.region_state:NAL ?= {
			save_temporary_scope_as = sb_klip_river_boer_muster_source
		}
		if = {
			limit = { exists = c:BUG }
			c:BUG.capital = { save_temporary_scope_as = sb_klip_river_servicemen_staging }
		}
		else = {
			c:ZUL.capital = { save_temporary_scope_as = sb_klip_river_servicemen_staging }
		}
		c:GBR.capital = { save_temporary_scope_as = sb_klip_river_officers_staging }
		if = {
			limit = {
				exists = scope:sb_klip_river_boer_muster_source
				exists = scope:sb_klip_river_servicemen_staging
				exists = scope:sb_klip_river_officers_staging
			}
			scope:sb_klip_river_boer_muster_source = {
				every_scope_pop = {
					limit = { culture = cu:boer }
					move_partial_pop = {
						state = scope:sb_klip_river_servicemen_staging
						population_ratio = 0.75
					}
				}
			}
			scope:sb_klip_river_servicemen_staging = {
				every_scope_pop = {
					limit = { culture = cu:boer }
					move_partial_pop = {
						state = scope:sb_klip_river_officers_staging
						population_ratio = 0.10
					}
				}
				every_scope_pop = {
					limit = { culture = cu:boer }
					change_poptype = pop_type:soldiers
				}
			}
			scope:sb_klip_river_officers_staging = {
				every_scope_pop = {
					limit = { culture = cu:boer }
					change_poptype = pop_type:officers
				}
			}
		}
		create_country = {
			tag = KLR
			origin = c:ORA
			province = p:xDE0EDE
		}
	}
	s:STATE_NATAL = {
		set_owner_of_provinces = {
			country = c:KLR
			provinces = { xBBCA32 xDE0EDE x552449 }
		}
	}
	c:KLR ?= {
		effect_starting_politics_conservative = yes
		set_country_type = unrecognized
		activate_law = law_type:law_presidential_republic
		activate_law = law_type:law_oligarchy
		activate_law = law_type:law_national_supremacy
		activate_law = law_type:law_homesteading
		activate_law = law_type:law_national_militia
		activate_law = law_type:law_discrete_inboekstelsel
		every_interest_group = { remove_ruling_interest_group = yes }
		ig:ig_landowners ?= { add_ruling_interest_group = yes }
		s:STATE_NATAL.region_state:KLR = {
			create_building = {
				building = building_maize_farm
				add_ownership = {
					building = {
						type = building_manor_house
						country = c:KLR
						levels = 1
						region = "STATE_NATAL"
					}
				}
				reserves = 1
				activate_production_methods = { "pm_simple_farming" "pm_no_secondary" "pm_tools_disabled" }
			}
		}
		if = {
			limit = { NOT = { has_claim = s:STATE_NATAL } }
			s:STATE_NATAL = { add_claim = c:KLR }
		}
		if = {
			limit = {
				NOT = { any_scope_character = { has_template = KLR_andries_theodorus_spies } }
			}
			create_character = { template = KLR_andries_theodorus_spies }
		}
		random_scope_character = {
			limit = {
				has_template = KLR_andries_theodorus_spies
				is_character_alive = yes
			}
			set_interest_group = c:KLR.ig:ig_landowners
			set_as_interest_group_leader = yes
			set_character_as_ruler = yes
		}
		if = {
			limit = {
				exists = scope:sb_klip_river_servicemen_staging
				exists = scope:sb_klip_river_officers_staging
				exists = s:STATE_NATAL.region_state:KLR
			}
			s:STATE_NATAL.region_state:KLR = {
				save_temporary_scope_as = sb_klip_river_muster_target
			}
			scope:sb_klip_river_servicemen_staging = {
				every_scope_pop = {
					limit = {
						culture = cu:boer
						is_pop_type = soldiers
					}
					move_partial_pop = {
						state = scope:sb_klip_river_muster_target
						population_ratio = 1
					}
				}
			}
			scope:sb_klip_river_officers_staging = {
				every_scope_pop = {
					limit = {
						culture = cu:boer
						is_pop_type = officers
					}
					move_partial_pop = {
						state = scope:sb_klip_river_muster_target
						population_ratio = 1
					}
				}
			}
		}
		if = {
			limit = { NOT = { has_variable = sb_klip_river_county_muster_created_var } }
			set_variable = sb_klip_river_county_muster_created_var
			create_military_formation = {
				type = army
				hq_region = sr:region_southern_africa
				name = klip_river_commando
				combat_unit = {
					type = unit_type:combat_unit_type_dragoons
					state_region = s:STATE_NATAL
					count = 1
				}
				combat_unit = {
					type = unit_type:combat_unit_type_irregular_infantry
					state_region = s:STATE_NATAL
					count = 1
				}
			}
		}
		set_variable = sb_klip_river_secession_active_var
	}
	set_variable = sb_klip_river_secession_active_var
	c:ZUL ?= { set_variable = sb_klip_river_secession_active_var }
}

				c:KLR ?= { sb_klip_river_coalition_assistance_enabled = yes }

			}
		}
		c:KLR ?= { set_variable = sb_klip_river_coalition_boost_rolled_var }
		if = {
			limit = { c:ZUL ?= { sb_klip_river_zulu_war_advantage = yes } }
			random_list = {
				70 = {
					c:KLR ?= {
						if = {
							limit = { sb_klip_river_coalition_assistance_enabled = yes }
							sb_klip_river_apply_coalition_laager = yes
						}
					}
					c:ZUL ?= {
						if = {
							limit = { sb_klip_river_coalition_assistance_enabled = yes }
							add_modifier = { name = sb_klip_river_zulu_muster months = 12 }
						}
					}
				}
				30 = { }
			}
		}
		else = {
			random_list = {
				30 = {
					c:KLR ?= {
						if = {
							limit = { sb_klip_river_coalition_assistance_enabled = yes }
							sb_klip_river_apply_coalition_laager = yes
						}

	}
}

sb_klip_river_start_secession_play = {
	sb_klip_river_roll_coalition_boost = yes
	if = {
		limit = { has_truce_with = c:ZUL }
		end_truce = c:ZUL
	}
	set_variable = sb_klip_river_secession_active_var
	c:KLR ?= { set_variable = sb_klip_river_secession_active_var }
	c:ZUL ?= { set_variable = sb_klip_river_secession_active_var }
	if = {
		limit = {
			c:ORA ?= {
				is_country_alive = yes
				is_subject = no
				has_variable = sb_klip_river_ora_backed_var
			}
		}
		c:ORA = {
			create_diplomatic_pact = {
				country = c:KLR
				type = protectorate
			}
		}
		c:ZUL = {
			create_diplomatic_play = {
				type = dp_sb_klip_river_secession
				target_country = c:NAL
				add_initiator_backers = { c:KLR c:ORA }
				add_target_backers = { c:GBR }
				add_war_goal = {
					holder = c:NAL
					target_country = c:KLR
					type = annex_country
				}
				add_war_goal = {
					holder = c:NAL
					target_country = c:ZUL
					type = annex_country
				}
			}
		}
	}
	else = {
		c:ZUL = {
			create_diplomatic_play = {
				type = dp_sb_klip_river_secession
				target_country = c:NAL
				add_initiator_backers = { c:KLR }
				add_target_backers = { c:GBR }
				add_war_goal = {
					holder = c:NAL
					target_country = c:KLR
					type = annex_country
				}
				add_war_goal = {
					holder = c:NAL
					target_country = c:ZUL
					type = annex_country
				}
			}
		}
	}
	c:ZUL ?= {
		if = {
			limit = { is_ai = yes }
			set_strategy = ai_strategy_sb_klip_river_secession_resistance
		}
	}
}

sb_klip_river_finalize_secession_white_peace = {
	if = {
		limit = { NOT = { has_variable = sb_klip_river_county_resolved_var } }
		set_variable = sb_klip_river_county_resolved_var
		# Only an enforced annexation of KLR displaces its Boer population.
		if = {
			limit = { has_variable = sb_klip_river_county_annexed_var }
			set_variable = sb_klip_river_boer_departure_pending_var
			trigger_event = { id = sb_klip_river_county.091 days = 1 popup = yes }
		}
		else = {
			c:KLR ?= {
				set_variable = sb_klip_river_county_resolved_var
				every_scope_character = {
					limit = { has_variable = sb_klip_river_deferred_exile_var }
					remove_variable = sb_klip_river_deferred_exile_var
				}

sb_klip_river_move_punitive_refugees = {

	if = {
		limit = { exists = scope:klip_refugee_receiver }
		if = {
			limit = {
				c:NAL ?= {
					is_country_alive = yes
					any_scope_character = { has_variable = sb_klip_river_deferred_exile_var }
				}
			}
			c:NAL = {
				random_scope_character = {
					limit = { has_variable = sb_klip_river_deferred_exile_var }
					transfer_character = scope:klip_refugee_receiver
					set_interest_group = scope:klip_refugee_receiver.ig:ig_landowners
					remove_variable = sb_klip_river_deferred_exile_var
				}
			}
		}
		else_if = {
			limit = { any_scope_character = { has_variable = sb_klip_river_deferred_exile_var } }
			random_scope_character = {
				limit = { has_variable = sb_klip_river_deferred_exile_var }
				transfer_character = scope:klip_refugee_receiver
				set_interest_group = scope:klip_refugee_receiver.ig:ig_landowners
				remove_variable = sb_klip_river_deferred_exile_var
			}
		}
		if = {
			limit = { c:NAL ?= { is_country_alive = yes } }
			c:NAL = {
				every_scope_state = {
					every_scope_pop = {
						limit = { culture = cu:boer }
						move_partial_pop = {
							state = scope:klip_refugee_receiver.capital
							population_ratio = 0.95
						}
					}
				}
			}
		}
		every_scope_state = {
			every_scope_pop = {
				limit = { culture = cu:boer }
				move_partial_pop = {
					state = scope:klip_refugee_receiver.capital
					population_ratio = 0.95
				}
			}
		}
		scope:klip_refugee_receiver = {
			trigger_event = { id = sb_klip_river_county.082 days = 1 popup = yes }
		}
	}
}

sb_klip_river_handle_wargoal_enforced = {
	# Timer flips are reversible. Record only a diagnostic; war-end reads owner.
	if = {
		limit = { exists = scope:enforced_by_timer scope:diplomatic_play ?= { OR = { is_diplomatic_play_type = dp_sb_klip_river_secession is_diplomatic_play_type = dp_sb_klip_river_punitive_expedition } } }
		c:NAL ?= { set_variable = sb_klip_river_timer_goal_seen_pending_var }
	}
	else_if = {
		limit = { scope:diplomatic_play ?= { is_diplomatic_play_type = dp_sb_klip_river_secession } c:NAL ?= { has_variable = sb_klip_river_secession_active_var } }
		if = { limit = { country_definition = cd:ZUL scope:target ?= { country_definition = cd:NAL } } c:NAL = { set_variable = sb_klip_river_zulu_demand_accepted_pending_var } }
		else_if = { limit = { country_definition = cd:NAL scope:target ?= { country_definition = cd:ZUL } } c:NAL = { set_variable = sb_klip_river_natal_demand_accepted_pending_var } }
		else_if = { limit = { country_definition = cd:NAL scope:target ?= { country_definition = cd:KLR } } c:NAL = { set_variable = sb_klip_river_county_annexation_accepted_pending_var } }
	}
	else_if = {
		limit = { scope:diplomatic_play ?= { is_diplomatic_play_type = dp_sb_klip_river_punitive_expedition } c:NAL ?= { has_variable = sb_klip_river_punitive_active_var } }
		if = { limit = { country_definition = cd:NAL } c:NAL = { set_variable = sb_klip_river_natal_demand_accepted_pending_var } }
		else_if = { limit = { country_definition = cd:ZUL } c:NAL = { set_variable = sb_klip_river_zulu_demand_accepted_pending_var } }
	}
}

sb_klip_river_handle_war_end = {
	if = {
		limit = { is_diplomatic_play_type = dp_sb_klip_river_secession c:NAL ?= { is_country_alive = yes has_variable = sb_klip_river_secession_active_var } }
		if = {
			limit = { p:x279045.state.owner = c:ZUL }
			c:KLR ?= { sb_klip_river_finalize_reduced_natalia = yes }
		}
		else_if = {
			limit = { c:NAL ?= { has_variable = sb_klip_river_natal_demand_accepted_pending_var } }
			if = { limit = { c:NAL ?= { has_variable = sb_klip_river_county_annexation_accepted_pending_var } NOT = { c:KLR ?= { is_country_alive = yes } } } c:NAL = { sb_klip_river_record_county_annexation = yes sb_klip_river_finalize_secession_white_peace = yes } }
			else = { c:NAL = { sb_klip_river_restore_colonial_status_quo = yes } }
		}
		else = { c:NAL = { sb_klip_river_finalize_secession_white_peace = yes } }
	}
	else_if = {
		limit = { is_diplomatic_play_type = dp_sb_klip_river_punitive_expedition c:NAL ?= { is_country_alive = yes has_variable = sb_klip_river_punitive_active_var } }
		if = { limit = { p:x279045.state.owner = c:ZUL } c:NAL = { sb_klip_river_finalize_punitive_zulu_victory = yes } }
		else_if = { limit = { c:NAL ?= { has_variable = sb_klip_river_natal_demand_accepted_pending_var } } c:NAL = { sb_klip_river_finalize_punitive_natal_victory = yes } }
		else = { c:NAL = { sb_klip_river_finalize_punitive_white_peace = yes } }
	}
	c:NAL ?= {
		if = { limit = { has_variable = sb_klip_river_zulu_demand_accepted_pending_var } remove_variable = sb_klip_river_zulu_demand_accepted_pending_var }
		if = { limit = { has_variable = sb_klip_river_natal_demand_accepted_pending_var } remove_variable = sb_klip_river_natal_demand_accepted_pending_var }
		if = { limit = { has_variable = sb_klip_river_county_annexation_accepted_pending_var } remove_variable = sb_klip_river_county_annexation_accepted_pending_var }
		if = { limit = { has_variable = sb_klip_river_timer_goal_seen_pending_var } remove_variable = sb_klip_river_timer_goal_seen_pending_var }
	}
}

sb_klip_river_county_monthly_housekeeping = {
	# Recover a partial settlement if the KLR annexation goal was enforced but
	# the terminating war callback no longer exposed its original actors.
	if = {
		limit = {
			country_definition = cd:NAL
			has_variable = sb_klip_river_secession_active_var
			has_variable = sb_klip_river_county_annexed_var
			NOT = { has_variable = sb_klip_river_county_resolved_var }
			is_at_war = no
			is_active_in_diplomatic_play = no
			c:ZUL ?= { is_country_alive = yes }
		}
		sb_klip_river_finalize_secession_white_peace = yes
	}
	if = {
		limit = {
			country_definition = cd:NAL
			has_variable = sb_natalia_british_colony_resolved_var
			sb_klip_river_county_setup_valid = yes
			NOT = { has_variable = sb_klip_river_county_pending_var }
			NOT = { has_variable = sb_klip_river_county_scheduled_var }
			NOT = { has_variable = sb_klip_river_county_resolved_var }
			NOT = { has_variable = sb_klip_river_boer_flight_started_var }
		}
		sb_mark_natalia_boer_leadership_for_deferred_exile = yes
		sb_schedule_klip_river_county_question = yes
	}
	if = {
		limit = {
			country_definition = cd:NAL
			has_variable = sb_klip_river_county_pending_var
			has_variable = sb_klip_river_county_ready_var
		}
			if = {
				limit = { sb_klip_river_county_ready_for_commission = yes }
				trigger_event = { id = sb_klip_river_county.010 days = 1 popup = yes }
		}
		else_if = {
			limit = { NOT = { sb_klip_river_county_setup_valid = yes } }
			sb_klip_river_prepare_standard_boer_flight = yes
		}
	}
}
