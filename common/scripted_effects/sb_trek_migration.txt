sb_move_mzilikazi_ndebele_to_zimbabwe = {
	s:STATE_ZAMBEZI = {
		region_state:MTB ?= {
			save_scope_as = mzilikazi_zimbabwe_target
		}
	}
	if = {
		limit = { exists = scope:mzilikazi_zimbabwe_target }
		every_scope_state = {
			limit = {
				NOT = { this = scope:mzilikazi_zimbabwe_target }
				any_scope_pop = {
					culture = cu:ndebele
				}
			}
			every_scope_pop = {
				limit = {
					culture = cu:ndebele
				}
				move_partial_pop = {
					state = scope:mzilikazi_zimbabwe_target
					population_ratio = 0.2
				}
			}
		}
	}
}

# Creates mass migration of Boer pops from Cape Colony to the calling country (DEPRECATED)
sb_trigger_boer_migration_effect = {
	# Find the Cape Colony as migration origin
	random_country = {
		limit = {
			country_definition = cd:CAP
			is_country_alive = yes
		}
		save_scope_as = boer_migration_origin
		random_scope_state = {
			limit = {
				is_incorporated = yes
				any_scope_culture = {
					this = cu:boer
				}
			}
			save_scope_as = boer_origin_state
		}
	}
	# Find best target state in the Boer republic
	ordered_scope_state = {
		limit = {
			is_incorporated = yes
		}
		order_by = migration_pull
		position = 0
		check_range_bounds = no
		save_scope_as = boer_migration_target
	}
	# Create the mass migration
	if = {
		limit = {
			exists = scope:boer_migration_origin
			exists = scope:boer_migration_target
		}
		scope:boer_migration_target = {
			create_mass_migration = {
				origin = scope:boer_migration_origin
				culture = cu:boer
			}
		}
		scope:boer_migration_origin = {
			add_modifier = {
				name = sb_trek_migration_cost
				months = 24
			}
		}
	}
}

sb_move_great_trek_completion_cape_boer_slice = {
	if = {
		limit = {
			NOT = { exists = scope:migration_target }
		}
		ordered_scope_state = {
			order_by = migration_pull
			position = 0
			check_range_bounds = no
			save_scope_as = migration_target
		}
	}
	if = {
		limit = {
			exists = c:CAP
			exists = scope:migration_target
		}
		c:CAP = {
			every_scope_state = {
				limit = {
					any_scope_pop = {
						culture = cu:boer
					}
				}
				every_scope_pop = {
					limit = {
						culture = cu:boer
					}
					move_partial_pop = {
						state = scope:migration_target
							population_ratio = 0.05 # 5% up from 3.3%
					}
				}
			}
		}
	}
}

sb_seed_trn_v2_start_package = {
	# Minimal scripted replacement for the old day-1 TRN history package.
	# This is applied only after the delayed split event creates the country.
	c:TRN ?= {
		random_scope_state = {
			limit = {
				state_region = s:STATE_TRANSVAAL
			}
			create_building = {
				building = building_trade_center
				add_ownership = {
					building = {
						type = building_trade_center
						country = c:TRN
						levels = 1
						region = "STATE_TRANSVAAL"
					}
				}
				reserves = 1
				activate_production_methods = { "pm_trade_center_trade_quantity_normal" }
				}
				create_building = {
					building = building_maize_farm
					add_ownership = {
						building = {
						type = building_manor_house
						country = c:TRN
						levels = 1
						region = "STATE_TRANSVAAL"
					}
				}
				reserves = 1
				activate_production_methods = { "pm_simple_farming" "pm_no_secondary" "pm_tools_disabled" }
			}
			create_building = {
				building = building_livestock_ranch
				add_ownership = {
					building = {
						type = building_manor_house
						country = c:TRN
						levels = 1
						region = "STATE_TRANSVAAL"
					}
				}
				reserves = 1
				activate_production_methods = { "pm_open_air_stockyards" "pm_simple_ranch" "pm_standard_fences" "pm_unrefrigerated" }
			}
		}
		sb_move_trn_v2_cape_trekkers_to_transvaal = yes
		if = {
			limit = { army_size <= 0 }
			create_military_formation = {
				type = army
				hq_region = sr:region_southern_africa
				name = pretorius_commando

				combat_unit = {
					type = unit_type:combat_unit_type_dragoons
					state_region = s:STATE_TRANSVAAL
					count = 1
				}
			}
		}
	}
}

sb_spawn_transvaal_republic_v2 = {

		#   ORA plants the first real Boer foothold beyond the Vaal
		#   -> a fresh TRN is carved out around that foothold
		#
		# Only ORA-held west-Transvaal core provinces are handed over here. That
		# keeps the split tied to the actual colony the player or AI just founded,
		# rather than giving the new republic land it has not settled yet.
	if = {
		limit = { has_variable = sb_trn_v2_spawn_pending_var }
		remove_variable = sb_trn_v2_spawn_pending_var
	}

		limit = { NOT = { has_variable = sb_trn_v2_spawned_var } }
		set_variable = sb_trn_v2_spawned_var
	}
		if = {
			limit = {
				NOT = { exists = c:TRN }
			}
			if = {
				limit = { p:xD0A010.state.owner = root }
				create_country = {
					tag = TRN
					origin = root
					province = p:xD0A010
				}
			}
			else_if = {
				limit = { p:x8BDC4D.state.owner = root }
				create_country = {
					tag = TRN
					origin = root
					province = p:x8BDC4D
				}
			}
			else_if = {
				limit = { p:xA22DC2.state.owner = root }
				create_country = {
					tag = TRN
					origin = root
					province = p:xA22DC2
				}
			}
			else_if = {
				limit = { p:xD02190.state.owner = root }
				create_country = {
					tag = TRN
					origin = root
					province = p:xD02190
				}
			}
			else_if = {
				limit = { p:x6C1786.state.owner = root }
				create_country = {
					tag = TRN
					origin = root
					province = p:x6C1786
				}
			}
			else = {
				create_country = {
					tag = TRN
					origin = root
					province = p:x9A137B
				}
			}
		}
		s:STATE_TRANSVAAL = {
			if = {
				limit = { p:xD0A010.state.owner = root }
				set_owner_of_provinces = {
					country = c:TRN
					provinces = { xD0A010 }
				}
			}
			if = {
				limit = { p:x8BDC4D.state.owner = root }
				set_owner_of_provinces = {
					country = c:TRN
					provinces = { x8BDC4D }
				}
			}
			if = {
				limit = { p:xA22DC2.state.owner = root }
				set_owner_of_provinces = {
					country = c:TRN
					provinces = { xA22DC2 }
				}

				limit = { p:xD02190.state.owner = root }
				set_owner_of_provinces = {
					country = c:TRN
					provinces = { xD02190 }
				}
			}
			if = {
				limit = { p:x6C1786.state.owner = root }
				set_owner_of_provinces = {
					country = c:TRN
					provinces = { x6C1786 }
				}
			}
			if = {
				limit = { p:x9A137B.state.owner = root }
				set_owner_of_provinces = {
					country = c:TRN

				}
