# Spes Bona - Trek and Boer republic scripted effects
#
# This file now does more than migration:
#   - mass trek migration rewards for Great Trek JE stages
#   - the delayed TRN spawn package used by the V2 ORA-first setup
#   - later inland republican splinters and convention helpers

# Reserve the authoritative Ndebele-retreat delivery before scheduling it.
# A timed queued receipt allows the monthly watchdog to retry a lost dispatch.
sb_great_trek_queue_outcome_delivery = {
	set_variable = sb_great_trek_outcome_delivery_pending_var
	set_variable = { name = sb_great_trek_outcome_delivery_queued_var days = 7 }
	trigger_event = { id = sb_great_trek.002 days = 3 popup = yes }
}

sb_move_mzilikazi_ndebele_to_zimbabwe = {
	s:STATE_ZAMBEZI = {
		region_state:MTB ?= {
			save_scope_as = mzilikazi_zimbabwe_target
		}
	}
	if = {
		limit = { exists = scope:mzilikazi_zimbabwe_target }
		every_scope_state = {
			limit = {
				NOT = { this = scope:mzilikazi_zimbabwe_target }
				any_scope_pop = {
					culture = cu:ndebele
				}
			}
			every_scope_pop = {
				limit = {
					culture = cu:ndebele
				}
				move_partial_pop = {
					state = scope:mzilikazi_zimbabwe_target
					population_ratio = 0.2
				}
			}
		}
	}
}

sb_move_great_trek_completion_cape_boer_slice = {
	if = {
		limit = {
			NOT = { exists = scope:migration_target }
		}
		ordered_scope_state = {
			order_by = migration_pull
			position = 0
			check_range_bounds = no
			save_scope_as = migration_target
		}
	}
	if = {
		limit = {
			exists = c:CAP
			exists = scope:migration_target
		}
		c:CAP = {
			every_scope_state = {
				limit = {
					any_scope_pop = {
						culture = cu:boer
					}
				}
				every_scope_pop = {
					limit = {
						culture = cu:boer
					}
					move_partial_pop = {
						state = scope:migration_target
							population_ratio = 0.05 # 5% up from 3.3%
					}
				}
			}
		}
	}
}

sb_great_trek_finalize_republic = {
	set_variable = sb_trek_stage2_complete
	if = {
		limit = { has_variable = sb_trek_stage2_pending_var }
		remove_variable = sb_trek_stage2_pending_var
	}
	if = {
		limit = { has_variable = sb_voortrekker_start_var }
		remove_variable = sb_voortrekker_start_var
	}
	if = {
		limit = { has_variable = sb_early_republic_var }
		remove_variable = sb_early_republic_var
	}
	if = {
		limit = { country_definition = cd:TRN }
		s:STATE_TRANSVAAL = { add_homeland = cu:boer }
	}
	if = {
		limit = { country_definition = cd:ORA }
		s:STATE_VRYSTAAT = { add_homeland = cu:boer }
	}
	if = {
		limit = { country_definition = cd:ZPB }
		s:STATE_NORTHERN_TRANSVAAL = { add_homeland = cu:boer }
	}
	if = {
		limit = { country_definition = cd:LYD }
		s:STATE_EAST_TRANSVAAL = { add_homeland = cu:boer }
	}
	if = {
		limit = {
			OR = {
				country_definition = cd:NAL
				country_definition = cd:KLR
			}
		}
		s:STATE_NATAL = { add_homeland = cu:boer }
	}
	if = {
		limit = { country_definition = cd:KLR }
		change_tag = NAL
	}
	if = {
		limit = { country_definition = cd:SGO }
		s:STATE_BECHUANALAND = { add_homeland = cu:boer }
	}
	if = {
		limit = { country_definition = cd:NRP }
		s:STATE_ZULULAND = { add_homeland = cu:boer }
	}
	if = {
		limit = {
			NOT = { has_modifier = sb_boer_principality_status }
		}
		add_modifier = { name = sb_boer_principality_status }
	}
	sb_great_trek_final_boer_slice_reward = yes
	add_modifier = {
		name = sb_prestige_gain_medium
		months = long_modifier_time
	}
	every_scope_state = {
		remove_modifier = sb_trek_migration_pull
	}
	if = {
		limit = {
			OR = {
				country_definition = cd:TRN
				country_definition = cd:ORA
				country_definition = cd:ZPB
				country_definition = cd:LYD
					country_definition = cd:NAL
					country_definition = cd:SGO
					country_definition = cd:NRP
			}
			country_has_primary_culture = cu:boer
		}
		trigger_event = { id = sb_boer_republics.130 days = 10 popup = yes }
	}
}

sb_cleanup_trn_generated_commanders = {
	if = {
		limit = { country_definition = cd:TRN }
		every_scope_character = {
			limit = {
				is_character_alive = yes
				has_role_of_type = general
				NOT = { has_template = TRN_andries_pretorius }
				NOT = { has_template = TRN_marthinus_wessel_pretorius }
				NOT = { has_template = TRN_stephanus_schoeman }
			}
			remove_character_role = general
		}
	}
}

sb_seed_trn_v2_start_package = {
	# Minimal scripted replacement for the old day-1 TRN history package.
	# This is applied only after the delayed split event creates the country.
	c:TRN ?= {
		sb_martinus_ensure_trn_pretorius_cast = yes
		random_scope_state = {
			limit = {
				state_region = s:STATE_TRANSVAAL
			}
			create_building = {
				building = building_trade_center
				add_ownership = {
					building = {
						type = building_trade_center
						country = c:TRN
						levels = 1
						region = "STATE_TRANSVAAL"
					}
				}
				reserves = 1
				activate_production_methods = { "pm_trade_center_trade_quantity_normal" }
				}
				create_building = {
					building = building_maize_farm
					add_ownership = {
						building = {
						type = building_manor_house
						country = c:TRN
						levels = 1
						region = "STATE_TRANSVAAL"
					}
				}
				reserves = 1
				activate_production_methods = { "pm_simple_farming" "pm_no_secondary" "pm_tools_disabled" }
			}
			create_building = {
				building = building_livestock_ranch
				add_ownership = {
					building = {
						type = building_manor_house
						country = c:TRN
						levels = 1
						region = "STATE_TRANSVAAL"
					}
				}
				reserves = 1
				activate_production_methods = { "pm_open_air_stockyards" "pm_simple_ranch" "pm_standard_fences" "pm_unrefrigerated" }
			}
		}
			sb_move_trn_v2_cape_trekkers_to_transvaal = yes
			if = {
				limit = {
					NOT = { has_variable = sb_trn_starting_commando_created_var }
				}
				set_variable = sb_trn_starting_commando_created_var
				create_military_formation = {
					type = army
					hq_region = sr:region_southern_africa
					name = pretorius_commando
					save_scope_as = trn_pretorius_commando_scope

					combat_unit = {
						type = unit_type:combat_unit_type_dragoons
						state_region = s:STATE_TRANSVAAL
						count = 1
					}
					combat_unit = {
						type = unit_type:combat_unit_type_irregular_infantry
						state_region = s:STATE_TRANSVAAL
						count = 1
					}
				}
				random_scope_character = {
					limit = {
						has_template = TRN_andries_pretorius
						is_character_alive = yes
					}
					if = {
						limit = {
							NOT = { has_role_of_type = general }
						}
						add_character_role = general
					}
					transfer_to_formation = scope:trn_pretorius_commando_scope
				}
				if = {
					limit = {
						NOT = {
							any_scope_character = {
								has_template = TRN_andries_pretorius
								is_character_alive = yes
								has_role_of_type = general
							}
						}
					}
					random_scope_character = {
						limit = {
							has_template = TRN_marthinus_wessel_pretorius
							is_character_alive = yes
						}
						if = {
							limit = { NOT = { has_role_of_type = general } }
							add_character_role = general
						}
						transfer_to_formation = scope:trn_pretorius_commando_scope
					}
				}
				sb_cleanup_trn_generated_commanders = yes
			}
		}
	}

sb_spawn_transvaal_republic_v2 = {
		# Potchefstroom / west-Transvaal split:
		#   ORA plants the first real Boer foothold beyond the Vaal
		#   -> a fresh TRN is carved out around that foothold
		#
		# Only ORA-held west-Transvaal core provinces are handed over here. That
		# keeps the split tied to the actual colony the player or AI just founded,
		# rather than giving the new republic land it has not settled yet.
	if = {
		limit = { has_variable = sb_trn_v2_spawn_pending_var }
		remove_variable = sb_trn_v2_spawn_pending_var
	}
	if = {
		limit = { NOT = { has_variable = sb_trn_v2_spawned_var } }
		set_variable = sb_trn_v2_spawned_var
	}
		if = {
			limit = {
				NOT = { exists = c:TRN }
			}
			if = {
				limit = { p:xD0A010.state.owner = root }
				create_country = {
					tag = TRN
					origin = c:ORA
					province = p:xD0A010
				}
			}
			else_if = {
				limit = { p:x8BDC4D.state.owner = root }
				create_country = {
					tag = TRN
					origin = c:ORA
					province = p:x8BDC4D
				}
			}
			else_if = {
				limit = { p:xA22DC2.state.owner = root }
				create_country = {
					tag = TRN
					origin = c:ORA
					province = p:xA22DC2
				}
			}
			else_if = {
				limit = { p:xD02190.state.owner = root }
				create_country = {
					tag = TRN
					origin = c:ORA
					province = p:xD02190
				}
			}
			else_if = {
				limit = { p:x6C1786.state.owner = root }
				create_country = {
					tag = TRN
					origin = c:ORA
					province = p:x6C1786
				}
			}
			else = {
				create_country = {
					tag = TRN
					origin = c:ORA
					province = p:x9A137B
				}
			}
		}
		s:STATE_TRANSVAAL = {
			if = {
				limit = { p:xD0A010.state.owner = root }
				set_owner_of_provinces = {
					country = c:TRN
					provinces = { xD0A010 }
				}
			}
			if = {
				limit = { p:x8BDC4D.state.owner = root }
				set_owner_of_provinces = {
					country = c:TRN
					provinces = { x8BDC4D }
				}
			}
			if = {
				limit = { p:xA22DC2.state.owner = root }
				set_owner_of_provinces = {
					country = c:TRN
					provinces = { xA22DC2 }
				}
			}
			if = {
				limit = { p:xD02190.state.owner = root }
				set_owner_of_provinces = {
					country = c:TRN
					provinces = { xD02190 }
				}
			}
			if = {
				limit = { p:x6C1786.state.owner = root }
				set_owner_of_provinces = {
					country = c:TRN
					provinces = { x6C1786 }
				}
			}
			if = {
				limit = { p:x9A137B.state.owner = root }
				set_owner_of_provinces = {
					country = c:TRN
					provinces = { x9A137B }
				}
			}
			add_claim = c:TRN
		}
	c:TRN ?= {
		# TRN starts as an early frontier republic again, with its own delayed
		# Great Trek / Lydenburg / ZPB lane instead of inheriting a day-1 shell.
		if = {
			limit = { c:ZUL ?= { is_country_alive = yes } }
			set_relations = { country = c:ZUL value = -30 }
		}
		set_country_type = unrecognized
		set_variable = sb_commandant_general_visible_var
		set_variable = sb_early_republic_var
		set_next_election_date = 1839.10.12
		effect_starting_politics_conservative = yes
		sb_apply_boer_republic_spawn_setup = yes
		activate_law = law_type:law_national_supremacy
		activate_law = law_type:law_elected_bureaucrats
		activate_law = law_type:law_agrarianism
		activate_law = law_type:law_land_based_taxation
		activate_law = law_type:law_homesteading
			activate_law = law_type:law_national_militia
			activate_law = law_type:law_discrete_inboekstelsel
			activate_law = law_type:law_frontier_colonization

							has_template = TRN_stephanus_schoeman
						}
				}
			}

				}
			}
				sb_martinus_ensure_trn_pretorius_cast = yes

					limit = {
						c:NAL ?= {
						is_country_alive = yes

					}
				}

						has_template = TRN_marthinus_wessel_pretorius
						is_character_alive = yes
					}
					set_interest_group = c:TRN.ig:ig_landowners
					set_as_interest_group_leader = yes
					set_character_as_ruler = yes
				}
			}
		else = {
			if = {
				limit = {
					NOT = {
						any_scope_character = {
							has_template = TRN_andries_pretorius
						}
					}
				}
				create_character = {
					template = TRN_andries_pretorius
					ruler = yes
					is_general = yes
					ig_leader = yes
					interest_group = ig_landowners
					ideology = ideology_sb_volksraad_elitism_leader
				}
			}
		}
		if = {
			limit = { NOT = { has_journal_entry = je_sb_great_trek } }
			add_journal_entry = { type = je_sb_great_trek }
		}
			sb_ensure_boer_anti_british_lobby = yes
			change_relations = {
				country = root
				value = 25
		}
		sb_seed_trn_v2_start_package = yes
		if = {
			limit = {
				c:ORA ?= {
					has_variable = sb_natal_war_active_var
				}
			}
			sb_join_trn_to_active_natal_war = yes
		}
		}
		change_relations = {
			country = c:TRN
			value = 25
		}
		c:TRN ?= {
			sb_ensure_gbr_transit_rights_with_root = yes
			create_bidirectional_truce = {
				country = c:GBR
				months = 240
			}
		}
		c:TRN ?= { sb_ensure_boer_republic_compacts_for_root = yes }
		c:TRN ?= { trigger_event = { id = sb_boer_republics.121 days = 1 popup = no } }
		c:ORA ?= {
		if = {
			limit = { has_variable = sb_voortrekker_start_var }
			remove_variable = sb_voortrekker_start_var
		}
	}
	c:GBR ?= {
		if = {
			limit = { is_country_alive = yes }
			trigger_event = { id = sb_boer_republics.007 days = sb_travel_london_southern_africa_message_days popup = yes }
		}
	}
}
