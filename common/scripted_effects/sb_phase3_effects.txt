# Spes Bona - Phase 3 Kimberley/Griqualand West helpers

# Kimberley uses 1.13 dynamic resource potential: the initial rush creates a
# small discovered field, while the settlement unlocks the remaining potential.
sb_phase3_discover_kimberley_diamonds = {
	if = {
		limit = { has_global_variable = sb_kimberley_discovery_pending_var }
		remove_global_variable = sb_kimberley_discovery_pending_var
	}
	set_global_variable = sb_kimberley_discovered_global_var
	s:STATE_NORTHERN_CAPE = {
		change_resource_potential = {
			type = building_gold_field
			amount = 4
		}
	}
	p:x68B5E8.state = {
		force_resource_discovery = building_gold_field
	}
	c:CAP ?= {
		if = {
			limit = { is_country_alive = yes }
			set_variable = sb_phase3_glw_question_active_var
		}
	}
}

sb_phase3_seed_remaining_kimberley_potential = {
	if = {
		limit = {
			NOT = { has_global_variable = sb_phase3_kimberley_remaining_potential_seeded_var }
		}
		set_global_variable = sb_phase3_kimberley_remaining_potential_seeded_var
		s:STATE_NORTHERN_CAPE = {
			change_resource_potential = {
				type = building_gold_field
				amount = 16
			}
		}
	}
}

sb_phase3_clear_griqualand_west_crisis_vars = {
	if = {
		limit = { has_global_variable = sb_kimberley_discovery_pending_var }
		remove_global_variable = sb_kimberley_discovery_pending_var
	}
	c:CAP ?= {
		if = { limit = { has_variable = sb_phase3_glw_question_active_var } remove_variable = sb_phase3_glw_question_active_var }
		if = { limit = { has_variable = sb_phase3_cap_ora_kimberley_dispute_var } remove_variable = sb_phase3_cap_ora_kimberley_dispute_var }
		if = { limit = { has_variable = sb_phase3_wbl_cap_crisis_war_var } remove_variable = sb_phase3_wbl_cap_crisis_war_var }
		if = { limit = { has_variable = sb_phase3_wbl_ora_crisis_war_var } remove_variable = sb_phase3_wbl_ora_crisis_war_var }
	}
	c:ORA ?= {
		if = { limit = { has_variable = sb_phase3_ora_pressed_kimberley_claim_var } remove_variable = sb_phase3_ora_pressed_kimberley_claim_var }
		if = { limit = { has_variable = sb_phase3_cap_ora_kimberley_dispute_var } remove_variable = sb_phase3_cap_ora_kimberley_dispute_var }
		if = { limit = { has_variable = sb_phase3_wbl_cap_crisis_war_var } remove_variable = sb_phase3_wbl_cap_crisis_war_var }
		if = { limit = { has_variable = sb_phase3_wbl_ora_crisis_war_var } remove_variable = sb_phase3_wbl_ora_crisis_war_var }
	}
	c:WBL ?= {
		if = { limit = { has_variable = sb_phase3_wbl_cap_protection_demand_var } remove_variable = sb_phase3_wbl_cap_protection_demand_var }
		if = { limit = { has_variable = sb_phase3_wbl_ora_protection_demand_var } remove_variable = sb_phase3_wbl_ora_protection_demand_var }
		if = { limit = { has_variable = sb_phase3_wbl_cap_crisis_war_var } remove_variable = sb_phase3_wbl_cap_crisis_war_var }
		if = { limit = { has_variable = sb_phase3_wbl_ora_crisis_war_var } remove_variable = sb_phase3_wbl_ora_crisis_war_var }
	}
}

sb_phase3_mark_griqualand_west_resolved = {
	c:CAP ?= {
		if = {
			limit = { is_country_alive = yes }
			set_variable = sb_phase3_glw_question_resolved_var
			if = { limit = { has_variable = sb_phase3_glw_question_active_var } remove_variable = sb_phase3_glw_question_active_var }
		}
	}
}

sb_phase3_finalize_kimberley_cap_route = {
	c:WBL ?= {
		if = {
			limit = { is_country_alive = yes }
			c:CAP ?= {
				if = {
					limit = { is_country_alive = yes }
					annex = c:WBL
				}
			}
		}
	}
	sb_phase3_seed_remaining_kimberley_potential = yes
	sb_phase3_mark_griqualand_west_resolved = yes
	sb_phase3_clear_griqualand_west_crisis_vars = yes
	c:CAP ?= {
		if = {
			limit = { is_country_alive = yes }
			trigger_event = { id = sb_griqualand_west.260 days = 14 popup = yes }
		}
	}
}

sb_phase3_finalize_kimberley_ora_route = {
	c:WBL ?= {
		if = {
			limit = {
				is_country_alive = yes
				NOT = { is_subject_of = c:ORA }
			}
			c:ORA ?= {
				if = {
					limit = {
						is_country_alive = yes
						c:WBL ?= { is_subject = yes }
					}
					transfer_subject = c:WBL
				}
				else_if = {
					limit = { is_country_alive = yes }
					create_diplomatic_pact = {
						country = c:WBL
						type = puppet
					}
				}
			}
		}
		if = {
			limit = {
				is_country_alive = yes
				is_subject_of = c:ORA
				NOT = { is_subject_type = subject_type_puppet }
			}
			change_subject_type = subject_type_puppet
		}
	}
	sb_phase3_seed_remaining_kimberley_potential = yes
	p:x68B5E8.state = {
		force_resource_discovery = building_gold_field
	}
	sb_phase3_mark_griqualand_west_resolved = yes
	sb_phase3_clear_griqualand_west_crisis_vars = yes
	c:WBL ?= {
		if = {
			limit = { is_country_alive = yes }
			trigger_event = { id = sb_griqualand_west.260 days = 14 popup = yes }
		}
	}
}

sb_phase3_annul_wbl_guarantee_frame = {
	c:GBR ?= {
		every_scope_treaty = {
			limit = {
				binds = c:WBL
				any_scope_article = {
					has_type = guarantee_independence
					target_country = c:WBL
				}
			}
			withdraw = {
				country = c:GBR
			}
		}
	}
	every_scope_treaty = {
		limit = {
			binds = c:WBL
			any_scope_article = {
				has_type = guarantee_independence
				target_country = c:WBL
			}
		}
		withdraw = {
			country = root
		}
	}
}

sb_phase3_prepare_wbl_cap_protection_demand = {
	sb_phase3_annul_wbl_guarantee_frame = yes
	c:WBL ?= {
		if = { limit = { has_variable = sb_phase3_wbl_ora_protection_demand_var } remove_variable = sb_phase3_wbl_ora_protection_demand_var }
		set_variable = sb_phase3_wbl_cap_protection_demand_var
		trigger_event = { id = sb_griqualand_west.025 days = 7 popup = yes }

}

sb_phase3_create_stellaland_goshen = {
	if = {
		limit = {
			NOT = { exists = c:SGO }
			s:STATE_NORTHERN_CAPE = {
				any_scope_state = { owner = root }
			}
		}
		random_scope_state = {
			limit = {
				state_region = s:STATE_NORTHERN_CAPE
				owner = root
			}
			save_scope_as = sb_phase3_sgo_colony_state
		}
		if = {
			limit = { exists = scope:sb_phase3_sgo_colony_state }
			create_country = {
				tag = SGO
				origin = root
				state = scope:sb_phase3_sgo_colony_state
				on_created = {
					set_country_type = unrecognized
					s:STATE_NORTHERN_CAPE = { add_claim = c:SGO }
					s:STATE_BOTSWANA = { add_claim = c:SGO }
				}
			}
		}
	}
}

sb_phase3_create_gle_route = {
	set_variable = sb_griqualand_east_resolved_var

	if = {
		limit = {
			NOT = { exists = c:GLE }
		}
		create_country = {
			tag = GLE
			origin = root
			province = p:x7524D4
		}
	}

	c:GLE ?= {
		set_country_type = unrecognized
		if = {
			limit = {
				NOT = { is_subject_of = root }
			}
			root = {
				create_diplomatic_pact = {
					country = c:GLE
					type = puppet
				}
			}
		}
		if = {
			limit = {
				c:XHG ?= { is_country_alive = yes }
			}
			s:STATE_EASTERN_CAPE = {
				set_owner_of_provinces = {
					country = c:GLE
					provinces = { x7524D4 x678D6E }
				}
			}
		}
		if = {
			limit = {
				c:XHR ?= { is_country_alive = yes }
			}
			annex = c:XHR
		}
		if = {
			limit = {
				c:XHT ?= { is_country_alive = yes }
			}
			annex = c:XHT
		}
		change_relations = {
			country = root
			value = 30
		}
	}

	p:xC5B1AF.state.owner = {
		save_scope_as = griqua_migration_source
	}
	cu:griqua = {
		save_scope_as = griqua_migration_culture
	}
	c:GLE ?= {
		random_scope_state = {
			limit = {
				state_region = s:STATE_EASTERN_CAPE
			}
			save_scope_as = griqua_migration_target
		}
	}
	scope:griqua_migration_target ?= {
		create_mass_migration = {
			origin = scope:griqua_migration_source
			culture = scope:griqua_migration_culture
		}
	}
	scope:griqua_migration_source ?= {
		random_scope_state = {
			limit = {
				state_region = s:STATE_VRYSTAAT
			}
			every_scope_pop = {
				limit = {
					culture = cu:griqua
				}
				move_partial_pop = {
					state = scope:griqua_migration_target
					population_ratio = 0.20
				}
			}
		}
	}

	if = {
		limit = {
			sb_cap_is_full_liberal_griqua_route = yes
			c:GLE ?= {
				is_country_alive = yes
			}
		}
		c:GLE ?= {
			set_variable = sb_gle_willing_annexation_pending_var
			trigger_event = { id = sb_griqualand_east.221 days = 1825 popup = yes }
		}
	}
}
