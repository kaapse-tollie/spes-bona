sb_court_calculate_backing = {

  set_variable = { name = sb_court_royal_political_weight_var value = 0 }
  set_variable = { name = sb_court_chiefly_political_weight_var value = 0 }
  every_in_list = {
   variable = sb_court_members

  }
  every_in_list = {
   variable = sb_court_members
   limit = { sb_court_representative_eligible = yes }
   save_scope_as = sb_court_candidate
   if = {
    limit = { var:sb_court_following_var = 0 }
    set_variable = sb_court_representative_var
    if = {
     limit = { OR = { sb_court_royal_candidate = yes is_ruler_of_own_country = yes has_variable = sb_court_claim_renounced_var } }
     set_variable = sb_court_house_leader_var
    }
   }
   else = {
    # These links were cleared above. Rank each house once, before any status
    # penalty changes; all its members receive the same original ordered choice.
    if = {
     limit = { NOT = { has_variable = sb_court_house_leader_scope } }
     scope:sb_court_country = {
      ordered_in_list = {
       variable = sb_court_members
       limit = { var:sb_court_following_var = scope:sb_court_candidate.var:sb_court_following_var sb_court_representative_eligible = yes }
       order_by = sb_court_house_leadership_score
       if = {
        limit = { scope:sb_court_candidate = { NOT = { has_variable = sb_court_house_leader_scope } } }
        set_variable = sb_court_representative_var
        set_variable = sb_court_house_leader_var
        save_scope_as = sb_court_house_head
        scope:sb_court_candidate = { set_variable = { name = sb_court_house_leader_scope value = scope:sb_court_house_head } }
       }
       set_variable = { name = sb_court_house_leader_scope value = scope:sb_court_house_head }
      }
     }
    }
    var:sb_court_house_leader_scope ?= { save_scope_as = sb_court_house_head }
   }
  }
  # The native prominence total now includes the one claimant bonus and status penalty.
  every_in_list = {
   variable = sb_court_members
   limit = { var:sb_court_home_scope ?= scope:sb_court_country }
   if = {
    limit = { scope:sb_court_country = { OR = { has_variable = sb_institutions_active_var sb_chiefs_contest_country = yes } } sb_court_representative_eligible = yes NOT = { has_variable = sb_court_house_leader_var } }

   }
   else_if = { limit = { has_modifier = sb_court_junior_status } remove_modifier = sb_court_junior_status }
  }
  every_in_list = {
   variable = sb_court_members
   limit = { sb_court_representative_eligible = yes }
   if = {
    limit = { var:sb_court_following_var > 0 }
    var:sb_court_house_leader_scope ?= {
     if = {
      limit = { has_variable = sb_court_represented_weight_var }
      change_variable = { name = sb_court_represented_weight_var add = prev.sb_court_member_weight }
      if = {
       limit = { has_variable = sb_court_attraction_weight_var prev = { NOT = { AND = { has_role_of_type = agitator owner ?= scope:sb_court_country } } } }
       change_variable = { name = sb_court_attraction_weight_var add = prev.sb_court_member_weight }
      }
      if = { limit = { NOT = { this = prev } } add_to_variable_list = { name = sb_court_house_members target = prev } }
     }
    }
   }
   else = {
    set_variable = { name = sb_court_represented_weight_var value = sb_court_member_weight }
    if = {
     limit = { NOT = { AND = { has_role_of_type = agitator owner ?= scope:sb_court_country } } }
     set_variable = { name = sb_court_attraction_weight_var value = sb_court_member_weight }
    }
   }
  }
  every_in_list = {
   variable = sb_court_members
   limit = { has_variable = sb_court_representative_var has_variable = sb_court_represented_weight_var sb_court_member_of_country = yes }
   sb_court_count_representative = yes
  }
  sb_court_calculate_coalitions = yes
  set_variable = { name = sb_prerogative_cooperation_var value = sb_court_cooperation_value }
  every_political_movement = {
   limit = { OR = { is_political_movement_type = movement_sb_zulu_royal is_political_movement_type = movement_sb_zulu_chiefly } }
   if = { limit = { has_modifier = sb_court_following_recruitment } remove_modifier = sb_court_following_recruitment }
   if = {
    limit = { scope:sb_court_country = { OR = { has_variable = sb_institutions_active_var sb_chiefs_contest_country = yes } } }
    if = { limit = { is_political_movement_type = movement_sb_zulu_royal owner.sb_court_royal_following_attraction > 0 } add_modifier = { name = sb_court_following_recruitment multiplier = owner.sb_court_royal_following_attraction } }
    else_if = { limit = { is_political_movement_type = movement_sb_zulu_chiefly owner.sb_court_chiefly_following_attraction > 0 } add_modifier = { name = sb_court_following_recruitment multiplier = owner.sb_court_chiefly_following_attraction } }
   }
  }
  if = { limit = { sb_chiefs_contest_country = yes } sb_chiefs_update_prominence_pressure = yes }
 }

 save_scope_as = sb_court_country
 sb_court_ensure_roster = yes
 heir ?= { if = { limit = { NOT = { has_variable = sb_court_registered_var } } sb_court_register_character = { FOLLOWING = 0 ALIGNMENT = 1 ENDORSEMENT = 0 } } }
 clear_saved_scope = sb_court_displaced_ruler
 clear_saved_scope = sb_court_displaced_heir
 if = { limit = { var:sb_court_minor_sovereign_scope ?= { is_character_alive = yes } }
  var:sb_court_minor_sovereign_scope = { save_scope_as = sb_court_displaced_ruler }
 }
 else_if = { limit = { NOT = { var:sb_chiefs_release_placeholder_scope ?= ruler } }
  ruler ?= { save_scope_as = sb_court_displaced_ruler }
 }
 heir ?= { save_scope_as = sb_court_displaced_heir }

 set_variable = sb_court_partition_setup_var
 every_in_list = {
  variable = sb_court_members
  limit = { sb_court_adult_member = yes var:sb_court_endorsement_var = 1 }
  if = {
   limit = { var:sb_court_alignment_var = 1 exists = scope:sb_court_displaced_ruler }
   set_variable = { name = sb_court_endorsement_var value = -1 }
   set_variable = { name = sb_court_claimant_scope value = scope:sb_court_displaced_ruler }
  }
  else = { set_variable = { name = sb_court_endorsement_var value = 0 } }
 }
}

# Preserve a recognized heir before selecting another living member of the existing royal cast.

  sb_chiefs_choose_family_claimant = yes

  limit = {
   OR = { has_variable = sb_zululand_usuthu_context_var sb_chiefs_contest_country = yes }
   NOT = { var:sb_court_restoration_claimant_scope ?= { sb_court_royal_candidate = yes } }
  }

  if = {
   limit = { var:sb_court_recognized_heir_scope ?= { sb_court_royal_candidate = yes } }
   set_variable = { name = sb_court_restoration_claimant_scope value = var:sb_court_recognized_heir_scope }
  }

   ordered_in_list = {
    variable = sb_court_members limit = { sb_court_royal_candidate = yes }
    order_by = prominence position = 0
    scope:sb_court_country = { set_variable = { name = sb_court_restoration_claimant_scope value = prev } }
   }
  }
 }
}
