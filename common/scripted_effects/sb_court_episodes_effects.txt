# Independent shuffle bag. A country timer expires while delivery is suspended.
# Saved card/delay survive exit, save/load and changed participant eligibility.
sb_court_deck_available = {
 set_variable = { name = sb_court_deck_available_var value = 0 }
 # Preview traversal skips the initialisation; do not read or increment an absent counter.
 if = {
  limit = { has_variable = sb_court_deck_available_var }
  if = { limit = { sb_court_card_210_eligible = yes NOT = { has_variable = sb_court_card_210_used_var } } change_variable = { name = sb_court_deck_available_var add = 1 } }
  if = { limit = { sb_court_card_211_eligible = yes NOT = { has_variable = sb_court_card_211_used_var } } change_variable = { name = sb_court_deck_available_var add = 1 } }
  if = { limit = { sb_court_card_212_eligible = yes NOT = { has_variable = sb_court_card_212_used_var } } change_variable = { name = sb_court_deck_available_var add = 1 } }
  if = { limit = { sb_court_card_213_eligible = yes NOT = { has_variable = sb_court_card_213_used_var } } change_variable = { name = sb_court_deck_available_var add = 1 } }
  if = { limit = { sb_court_card_214_eligible = yes NOT = { has_variable = sb_court_card_214_used_var } } change_variable = { name = sb_court_deck_available_var add = 1 } }
  if = { limit = { sb_court_card_215_eligible = yes NOT = { has_variable = sb_court_card_215_used_var } } change_variable = { name = sb_court_deck_available_var add = 1 } }
 }
}
sb_court_deck_draw = {
 sb_court_deck_available = yes
 if = {
  limit = { has_variable = sb_court_deck_available_var }
  if = {
   limit = { var:sb_court_deck_available_var = 0 }
   remove_variable = sb_court_card_210_used_var
   remove_variable = sb_court_card_211_used_var
   remove_variable = sb_court_card_212_used_var
   remove_variable = sb_court_card_213_used_var
   remove_variable = sb_court_card_214_used_var
   remove_variable = sb_court_card_215_used_var
   sb_court_deck_available = yes
  }
  if = {
   limit = { var:sb_court_deck_available_var > 0 }
   random_list = {
    1 = { trigger = { sb_court_card_210_eligible = yes NOT = { has_variable = sb_court_card_210_used_var } OR = { NOT = { var:sb_court_deck_last_var ?= 210 } var:sb_court_deck_available_var = 1 } } set_variable = { name = sb_court_deck_card_var value = 210 } }
    1 = { trigger = { sb_court_card_211_eligible = yes NOT = { has_variable = sb_court_card_211_used_var } OR = { NOT = { var:sb_court_deck_last_var ?= 211 } var:sb_court_deck_available_var = 1 } } set_variable = { name = sb_court_deck_card_var value = 211 } }
    1 = { trigger = { sb_court_card_212_eligible = yes NOT = { has_variable = sb_court_card_212_used_var } OR = { NOT = { var:sb_court_deck_last_var ?= 212 } var:sb_court_deck_available_var = 1 } } set_variable = { name = sb_court_deck_card_var value = 212 } }
    1 = { trigger = { sb_court_card_213_eligible = yes NOT = { has_variable = sb_court_card_213_used_var } OR = { NOT = { var:sb_court_deck_last_var ?= 213 } var:sb_court_deck_available_var = 1 } } set_variable = { name = sb_court_deck_card_var value = 213 } }
    1 = { trigger = { sb_court_card_214_eligible = yes NOT = { has_variable = sb_court_card_214_used_var } OR = { NOT = { var:sb_court_deck_last_var ?= 214 } var:sb_court_deck_available_var = 1 } } set_variable = { name = sb_court_deck_card_var value = 214 } }

   }
  }
  remove_variable = sb_court_deck_available_var
 }
}

sb_court_deck_monthly = {
 save_scope_as = sb_court_country
 # Event expiry selects its default option. Recover a lost popup after its lease.
 if = { limit = { has_variable = sb_court_episode_open_var NOT = { has_variable = sb_court_episode_lease_var } } sb_court_episode_cleanup = yes }
 if = {
  limit = { NOT = { has_variable = sb_court_deck_started_var } }
  sb_court_deck_draw = yes sb_court_deck_delay = yes
 }
 if = {
  limit = {
   has_variable = sb_institutions_active_var exists = ruler
   NOR = { has_variable = sb_court_deck_wait_var has_variable = sb_court_episode_open_var has_variable = sb_court_action_open_var has_variable = sb_court_accession_open_var has_variable = sb_court_dynastic_war_scope has_variable = sb_court_civil_war_var has_variable = sb_court_dynastic_crisis_var }
  }
  if = { limit = { var:sb_court_deck_card_var ?= 210 sb_court_card_210_eligible = no } remove_variable = sb_court_deck_card_var }
  if = { limit = { var:sb_court_deck_card_var ?= 211 sb_court_card_211_eligible = no } remove_variable = sb_court_deck_card_var }
  if = { limit = { var:sb_court_deck_card_var ?= 212 sb_court_card_212_eligible = no } remove_variable = sb_court_deck_card_var }
  if = { limit = { var:sb_court_deck_card_var ?= 213 sb_court_card_213_eligible = no } remove_variable = sb_court_deck_card_var }
  if = { limit = { var:sb_court_deck_card_var ?= 214 sb_court_card_214_eligible = no } remove_variable = sb_court_deck_card_var }
  if = { limit = { var:sb_court_deck_card_var ?= 215 sb_court_card_215_eligible = no } remove_variable = sb_court_deck_card_var }
  if = { limit = { NOT = { has_variable = sb_court_deck_card_var } } sb_court_deck_draw = yes }
  if = {
   limit = { has_variable = sb_court_deck_card_var }
   set_variable = sb_court_episode_open_var
   set_variable = { name = sb_court_episode_lease_var months = 4 }
   set_variable = { name = sb_court_episode_card_var value = var:sb_court_deck_card_var }
   set_variable = { name = sb_court_deck_last_var value = var:sb_court_deck_card_var }
   set_variable = { name = sb_court_episode_ruler_scope value = ruler }
   sb_court_deck_deliver = yes
   remove_variable = sb_court_deck_card_var
  }
 }
}
# Country scope: select once when reserving the episode, using the harvest's
# Popularity curve. Scale both reservoir weights for native integer rounding.
sb_court_pick_episode_prince = {
 save_scope_as = sb_court_country
 remove_variable = $DESTINATION$
 set_variable = { name = sb_court_episode_draw_weight_var value = 0 }
 every_in_list = {
  variable = sb_court_members
  limit = {
   sb_court_episode_prince = yes
   trigger_if = { limit = { scope:sb_court_country = { var:sb_court_deck_card_var ?= 214 } } sb_court_current_opposition_prince = yes }
   sb_court_patron_attraction > 0
   trigger_if = {
    limit = { always = $EXCLUDE_FIRST$ exists = scope:sb_court_country.var:sb_court_episode_candidate_scope }
    NOT = { this = scope:sb_court_country.var:sb_court_episode_candidate_scope }
   }
  }
  save_scope_as = sb_court_episode_draw_prince
  random_list = {
   0 = {
    modifier = { add = { value = sb_court_patron_attraction multiply = 100000 } }
    scope:sb_court_country = { set_variable = { name = $DESTINATION$ value = scope:sb_court_episode_draw_prince } }
   }
   0 = { modifier = { add = { value = scope:sb_court_country.var:sb_court_episode_draw_weight_var multiply = 100000 } } }
  }
  scope:sb_court_country = { change_variable = { name = sb_court_episode_draw_weight_var add = prev.sb_court_patron_attraction } }
 }
 # At -100 Popularity every weight is zero; retain a usable choice among the
 # eligible princes rather than opening an episode with a missing participant.
 if = {
  limit = { NOT = { has_variable = $DESTINATION$ } }
  random_in_list = {
   variable = sb_court_members
   limit = {
    sb_court_episode_prince = yes
   trigger_if = { limit = { scope:sb_court_country = { var:sb_court_deck_card_var ?= 214 } } sb_court_current_opposition_prince = yes }
    trigger_if = {
     limit = { always = $EXCLUDE_FIRST$ exists = scope:sb_court_country.var:sb_court_episode_candidate_scope }
     NOT = { this = scope:sb_court_country.var:sb_court_episode_candidate_scope }
    }
   }
   scope:sb_court_country = { set_variable = { name = $DESTINATION$ value = prev } }
  }
 }
 remove_variable = sb_court_episode_draw_weight_var
}
