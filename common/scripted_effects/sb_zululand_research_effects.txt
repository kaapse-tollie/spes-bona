# Keep the last main ZUL's completed inventions across annexation, delayed handover and reload.
sb_zululand_refresh_research_archive = {
	if = {
		limit = {
			this = c:ZUL
			is_country_alive = yes
			NOT = { has_global_variable = sb_zululand_release_law_source }
			NOT = { c:GBR ?= { has_variable = sb_zululand_postwar_handoff_resolved_var } }
			OR = {
				NOT = { has_global_variable = sb_zululand_archived_tech_count }
				NOT = { global_var:sb_zululand_archived_tech_country ?= this }
				NOT = { techs_researched = global_var:sb_zululand_archived_tech_count }
			}
		}
		if = { limit = { has_global_variable_list = sb_zululand_archived_technologies } clear_global_variable_list = sb_zululand_archived_technologies }
		set_global_variable = { name = sb_zululand_archived_tech_country value = this }
		set_global_variable = { name = sb_zululand_archived_tech_count value = techs_researched }
		sb_zululand_for_each_technology = { ACTION = sb_zululand_record_technology }
	}
}

sb_zululand_record_technology = {
	if = {
		limit = { has_technology_researched = $TECH$ }
		add_to_global_variable_list = { name = sb_zululand_archived_technologies target = flag:$TECH$ }
	}
}

sb_zululand_restore_technology = {
	if = {
		limit = { is_target_in_global_variable_list = { name = sb_zululand_archived_technologies target = flag:$TECH$ } }
		add_technology_researched = $TECH$
	}
}

sb_zululand_clear_research_archive = {
	if = { limit = { has_global_variable_list = sb_zululand_archived_technologies } clear_global_variable_list = sb_zululand_archived_technologies }
	remove_global_variable = sb_zululand_archived_tech_count
	remove_global_variable = sb_zululand_archived_tech_country
}
