			country_definition = cd:GER

		}
		c:SWA ?= {
			if = {
				limit = { game_date < 1891.1.1 }
				if = {
					limit = {
						NOT = { any_scope_character = { has_template = SWA_heinrich_ernst_goring } }
					}
					create_character = { template = SWA_heinrich_ernst_goring }

sb_nam_finalize_coastal_expedition_success = {
	if = {
		limit = {
			p:xDEAA3C.state.owner = { is_country_type = decentralized }
		}
		s:STATE_NAMAQUALAND = {
			set_owner_of_provinces = {
				country = root
				provinces = { xDEAA3C }
			}
			add_claim = root
		}
		set_variable = sb_nam_coast_secured_luderitz_var
		random_scope_state = {
			limit = { state_region = s:STATE_NAMAQUALAND }
			if = {
				limit = { NOT = { has_building = building_port } }
				create_building = {
					building = building_port
					level = 1
					reserves = 1
				}
			}
			if = {
				limit = { NOT = { has_building = building_trade_center } }
				create_building = {
					building = building_trade_center
					level = 1
					reserves = 1
					activate_production_methods = { "pm_trade_center_trade_quantity_normal" }
				}
			}
		}
	}
	if = {
		limit = {
			p:xB04050.state.owner = { is_country_type = decentralized }
		}
		s:STATE_HEREROLAND = {
			set_owner_of_provinces = {
				country = root
				provinces = { xB04050 }
			}
			add_claim = root
		}
		set_variable = sb_nam_coast_secured_swakopmund_var
		random_scope_state = {
			limit = { state_region = s:STATE_HEREROLAND }
			if = {
				limit = { NOT = { has_building = building_port } }
				create_building = {
					building = building_port
					level = 1
					reserves = 1
				}
			}
			if = {
				limit = { NOT = { has_building = building_trade_center } }
				create_building = {
					building = building_trade_center
					level = 1
					reserves = 1
					activate_production_methods = { "pm_trade_center_trade_quantity_normal" }
				}
			}
		}
	}
	if = {
		limit = {
			OR = {
				has_variable = sb_nam_coast_secured_luderitz_var
				has_variable = sb_nam_coast_secured_swakopmund_var
			}
		}
		if = {
			limit = { NOT = { has_technology_researched = sb_namibian_coastal_access } }
			add_technology_researched = sb_namibian_coastal_access
		}
		# The mapping claim is retained even after administration passes to SWA,
		# permanently sustaining Engaged status. Keep the immediate grant modest.
		set_variable = sb_nam_mapping_permanent_engagement_var
		add_involvement = {
			strategic_region = sr:region_southern_africa
			value = 1100
		}
		set_global_variable = sb_namibian_coast_race_closed_var
	}
	remove_variable = sb_nam_coast_secured_luderitz_var
	remove_variable = sb_nam_coast_secured_swakopmund_var
}

sb_nam_finalize_swa_administration = {
	create_diplomatic_pact = {
		country = c:SWA
		type = colony
	}
	c:SWA ?= {
		remove_primary_culture = cu:british
	}
	every_primary_culture = {
		limit = {
			has_discrimination_trait_group = heritage_group_european
			NOT = { is_primary_culture_of = c:SWA }
		}
		c:SWA ?= {
			add_primary_culture = prev
		}
	}
	transfer_state = {
		STATE = s:STATE_NAMAQUALAND
		GIVER = root
		TAKER = c:SWA
	}
	s:STATE_NAMAQUALAND = {
		region_state:SWA ?= {
			set_state_type = incorporated
		}
	}
	transfer_state = {
		STATE = s:STATE_HEREROLAND
		GIVER = root
		TAKER = c:SWA
	}
	# Mapping winners retain their expedition claims after delegating government;
	# either permanent claim keeps their Southern African interest Engaged.
	if = {
		limit = {
			has_claim = s:STATE_NAMAQUALAND
			NOT = { has_variable = sb_nam_mapping_permanent_engagement_var }
		}
		s:STATE_NAMAQUALAND = {
			remove_claim = root
		}
	}
	if = {
		limit = {
			has_claim = s:STATE_HEREROLAND
			NOT = { has_variable = sb_nam_mapping_permanent_engagement_var }
		}
		s:STATE_HEREROLAND = {
			remove_claim = root
		}
	}
	if = {
		limit = {
			c:SWA ?= {
				NOT = { has_claim = s:STATE_NAMAQUALAND }
			}
		}
		s:STATE_NAMAQUALAND = {
			add_claim = c:SWA
		}
	}
	if = {
		limit = {
			c:SWA ?= {
				NOT = { has_claim = s:STATE_HEREROLAND }
			}
		}
		s:STATE_HEREROLAND = {
			add_claim = c:SWA
		}
	}
	sb_nam_assign_german_swa_ruler = yes
	c:SWA ?= {
		sb_nam_copy_final_harshness_flags_from_root = yes
		sb_nam_apply_swa_administration_package = yes
		trigger_event = { id = sb_nam.150 days = 1 popup = yes }
	}
	sb_nam_offer_swa_play_as_to_player_overlord = yes
	sb_nam_clear_final_harshness_flags = yes
	sb_nam_clear_administration_package_flags = yes
}

sb_nam_sync_consolidation_modifiers = {
	sb_nam_sync_consolidation_harshness_value = yes
	set_variable = { name = sb_nam_consolidation_target_tier_var value = 4 }
	if = {
		limit = { global_var:sb_nam_rule_harshness_current_var < 2 }
		set_variable = { name = sb_nam_consolidation_target_tier_var value = 0 }
	}
	else_if = {
		limit = { global_var:sb_nam_rule_harshness_current_var < 5 }
		set_variable = { name = sb_nam_consolidation_target_tier_var value = 1 }
	}
	else_if = {
		limit = { global_var:sb_nam_rule_harshness_current_var < 7 }
		set_variable = { name = sb_nam_consolidation_target_tier_var value = 2 }
	}
	else_if = {
		limit = { global_var:sb_nam_rule_harshness_current_var < 10 }
		set_variable = { name = sb_nam_consolidation_target_tier_var value = 3 }
	}
	if = {
		limit = {
			OR = {
				NOT = { has_variable = sb_nam_consolidation_applied_tier_var }
				NOT = { var:sb_nam_consolidation_applied_tier_var = var:sb_nam_consolidation_target_tier_var }
			}
		}
		sb_nam_clear_consolidation_modifiers = yes
	if = {
		limit = {
			global_var:sb_nam_rule_harshness_current_var < 2
		}
		add_modifier = {
			name = sb_nam_rule_accommodating
			months = -1
		}
	}
	else_if = {
		limit = {
			global_var:sb_nam_rule_harshness_current_var < 5
		}
		add_modifier = {
			name = sb_nam_rule_supervised
			months = -1
		}
		sb_nam_add_supervised_state_modifier = yes
	}
	else_if = {
		limit = {
			global_var:sb_nam_rule_harshness_current_var < 7
		}
		add_modifier = {
			name = sb_nam_rule_coercive
			months = -1
		}
		sb_nam_add_coercive_state_modifier = yes
	}
	else_if = {
		limit = {
			global_var:sb_nam_rule_harshness_current_var < 10
		}
		add_modifier = {
			name = sb_nam_rule_brutal
			months = -1
		}
		sb_nam_add_brutal_state_modifier = yes
	}
	else = {
		add_modifier = {
			name = sb_nam_rule_exterminatory
			months = -1
		}
		sb_nam_add_exterminatory_state_modifier = yes
	}
		set_variable = {
			name = sb_nam_consolidation_applied_tier_var
			value = var:sb_nam_consolidation_target_tier_var
		}
	}
	remove_variable = sb_nam_consolidation_target_tier_var
}
