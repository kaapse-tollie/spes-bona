# Shared subject-autonomy helpers for mid-century constitutional content.

sb_apply_responsible_colony_subject_type_from_overlord = {
	# Non-British monarchical grants use a dynastic responsible-colony variant
	# that shares the overlord monarch as head of state. This keeps the colony
	# inside the parent dynasty without inventing a cadet branch in script.
	if = {
		limit = {
			is_subject_of = c:GBR
		}
		change_subject_type = subject_type_sb_responsible_colony
	}
	else_if = {
		limit = {
			overlord = {
				has_law_or_variant = law_type:law_monarchy
				NOT = { country_definition = cd:GBR }
			}
		}
		change_subject_type = subject_type_sb_responsible_colony_monarchy
	}
	else = {
		change_subject_type = subject_type_sb_responsible_colony
	}
}

sb_apply_responsible_colony_governance_from_overlord = {
	# Responsible government should move the subject out of the colonial
	# administration family and into the ordinary governance-principles family.
	# British responsible colonies use parliamentary republic so Victoria does
	# not become their head of state; other monarchies remain monarchies.
	if = {
		limit = {
			is_subject_of = c:GBR
		}
		activate_law = law_type:law_parliamentary_republic
	}
	else_if = {
		limit = {
			overlord = {
				has_law_or_variant = law_type:law_council_republic
			}
		}
		activate_law = law_type:law_council_republic
	}
	else_if = {
		limit = {
			overlord = {
				has_law_or_variant = law_type:law_theocracy
			}
		}
		activate_law = law_type:law_theocracy
	}
	else_if = {
		limit = {
			overlord = {
				has_law_or_variant = law_type:law_presidential_republic
			}
		}
		activate_law = law_type:law_presidential_republic
	}
	else_if = {
		limit = {
			overlord = {
				has_law_or_variant = law_type:law_parliamentary_republic
			}
		}
		activate_law = law_type:law_parliamentary_republic
	}
	else_if = {
		limit = {
			overlord = {
				country_definition = cd:GBR
				has_law_or_variant = law_type:law_monarchy
			}
		}
		activate_law = law_type:law_parliamentary_republic
	}
	else_if = {
		limit = {
			overlord = {
				has_law_or_variant = law_type:law_monarchy
			}
		}
		activate_law = law_type:law_monarchy
	}
}

sb_grant_responsible_government_effect = {
	if = {
		limit = {
			is_subject_type = subject_type_colony
		}
		sb_apply_responsible_colony_subject_type_from_overlord = yes
		sb_apply_responsible_colony_governance_from_overlord = yes
	}

	add_liberty_desire = -50
	change_relations = {
		country = scope:actor
		value = 20
	}
}

sb_refresh_cap_aby_subject_state = {
	if = {
		limit = {
			country_definition = cd:CAP
			is_subject = yes
			OR = {
				is_subject_type = subject_type_dominion
				is_subject_type = subject_type_sb_dominion
			}
			NOT = { has_variable = sb_cap_dominion_map_color_var }
		}
		set_variable = sb_cap_dominion_map_color_var
	}
	else_if = {
		limit = {
			country_definition = cd:CAP
			has_variable = sb_cap_dominion_map_color_var
			OR = {
				is_subject = no
				NOT = {
					OR = {
						is_subject_type = subject_type_dominion
						is_subject_type = subject_type_sb_dominion
					}
				}
			}
		}
		remove_variable = sb_cap_dominion_map_color_var
	}
	if = {
		limit = {
			country_definition = cd:ABY
			is_subject = yes
			OR = {
				is_subject_type = subject_type_dominion
				is_subject_type = subject_type_sb_dominion
			}
			NOT = { has_variable = sb_aby_dominion_map_color_var }
		}
		set_variable = sb_aby_dominion_map_color_var
	}
	else_if = {
		limit = {
			country_definition = cd:ABY
			has_variable = sb_aby_dominion_map_color_var
			OR = {
				is_subject = no
				NOT = {
					OR = {
						is_subject_type = subject_type_dominion
						is_subject_type = subject_type_sb_dominion
					}
				}
			}
		}
		remove_variable = sb_aby_dominion_map_color_var
	}
	if = {
		limit = { is_subject_type = subject_type_dominion }
		change_subject_type = subject_type_sb_dominion
	}
	if = {
		limit = {
			country_definition = cd:CAP
			has_modifier = sb_cape_autonomy_pressure
			NOT = { has_variable = sb_cape_autonomy_pressure_resolution_pending_var }
			OR = {
				NOT = { is_subject_of = c:GBR }
				is_subject_type = subject_type_dominion
				is_subject_type = subject_type_sb_dominion
				is_subject_type = subject_type_sb_responsible_colony
				is_subject_type = subject_type_sb_responsible_colony_monarchy
			}
		}
		set_variable = sb_cape_autonomy_pressure_resolution_pending_var
		trigger_event = { id = sb_cape.031 days = 1 popup = yes }
	}
	if = {
		limit = {
			country_definition = cd:CAP
			has_variable = sb_cape_autonomy_pressure_pending_var
			OR = {
				NOT = { is_subject_of = c:GBR }
				is_subject_type = subject_type_dominion
				is_subject_type = subject_type_sb_dominion
				is_subject_type = subject_type_sb_responsible_colony
				is_subject_type = subject_type_sb_responsible_colony_monarchy
			}
		}
		remove_variable = sb_cape_autonomy_pressure_pending_var
	}
	if = {
		limit = {
			is_subject = yes
			NOT = { is_country_type = colonial }
		}
		set_country_type = colonial
	}
	if = {
		limit = {
			is_subject = no
			is_country_type = colonial
		}
		set_country_type = recognized
	}
	if = {
		limit = {
			country_definition = cd:CAP
			is_subject = no
			has_variable = southern_africa_subject_var
		}
		remove_variable = southern_africa_subject_var
	}
}
