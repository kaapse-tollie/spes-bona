# Spes Bona - technology-gated resource potential
#
# Gold and oil stages add undiscovered potential and deliberately leave discovery
# to the standard engine. Other stages add usable capacity immediately and post a
# one-time owner notification. Global flags make each geographical stage idempotent.

		limit = {
			has_technology_researched = nitroglycerin
			any_scope_state = { state_region = s:STATE_BECHUANALAND }

		}
		set_global_variable = sb_resource_bechuanaland_gold_nitroglycerin_added_var
