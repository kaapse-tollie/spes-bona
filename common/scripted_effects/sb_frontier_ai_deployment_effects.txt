# Spes Bona - AI frontier deployment helpers
#
# These effects are intentionally narrow. They mobilize AI Transvaal in scripted
# frontier wars; direct teleporting is avoided because dynamic fronts can reject
# repeat deploy orders or lack a valid base camp in some runtime states.

sb_trn_deploy_armies_to_phl_front = {
	if = {
		limit = {
			country_definition = cd:TRN
			is_ai = yes
			has_war_with = c:PHL
		}
		every_military_formation = {
			limit = { is_army = yes }
			fully_mobilize_army = yes
		}
	}
}

sb_trn_deploy_armies_to_active_frontier_war = {
	sb_trn_deploy_armies_to_phl_front = yes
	sb_trn_deploy_armies_to_bst_front = yes
	sb_trn_deploy_armies_to_swz_front = yes

}

sb_schedule_trn_frontier_deployment_for_play = {
	if = {
		limit = {
			NOT = { has_variable = sb_trn_frontier_deployment_train_active_var }
			sb_frontier_play_artificial_assistance_enabled = { RECIPIENT = c:TRN }
			any_scope_play_involved = {
				OR = {
					country_definition = cd:PHL
					country_definition = cd:BST
					country_definition = cd:SWZ
					country_definition = cd:ZUL
				}
				is_diplomatic_play_committed_participant = yes
				is_diplomatic_play_enemy_of = c:TRN
			}
		}
		set_variable = sb_trn_frontier_deployment_train_active_var
		save_scope_as = sb_trn_frontier_deployment_play_scope
		c:TRN ?= {
			trigger_event = { id = sb_frontier_ai_wars.900 days = 1 }
			trigger_event = { id = sb_frontier_ai_wars.900 days = 7 }
			trigger_event = { id = sb_frontier_ai_wars.900 days = 21 }
			trigger_event = { id = sb_frontier_ai_wars.900 days = 45 }
			trigger_event = { id = sb_frontier_ai_wars.900 days = 90 }
			trigger_event = { id = sb_frontier_ai_wars.900 days = 180 }
		}
	}
}

sb_cleanup_trn_frontier_deployment_train_for_play = {
	if = {
		limit = { has_variable = sb_trn_frontier_deployment_train_active_var }
		remove_variable = sb_trn_frontier_deployment_train_active_var
	}
}
