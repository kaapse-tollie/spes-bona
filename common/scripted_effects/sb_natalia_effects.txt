# Shared scripted effects for the Natalia republican / Blood River chain.
#
# NAL can now appear either as:
#   - a Boer republic created from ORA's Blood River victory
#   - a British-organized crown colony after the later coast crisis
#
# V2 note:
#   Blood River itself is ORA-only in fresh starts. Some TRN-related helpers are
#   intentionally retained below as compatibility fallbacks for older saves or
#   partial branches that still have a live TRN participant.
#
# File split:
#   - this file owns republic creation, Blood River support, and Natalia's Boer
#     founding path
#   - British submission / colony conversion / refugee fallout lives in
#     `sb_natalia_colony_effects.txt`

sb_create_natalia_republic_if_missing = {
	if = {
		limit = {
			NOT = { exists = c:NAL }
		}
		# Country creation generates its default political cast from the seed state.
		# Stage the Boer founders there before NAL exists so their appearance follows
		# the republic's actual settler population rather than the Zulu majority.
		p:x5B124F.state = {
			save_temporary_scope_as = sb_natalia_boer_founder_seed_state
		}
		hidden_effect = {
			s:STATE_CAPE_COLONY = {
				region_state:CAP ?= {
					every_scope_pop = {
						limit = {
							culture = cu:boer
						}
						move_partial_pop = {
							state = scope:sb_natalia_boer_founder_seed_state
							population_ratio = 0.05
						}
					}
				}
			}
		}
		create_country = {
			tag = NAL
			origin = root
			province = p:x5B124F
		}
	}
}

sb_assign_natalia_republic_territory = {
	s:STATE_NATAL = {
		set_owner_of_provinces = {
			country = c:NAL
			# Restore the full former-NGI interior after the Blood River settlement.
			# Port Natal remains with CAP and xBBCA32 remains with ZUL.
			provinces = {
				x5B124F xFF0EF1 x552449 xE0EB02 x85695F xDE0EDE
				x7ACC38 xB1F868 x3CED3D x11A090 xCD31DB
			}
		}
		add_claim = c:NAL
	}
}

# Blood River must not hand Zulu levy infrastructure to the new republic.
# Removing the inherited conscription center also removes any attached levy
# units before Natalia receives the standard two-unit Boer commando.
sb_apply_natalia_blood_river_military_setup = {
	every_scope_state = {
		limit = {
			state_region = s:STATE_NATAL
			any_scope_building = {
				is_building_type = building_conscription_center
			}
		}
		remove_building = building_conscription_center
	}
	create_military_formation = {
		type = army
		hq_region = sr:region_southern_africa
		name = natalia_commando
		combat_unit = {
			type = unit_type:combat_unit_type_irregular_infantry
			state_region = s:STATE_NATAL
			count = 1
		}
		combat_unit = {
			type = unit_type:combat_unit_type_dragoons
			state_region = s:STATE_NATAL
			count = 1
		}
	}
}

sb_apply_natalia_boer_republic_setup = {
	c:ORA ?= {
		if = {
			limit = { has_variable = sb_voortrekker_start_var }
			remove_variable = sb_voortrekker_start_var
		}
	}
	if = {
		limit = { has_variable = southern_africa_subject_var }
		remove_variable = southern_africa_subject_var
	}
	if = {
		limit = { has_variable = sb_natalia_british_colony_resolved_var }
		remove_variable = sb_natalia_british_colony_resolved_var
	}
	if = {
		limit = { has_variable = sb_natalia_ultimatum_truce_consumed_var }
		remove_variable = sb_natalia_ultimatum_truce_consumed_var
	}
	if = {
		limit = {
			c:GBR ?= {
				is_country_alive = yes
			}
			NOT = { has_truce_with = c:GBR }
		}
		create_bidirectional_truce = {
			country = c:GBR
			months = 240
		}
		set_variable = sb_natalia_ultimatum_truce_var
	}
	effect_starting_technology_tier_3_tech = yes
	add_technology_researched = napoleonic_warfare
	set_country_type = unrecognized
	set_next_election_date = 1840.1.1
	if = {
		limit = {
			NOT = { country_has_primary_culture = cu:boer }
		}
		add_primary_culture = cu:boer
	}
	if = {
		limit = {
			country_has_primary_culture = cu:british
		}
		remove_primary_culture = cu:british
	}
	if = {
		limit = {
			country_has_primary_culture = cu:anglo_african
		}
		remove_primary_culture = cu:anglo_african
	}
	effect_starting_politics_conservative = yes
	sb_apply_boer_commandant_general_baseline = yes
	activate_law = law_type:law_national_supremacy
	sb_ensure_boer_supremacist_movement = yes
	activate_law = law_type:law_elected_bureaucrats
	activate_law = law_type:law_agrarianism
	activate_law = law_type:law_land_based_taxation
	activate_law = law_type:law_homesteading
	activate_law = law_type:law_national_militia
	activate_law = law_type:law_discrete_inboekstelsel
	activate_law = law_type:law_frontier_colonization
	s:STATE_NATAL.region_state:NAL ?= {
		if = {
			limit = { NOT = { has_building = building_maize_farm } }
			create_building = {
				building = building_maize_farm
				level = 1
				add_ownership = {
					building = {
						type = building_manor_house
						country = c:NAL
						levels = 1
						region = "STATE_NATAL"
					}
				}
				reserves = 1
				activate_production_methods = { "pm_simple_farming" "pm_no_secondary" "pm_tools_disabled" }
			}
		}
	}
	add_modifier = {
		name = sb_trek_frontier_drive
		months = 120
	}
	sb_apply_boer_republic_interest_group_identity = yes
	if = {
		limit = {
			country_has_primary_culture = cu:boer
			NOT = { has_journal_entry = je_sb_great_trek }
		}
		add_journal_entry = { type = je_sb_great_trek }
	}
	active_law:lawgroup_army_model ?= {
		if = {
			limit = {
				NOT = { has_amendment = amendment_type:amendment_sb_commando_corps }
			}
			add_amendment = {
				type = amendment_sb_commando_corps
				sponsor = PREV.ig:ig_rural_folk
			}
		}
	}
	every_interest_group = {
		remove_ruling_interest_group = yes
	}
	ig:ig_armed_forces ?= { add_ruling_interest_group = yes }
	ig:ig_landowners ?= { add_ruling_interest_group = yes }
	ig:ig_rural_folk ?= { add_ruling_interest_group = yes }
	ig:ig_devout ?= { add_ruling_interest_group = yes }
	sb_refresh_boer_commandant_general_government = yes
	sb_ensure_commandant_general_ruler_command = yes
	sb_ensure_boer_anti_british_lobby = yes
}

# Late-game counterpart to the founding setup. Restore republican Natalia after
# a Boer corridor victory without replaying starting tech, an 1840 election,
# the Great Trek journal, or a new twenty-year British truce.
sb_restore_natalia_after_boer_corridor_victory = {
	random_scope_character = {
		limit = { has_variable = sb_martin_west_var }
		transfer_character = c:GBR
	}
	if = {
		limit = { has_variable = southern_africa_subject_var }
		remove_variable = southern_africa_subject_var
	}
	if = {
		limit = { has_variable = sb_natalia_british_colony_resolved_var }
		remove_variable = sb_natalia_british_colony_resolved_var
	}
	# A side victory can settle before every secondary goal is formally enforced.
	# Never leave the restored republic inside the defeated British hierarchy.
	if = {
		limit = { is_subject_of = c:GBR }
		make_independent = yes
	}
	# Liberation restores an unrecognized republic. If a transfer goal instead
	# placed Natalia under TRN, preserve the validated subject relationship's
	# country type rather than invalidating that peace term after the fact.
	if = {
		limit = { is_subject = no }
		set_country_type = unrecognized
	}
	if = {
		limit = { NOT = { country_has_primary_culture = cu:boer } }
		add_primary_culture = cu:boer
	}
	if = {
		limit = { country_has_primary_culture = cu:british }
		remove_primary_culture = cu:british
	}
	if = {
		limit = { country_has_primary_culture = cu:anglo_african }
		remove_primary_culture = cu:anglo_african
	}
	sb_apply_boer_commandant_general_baseline = yes
	activate_law = law_type:law_national_supremacy
	sb_ensure_boer_supremacist_movement = yes
	activate_law = law_type:law_elected_bureaucrats
	activate_law = law_type:law_agrarianism
	activate_law = law_type:law_land_based_taxation
	activate_law = law_type:law_homesteading
	activate_law = law_type:law_national_militia
	activate_law = law_type:law_discrete_inboekstelsel
	activate_law = law_type:law_frontier_colonization
	add_modifier = {
		name = sb_trek_frontier_drive
		months = 120
	}
	sb_apply_boer_republic_interest_group_identity = yes
	active_law:lawgroup_army_model ?= {
		if = {
			limit = { NOT = { has_amendment = amendment_type:amendment_sb_commando_corps } }
			add_amendment = {
				type = amendment_sb_commando_corps
				sponsor = PREV.ig:ig_rural_folk
			}
		}
	}
	every_interest_group = { remove_ruling_interest_group = yes }
	ig:ig_armed_forces ?= { add_ruling_interest_group = yes }
	ig:ig_landowners ?= { add_ruling_interest_group = yes }
	ig:ig_rural_folk ?= { add_ruling_interest_group = yes }
	ig:ig_devout ?= { add_ruling_interest_group = yes }
	sb_refresh_boer_commandant_general_government = yes
	sb_ensure_commandant_general_ruler_command = yes
	sb_ensure_boer_anti_british_lobby = yes
}

	effect_starting_technology_tier_2_tech = yes

		limit = { NOT = { country_has_primary_culture = cu:anglo_african } }
		add_primary_culture = cu:anglo_african
	}
	if = {
		limit = { country_has_primary_culture = cu:british }

sb_ensure_ora_ai_natal_front_commitment = {
	if = {
		limit = {
			country_definition = cd:ORA
			is_ai = yes
			has_variable = sb_natal_war_active_var
			is_at_war = yes
		}
		if = {
			limit = { NOT = { has_variable = sb_ora_blood_river_force_floor_created_var } }
			set_variable = sb_ora_blood_river_force_floor_created_var
			create_military_formation = {
				type = army
				hq_region = sr:region_southern_africa
				name = ora_blood_river_commando
				combat_unit = {
					type = unit_type:combat_unit_type_dragoons
					state_region = s:STATE_VRYSTAAT
					count = 1
				}
				combat_unit = {
					type = unit_type:combat_unit_type_line_infantry
					state_region = s:STATE_VRYSTAAT
					count = 1
				}
			}
		}
		every_military_formation = {
			limit = {
				is_army = yes
			}
			fully_mobilize_army = yes
		}
	}
}

sb_apply_ora_blood_river_player_laager = {
	if = {
		limit = {
			country_definition = cd:ORA
			is_ai = yes
		}
		if = {
			limit = {
				has_variable = sb_ora_blood_river_laager_active_var
				has_modifier = sb_laager_defence_ai
			}
			remove_modifier = sb_laager_defence_ai
			remove_variable = sb_ora_blood_river_laager_active_var
		}
		if = {
			limit = { NOT = { has_modifier = sb_laager_defence } }
			add_modifier = {
				name = sb_laager_defence
				months = 15
			}
			set_variable = {
				name = sb_ora_blood_river_laager_active_var
				months = 15
			}
		}
	}
}

# Diplomatic-play scope. Only committed participants in this Blood River play
# affect routing. The original AI-AI roll is preserved across later joins.
sb_reconcile_blood_river_assistance_for_play = {
	if = {
		limit = { is_diplomatic_play_type = dp_sb_natal_crisis }
		c:ORA ?= {
			if = {
				limit = {
					is_ai = yes
					root = {
						sb_frontier_play_artificial_assistance_enabled = { RECIPIENT = c:ORA }
					}
				}
				sb_ensure_ora_ai_natal_front_commitment = yes
			}
			if = {
				limit = {
					is_ai = yes
					root = {
						sb_frontier_play_has_committed_player_enemy = { RECIPIENT = c:ORA }
						sb_frontier_player_challenge_enabled = yes
					}
				}
				sb_apply_ora_blood_river_player_laager = yes
			}
			else_if = {
				limit = {
					is_ai = yes
					c:ZUL ?= {
						is_country_alive = yes
						is_ai = yes
						NOT = { has_variable = sb_zul_blood_river_positive_roll_var }
					}
					root = {
						sb_frontier_ai_scripting_enabled = yes
						NOT = {
							sb_frontier_play_has_committed_player_enemy = { RECIPIENT = c:ORA }
						}
						NOT = {
							sb_frontier_play_has_committed_player_enemy = { RECIPIENT = c:ZUL }
						}
					}
				}
				sb_apply_ora_blood_river_ai_laager = yes
			}
			else = {
				sb_clear_ora_blood_river_laager = yes
			}
		}
		c:ZUL ?= {
			if = {
				limit = {
					is_ai = yes
					root = {
						sb_frontier_play_has_committed_player_enemy = { RECIPIENT = c:ZUL }
					}
				}
				if = {
					limit = {
						root = { sb_frontier_player_challenge_enabled = yes }
					}
					if = {
						limit = { has_modifier = sb_blood_river_zulu_no_relief }
						remove_modifier = sb_blood_river_zulu_no_relief
					}
					if = {
						limit = { has_modifier = sb_blood_river_zulu_limited_support }
						remove_modifier = sb_blood_river_zulu_limited_support
					}
					sb_apply_zul_blood_river_readiness = yes
				}
				else = {
					if = {
						limit = { has_modifier = sb_blood_river_zulu_no_relief }
						remove_modifier = sb_blood_river_zulu_no_relief
					}
					if = {
						limit = { has_modifier = sb_blood_river_zulu_limited_support }
						remove_modifier = sb_blood_river_zulu_limited_support
					}
					if = {
						limit = { has_modifier = sb_blood_river_zulu_relief_mobilization }
						remove_modifier = sb_blood_river_zulu_relief_mobilization
					}
				}
			}
			else_if = {
				limit = {
					is_ai = yes
					has_variable = sb_zul_blood_river_positive_roll_var
					root = { sb_frontier_ai_scripting_enabled = yes }
				}
				if = {
					limit = { has_modifier = sb_blood_river_zulu_no_relief }
					remove_modifier = sb_blood_river_zulu_no_relief
				}
				if = {
					limit = { NOT = { has_modifier = sb_blood_river_zulu_limited_support } }
					add_modifier = {
						name = sb_blood_river_zulu_limited_support
						months = 15
					}
				}
				if = {
					limit = { has_modifier = sb_blood_river_zulu_relief_mobilization }
					remove_modifier = sb_blood_river_zulu_relief_mobilization
				}
			}
			else_if = {
				limit = {
					is_ai = yes
					has_variable = sb_zul_blood_river_no_relief_roll_var
					root = { sb_frontier_ai_scripting_enabled = yes }
				}
				if = {
					limit = { has_modifier = sb_blood_river_zulu_limited_support }
					remove_modifier = sb_blood_river_zulu_limited_support
				}
				if = {
					limit = { has_modifier = sb_blood_river_zulu_relief_mobilization }
					remove_modifier = sb_blood_river_zulu_relief_mobilization
				}
				if = {
					limit = { NOT = { has_modifier = sb_blood_river_zulu_no_relief } }
					if = {
						limit = { root = { sb_frontier_ai_behavior_strict_historical = yes } }
						add_modifier = {
							name = sb_blood_river_zulu_no_relief
							months = 15
						}
					}
					else = {
						add_modifier = {
							name = sb_blood_river_zulu_no_relief
							months = 9
						}
					}
				}
			}
			else_if = {
				limit = { is_ai = yes }
				if = {
					limit = { has_modifier = sb_blood_river_zulu_no_relief }
					remove_modifier = sb_blood_river_zulu_no_relief
				}
				if = {
					limit = { has_modifier = sb_blood_river_zulu_limited_support }
					remove_modifier = sb_blood_river_zulu_limited_support
				}
				if = {
					limit = { has_modifier = sb_blood_river_zulu_relief_mobilization }
					remove_modifier = sb_blood_river_zulu_relief_mobilization
				}
			}
		}
	}
}

sb_join_trn_to_active_natal_war = {
	if = {
		limit = {
			c:TRN ?= {
				is_country_alive = yes
				is_ai = yes
				NOT = { has_variable = sb_trn_joined_natal_war_var }
			}
			c:ORA ?= {
				is_country_alive = yes
				has_variable = sb_natal_war_active_var
			}
			c:ZUL ?= {
				is_country_alive = yes
			}
		}
		c:TRN ?= { sb_ensure_boer_republic_compacts_for_root = yes }
		c:TRN ?= {
			set_variable = sb_trn_joined_natal_war_var
			change_relations = {
				country = c:ORA
				value = 20
			}
		}
		c:ORA ?= {
			set_variable = sb_trn_joined_natal_war_var
			change_relations = {
				country = c:TRN
				value = 20
			}
		}
		random_diplomatic_play = {
			limit = {
				initiator = c:ZUL
				target = c:ORA
				is_diplomatic_play_type = dp_sb_natal_crisis
				NOT = {
					any_scope_play_involved = {
						c:TRN ?= THIS
						is_diplomatic_play_committed_participant = yes
					}
				}
			}
			add_target_backers = { c:TRN }
		}
	}
}

			create_character = {
				template = ORA_piet_retief
				ruler = yes
				is_general = yes
				interest_group = ig_landowners
			}
		}
		random_scope_character = {
			limit = {
				has_template = ORA_piet_retief
			}
			if = {
				limit = {
					NOT = { has_role_of_type = general }

# ROOT = NAL. Consultation state is centralized here so invalidation and timeout
# recovery cannot leave one Boer player's reply marker blocking the ultimatum.
sb_natalia_clear_player_consultation = {
	if = {
		limit = { has_variable = sb_natalia_player_consultation_pending_var }
		remove_variable = sb_natalia_player_consultation_pending_var
	}
	if = {
		limit = { has_variable = sb_natalia_player_consultation_ready_var }
		remove_variable = sb_natalia_player_consultation_ready_var
	}
	every_country = {
		limit = { has_variable = sb_natalia_player_appeal_outstanding_var }
		remove_variable = sb_natalia_player_appeal_outstanding_var
	}
}

# ROOT = NAL. The nine-day aggregation event opens this gate; the last player
# response, or the monthly timeout repair, then advances the chain immediately.
sb_natalia_try_finish_player_consultation = {
	if = {
		limit = {
			has_variable = sb_natalia_player_consultation_pending_var
			has_variable = sb_natalia_player_consultation_ready_var
			NOT = {
				any_country = {
					has_variable = sb_natalia_player_appeal_outstanding_var
				}
			}
		}
		if = {
			limit = {
				is_country_alive = yes
				sb_natalia_has_harbour = yes
				is_subject = no
				is_at_war = no
				is_active_in_diplomatic_play = no
				has_variable = sb_british_natal_ultimatum_var
				NOT = { has_diplomatic_pact = { who = c:GBR type = colony } }
				NOT = { has_variable = sb_natalia_british_colony_resolved_var }
				c:GBR ?= {
					is_country_alive = yes
					has_variable = sb_british_natal_ultimatum_var
				}
			}
			sb_natalia_clear_player_consultation = yes
			c:GBR ?= { trigger_event = { id = sb_natal_crisis.108 days = 1 popup = no } }
		}
		else = {
			sb_natalia_clear_player_consultation = yes
			if = {
				limit = { has_variable = sb_british_natal_ultimatum_var }
				remove_variable = sb_british_natal_ultimatum_var
			}
			c:GBR ?= {
				if = {
					limit = { has_variable = sb_british_natal_ultimatum_var }
					remove_variable = sb_british_natal_ultimatum_var
				}
			}
		}
	}
}
