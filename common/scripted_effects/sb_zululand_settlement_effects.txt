# Spes Bona - postwar Zululand settlement and Nieuwe Republiek effects
#
# The shared Chiefdoms story owns its runtime state in one named container.
# Country variables are limited to route markers, button toggles, and delayed
# event leases that must survive JE projection.

# -----------------------------------------------------------------------------
# British victory handoff
# -----------------------------------------------------------------------------

sb_zululand_queue_british_handoff = {
	if = {
		limit = {
			country_definition = cd:GBR
			NOT = { has_variable = sb_zululand_postwar_handoff_resolved_var }
			NOT = { has_variable = sb_zululand_postwar_handoff_pending_var }
		}
		set_variable = sb_zululand_postwar_handoff_pending_var
		set_variable = { name = sb_zululand_postwar_handoff_not_before_var days = 7 }
	}
}

# Preserve the actual crown before annexation destroys ZUL. Character-valued
# variables keep the exact ruler and heir without removing either from office
# during the diplomatic play.
sb_zululand_archive_preannexation_royal_house = {
	if = {
		limit = {
			country_definition = cd:ZUL
			is_country_alive = yes
			c:NAL ?= {
				is_country_alive = yes
			}
		}
		sb_zulu_reconcile_secured_mpande_heir = yes
		c:NAL = { set_variable = sb_zululand_preannexation_royal_house_archived_var }
		c:NAL = {
			if = {
				limit = { has_variable = sb_zululand_preannexation_mbuyazi_line_var }
				remove_variable = sb_zululand_preannexation_mbuyazi_line_var
			}
			if = {
				limit = { has_variable = sb_zululand_preannexation_cetshwayo_line_var }
				remove_variable = sb_zululand_preannexation_cetshwayo_line_var
			}
		}
		if = {
			limit = {
				OR = {
					has_variable = sb_zulu_mpande_mbuyazi_heir_var
					has_global_variable = sb_zulu_mbuyazi_succession_global_var
					ruler ?= { has_template = ZUL_mbuyazi_heir }
					heir ?= { has_template = ZUL_mbuyazi_heir }
				}
			}
			c:NAL = { set_variable = sb_zululand_preannexation_mbuyazi_line_var }
		}
		else = {
			c:NAL = { set_variable = sb_zululand_preannexation_cetshwayo_line_var }
		}
		if = {
			limit = {
				exists = ruler
				ruler = { is_character_alive = yes }
			}
			ruler = { save_scope_as = sb_zululand_preannexation_ruler_scope }
			c:NAL = {
				set_variable = {
					name = sb_zululand_preannexation_ruler_scope_var
					value = scope:sb_zululand_preannexation_ruler_scope
				}
			}
		}
		if = {
			limit = {
				exists = heir
				heir = { is_character_alive = yes }
			}
			heir = { save_scope_as = sb_zululand_preannexation_heir_scope }
			c:NAL = {
				set_variable = {
					name = sb_zululand_preannexation_heir_scope_var
					value = scope:sb_zululand_preannexation_heir_scope
				}
			}
		}
	}
}

# ROOT = live ZUL. Discard an annexation snapshot without mutating a royal
# succession that changed legitimately while the diplomatic play was open.
sb_zululand_discard_preannexation_royal_house_archive = {
	c:NAL ?= {
		if = {
			limit = { exists = var:sb_zululand_preannexation_ruler_scope_var }
			remove_variable = sb_zululand_preannexation_ruler_scope_var
		}
		if = {
			limit = { exists = var:sb_zululand_preannexation_heir_scope_var }
			remove_variable = sb_zululand_preannexation_heir_scope_var
		}
		if = {
			limit = { has_variable = sb_zululand_preannexation_royal_house_archived_var }
			remove_variable = sb_zululand_preannexation_royal_house_archived_var
		}
		if = {
			limit = { has_variable = sb_zululand_preannexation_mbuyazi_line_var }
			remove_variable = sb_zululand_preannexation_mbuyazi_line_var
		}
		if = {
			limit = { has_variable = sb_zululand_preannexation_cetshwayo_line_var }
			remove_variable = sb_zululand_preannexation_cetshwayo_line_var
		}
	}
}

# ROOT = the live or newly restored ZUL. Reinstall the precise ruler and heir
# cached above; this deliberately does not depend on historical templates.
sb_zululand_restore_preannexation_royal_house = {
	if = {
		limit = {
			country_definition = cd:ZUL
			c:NAL ?= {
				exists = var:sb_zululand_preannexation_ruler_scope_var
				var:sb_zululand_preannexation_ruler_scope_var ?= {
					is_character_alive = yes
				}
			}
		}
		c:NAL = {
			var:sb_zululand_preannexation_ruler_scope_var ?= {
				save_scope_as = summoned_character_scope
			}
		}
		scope:summoned_character_scope ?= {
			if = {
				limit = {
					NOT = { is_in_void = yes }
					NOT = { is_in_exile_pool = yes }
					NOT = { owner ?= PREV }
				}
				transfer_character = PREV
			}
		}
		sb_zulu_install_summoned_claimant_as_ruler = yes
	}
	if = {
		limit = {
			country_definition = cd:ZUL
			c:NAL ?= {
				exists = var:sb_zululand_preannexation_heir_scope_var
				var:sb_zululand_preannexation_heir_scope_var ?= {
					is_character_alive = yes
				}
			}
		}
		c:NAL = {
			var:sb_zululand_preannexation_heir_scope_var ?= {
				save_scope_as = new_heir_scope
			}
		}
		scope:new_heir_scope ?= {
			if = {
				limit = {
					NOT = { is_in_void = yes }
					NOT = { is_in_exile_pool = yes }
					NOT = { owner ?= PREV }
				}
				transfer_character = PREV
			}
		}
		sb_zulu_assign_new_heir_scope = yes
	}
	c:NAL ?= {
		if = {
			limit = { exists = var:sb_zululand_preannexation_ruler_scope_var }
			remove_variable = sb_zululand_preannexation_ruler_scope_var
		}
		if = {
			limit = { exists = var:sb_zululand_preannexation_heir_scope_var }
			remove_variable = sb_zululand_preannexation_heir_scope_var
		}
		if = {
			limit = { has_variable = sb_zululand_preannexation_royal_house_archived_var }
			remove_variable = sb_zululand_preannexation_royal_house_archived_var
		}
		if = {
			limit = { has_variable = sb_zululand_preannexation_mbuyazi_line_var }
			remove_variable = sb_zululand_preannexation_mbuyazi_line_var
		}
		if = {
			limit = { has_variable = sb_zululand_preannexation_cetshwayo_line_var }
			remove_variable = sb_zululand_preannexation_cetshwayo_line_var
		}
	}
}

# ROOT = the engine-created ZUL secession. Restore a named historical ruler or
# heir before falling back to the chosen Mpande line's named successor. If no
# named candidate can be restored, the engine-generated monarch remains.
sb_zululand_restore_secession_dynasty = {
	if = {
		limit = {
			country_definition = cd:ZUL
			c:NAL ?= {
				exists = var:sb_zululand_preannexation_ruler_scope_var
				var:sb_zululand_preannexation_ruler_scope_var ?= {
					is_character_alive = yes
					OR = {
						has_template = ZUL_dingane
						has_template = ZUL_mpande_zulu
						has_template = ZUL_gqugqu_claimant
						has_template = ZUL_cetshwayo_heir
						has_template = ZUL_mbuyazi_heir
						has_template = ZUL_dinuzulu_heir
						has_template = ZUL_solomon_heir
					}
				}
			}
		}
		c:NAL = {
			var:sb_zululand_preannexation_ruler_scope_var ?= {
				save_scope_as = summoned_character_scope
			}
		}
		scope:summoned_character_scope ?= {
			if = {
				limit = {
					NOT = { is_in_void = yes }
					NOT = { is_in_exile_pool = yes }
					NOT = { owner ?= PREV }
				}
				transfer_character = PREV
			}
		}
		sb_zulu_install_summoned_claimant_as_ruler = yes
		set_variable = sb_zululand_secession_named_ruler_restored_var
	}
	else_if = {
		limit = {
			country_definition = cd:ZUL
			c:NAL ?= {
				exists = var:sb_zululand_preannexation_heir_scope_var
				var:sb_zululand_preannexation_heir_scope_var ?= {
					is_character_alive = yes
					OR = {
						has_template = ZUL_mpande_zulu
						has_template = ZUL_gqugqu_claimant
						has_template = ZUL_cetshwayo_heir
						has_template = ZUL_mbuyazi_heir
						has_template = ZUL_dinuzulu_heir
						has_template = ZUL_solomon_heir
					}
				}
			}
		}
		c:NAL = {
			var:sb_zululand_preannexation_heir_scope_var ?= {
				save_scope_as = summoned_character_scope
			}
		}
		scope:summoned_character_scope ?= {
			if = {
				limit = {
					NOT = { is_in_void = yes }
					NOT = { is_in_exile_pool = yes }
					NOT = { owner ?= PREV }
				}
				transfer_character = PREV
			}
		}
		sb_zulu_install_summoned_claimant_as_ruler = yes
		set_variable = sb_zululand_secession_named_ruler_restored_var
	}
	if = {
		limit = {
			NOT = { has_variable = sb_zululand_secession_named_ruler_restored_var }
			c:NAL ?= { has_variable = sb_zululand_preannexation_mbuyazi_line_var }
		}
		if = {
			limit = {
				any_country = {
					any_scope_character = {
						has_template = ZUL_mbuyazi_successor
						is_character_alive = yes
					}
				}
			}
			random_country = {
				limit = { any_scope_character = { has_template = ZUL_mbuyazi_successor is_character_alive = yes } }
				random_scope_character = {
					limit = { has_template = ZUL_mbuyazi_successor is_character_alive = yes }
					save_scope_as = summoned_character_scope
				}
			}
		}
		else_if = {
			limit = { any_character_in_exile_pool = { has_template = ZUL_mbuyazi_successor is_character_alive = yes } }
			random_character_in_exile_pool = {
				limit = { has_template = ZUL_mbuyazi_successor is_character_alive = yes }
				save_scope_as = summoned_character_scope
			}
		}
		else_if = {
			limit = {
				NOT = { is_template_used = ZUL_mbuyazi_successor }
				game_date >= 1852.1.1
			}
			create_character = {
				template = ZUL_mbuyazi_successor
				on_created = { save_scope_as = summoned_character_scope }
			}
		}
	}
	else_if = {
		limit = {
			NOT = { has_variable = sb_zululand_secession_named_ruler_restored_var }
			c:NAL ?= { has_variable = sb_zululand_preannexation_cetshwayo_line_var }
		}
		if = {
			limit = {
				any_country = {
					any_scope_character = {
						has_template = ZUL_dinuzulu_heir
						is_character_alive = yes
					}
				}
			}
			random_country = {
				limit = { any_scope_character = { has_template = ZUL_dinuzulu_heir is_character_alive = yes } }
				random_scope_character = {
					limit = { has_template = ZUL_dinuzulu_heir is_character_alive = yes }
					save_scope_as = summoned_character_scope
				}
			}
		}
		else_if = {
			limit = { any_character_in_exile_pool = { has_template = ZUL_dinuzulu_heir is_character_alive = yes } }
			random_character_in_exile_pool = {
				limit = { has_template = ZUL_dinuzulu_heir is_character_alive = yes }
				save_scope_as = summoned_character_scope
			}
		}
		else_if = {
			limit = {
				NOT = { is_template_used = ZUL_dinuzulu_heir }
				game_date >= 1868.1.1
			}
			create_character = {
				template = ZUL_dinuzulu_heir
				on_created = { save_scope_as = summoned_character_scope }
			}
		}
	}
	if = {
		limit = {
			NOT = { has_variable = sb_zululand_secession_named_ruler_restored_var }
			exists = scope:summoned_character_scope
		}
		scope:summoned_character_scope ?= {
			if = {
				limit = {
					NOT = { is_in_void = yes }
					NOT = { is_in_exile_pool = yes }
					NOT = { owner ?= PREV }
				}
				transfer_character = PREV
			}
		}
		sb_zulu_install_summoned_claimant_as_ruler = yes
		set_variable = sb_zululand_secession_named_ruler_restored_var
	}
	if = {
		limit = { has_variable = sb_zululand_secession_named_ruler_restored_var }
		remove_variable = sb_zululand_secession_named_ruler_restored_var
	}
	c:NAL ?= {
		if = { limit = { exists = var:sb_zululand_preannexation_ruler_scope_var } remove_variable = sb_zululand_preannexation_ruler_scope_var }
		if = { limit = { exists = var:sb_zululand_preannexation_heir_scope_var } remove_variable = sb_zululand_preannexation_heir_scope_var }
		if = { limit = { has_variable = sb_zululand_preannexation_royal_house_archived_var } remove_variable = sb_zululand_preannexation_royal_house_archived_var }
		if = { limit = { has_variable = sb_zululand_preannexation_mbuyazi_line_var } remove_variable = sb_zululand_preannexation_mbuyazi_line_var }
		if = { limit = { has_variable = sb_zululand_preannexation_cetshwayo_line_var } remove_variable = sb_zululand_preannexation_cetshwayo_line_var }
	}
}

sb_zululand_normalize_live_subject_for_natal = {
	if = {
		limit = {
			c:ZUL ?= {
				is_country_alive = yes
				is_subject = yes
				NOT = { is_direct_subject_of = c:NAL }
			}
		}
		c:NAL ?= { transfer_subject = c:ZUL }
	}
	else_if = {
		limit = {
			c:ZUL ?= {
				is_country_alive = yes
				is_subject = no
			}
		}
		c:NAL ?= {
			create_diplomatic_pact = {
				country = c:ZUL
				type = protectorate
			}
		}
	}
	if = {
		limit = {
			c:ZUL ?= {
				is_country_alive = yes
				is_direct_subject_of = c:NAL
				NOT = { is_subject_type = subject_type_protectorate }
			}
		}
		c:ZUL = { change_subject_type = subject_type_protectorate }
	}
	c:ZUL ?= { sb_natal_restore_archived_zulu_firearms = yes }
	c:NAL ?= { sb_natal_schedule_zulu_settlement = yes }
}

sb_zululand_restore_protectorate_for_natal = {
	# Ordinary Release Country uses the same ZUL country-creation footprint, but
	# is a UI action and cannot carry this postwar branch's bespoke setup. Create
	# the complete owner state atomically, as in Vanilla and Gates country splits.
	p:xBE6FEE.state = { save_scope_as = sb_zululand_restoration_state }
	p:xBE6FEE.state.owner = {
		create_country = {
			tag = ZUL
			origin = c:GBR
			state = scope:sb_zululand_restoration_state
			on_created = { sb_natal_apply_restored_zulu_baseline = yes }
		}
	}
	sb_zululand_normalize_live_subject_for_natal = yes
}

sb_zululand_handoff_to_natal = {
	if = {
		limit = { c:ZUL ?= { is_country_alive = yes } }
		sb_zululand_normalize_live_subject_for_natal = yes
	}
	else = { sb_zululand_restore_protectorate_for_natal = yes }
	if = {
		limit = { c:NAL ?= { sb_zululand_normalized_nal_protectorate = yes } }
		set_variable = sb_zululand_postwar_handoff_resolved_var
		if = { limit = { has_variable = sb_zululand_postwar_handoff_pending_var } remove_variable = sb_zululand_postwar_handoff_pending_var }
		if = { limit = { has_variable = sb_zululand_postwar_handoff_not_before_var } remove_variable = sb_zululand_postwar_handoff_not_before_var }
		if = { limit = { has_variable = sb_zululand_postwar_handoff_event_open_var } remove_variable = sb_zululand_postwar_handoff_event_open_var }
		set_variable = sb_anglo_zulu_pressure_resolved_var
	}
}

			country = c:GBR

				xBE6FEE x1A084B xBFA16B x9E9742 x88FAD4

			}
		}
	}
	add_radicals = { value = 0.25 culture = cu:zulu }
	sb_natal_create_zulu_national_movement = yes
	set_variable = sb_zululand_postwar_handoff_resolved_var
	set_variable = sb_zululand_direct_british_administration_var

	if = { limit = { has_variable = sb_zululand_postwar_handoff_event_open_var } remove_variable = sb_zululand_postwar_handoff_event_open_var }
	set_variable = sb_anglo_zulu_pressure_resolved_var
}

sb_zululand_postwar_monthly_housekeeping = {
	if = {
		limit = {
			country_definition = cd:GBR
			NOT = { has_variable = sb_zululand_postwar_handoff_resolved_var }
			NOT = { has_variable = sb_zululand_postwar_handoff_pending_var }
			sb_zululand_under_british_postwar_control = yes
		}
		sb_zululand_queue_british_handoff = yes
	}
	if = {
		limit = {
			country_definition = cd:GBR
			has_variable = sb_zululand_postwar_handoff_pending_var
			NOT = { has_variable = sb_zululand_postwar_handoff_not_before_var }
			NOT = { has_variable = sb_zululand_postwar_handoff_event_open_var }
			sb_zululand_under_british_postwar_control = yes
		}
		set_variable = { name = sb_zululand_postwar_handoff_event_open_var days = 120 }
		trigger_event = { id = sb_zululand_settlement.001 days = 1 popup = yes }
	}
	if = {
		limit = {
			country_definition = cd:NRP
			is_country_alive = yes
		}
		sb_nrp_monthly_housekeeping = yes
	}
}

sb_zululand_begin_natal_incorporation = {
	if = {
		limit = {
			country_definition = cd:NAL
			s:STATE_ZULULAND.region_state:NAL ?= { is_incorporated = no }
		}
		set_variable = sb_zululand_incorporation_requested_var
	}
}

sb_zululand_choose_direct_natal_administration = {
	if = {
		limit = { c:ZUL ?= { is_country_alive = yes is_direct_subject_of = ROOT } }
		c:ZUL = { sb_natal_archive_zulu_firearms = yes sb_zululand_archive_preannexation_royal_house = yes }
		annex = c:ZUL
	}
	if = { limit = { has_variable = sb_natal_zulu_settlement_event_open_var } remove_variable = sb_natal_zulu_settlement_event_open_var }
	set_variable = sb_zululand_direct_natal_administration_var
	set_variable = sb_natal_zulu_settlement_resolved_var
	add_radicals = { value = 0.25 culture = cu:zulu }
	sb_natal_create_zulu_national_movement = yes
	sb_zululand_begin_natal_incorporation = yes
}

sb_zululand_mark_and_exile_royal_house = {
	if = {
		limit = {
			c:ZUL ?= {
				OR = {
					ruler ?= { has_template = ZUL_uthumbu_claimed_son }
					heir ?= { has_template = ZUL_uthumbu_claimed_son }
				}
			}
		}
		set_variable = sb_zululand_uthumbu_line_var
	}
	if = {
		limit = {
			OR = {
				has_global_variable = sb_zulu_mbuyazi_succession_global_var
				c:ZUL ?= {
					OR = {
						ruler ?= { has_template = ZUL_mbuyazi_heir }
						heir ?= { has_template = ZUL_mbuyazi_heir }
					}
				}
			}
		}
		set_variable = sb_zululand_mbuyazi_line_var
	}
	c:ZUL ?= {
		ruler ?= {
			set_variable = sb_zululand_exiled_claimant_var
			exile_character_with_role_cleanup = yes
		}
		heir ?= {
			set_variable = sb_zululand_exiled_successor_var
			exile_character_with_role_cleanup = yes
		}
		create_character = {
			template = ZUL_zibhebhu_ka_mapitha
			on_created = { save_scope_as = summoned_character_scope }
		}
		sb_zulu_install_summoned_claimant_as_ruler = yes
	}
}

sb_zululand_update_bureaucracy_drift = {
	container:sb_zululand_chiefdoms_state ?= {
		set_variable = { name = sb_zululand_bureaucracy_x_var value = c:NAL.relative_bureaucracy }
		if = {
			limit = { var:sb_zululand_bureaucracy_x_var < 0 }
			change_variable = { name = sb_zululand_bureaucracy_x_var multiply = -1 }
			set_variable = { name = sb_zululand_bureaucracy_sign_var value = -1 }
		}
		else_if = {
			limit = { var:sb_zululand_bureaucracy_x_var > 0 }
			set_variable = { name = sb_zululand_bureaucracy_sign_var value = 1 }
		}
		else = { set_variable = { name = sb_zululand_bureaucracy_sign_var value = 0 } }
		clamp_variable = { name = sb_zululand_bureaucracy_x_var min = 0 max = 1 }
		set_variable = { name = sb_zululand_bureaucracy_square_var value = var:sb_zululand_bureaucracy_x_var }
		change_variable = { name = sb_zululand_bureaucracy_square_var multiply = var:sb_zululand_bureaucracy_x_var }
		set_variable = { name = sb_zululand_bureaucracy_drift_var value = var:sb_zululand_bureaucracy_x_var }
		change_variable = { name = sb_zululand_bureaucracy_drift_var multiply = 2 }
		change_variable = { name = sb_zululand_bureaucracy_drift_var subtract = var:sb_zululand_bureaucracy_square_var }
		change_variable = { name = sb_zululand_bureaucracy_drift_var multiply = 0.9 }
		if = {
			limit = { var:sb_zululand_bureaucracy_sign_var != 0 }
			change_variable = { name = sb_zululand_bureaucracy_drift_var add = 0.1 }
			change_variable = { name = sb_zululand_bureaucracy_drift_var multiply = var:sb_zululand_bureaucracy_sign_var }
			change_variable = { name = sb_zululand_balance_var add = var:sb_zululand_bureaucracy_drift_var }
		}
		remove_variable = sb_zululand_bureaucracy_x_var
		remove_variable = sb_zululand_bureaucracy_square_var
		remove_variable = sb_zululand_bureaucracy_sign_var
		remove_variable = sb_zululand_bureaucracy_drift_var
	}
}

sb_zululand_apply_monthly_action_drift = {
	container:sb_zululand_chiefdoms_state ?= {
		if = {
			limit = { c:NAL ?= { has_variable = sb_zululand_policing_active_var } }
			change_variable = { name = sb_zululand_balance_var add = 0.5 }
		}
		if = {
			limit = { c:NAL ?= { has_variable = sb_zululand_claimant_returned_var } }
			change_variable = { name = sb_zululand_balance_var subtract = 0.5 }
		}
		if = {
			limit = { c:TRN ?= { has_variable = sb_zululand_usuthu_sponsorship_active_var } }
			change_variable = { name = sb_zululand_balance_var subtract = 0.25 }
			change_variable = { name = sb_zululand_boer_commitment_var add = 2 }
		}
		if = {
			limit = { c:TRN ?= { has_variable = sb_zululand_boer_volunteers_active_var } }
			change_variable = { name = sb_zululand_balance_var subtract = 0.5 }
			change_variable = { name = sb_zululand_boer_commitment_var add = 3 }
		}
		clamp_variable = { name = sb_zululand_balance_var min = 0 max = 100 }
		clamp_variable = { name = sb_zululand_boer_commitment_var min = 0 max = 100 }
		if = {
			limit = {
				var:sb_zululand_boer_commitment_var >= 100
				NOT = { has_variable = sb_zululand_boer_committed_var }
			}
			set_variable = sb_zululand_boer_committed_var
			if = {
				limit = { c:TRN ?= { is_country_alive = yes } }
				set_variable = { name = sb_zululand_boer_patron_scope value = c:TRN }
			}
			c:TRN ?= {
				sb_zululand_stop_usuthu_sponsorship = yes
				sb_zululand_stop_boer_volunteers = yes
			}
		}
	}
}

# One terminal container is one delivery generation. Its saved event scope
# prevents a held .120/.121 from becoming an authority for a later situation.
sb_zululand_begin_zibhebhu_terminal_generation = {
	if = {
		limit = {
			sb_zululand_chiefdoms_runtime_valid = yes
			NOT = { has_variable = sb_zululand_terminal_event_pending_var }
			NOT = { container_exists = sb_zululand_terminal_generation_state }
			NOT = { has_variable = sb_zululand_terminal_generation_scope }
		}
		create_container = {
			name = sb_zululand_terminal_generation_state
			tags = { sb_story sb_zululand_terminal_generation }
			parent = root
			on_created = { set_variable = sb_zululand_terminal_generation_pending_var }
		}
		container:sb_zululand_terminal_generation_state = {
			set_variable = sb_zululand_terminal_zibhebhu_outcome_var
			save_scope_as = sb_zululand_terminal_receipt_scope
		}
		set_variable = { name = sb_zululand_terminal_generation_scope value = scope:sb_zululand_terminal_receipt_scope }
		set_variable = sb_zululand_terminal_event_pending_var
		trigger_event = { id = sb_zululand_settlement.120 days = 1 popup = yes }
	}
}

sb_zululand_queue_terminal_outcome = {
	if = {
		limit = {
			NOT = { has_variable = sb_zululand_terminal_event_pending_var }
			NOT = { has_variable = sb_zululand_event_spacing_var }
			NOT = { has_variable = sb_zululand_incident_pending_var }
			NOT = { has_variable = sb_zululand_claimant_fate_pending_var }
			NOT = { has_variable = sb_zululand_boer_appeal_pending_var }
			OR = {
				NOT = { has_variable = sb_zululand_claimant_returned_var }
				has_variable = sb_zululand_claimant_fate_resolved_var
			}
			container:sb_zululand_chiefdoms_state.var:sb_zululand_situation_months_var >= 18
			container:sb_zululand_chiefdoms_state.var:sb_zululand_balance_var <= 0
			has_variable = sb_zululand_skirmish_fired_var
			has_variable = sb_zululand_hut_tax_fired_var
			has_variable = sb_zululand_defection_fired_var
		}
		# id = sb_zululand_settlement.121 (created by the bound Crown receipt)
		sb_zululand_begin_crown_terminal_generation = yes
	}
	else_if = {
		limit = {
			NOT = { has_variable = sb_zululand_terminal_event_pending_var }
			NOT = { has_variable = sb_zululand_event_spacing_var }
			NOT = { has_variable = sb_zululand_incident_pending_var }
			NOT = { has_variable = sb_zululand_claimant_fate_pending_var }
			NOT = { has_variable = sb_zululand_boer_appeal_pending_var }
			OR = {
				NOT = { has_variable = sb_zululand_claimant_returned_var }
				has_variable = sb_zululand_claimant_fate_resolved_var
			}
			container:sb_zululand_chiefdoms_state.var:sb_zululand_situation_months_var >= 18
			container:sb_zululand_chiefdoms_state.var:sb_zululand_balance_var >= 100
			has_variable = sb_zululand_skirmish_fired_var
			has_variable = sb_zululand_hut_tax_fired_var
			has_variable = sb_zululand_defection_fired_var
		}
		sb_zululand_begin_zibhebhu_terminal_generation = yes
	}
	else_if = {
		limit = {
			NOT = { has_variable = sb_zululand_terminal_event_pending_var }
			NOT = { has_variable = sb_zululand_event_spacing_var }
			NOT = { has_variable = sb_zululand_incident_pending_var }
			NOT = { has_variable = sb_zululand_claimant_fate_pending_var }
			container:sb_zululand_chiefdoms_state.var:sb_zululand_situation_months_var >= 60
		}
		if = {
			limit = { container:sb_zululand_chiefdoms_state.var:sb_zululand_balance_var > 50 }
			sb_zululand_begin_zibhebhu_terminal_generation = yes
		}
		else = {
			# Parity at the hard timeout restores the Crown.
			sb_zululand_begin_crown_terminal_generation = yes
		}
	}
}

sb_zululand_restore_dynastic_ruler = {
	if = {
		limit = {
			NOT = {
				c:ZUL ?= {
					any_scope_character = {
						has_variable = sb_zululand_exiled_claimant_var
						is_character_alive = yes
					}
				}
			}
			any_character_in_exile_pool = {
				has_variable = sb_zululand_exiled_claimant_var
				is_character_alive = yes
			}
		}
		random_character_in_exile_pool = {
			limit = { has_variable = sb_zululand_exiled_claimant_var is_character_alive = yes }
			transfer_character = c:ZUL
		}
	}
	if = {
		limit = {
			c:ZUL ?= {
				NOT = {
					any_scope_character = {
						OR = {
							has_variable = sb_zululand_exiled_claimant_var
							has_variable = sb_zululand_crown_successor_var
						}
						is_character_alive = yes
					}
				}
			}
		}
		sb_zululand_prepare_crown_successor = yes
	}
	c:ZUL ?= {
		if = {
			limit = {
				any_scope_character = {
					has_variable = sb_zululand_exiled_claimant_var
					is_character_alive = yes
				}
			}
			random_scope_character = {
				limit = { has_variable = sb_zululand_exiled_claimant_var is_character_alive = yes }
				save_scope_as = summoned_character_scope
			}
			sb_zulu_install_summoned_claimant_as_ruler = yes
		}
		else_if = {
			limit = {
				any_scope_character = {
					has_variable = sb_zululand_crown_successor_var
					is_character_alive = yes
				}
			}
			random_scope_character = {
				limit = { has_variable = sb_zululand_crown_successor_var is_character_alive = yes }
				save_scope_as = summoned_character_scope
			}
			sb_zulu_install_summoned_claimant_as_ruler = yes
		}
	}
}

	s:STATE_ZULULAND.region_state:NRP ?= {
		if = {
			limit = { NOT = { has_building = building_livestock_ranch } }
			create_building = {
				building = building_livestock_ranch
				add_ownership = {
					building = {
						type = building_manor_house
						country = c:NRP
						levels = 1
						region = "STATE_ZULULAND"
					}
				}
				reserves = 1
				activate_production_methods = {
					"pm_open_air_stockyards"
					"pm_simple_ranch"
					"pm_standard_fences"
					"pm_unrefrigerated"
				}
			}
		}
	}
}

# BUILDING = a literal building type. Province-level state partitioning assigns
# the existing economy to the larger NRP fragment, so merge each inherited
# level back into surviving ZUL before NRP receives its own starter economy.
sb_nrp_return_building_type_to_zul = {
	s:STATE_ZULULAND.region_state:NRP = {
		if = {
			limit = { has_building = $BUILDING$ }
			every_scope_building = {
				limit = { is_building_type = $BUILDING$ }
				ROOT = {
					set_variable = {
						name = sb_nrp_relocated_building_levels_var
						value = PREV.level
					}
				}
			}
			s:STATE_ZULULAND.region_state:ZUL = {
				if = {
					limit = { has_building = $BUILDING$ }
					every_scope_building = {
						limit = { is_building_type = $BUILDING$ }
						ROOT = {
							change_variable = {
								name = sb_nrp_relocated_building_levels_var
								add = PREV.level
							}
						}
					}
					remove_building = $BUILDING$
				}
			}
			remove_building = $BUILDING$
			s:STATE_ZULULAND.region_state:ZUL = {
				create_building = {
					building = $BUILDING$
					level = ROOT.var:sb_nrp_relocated_building_levels_var
					reserves = 1
				}
			}
			ROOT = { remove_variable = sb_nrp_relocated_building_levels_var }
		}
	}
}

sb_nrp_create_greater_republic = {
	if = {
		limit = {
			country_definition = cd:NAL
			c:TRN ?= { is_country_alive = yes }
			c:ZUL ?= { is_country_alive = yes }
			NOT = { c:NRP ?= { is_country_alive = yes } }
			p:xE1E455.state.owner = c:ZUL
		}
		set_variable = sb_nrp_setup_pending_var
		c:ZUL = {
			create_country = {
				tag = NRP
				origin = c:TRN
				province = p:xE1E455
				on_created = {
					c:NAL = { sb_nrp_finalize_greater_republic = yes }
				}
			}
		}
		# Independent lease cleanup: normally the on_created callback resolves
		# immediately; this fallback finalizes a live NRP or clears a failed setup.
		trigger_event = { id = sb_zululand_settlement.201 days = 1 popup = no }
	}
}

sb_nrp_finalize_greater_republic = {
	if = {
		limit = {
			country_definition = cd:NAL
			has_variable = sb_nrp_setup_pending_var
			c:NRP ?= { is_country_alive = yes }
			OR = {
				c:ZUL ?= { is_country_alive = yes }
				AND = {
					has_variable = sb_nrp_direct_annexation_rebellion_pending_var
					is_country_alive = yes
					exists = s:STATE_ZULULAND.region_state:NAL
				}
			}
		}
		remove_variable = sb_nrp_setup_pending_var
		# Restore ZUL's inherited buildable levels in its residual state. The split
		# assigns the objects to NRP, so the helper rebuilds those levels for ZUL;
		# NRP then receives only the agreed one-ranch starter economy.
		s:STATE_ZULULAND = {
			set_owner_of_provinces = {
				country = c:NRP
				provinces = { xE1E455 xE882CE xBE6FEE x904EBE x9E9742 }
			}
			add_claim = c:NRP
			region_state:NRP ?= { set_state_type = incorporated }
		}
		sb_nrp_return_inherited_buildings_to_zul = yes
		c:NRP = {
			sb_nrp_apply_boer_republic_setup = yes
			set_variable = sb_nrp_greater_concession_var
				if = {
					limit = { c:NAL ?= { has_variable = sb_nrp_direct_annexation_rebellion_pending_var } }
					set_variable = sb_nrp_direct_annexation_rebellion_var
					if = {
						limit = { NOT = { c:ZUL ?= { is_country_alive = yes } } }
						set_variable = sb_nrp_boundary_pending_var
						set_variable = { name = sb_nrp_boundary_not_before_var days = 30 }
						trigger_event = { id = sb_zululand_settlement.209 days = 30 popup = no }
					}
				}
			else = {
				set_variable = sb_nrp_boundary_pending_var
				set_variable = { name = sb_nrp_boundary_not_before_var days = 30 }
				trigger_event = { id = sb_zululand_settlement.209 days = 30 popup = no }
			}
		}
		if = {
			limit = { has_variable = sb_nrp_direct_annexation_rebellion_pending_var }
			remove_variable = sb_nrp_direct_annexation_rebellion_pending_var
		}
		sb_nrp_seed_livestock_ranch = yes
		c:TRN ?= {
			if = {
				limit = { is_player = yes }
				trigger_event = { id = sb_zululand_settlement.200 days = 1 popup = yes }
			}
		}
	}
	else_if = {
		limit = { has_variable = sb_nrp_setup_pending_var }
		remove_variable = sb_nrp_setup_pending_var
		if = {
			limit = { has_variable = sb_nrp_direct_annexation_rebellion_pending_var }
			remove_variable = sb_nrp_direct_annexation_rebellion_pending_var
		}
	}
}

sb_nrp_select_boundary_initiator = {
	if = {
		limit = {
			c:NAL ?= {
				sb_zululand_nal_is_self_governing = yes
				is_active_in_diplomatic_play = no
				NOT = { has_truce_with = c:NRP }
			}
		}
		set_variable = { name = sb_nrp_boundary_initiator_scope value = c:NAL }
	}
	else_if = {
		limit = {
			c:NAL ?= {
				is_country_alive = yes
				NOT = { sb_zululand_nal_is_self_governing = yes }
			}
			c:GBR ?= {
				is_country_alive = yes
				is_active_in_diplomatic_play = no
				NOT = { has_truce_with = c:NRP }
			}
		}
		set_variable = { name = sb_nrp_boundary_initiator_scope value = c:GBR }
	}
}

sb_nrp_open_boundary_demand = {
	if = {
		limit = { sb_nrp_boundary_demand_can_open = yes }

			}
			else = { trigger_event = { id = sb_zululand_settlement.210 popup = yes } }
		}
	}
}

sb_nrp_calculate_boundary_strength = {
	set_variable = { name = sb_nrp_boer_strength_var value = 0 }
	set_variable = { name = sb_nrp_opposition_strength_var value = 0 }
	if = {
		limit = { c:TRN ?= { is_country_alive = yes } }
		change_variable = { name = sb_nrp_boer_strength_var add = c:TRN.army_power_projection }
	}
	if = {
		limit = {
			c:ORA ?= {
				is_country_alive = yes
				is_direct_subject_of = c:TRN
				is_subject_type = subject_type_sb_boer_confederal_partner
			}
		}
		change_variable = { name = sb_nrp_boer_strength_var add = c:ORA.army_power_projection }
	}
	if = {
		limit = { c:NAL ?= { is_country_alive = yes } }
		change_variable = { name = sb_nrp_opposition_strength_var add = c:NAL.army_power_projection }
	}
	if = {
		limit = { c:ZUL ?= { is_country_alive = yes } }
		change_variable = { name = sb_nrp_opposition_strength_var add = c:ZUL.army_power_projection }
	}
	if = {
		limit = { NOT = { c:NAL ?= { sb_zululand_nal_is_self_governing = yes } } }
		if = {
			limit = {
				c:CAP ?= {
					is_country_alive = yes
					is_subject_of = c:GBR
				}
			}
			change_variable = { name = sb_nrp_opposition_strength_var add = c:CAP.army_power_projection }
		}
		if = {
			limit = { c:GBR ?= { is_country_alive = yes } }
			set_variable = { name = sb_nrp_british_fraction_var value = c:GBR.army_power_projection }
			change_variable = { name = sb_nrp_british_fraction_var multiply = 0.10 }
			change_variable = { name = sb_nrp_opposition_strength_var add = var:sb_nrp_british_fraction_var }
			remove_variable = sb_nrp_british_fraction_var
		}
	}
	set_variable = { name = sb_nrp_strength_ratio_var value = var:sb_nrp_boer_strength_var }
	if = {
		limit = { var:sb_nrp_opposition_strength_var > 0 }
		change_variable = { name = sb_nrp_strength_ratio_var divide = var:sb_nrp_opposition_strength_var }
	}
	else = { set_variable = { name = sb_nrp_strength_ratio_var value = 2 } }
	if = {
		limit = { var:sb_nrp_strength_ratio_var < 0.50 }
		set_variable = { name = sb_nrp_refusal_chance_var value = 5 }
	}
	else_if = {
		limit = { var:sb_nrp_strength_ratio_var < 0.75 }
		set_variable = { name = sb_nrp_refusal_chance_var value = 15 }
	}
	else_if = {
		limit = { var:sb_nrp_strength_ratio_var < 1 }
		set_variable = { name = sb_nrp_refusal_chance_var value = 35 }
	}
	else_if = {
		limit = { var:sb_nrp_strength_ratio_var < 1.50 }
		set_variable = { name = sb_nrp_refusal_chance_var value = 60 }
	}
	else = { set_variable = { name = sb_nrp_refusal_chance_var value = 80 } }
	if = {
		limit = { c:TRN ?= { is_player = yes } }
		change_variable = { name = sb_nrp_refusal_chance_var add = 10 }
		clamp_variable = { name = sb_nrp_refusal_chance_var min = 0 max = 95 }
	}
}

sb_nrp_roll_refusal_from_chance = {
	if = {
		limit = { var:sb_nrp_refusal_chance_var <= 5 }
		random_list = { 5 = { set_variable = sb_nrp_should_refuse_boundary_var } 95 = {} }
	}
	else_if = {
		limit = { var:sb_nrp_refusal_chance_var <= 15 }
		random_list = { 15 = { set_variable = sb_nrp_should_refuse_boundary_var } 85 = {} }
	}
	else_if = {
		limit = { var:sb_nrp_refusal_chance_var <= 25 }
		random_list = { 25 = { set_variable = sb_nrp_should_refuse_boundary_var } 75 = {} }
	}
	else_if = {
		limit = { var:sb_nrp_refusal_chance_var <= 35 }
		random_list = { 35 = { set_variable = sb_nrp_should_refuse_boundary_var } 65 = {} }
	}
	else_if = {
		limit = { var:sb_nrp_refusal_chance_var <= 45 }
		random_list = { 45 = { set_variable = sb_nrp_should_refuse_boundary_var } 55 = {} }
	}
	else_if = {
		limit = { var:sb_nrp_refusal_chance_var <= 60 }
		random_list = { 60 = { set_variable = sb_nrp_should_refuse_boundary_var } 40 = {} }
	}
	else_if = {
		limit = { var:sb_nrp_refusal_chance_var <= 70 }
		random_list = { 70 = { set_variable = sb_nrp_should_refuse_boundary_var } 30 = {} }
	}
	else_if = {
		limit = { var:sb_nrp_refusal_chance_var <= 80 }
		random_list = { 80 = { set_variable = sb_nrp_should_refuse_boundary_var } 20 = {} }
	}
	else = { random_list = { 90 = { set_variable = sb_nrp_should_refuse_boundary_var } 10 = {} } }
}

sb_nrp_roll_boundary_response = {
	if = { limit = { has_variable = sb_nrp_should_refuse_boundary_var } remove_variable = sb_nrp_should_refuse_boundary_var }
	if = {
		limit = { NOT = { sb_nrp_subject_hard_safety = yes } }
		if = {
			limit = { sb_frontier_ai_behavior_off = yes }
			random_list = { 50 = { set_variable = sb_nrp_should_refuse_boundary_var } 50 = {} }
		}
		else_if = {
			limit = {
				sb_frontier_ai_behavior_dynamic_historical = yes
				NOT = { sb_nrp_dynamic_hard_safety = yes }
			}
			sb_nrp_calculate_boundary_strength = yes
			sb_nrp_roll_refusal_from_chance = yes
		}
		# Strict Historical and Dynamic hard-safety cases deliberately leave the
		# refusal marker unset.
	}
}

	if = {
		limit = { c:NRP ?= { has_variable = sb_nrp_direct_annexation_rebellion_var } }

		if = { limit = { c:NAL ?= { is_country_alive = yes NOT = { this = scope:initiator } } } add_initiator_backers = { c:NAL } }
		# Replacement goals are installed before the engine-required placeholder.
		add_war_goal = { holder = c:NAL type = annex_country target_country = c:NRP primary_demand = yes }
		add_war_goal = { holder = c:NRP type = return_state target_country = c:NAL target_state = scope:sb_nrp_boundary_natal_state primary_demand = yes }
	}
	else = {
		s:STATE_ZULULAND.region_state:ZUL ?= { save_temporary_scope_as = sb_nrp_boundary_zul_state }
		add_initiator_backers = { c:ZUL }
		if = { limit = { c:NAL ?= { is_country_alive = yes NOT = { this = scope:initiator } } } add_initiator_backers = { c:NAL } }
		add_war_goal = { holder = c:ZUL type = return_state target_country = c:NRP target_state = scope:sb_nrp_boundary_nrp_state primary_demand = yes }
		add_war_goal = { holder = c:NRP type = return_state target_country = c:ZUL target_state = scope:sb_nrp_boundary_zul_state primary_demand = yes }
	}
	remove_war_goal = { who = initiator type = sb_story_humiliation }
}

# Root is the actual new play. Do not select a same-type play after creation.
sb_nrp_configure_started_boundary_play = {
	if = {
		limit = {
			is_diplomatic_play_type = dp_sb_nrp_boundary_confrontation
			scope:initiator ?= { has_variable = sb_nrp_boundary_launch_lease_var }
			c:NRP ?= {
				sb_nrp_boundary_current_political_authority = yes
				has_variable = sb_nrp_boundary_refused_var
				var:sb_nrp_boundary_initiator_scope = scope:initiator
			}
			OR = {
				scope:target ?= { country_definition = cd:NRP }
				AND = { scope:target ?= { country_definition = cd:TRN } c:NRP ?= { is_direct_subject_of = scope:target is_subject_type = subject_type_sb_boer_confederal_partner } }
			}
		}
		if = { limit = { scope:target ?= { country_definition = cd:TRN } } add_target_backers = { c:NRP } }
		if = {
			limit = { scope:target ?= { country_definition = cd:NRP } c:NRP ?= { has_variable = sb_nrp_trn_defence_accepted_pending_var } c:TRN ?= { is_country_alive = yes is_subject = no } }
			add_target_backers = { c:TRN }
		}
		if = {
			limit = { c:NRP ?= { has_variable = sb_nrp_trn_defence_declined_pending_var } c:TRN ?= { is_country_alive = yes } }
			c:TRN = { change_relations = { country = c:NRP value = -50 } }
			c:NRP = {
				every_scope_treaty = {
					limit = { binds = c:TRN any_scope_article = { has_type = defensive_pact } }
					withdraw = { country = c:NRP }
				}
			}
		}
		c:NRP = {
			set_variable = sb_nrp_boundary_confrontation_active_var
			set_variable = { name = sb_nrp_boundary_play_scope value = root }
			if = { limit = { has_variable = sb_nrp_boundary_refused_var } remove_variable = sb_nrp_boundary_refused_var }
			if = { limit = { has_variable = sb_nrp_boundary_creation_failed_var } remove_variable = sb_nrp_boundary_creation_failed_var }
		}
		sb_nrp_add_boundary_war_goals = yes
		scope:initiator = { remove_variable = sb_nrp_boundary_launch_lease_var }
	}
	# A real leased root may have raced into an obsolete target shape. It remains
	# deliberately unbound: remove only launch scaffolding and retain the refusal,
	# saved initiator, and accept/decline receipts for a later exact retry.
	else_if = {
		limit = {
			is_diplomatic_play_type = dp_sb_nrp_boundary_confrontation
			scope:initiator ?= { has_variable = sb_nrp_boundary_launch_lease_var }
			c:NRP ?= {
				sb_nrp_boundary_current_political_authority = yes
				has_variable = sb_nrp_boundary_refused_var
				var:sb_nrp_boundary_initiator_scope = scope:initiator
			}
		}
		scope:initiator = { remove_variable = sb_nrp_boundary_launch_lease_var }
		c:NRP = {
			if = { limit = { has_variable = sb_nrp_boundary_creation_failed_var } remove_variable = sb_nrp_boundary_creation_failed_var }
			if = { limit = { has_variable = sb_nrp_boundary_launch_refused_var } remove_variable = sb_nrp_boundary_launch_refused_var }
		}
	}
	# The lease still authenticates this engine root, but durable route authority
	# is gone. Never configure that root; retire its route instead of leaking it.
	else_if = {
		limit = {
			is_diplomatic_play_type = dp_sb_nrp_boundary_confrontation
			scope:initiator ?= { has_variable = sb_nrp_boundary_launch_lease_var }
		}
		scope:initiator = { remove_variable = sb_nrp_boundary_launch_lease_var }
		c:NRP ?= { sb_nrp_clear_boundary_confrontation_runtime = yes }
	}
}


sb_nrp_launch_boundary_confrontation = {
	if = {
		limit = { sb_nrp_boundary_current_launch_authority = yes has_variable = sb_nrp_boundary_refused_var NOT = { has_variable = sb_nrp_boundary_play_scope } }
		var:sb_nrp_boundary_initiator_scope = { save_temporary_scope_as = sb_nrp_boundary_initiator }
		if = {
			limit = { is_direct_subject_of = c:TRN is_subject_type = subject_type_sb_boer_confederal_partner scope:sb_nrp_boundary_initiator ?= { NOT = { any_diplomatic_play = { is_diplomatic_play_type = dp_sb_nrp_boundary_confrontation initiator = scope:sb_nrp_boundary_initiator target = c:TRN } } } }
			scope:sb_nrp_boundary_initiator = { set_variable = sb_nrp_boundary_launch_lease_var create_diplomatic_play = { type = dp_sb_nrp_boundary_confrontation target_country = c:TRN } }
		}
		else_if = {
			limit = { NOT = { AND = { is_direct_subject_of = c:TRN is_subject_type = subject_type_sb_boer_confederal_partner } } scope:sb_nrp_boundary_initiator ?= { NOT = { any_diplomatic_play = { is_diplomatic_play_type = dp_sb_nrp_boundary_confrontation initiator = scope:sb_nrp_boundary_initiator target = c:NRP } } } }
			scope:sb_nrp_boundary_initiator = { set_variable = sb_nrp_boundary_launch_lease_var create_diplomatic_play = { type = dp_sb_nrp_boundary_confrontation target_country = c:NRP } }
		}
		if = {
			limit = { scope:sb_nrp_boundary_initiator ?= { has_variable = sb_nrp_boundary_launch_lease_var } }
			scope:sb_nrp_boundary_initiator = { remove_variable = sb_nrp_boundary_launch_lease_var }
			set_variable = sb_nrp_boundary_creation_failed_var
		}
	}
}

sb_nrp_refuse_boundary_demand = {
	if = { limit = { has_variable = sb_nrp_boundary_pending_var } remove_variable = sb_nrp_boundary_pending_var }
	if = { limit = { has_variable = sb_nrp_boundary_event_open_var } remove_variable = sb_nrp_boundary_event_open_var }
	set_variable = sb_nrp_boundary_refused_var

	c:TRN ?= {
		if = { limit = { has_variable = sb_nrp_shared_refusal_commitment_var } remove_variable = sb_nrp_shared_refusal_commitment_var }

}

sb_nrp_monthly_housekeeping = {
	if = {
		limit = {
			country_definition = cd:NRP
			OR = { has_variable = sb_nrp_union_petition_pending_var has_variable = sb_nrp_union_petition_response_pending_var }
			OR = { NOT = { sb_nrp_is_reduced_republic = yes } AND = { has_variable = sb_nrp_union_petition_response_pending_var NOT = { c:TRN ?= { is_country_alive = yes is_subject = no } } } }
		}
		if = { limit = { has_variable = sb_nrp_union_petition_pending_var } remove_variable = sb_nrp_union_petition_pending_var }
		if = { limit = { has_variable = sb_nrp_union_petition_response_pending_var } remove_variable = sb_nrp_union_petition_response_pending_var }
		set_variable = sb_nrp_union_petition_lapsed_var
	}

	if = {
		limit = { country_definition = cd:NRP has_variable = sb_nrp_deferred_defence_pact_pending_var }
		if = { limit = { c:TRN ?= { is_country_alive = yes is_subject = no } is_country_alive = yes is_subject = no } sb_nrp_commit_deferred_defence_pact = yes }
		else = { remove_variable = sb_nrp_deferred_defence_pact_pending_var }
	}

	if = {
		limit = { country_definition = cd:NRP has_variable = sb_nrp_boundary_refused_var NOT = { has_variable = sb_nrp_boundary_play_scope } }
		if = {
			limit = { sb_nrp_boundary_current_political_authority = yes }
			if = {
				limit = { c:TRN ?= { is_country_alive = yes is_subject = no } NOT = { has_variable = sb_nrp_trn_defence_request_queued_var } NOT = { has_variable = sb_nrp_trn_defence_accepted_pending_var } NOT = { has_variable = sb_nrp_trn_defence_declined_pending_var } }
				set_variable = { name = sb_nrp_trn_defence_request_queued_var months = 4 }
				c:TRN = { trigger_event = { id = sb_zululand_settlement.211 days = 1 popup = yes } }
			}
			else_if = { limit = { sb_nrp_boundary_current_launch_authority = yes } sb_nrp_launch_boundary_confrontation = yes }
		}
		else = { sb_nrp_clear_boundary_confrontation_runtime = yes }
	}

	if = {
		limit = {
			country_definition = cd:NRP
			has_variable = sb_nrp_boundary_pending_var
			NOT = { has_variable = sb_nrp_boundary_not_before_var }
			NOT = { has_variable = sb_nrp_boundary_event_open_var }
		}
		if = {
			limit = { sb_nrp_boundary_demand_can_open = yes }
			sb_nrp_open_boundary_demand = yes
		}
		else_if = {
			limit = {
				OR = {
					NOT = { sb_nrp_is_greater_republic = yes }
					AND = {
						NOT = { has_variable = sb_nrp_direct_annexation_rebellion_var }
						OR = {
							NOT = { c:ZUL ?= { is_country_alive = yes } }
							NOT = { exists = s:STATE_ZULULAND.region_state:ZUL }
						}
					}
					AND = {
						has_variable = sb_nrp_direct_annexation_rebellion_var
						NOT = { exists = s:STATE_ZULULAND.region_state:NAL }
					}
					NOT = {
						OR = {
							c:NAL ?= { is_country_alive = yes }
							c:GBR ?= { is_country_alive = yes }
						}
					}
				}
			}
			remove_variable = sb_nrp_boundary_pending_var
			set_variable = sb_nrp_boundary_demand_lapsed_var
		}
	}
	if = {
		limit = {
			has_variable = sb_nrp_boundary_cutback_accepted_var
			NOT = { has_variable = sb_nrp_union_wait_var }
			NOT = { has_variable = sb_nrp_union_petition_pending_var }
			NOT = { has_variable = sb_nrp_union_petition_response_pending_var }
			sb_nrp_is_reduced_republic = yes
		}
		set_variable = sb_nrp_union_petition_pending_var
		trigger_event = { id = sb_zululand_settlement.220 days = 1 popup = yes }
	}
}
