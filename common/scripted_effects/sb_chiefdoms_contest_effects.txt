# Shared controller. No observer country owns the political result or the clock.
sb_chiefs_start_clock = {
 container:sb_zululand_chiefdoms_state ?= {
  if = {
   limit = { NOT = { has_variable = sb_chiefs_clock_started_var } }
   set_global_variable = { name = sb_chiefs_state_scope value = this }
   set_variable = sb_chiefs_clock_started_var
   set_variable = { name = sb_chiefs_origin_scope value = global_var:sb_zululand_live_zul_scope }
   add_to_variable_list = { name = sb_chiefs_countries target = global_var:sb_zululand_live_zul_scope }
  }
 }
}

# Native JE age advances independently of container-variable expiry. Record the
# last processed day before evaluation so duplicate callbacks cannot repeat it.
sb_chiefs_weekly_update = {
 scope:journal_entry ?= {
  if = {
   limit = { NOT = { has_variable = sb_chiefs_last_weekly_age_var } }
   set_variable = { name = sb_chiefs_last_weekly_age_var value = -1 }
  }
  if = {
   limit = { sb_zululand_chiefdoms_situation_active = yes journal_entry_age > var:sb_chiefs_last_weekly_age_var }
   set_variable = { name = sb_chiefs_last_weekly_age_var value = journal_entry_age }
   sb_chiefs_evaluate_contest = yes
   if = {
    limit = { sb_zululand_chiefdoms_situation_active = yes }
    sb_chiefs_ai_review_stops = yes
    sb_chiefs_sync_intervention_movements = yes
   }
  }
 }
}

# The JE reads each movement on its rebel side if that side exists, otherwise on the live country.
sb_chiefs_save_intervention_sides = {
 global_var:sb_zululand_live_zul_scope ?= {
  save_temporary_scope_as = sb_chiefs_royal_side
  save_temporary_scope_as = sb_chiefs_chiefly_side
 }
 container:sb_zululand_chiefdoms_state.var:sb_chiefs_rebels_scope ?= {
  if = { limit = { is_country_alive = yes var:sb_court_war_alignment_var ?= 1 } save_temporary_scope_as = sb_chiefs_royal_side }
  else_if = { limit = { is_country_alive = yes var:sb_court_war_alignment_var ?= 2 } save_temporary_scope_as = sb_chiefs_chiefly_side }
 }
}

sb_chiefs_sync_intervention_movements = {
 sb_chiefs_save_intervention_sides = yes
 global_var:sb_zululand_live_zul_scope ?= { sb_chiefs_sync_country_interventions = yes }
 container:sb_zululand_chiefdoms_state.var:sb_chiefs_origin_scope ?= {
  if = { limit = { NOT = { this = global_var:sb_zululand_live_zul_scope } } sb_chiefs_sync_country_interventions = yes }
 }
 container:sb_zululand_chiefdoms_state.var:sb_chiefs_rebels_scope ?= {
  if = { limit = { NOT = { this = global_var:sb_zululand_live_zul_scope } } sb_chiefs_sync_country_interventions = yes }
 }
}

sb_chiefs_sync_country_interventions = {
 every_political_movement = {
  limit = { is_political_movement_type = movement_sb_zulu_royal }
  sb_chiefs_sync_movement_modifier = { SIDE = sb_chiefs_royal_side PAYER = NAL FLAG = sb_zululand_policing_active_var MODIFIER = sb_usuthu_policing }
  sb_chiefs_sync_movement_modifier = { SIDE = sb_chiefs_royal_side PAYER = TRN FLAG = sb_zululand_usuthu_sponsorship_active_var MODIFIER = sb_usuthu_sponsorship }
  sb_chiefs_sync_movement_modifier = { SIDE = sb_chiefs_royal_side PAYER = TRN FLAG = sb_zululand_boer_volunteers_active_var MODIFIER = sb_usuthu_volunteers }
 }
 every_political_movement = {
  limit = { is_political_movement_type = movement_sb_zulu_chiefly }
  sb_chiefs_sync_movement_modifier = { SIDE = sb_chiefs_chiefly_side PAYER = NAL FLAG = sb_zululand_chief_patronage_active_var MODIFIER = sb_zululand_chiefly_patronage }
 }
}

sb_chiefs_sync_movement_modifier = {
 if = {
  limit = { owner ?= scope:$SIDE$ c:$PAYER$ ?= { has_variable = $FLAG$ } }
  if = { limit = { NOT = { has_modifier = $MODIFIER$ } } add_modifier = { name = $MODIFIER$ } }
 }
 else_if = { limit = { has_modifier = $MODIFIER$ } remove_modifier = $MODIFIER$ }
}

# The movement's native civil_war link identifies the organising object exactly.
sb_chiefs_refresh_organiser = {
 container:sb_zululand_chiefdoms_state ?= { remove_variable = sb_chiefs_organising_war_scope }
 global_var:sb_zululand_live_zul_scope ?= {
  every_political_movement = {
   limit = {
    sb_chiefs_native_mode = yes
    OR = { is_political_movement_type = movement_sb_zulu_royal is_political_movement_type = movement_sb_zulu_chiefly }
    exists = civil_war
    civil_war = { is_civil_war_type = revolution civil_war_progress < 1 }
   }
   civil_war = { save_scope_as = sb_chiefs_current_organiser }
   container:sb_zululand_chiefdoms_state = {
    set_variable = { name = sb_chiefs_organising_war_scope value = scope:sb_chiefs_current_organiser }
   }
  }
 }
}

sb_chiefs_evaluate_contest = {
 if = {
  limit = { sb_zululand_chiefdoms_situation_active = yes }
  sb_chiefs_check_house_extinction = yes
 }
 if = {
  limit = { sb_zululand_chiefdoms_situation_active = yes }
  global_var:sb_zululand_live_zul_scope ?= {
   if = { limit = { c:NAL ?= { has_variable = sb_chiefs_house_recorded_var } }
    sb_chiefs_choose_family_claimant = yes
    if = { limit = { c:NAL = { has_variable = sb_zululand_claimant_returned_var } } sb_chiefs_appoint_spokesperson = yes }
   }
  }
  sb_zululand_enforce_boer_withdrawal = yes
  sb_chiefs_refresh_organiser = yes
  if = {
   limit = { container:sb_zululand_chiefdoms_state = { has_variable = sb_chiefs_winner_scope } }
   # An authenticated result waits for native territorial reunion. It must not
   # transfer somebody else's subject or give away an uninvolved state's land.
   if = {
    limit = {
     OR = {
      NOT = { c:NAL ?= { is_country_alive = yes } }
      NOT = { container:sb_zululand_chiefdoms_state.var:sb_chiefs_winner_scope ?= { is_country_alive = yes OR = { is_direct_subject_of = c:NAL AND = { is_subject = no container:sb_zululand_chiefdoms_state.var:sb_chiefs_rebels_scope ?= this } } } }
     }
    }
    sb_chiefs_interrupt_contest = yes
   }
   else = {
   container:sb_zululand_chiefdoms_state.var:sb_chiefs_winner_scope ?= {
    if = {
     limit = { sb_chiefs_winner_ready = yes }
     if = {
      limit = { container:sb_zululand_chiefdoms_state.var:sb_chiefs_winner_alignment_var = 1 sb_chiefs_valid_claimant = yes }
      sb_zululand_resolve_royal_contest = yes
     }
     else_if = {
      limit = { container:sb_zululand_chiefdoms_state.var:sb_chiefs_winner_alignment_var = 2 }
      sb_zululand_resolve_chiefly_contest = yes
     }
    }
   }
   }
  }
  else_if = {
   limit = {
    OR = {
     container:sb_zululand_chiefdoms_state = { exists = var:sb_chiefs_organising_war_scope }
     sb_chiefs_active_split = yes
    }
   }
   # Organising or fighting is neither defeat nor a missing constitutional pair.
  }
  else_if = {
   limit = { NOT = { global_var:sb_zululand_live_zul_scope ?= { sb_chiefs_partition_intact = yes } } }
   sb_chiefs_interrupt_contest = yes
  }
  else = {
   global_var:sb_zululand_live_zul_scope = {
    if = {
     limit = {
      NOT = { sb_chiefs_native_mode = yes }
      var:sb_court_restoration_claimant_scope ?= { sb_chiefs_restorable_relative = yes }
      any_political_movement = { is_political_movement_type = movement_sb_zulu_royal political_movement_radicalism >= 1 }
     }
     if = { limit = { sb_chiefs_valid_claimant = yes } sb_zululand_resolve_royal_contest = yes }
     else = {
      # A child without a regent keeps the same durable victory receipt as a military winner.
      container:sb_zululand_chiefdoms_state = {
       set_variable = { name = sb_chiefs_winner_scope value = prev }
       set_variable = { name = sb_chiefs_winner_alignment_var value = 1 }
      }
     }
    }
    else_if = { limit = { sb_chiefs_royal_restored = yes } sb_zululand_resolve_royal_contest = yes }
    else_if = {
     limit = {
      sb_chiefs_chiefly_constitution = yes
      sb_chiefs_review_due = yes
     }
     sb_zululand_resolve_chiefly_contest = yes
    }
    # Mixed laws remain pending beyond year ten; the clock is never restarted.
   }
  }
 }
}

 if = { limit = { NOT = { has_global_variable = sb_nrp_chiefdoms_earned_var } } remove_global_variable = sb_nrp_chiefdoms_promise_var }

}

# Any caller, including foreign events and the contextless journal.
sb_chiefs_clear_prominence_pressure = {
 global_var:sb_zululand_live_zul_scope ?= { sb_chiefs_clear_country_prominence_pressure = yes }
 container:sb_zululand_chiefdoms_state ?= {
  var:sb_chiefs_origin_scope ?= { sb_chiefs_clear_country_prominence_pressure = yes }
  var:sb_chiefs_rebels_scope ?= { sb_chiefs_clear_country_prominence_pressure = yes }
 }
}

sb_chiefs_update_prominence_pressure = {
 sb_chiefs_clear_prominence_pressure = yes
 if = {
  limit = { sb_zululand_chiefdoms_situation_active = yes NOT = { sb_chiefs_prominence_pressure = 0 } }
  if = {
   limit = { container:sb_zululand_chiefdoms_state ?= { var:sb_chiefs_rebels_scope ?= { is_country_alive = yes var:sb_court_war_alignment_var ?= 1 } } }
   container:sb_zululand_chiefdoms_state.var:sb_chiefs_rebels_scope = {
    every_political_movement = {
     limit = { is_political_movement_type = movement_sb_zulu_royal }
     add_modifier = { name = sb_chiefs_prominence_pressure multiplier = sb_chiefs_prominence_pressure }
    }
   }
  }
  else = {
   global_var:sb_zululand_live_zul_scope ?= {
    if = { limit = { is_country_alive = yes }
     every_political_movement = {
      limit = { is_political_movement_type = movement_sb_zulu_royal }
      add_modifier = { name = sb_chiefs_prominence_pressure multiplier = sb_chiefs_prominence_pressure }
     }
    }
   }
  }
 }
}

# Do not infer extinction from a missing adult, claimant reference, or court in exile.
sb_chiefs_check_house_extinction = {
 if = {
  limit = {
   sb_zululand_chiefdoms_situation_active = yes
   NOT = { has_global_variable = sb_chiefs_house_extinct_var }
   c:NAL ?= {
    has_variable = sb_chiefs_house_recorded_var
    NOT = { any_in_list = { variable = sb_chiefs_royal_house is_character_alive = yes NOT = { has_variable = sb_chiefs_death_recorded_var } } }
   }
  }
  set_global_variable = sb_chiefs_house_extinct_var
  clear_saved_scope = sb_chiefs_settled_country
  container:sb_zululand_chiefdoms_state = {
   if = {
    limit = { var:sb_chiefs_rebels_scope ?= { is_country_alive = yes } var:sb_chiefs_origin_scope ?= { is_country_alive = yes } }
    set_global_variable = sb_chiefs_extinction_awaiting_reunion_var
    var:sb_chiefs_origin_scope = { set_variable = { name = sb_chiefs_extinction_reunion_var value = prev.var:sb_chiefs_rebels_scope } }
    var:sb_chiefs_rebels_scope = { set_variable = { name = sb_chiefs_extinction_reunion_var value = prev.var:sb_chiefs_origin_scope } }
   }
   if = { limit = { var:sb_chiefs_origin_scope ?= { is_country_alive = yes is_direct_subject_of = c:NAL } }
    var:sb_chiefs_origin_scope = { save_scope_as = sb_chiefs_settled_country }
   }
   else_if = { limit = { var:sb_chiefs_rebels_scope ?= { is_country_alive = yes } }
    var:sb_chiefs_rebels_scope = { save_scope_as = sb_chiefs_settled_country }
   }
   else = { global_var:sb_zululand_live_zul_scope ?= { save_scope_as = sb_chiefs_settled_country } }
  }
  scope:sb_chiefs_settled_country ?= {
   set_variable = { name = sb_zululand_contest_result_var value = 2 }
   sb_zululand_finalize_political_contest = yes
  }
 }
}

# Territorial reunion can complete after either native military callback.
sb_chiefs_finish_extinction_reunion = {
 if = {
  limit = {
   has_global_variable = sb_chiefs_extinction_awaiting_reunion_var
   OR = { country_definition = cd:ZUL has_variable = sb_chiefs_extinction_reunion_var }
   is_country_alive = yes
   OR = {
    owns_entire_state_region = STATE_ZULULAND
    AND = {
     any_scope_state = { state_region = s:STATE_ZULULAND }
     NOT = { var:sb_chiefs_extinction_reunion_var ?= { is_country_alive = yes } }
    }
   }
   c:NAL ?= { is_country_alive = yes }
  }
  remove_global_variable = sb_chiefs_extinction_awaiting_reunion_var
  set_global_variable = { name = sb_zululand_live_zul_scope value = this }
  set_variable = { name = sb_zululand_contest_result_var value = 2 }
  sb_chiefs_install_council = yes
  sb_chiefs_normalize_settled_subject = yes
  every_scope_state = {
   limit = { state_region = s:STATE_ZULULAND NOT = { has_modifier = sb_zululand_chiefdom_reserves } }
   add_modifier = { name = sb_zululand_chiefdom_reserves }
  }
  sb_institutions_reconcile = yes
 }
