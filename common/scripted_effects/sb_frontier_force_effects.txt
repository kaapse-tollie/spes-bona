# Restores the intended levy floor without replacing later unit upgrades or
# creating duplicate formations. The AI strategy prevents repeated trimming;
# this effect handles old saves and the first AI tick after a play begins.

sb_save_frontier_recruitment_state = {
	if = {
		limit = {
			country_definition = cd:ZUL
			any_scope_state = { state_region = s:STATE_ZULULAND }
		}
		random_scope_state = {
			limit = { state_region = s:STATE_ZULULAND }
			save_temporary_scope_as = sb_frontier_recruitment_state
		}
	}
	else_if = {
		limit = {
			country_definition = cd:SWZ
			any_scope_state = { state_region = s:STATE_EAST_TRANSVAAL }
		}
		random_scope_state = {
			limit = { state_region = s:STATE_EAST_TRANSVAAL }
			save_temporary_scope_as = sb_frontier_recruitment_state
		}
	}
	else_if = {
		limit = {
			country_definition = cd:GZA
			any_scope_state = { state_region = s:STATE_LOURENCO_MARQUES }
		}
		random_scope_state = {
			limit = { state_region = s:STATE_LOURENCO_MARQUES }
			save_temporary_scope_as = sb_frontier_recruitment_state
		}
	}
	else_if = {
		limit = {
			country_definition = cd:BST
			any_scope_state = { state_region = s:STATE_DRAKENSBERG }
		}
		random_scope_state = {
			limit = { state_region = s:STATE_DRAKENSBERG }
			save_temporary_scope_as = sb_frontier_recruitment_state
		}
	}

	if = {
		limit = { NOT = { exists = scope:sb_frontier_recruitment_state } }
		capital = { save_temporary_scope_as = sb_frontier_recruitment_state }
	}
}

sb_restore_ai_frontier_force_floor = {
	if = {
		limit = { sb_frontier_ai_force_floor_eligible = yes }
		save_temporary_scope_as = sb_frontier_force_country
		sb_save_frontier_recruitment_state = yes

		scope:sb_frontier_recruitment_state = {
			if = {
				limit = {
					scope:sb_frontier_force_country = {
						sb_frontier_regular_force_shortfall > 0
					}
				}
				create_building = {
					building = building_barrack
					level = scope:sb_frontier_force_country.sb_frontier_regular_force_shortfall
					reserves = 1
				}
			}
			if = {
				limit = {
					scope:sb_frontier_force_country = {
						sb_frontier_conscript_force_shortfall > 0
					}
				}
				create_building = {
					building = building_conscription_center
					level = scope:sb_frontier_force_country.sb_frontier_conscript_force_shortfall
					reserves = 1
				}
			}
		}

		if = {
			limit = { has_modifier = sb_frontier_levy_reconstitution }
			remove_modifier = sb_frontier_levy_reconstitution
		}
		if = {
			limit = { has_modifier = sb_amabutho_levy_reconstitution }
			remove_modifier = sb_amabutho_levy_reconstitution
		}
		if = {
			limit = {
				OR = {
					country_definition = cd:ZUL
					country_definition = cd:SWZ
					country_definition = cd:GZA
				}
			}
			add_modifier = {
				name = sb_amabutho_levy_reconstitution
				months = 3
			}
		}
		else = {
			add_modifier = {
				name = sb_frontier_levy_reconstitution
				months = 3
			}
		}
	}
}

sb_restore_ai_frontier_force_floors_for_play = {
	if = {
		limit = { any_scope_play_involved = { country_definition = cd:ZUL } }
		c:ZUL ?= { sb_restore_ai_frontier_force_floor = yes }
	}
	if = {
		limit = { any_scope_play_involved = { country_definition = cd:SWZ } }
		c:SWZ ?= { sb_restore_ai_frontier_force_floor = yes }
	}
	if = {
		limit = { any_scope_play_involved = { country_definition = cd:GZA } }
		c:GZA ?= { sb_restore_ai_frontier_force_floor = yes }
	}
	if = {
		limit = { any_scope_play_involved = { country_definition = cd:BST } }
		c:BST ?= { sb_restore_ai_frontier_force_floor = yes }
	}
}

sb_cleanup_ai_frontier_force_support = {
	if = {
		limit = {
			has_modifier = sb_frontier_levy_reconstitution
			NOR = {
				is_active_in_diplomatic_play = yes
				is_at_war = yes
			}
		}
		remove_modifier = sb_frontier_levy_reconstitution
	}
	if = {
		limit = {
			has_modifier = sb_amabutho_levy_reconstitution
			NOR = {
				is_active_in_diplomatic_play = yes
				is_at_war = yes
			}
		}
		remove_modifier = sb_amabutho_levy_reconstitution
	}
}

sb_cleanup_ai_frontier_force_support_for_play = {
	c:ZUL ?= { sb_cleanup_ai_frontier_force_support = yes }
	c:SWZ ?= { sb_cleanup_ai_frontier_force_support = yes }
	c:GZA ?= { sb_cleanup_ai_frontier_force_support = yes }
	c:BST ?= { sb_cleanup_ai_frontier_force_support = yes }
}
