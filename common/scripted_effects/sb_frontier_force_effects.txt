# Restores historical frontier force structures without replacing later unit
# upgrades. Play-scoped routing selects Frontier AI History for AI-AI wars and
# Player Challenge against a committed player opponent.

sb_save_frontier_recruitment_state = {
	if = {
		limit = {
			country_definition = cd:ZUL
			any_scope_state = { state_region = s:STATE_ZULULAND }
		}
		random_scope_state = {
			limit = { state_region = s:STATE_ZULULAND }
			save_temporary_scope_as = sb_frontier_recruitment_state
		}
	}
	else_if = {
		limit = {
			country_definition = cd:SWZ
			any_scope_state = { state_region = s:STATE_EAST_TRANSVAAL }
		}
		random_scope_state = {
			limit = { state_region = s:STATE_EAST_TRANSVAAL }
			save_temporary_scope_as = sb_frontier_recruitment_state
		}
	}
	else_if = {
		limit = {
			country_definition = cd:GZA
			any_scope_state = { state_region = s:STATE_LOURENCO_MARQUES }
		}
		random_scope_state = {
			limit = { state_region = s:STATE_LOURENCO_MARQUES }
			save_temporary_scope_as = sb_frontier_recruitment_state
		}
	}
	else_if = {
		limit = {
			country_definition = cd:BST
			any_scope_state = { state_region = s:STATE_DRAKENSBERG }
		}
		random_scope_state = {
			limit = { state_region = s:STATE_DRAKENSBERG }
			save_temporary_scope_as = sb_frontier_recruitment_state
		}
	}
	else_if = {
		limit = {
			country_definition = cd:XHO
			any_scope_state = { state_region = s:STATE_EASTERN_CAPE }
		}
		random_scope_state = {
			limit = { state_region = s:STATE_EASTERN_CAPE }
			save_temporary_scope_as = sb_frontier_recruitment_state
		}
	}
	if = {
		limit = { NOT = { exists = scope:sb_frontier_recruitment_state } }
		capital = { save_temporary_scope_as = sb_frontier_recruitment_state }
	}
}

sb_restore_ai_frontier_force_floor = {
	if = {
		limit = {
			is_ai = yes
			is_country_alive = yes
			OR = {
				country_definition = cd:ZUL
				country_definition = cd:SWZ
				country_definition = cd:GZA
				country_definition = cd:BST
				country_definition = cd:XHO
			}
			OR = {
				is_active_in_diplomatic_play = yes
				is_at_war = yes
			}
		}
		save_temporary_scope_as = sb_frontier_force_country
		sb_save_frontier_recruitment_state = yes

		scope:sb_frontier_recruitment_state = {
			if = {
				limit = {
					scope:sb_frontier_force_country = {
						sb_frontier_regular_force_shortfall > 0
					}
				}
				create_building = {
					building = building_barrack
					level = scope:sb_frontier_force_country.sb_frontier_regular_force_shortfall
					reserves = 1
				}
			}
			if = {
				limit = {
					scope:sb_frontier_force_country = {
						sb_frontier_conscript_force_shortfall > 0
					}
				}
				create_building = {
					building = building_conscription_center
					level = scope:sb_frontier_force_country.sb_frontier_conscript_force_shortfall
					reserves = 1
				}
			}
		}
	}
}

sb_restore_ai_frontier_force_floors_for_play = {
	if = {
		limit = {
			sb_frontier_play_artificial_assistance_enabled = { RECIPIENT = c:ZUL }
		}
		c:ZUL ?= { sb_restore_ai_frontier_force_floor = yes }
	}
	if = {
		limit = {
			sb_frontier_play_artificial_assistance_enabled = { RECIPIENT = c:SWZ }
		}
		c:SWZ ?= { sb_restore_ai_frontier_force_floor = yes }
	}
	if = {
		limit = {
			sb_frontier_play_artificial_assistance_enabled = { RECIPIENT = c:GZA }
		}
		c:GZA ?= { sb_restore_ai_frontier_force_floor = yes }
	}
	if = {
		limit = {
			sb_frontier_play_artificial_assistance_enabled = { RECIPIENT = c:BST }
		}
		c:BST ?= { sb_restore_ai_frontier_force_floor = yes }
	}
	if = {
		limit = {
			sb_frontier_play_artificial_assistance_enabled = { RECIPIENT = c:XHO }
		}
		c:XHO ?= { sb_restore_ai_frontier_force_floor = yes }
	}
}

sb_mobilize_ai_frontier_historical_formation = {
	if = {
		limit = {
			is_country_alive = yes
			is_ai = yes
			OR = {
				country_definition = cd:ZUL
				country_definition = cd:SWZ
				country_definition = cd:GZA
				country_definition = cd:BST
				country_definition = cd:XHO
			}
		}
		every_military_formation = {
			limit = { is_army = yes }
			fully_mobilize_army = yes
		}
	}
}
