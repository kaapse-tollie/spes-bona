# Spes Bona - Swazi frontier and accommodation helpers
#
# Chain order:
#   1. TRN completes Transvaal Unity
#   2. TRN chooses whether to press the western border claims
#   3. SWZ answers Boer consolidation pressure (cede land or resist)
#   4. only after that, and only if Natal is British, SWZ can choose whether to
#      seek British shelter or stand apart

sb_swazi_raise_defensive_muster = {
	if = {
		limit = {
			NOT = { has_variable = sb_swazi_defensive_muster_var }
		}
		set_variable = {
			name = sb_swazi_defensive_muster_var
			months = 24
		}
		add_treasury = 10000
		if = {
			limit = { NOT = { has_modifier = sb_swazi_frontier_muster } }
			add_modifier = {
				name = sb_swazi_frontier_muster
				months = 24
			}
		}
	}
}

sb_swazi_clear_defensive_muster = {
	if = {
		limit = { has_variable = sb_swazi_defensive_muster_var }
		remove_variable = sb_swazi_defensive_muster_var
		# Reverse the one-off artificial grant when a late player join routes the
		# play to a disabled Player Challenge package.
		add_treasury = -10000
		if = {
			limit = { has_modifier = sb_swazi_frontier_muster }
			remove_modifier = sb_swazi_frontier_muster
		}
	}
}

sb_swazi_end_defensive_muster = {
	if = {
		limit = { has_variable = sb_swazi_defensive_muster_var }
		remove_variable = sb_swazi_defensive_muster_var
		if = {
			limit = { has_modifier = sb_swazi_frontier_muster }
			remove_modifier = sb_swazi_frontier_muster
		}
	}
}

# Pre-play AI-history roll marker. The visible assistance is not applied until
# the diplomatic-play start hook confirms that SWZ is the primary defender.
sb_zulu_face_swazi_defensive_muster = {
	c:SWZ ?= {
		set_variable = {
			name = sb_swazi_ai_history_muster_selected_var
			months = 24
		}
	}
}

sb_apply_swazi_defensive_muster_for_play = {
	c:SWZ ?= { sb_swazi_raise_defensive_muster = yes }
	initiator ?= {
		if = {
			limit = {
				country_definition = cd:ZUL
				is_diplomatic_play_enemy_of = c:SWZ
				NOT = { has_modifier = sb_zulu_swazi_frontier_disarray }
			}
			add_modifier = {
				name = sb_zulu_swazi_frontier_disarray
				months = 12
			}
			set_variable = {
				name = sb_zulu_swazi_disarray_from_muster_var
				months = 12
			}
		}
	}
}

sb_clear_swazi_defensive_muster_for_play = {
	c:SWZ ?= { sb_swazi_clear_defensive_muster = yes }
	initiator ?= {
		if = {
			limit = { has_variable = sb_zulu_swazi_disarray_from_muster_var }
			remove_variable = sb_zulu_swazi_disarray_from_muster_var
			if = {
				limit = { has_modifier = sb_zulu_swazi_frontier_disarray }
				remove_modifier = sb_zulu_swazi_frontier_disarray
			}
		}
	}
}

# Diplomatic-play scope. SWZ must be the primary defender; interested countries
# and secondary backers never qualify. A player must be a committed enemy in
# this same play before Player Challenge can replace the AI-History route.
sb_route_swazi_defensive_muster_for_play = {
	if = {
		limit = {
			is_diplomatic_play_type = dp_conquer_state
			initiator = c:ZUL
			target = c:SWZ
			c:ZUL ?= { has_variable = sb_zulu_swazi_play_scope var:sb_zulu_swazi_play_scope = root }
			c:SWZ ?= { is_country_alive = yes is_ai = yes }
		}
		if = {
			limit = {
				sb_frontier_play_has_committed_player_enemy = { RECIPIENT = c:SWZ }
			}
			if = {
				limit = { sb_frontier_player_challenge_enabled = yes }
				sb_apply_swazi_defensive_muster_for_play = yes
			}
			else = {
				sb_clear_swazi_defensive_muster_for_play = yes
			}
		}
		else_if = {
			limit = {
				sb_frontier_ai_scripting_enabled = yes
				c:SWZ ?= { has_variable = sb_swazi_ai_history_muster_selected_var }
			}
			sb_apply_swazi_defensive_muster_for_play = yes
		}
		else = {
			sb_clear_swazi_defensive_muster_for_play = yes
		}
	}
}

sb_cleanup_swazi_defensive_muster_for_play = {
	if = {
		limit = {
			is_diplomatic_play_type = dp_conquer_state
			exists = initiator
			exists = target
			initiator ?= { country_definition = cd:ZUL has_variable = sb_zulu_swazi_play_scope var:sb_zulu_swazi_play_scope = root }
			target ?= { country_definition = cd:SWZ }
		}
		# The play retains its participants even when conquest removes the SWZ tag.
		target ?= {
			sb_swazi_end_defensive_muster = yes
			if = {
				limit = { has_variable = sb_swazi_ai_history_muster_selected_var }
				remove_variable = sb_swazi_ai_history_muster_selected_var
			}
		}
		initiator ?= {
			if = {
				limit = { has_variable = sb_zulu_swazi_disarray_from_muster_var }
				remove_variable = sb_zulu_swazi_disarray_from_muster_var
				if = {
					limit = { has_modifier = sb_zulu_swazi_frontier_disarray }
					remove_modifier = sb_zulu_swazi_frontier_disarray
				}
			}
		}
	}
}

sb_swazi_apply_guns_for_land_bonus = {
	add_modifier = { name = sb_swazi_frontier_rearmed months = 60 }
	if = {
		limit = {
			has_journal_entry = je_sb_adopt_firearms
			NOT = { has_variable = sb_firearms_modernized_var }
		}
		if = {
			limit = { NOT = { has_variable = sb_firearms_supply_weeks_var } }
			set_variable = { name = sb_firearms_supply_weeks_var value = 0 }
		}
		change_variable = { name = sb_firearms_supply_weeks_var add = 104 }
		sb_update_firearms_progress = yes
		sb_sync_iron_age_weapons_penalty = yes
	}
}

# Pop scope. Each culture holds its census by religion for the duration of the sale.
sb_swazi_count_cession_pop = {
	religion = { save_temporary_scope_as = sb_swazi_cession_faith }
	set_local_variable = {
		name = sb_swazi_census_total
		value = {
			value = 0
			if = {
				limit = { culture = { is_key_in_variable_map = { name = $MAP$ target = scope:sb_swazi_cession_faith } } }
				add = "culture.variable_map($MAP$|scope:sb_swazi_cession_faith)"
			}
			$OPERATION$ = total_size
		}
	}
	culture = {
		if = {
			limit = { NOR = { has_variable_map = sb_swazi_seller_loss has_variable_map = sb_swazi_buyer_gain } }
			add_to_temporary_list = sb_swazi_census_cultures
		}
		if = {
			limit = { is_key_in_variable_map = { name = $MAP$ target = scope:sb_swazi_cession_faith } }
			remove_from_variable_map = { name = $MAP$ key = scope:sb_swazi_cession_faith }
		}
		add_to_variable_map = { name = $MAP$ key = scope:sb_swazi_cession_faith value = local_var:sb_swazi_census_total }
	}
}

# TRN Pop scope. Protect both the established Pop and the resident total of its culture and faith.
sb_swazi_return_cession_pop = {
	religion = { save_temporary_scope_as = sb_swazi_cession_faith }
	if = {
		limit = { culture = { is_key_in_variable_map = { name = sb_swazi_seller_loss target = scope:sb_swazi_cession_faith } } }
		set_local_variable = {
			name = sb_swazi_unmatched
			value = {
				value = "culture.variable_map(sb_swazi_seller_loss|scope:sb_swazi_cession_faith)"
				max = "culture.variable_map(sb_swazi_buyer_gain|scope:sb_swazi_cession_faith)"
			}
		}
		set_local_variable = {
			name = sb_swazi_arrivals
			value = {
				value = total_size
				if = {
					limit = { is_key_in_local_variable_map = { name = sb_swazi_trn_residents target = this } }
					subtract = "local_variable_map(sb_swazi_trn_residents|this)"
				}
				max = local_var:sb_swazi_unmatched
				min = 0
			}
		}
		if = {
			limit = { local_var:sb_swazi_arrivals > 0 }
			change_local_variable = { name = sb_swazi_unmatched subtract = local_var:sb_swazi_arrivals }
			culture = {
				remove_from_variable_map = { name = sb_swazi_seller_loss key = scope:sb_swazi_cession_faith }
				add_to_variable_map = { name = sb_swazi_seller_loss key = scope:sb_swazi_cession_faith value = local_var:sb_swazi_unmatched }
			}
			# Carry an odd person into the next Pop, even across cultures or faiths.
			change_local_variable = { name = sb_swazi_remainder add = local_var:sb_swazi_arrivals }
			set_local_variable = { name = sb_swazi_returning value = { value = local_var:sb_swazi_remainder divide = 2 floor = yes } }
			change_local_variable = { name = sb_swazi_remainder modulo = 2 }
			if = {
				limit = { local_var:sb_swazi_returning > 0 }
				move_partial_pop = { state = scope:sb_swazi_retained_state population = local_var:sb_swazi_returning }
			}
		}
	}
}

# Keep all third-party districts in one native grant. Splitting one owner's land
# a province at a time would let native Pop and building rounding compound.
sb_swazi_settle_outside_districts = {
	if = {
		limit = { sb_swazi_district_has_outside_owner = { DISTRICT = x73CDE8 } sb_swazi_district_has_outside_owner = { DISTRICT = x25A652 } }
		sb_swazi_settle_remaining_outside_districts = { DISTRICTS = "x73CDE8 x25A652" }
	}
	else_if = {
		limit = { sb_swazi_district_has_outside_owner = { DISTRICT = x73CDE8 } }
		sb_swazi_settle_remaining_outside_districts = { DISTRICTS = x73CDE8 }
	}
	else_if = {
		limit = { sb_swazi_district_has_outside_owner = { DISTRICT = x25A652 } }
		sb_swazi_settle_remaining_outside_districts = { DISTRICTS = x25A652 }
	}
	else_if = {
		limit = { sb_swazi_district_has_outside_owner = { DISTRICT = x50A011 } sb_swazi_district_has_outside_owner = { DISTRICT = x452304 } }
		set_owner_of_provinces = { country = c:TRN provinces = { x50A011 x452304 } }
	}
	else_if = {
		limit = { sb_swazi_district_has_outside_owner = { DISTRICT = x50A011 } }
		set_owner_of_provinces = { country = c:TRN provinces = { x50A011 } }
	}
	else_if = {
		limit = { sb_swazi_district_has_outside_owner = { DISTRICT = x452304 } }
		set_owner_of_provinces = { country = c:TRN provinces = { x452304 } }
	}
}

sb_swazi_settle_remaining_outside_districts = {
	if = {
		limit = { sb_swazi_district_has_outside_owner = { DISTRICT = x50A011 } sb_swazi_district_has_outside_owner = { DISTRICT = x452304 } }
		set_owner_of_provinces = { country = c:TRN provinces = { $DISTRICTS$ x50A011 x452304 } }
	}
	else_if = {
		limit = { sb_swazi_district_has_outside_owner = { DISTRICT = x50A011 } }
		set_owner_of_provinces = { country = c:TRN provinces = { $DISTRICTS$ x50A011 } }
	}
	else_if = {
		limit = { sb_swazi_district_has_outside_owner = { DISTRICT = x452304 } }
		set_owner_of_provinces = { country = c:TRN provinces = { $DISTRICTS$ x452304 } }
	}
	else = { set_owner_of_provinces = { country = c:TRN provinces = { $DISTRICTS$ } } }
}

sb_swazi_apply_boer_buy_time_settlement = {
	if = {
		limit = { NOT = { has_variable = sb_swazi_boer_pressure_resolved_var } }
		sb_swazi_settle_boer_frontier = yes
	}
}

sb_swazi_settle_boer_frontier = {
	# "Buy Time" is deliberately narrow: only the disputed frontier pocket changes
	# hands, and the payoff is security + firearms progress rather than a full
	# subject relationship.
	set_variable = sb_swazi_boer_pressure_resolved_var
	set_global_variable = sb_swazi_boer_border_resolved_global_var
	set_variable = sb_swazi_boer_land_sold_var
	if = {
		limit = { has_variable = sb_swazi_boer_resisted_var }
		remove_variable = sb_swazi_boer_resisted_var
	}

	s:STATE_EAST_TRANSVAAL = {
		hidden_effect = {
			# Only the SWZ-to-TRN handover belongs between the two censuses.
			sb_swazi_settle_outside_districts = yes
			region_state:SWZ ?= {
				every_scope_pop = { sb_swazi_count_cession_pop = { MAP = sb_swazi_seller_loss OPERATION = add } }
			}
			region_state:TRN ?= {
				every_scope_pop = {
					add_to_local_variable_map = { name = sb_swazi_trn_residents key = this value = total_size }
					sb_swazi_count_cession_pop = { MAP = sb_swazi_buyer_gain OPERATION = subtract }
				}
			}
		}
		set_owner_of_provinces = {
			country = c:TRN
			provinces = {
				x73CDE8 x25A652 x50A011 x452304
			}
		}
		hidden_effect = {
			if = {
				limit = { exists = region_state:SWZ exists = region_state:TRN }
				region_state:SWZ = {
					save_temporary_scope_as = sb_swazi_retained_state
					every_scope_pop = { sb_swazi_count_cession_pop = { MAP = sb_swazi_seller_loss OPERATION = subtract } }
				}
				region_state:TRN = {
					every_scope_pop = { sb_swazi_count_cession_pop = { MAP = sb_swazi_buyer_gain OPERATION = add } }
					set_local_variable = { name = sb_swazi_remainder value = 0 }
					every_scope_pop = { sb_swazi_return_cession_pop = yes }
				}
			}
			every_in_list = {
				list = sb_swazi_census_cultures
				if = { limit = { has_variable_map = sb_swazi_seller_loss } clear_variable_map = sb_swazi_seller_loss }
				if = { limit = { has_variable_map = sb_swazi_buyer_gain } clear_variable_map = sb_swazi_buyer_gain }
			}
		}
		remove_claim = c:SWZ
	}

	change_relations = {
		country = c:TRN
		value = 25
	}
	hidden_effect = {
		create_bidirectional_truce = {
			country = c:TRN
			months = 300
		}
	}
	custom_description = {
		text = sb_swzai_frontier_concede_truce
	}
	hidden_effect = {
		sb_create_trn_swz_firearms_treaty = yes
		sb_swazi_apply_guns_for_land_bonus = yes
	}
	custom_description = {
		text = sb_swzai_frontier_concede_guns
	}



}

sb_trn_apply_swazi_reject_posture = {
	# Attitudes are derived, so make the inputs unambiguous: a rejected frontier
	# settlement leaves TRN hostile and, for AI TRN, actively claim-focused.
	if = {
		limit = {
			c:SWZ ?= { is_country_alive = yes }
		}
		hidden_effect = {
			change_relations = {
				country = c:SWZ
				value = -100
			}
		}
	}
	if = {
		limit = { is_ai = yes }
		set_strategy = ai_strategy_sb_boer_frontier_republic
	}
}

sb_swazi_apply_boer_resistance = {
	# Resistance does not immediately force a war; it creates hostility and a
	# grievance window for later pressure instead.
	set_variable = sb_swazi_boer_pressure_resolved_var
	set_global_variable = sb_swazi_boer_border_resolved_global_var
	set_variable = sb_swazi_boer_resisted_var
	if = {
		limit = { has_variable = sb_swazi_boer_land_sold_var }
		remove_variable = sb_swazi_boer_land_sold_var
	}

	c:TRN ?= {
		sb_trn_apply_swazi_reject_posture = yes
	}

}
