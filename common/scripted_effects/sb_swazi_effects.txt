# Spes Bona - Swazi frontier and accommodation helpers
#
# Chain order:
#   1. TRN completes Transvaal Unity
#   2. TRN chooses whether to press the western border claims
#   3. SWZ answers Boer consolidation pressure (cede land or resist)
#   4. only after that, and only if Natal is British, SWZ can choose whether to
#      seek British shelter or stand apart

sb_swazi_raise_defensive_muster = {
	if = {
		limit = {
			NOT = { has_variable = sb_swazi_defensive_muster_var }
		}
		set_variable = {
			name = sb_swazi_defensive_muster_var
			months = 24
		}
		add_treasury = 10000
		if = {
			limit = { NOT = { has_modifier = sb_swazi_frontier_muster } }
			add_modifier = {
				name = sb_swazi_frontier_muster
				months = 24
			}
		}
	}
}

sb_swazi_clear_defensive_muster = {
	if = {
		limit = { has_variable = sb_swazi_defensive_muster_var }
		remove_variable = sb_swazi_defensive_muster_var
		# Reverse the one-off artificial grant when a late player join routes the
		# play to a disabled Player Challenge package.
		add_treasury = -10000
		if = {
			limit = { has_modifier = sb_swazi_frontier_muster }
			remove_modifier = sb_swazi_frontier_muster
		}
	}
}

sb_swazi_end_defensive_muster = {
	if = {
		limit = { has_variable = sb_swazi_defensive_muster_var }
		remove_variable = sb_swazi_defensive_muster_var
		if = {
			limit = { has_modifier = sb_swazi_frontier_muster }
			remove_modifier = sb_swazi_frontier_muster
		}
	}
}

# Pre-play AI-history roll marker. The visible assistance is not applied until
# the diplomatic-play start hook confirms that SWZ is the primary defender.
sb_zulu_face_swazi_defensive_muster = {
	c:SWZ ?= {
		set_variable = {
			name = sb_swazi_ai_history_muster_selected_var
			months = 24
		}
	}
}

sb_apply_swazi_defensive_muster_for_play = {
	c:SWZ ?= { sb_swazi_raise_defensive_muster = yes }
	initiator ?= {
		if = {
			limit = {
				country_definition = cd:ZUL
				is_diplomatic_play_enemy_of = c:SWZ
				NOT = { has_modifier = sb_zulu_swazi_frontier_disarray }
			}
			add_modifier = {
				name = sb_zulu_swazi_frontier_disarray
				months = 12
			}
			set_variable = {
				name = sb_zulu_swazi_disarray_from_muster_var
				months = 12
			}
		}
	}
}

sb_clear_swazi_defensive_muster_for_play = {
	c:SWZ ?= { sb_swazi_clear_defensive_muster = yes }
	initiator ?= {
		if = {
			limit = { has_variable = sb_zulu_swazi_disarray_from_muster_var }
			remove_variable = sb_zulu_swazi_disarray_from_muster_var
			if = {
				limit = { has_modifier = sb_zulu_swazi_frontier_disarray }
				remove_modifier = sb_zulu_swazi_frontier_disarray
			}
		}
	}
}

# Diplomatic-play scope. SWZ must be the primary defender; interested countries
# and secondary backers never qualify. A player must be a committed enemy in
# this same play before Player Challenge can replace the AI-History route.
sb_route_swazi_defensive_muster_for_play = {
	if = {
		limit = {
			is_diplomatic_play_type = dp_conquer_state
			initiator = c:ZUL
			target = c:SWZ
			c:ZUL ?= { has_variable = sb_zulu_swazi_play_scope var:sb_zulu_swazi_play_scope = root }
			c:SWZ ?= { is_country_alive = yes is_ai = yes }
		}
		if = {
			limit = {
				sb_frontier_play_has_committed_player_enemy = { RECIPIENT = c:SWZ }
			}
			if = {
				limit = { sb_frontier_player_challenge_enabled = yes }
				sb_apply_swazi_defensive_muster_for_play = yes
			}
			else = {
				sb_clear_swazi_defensive_muster_for_play = yes
			}
		}
		else_if = {
			limit = {
				sb_frontier_ai_scripting_enabled = yes
				c:SWZ ?= { has_variable = sb_swazi_ai_history_muster_selected_var }
			}
			sb_apply_swazi_defensive_muster_for_play = yes
		}
		else = {
			sb_clear_swazi_defensive_muster_for_play = yes
		}
	}
}

sb_cleanup_swazi_defensive_muster_for_play = {
	if = {
		limit = {
			is_diplomatic_play_type = dp_conquer_state
			c:ZUL ?= { has_variable = sb_zulu_swazi_play_scope var:sb_zulu_swazi_play_scope = root }
			initiator = c:ZUL
			target = c:SWZ
		}
		c:SWZ ?= {
			sb_swazi_end_defensive_muster = yes
			if = {
				limit = { has_variable = sb_swazi_ai_history_muster_selected_var }
				remove_variable = sb_swazi_ai_history_muster_selected_var
			}
		}
		c:ZUL ?= {
			if = {
				limit = { has_variable = sb_zulu_swazi_disarray_from_muster_var }
				remove_variable = sb_zulu_swazi_disarray_from_muster_var
				if = {
					limit = { has_modifier = sb_zulu_swazi_frontier_disarray }
					remove_modifier = sb_zulu_swazi_frontier_disarray
				}
			}
		}
	}
}

sb_swazi_apply_guns_for_land_bonus = {
	add_modifier = {
		name = sb_swazi_frontier_rearmed
		months = 60
	}

	if = {
		limit = {
			has_journal_entry = je_sb_adopt_firearms
			NOT = { has_variable = sb_firearms_modernized_var }
		}
		if = {
			limit = {
				NOT = { has_variable = sb_firearms_treaty_progress_months_var }
			}
			set_variable = {
				name = sb_firearms_treaty_progress_months_var
				value = 0
			}
		}
		if = {
			limit = {
				NOT = { has_variable = sb_firearms_industry_progress_months_var }
			}
			set_variable = {
				name = sb_firearms_industry_progress_months_var
				value = 0
			}
		}
		if = {
			limit = { var:sb_firearms_treaty_progress_months_var < 24 }
			change_variable = {
				name = sb_firearms_treaty_progress_months_var
				add = 9.6
			}
		}
		if = {
			limit = { var:sb_firearms_industry_progress_months_var < 24 }
			change_variable = {
				name = sb_firearms_industry_progress_months_var
				add = 9.6
			}
		}
		sb_update_firearms_progress = yes
		sb_sync_iron_age_weapons_penalty = yes
	}
}

sb_swazi_apply_boer_buy_time_settlement = {
	# "Buy Time" is deliberately narrow: only the disputed frontier pocket changes
	# hands, and the payoff is security + firearms progress rather than a full
	# subject relationship.
	set_variable = sb_swazi_boer_pressure_resolved_var
	set_variable = sb_swazi_boer_land_sold_var
	if = {
		limit = { has_variable = sb_swazi_boer_resisted_var }
		remove_variable = sb_swazi_boer_resisted_var
	}

	s:STATE_EAST_TRANSVAAL = {
		set_owner_of_provinces = {
			country = c:TRN
			provinces = {
				x73CDE8 x25A652 x50A011 x452304
			}
		}
		remove_claim = c:SWZ
	}

	change_relations = {
		country = c:TRN
		value = 25
	}
	hidden_effect = {
		create_bidirectional_truce = {
			country = c:TRN
			months = 300
		}
	}
	custom_description = {
		text = sb_swzai_frontier_concede_truce
	}
	hidden_effect = {
		sb_create_trn_swz_firearms_treaty = yes
		sb_swazi_apply_guns_for_land_bonus = yes
	}
	custom_description = {
		text = sb_swzai_frontier_concede_guns
	}



}

sb_trn_apply_swazi_reject_posture = {
	# Attitudes are derived, so make the inputs unambiguous: a rejected frontier
	# settlement leaves TRN hostile and, for AI TRN, actively claim-focused.
	if = {
		limit = {
			c:SWZ ?= { is_country_alive = yes }
		}
		hidden_effect = {
			change_relations = {
				country = c:SWZ
				value = -100
			}
		}
	}
	if = {
		limit = { is_ai = yes }
		set_strategy = ai_strategy_sb_boer_frontier_republic
	}
}

sb_swazi_apply_boer_resistance = {
	# Resistance does not immediately force a war; it creates hostility and a
	# grievance window for later pressure instead.
	set_variable = sb_swazi_boer_pressure_resolved_var
	set_variable = sb_swazi_boer_resisted_var
	if = {
		limit = { has_variable = sb_swazi_boer_land_sold_var }
		remove_variable = sb_swazi_boer_land_sold_var
	}

	c:TRN ?= {
		sb_trn_apply_swazi_reject_posture = yes
	}

}
