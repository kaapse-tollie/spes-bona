# One container object is one coercive delivery generation. Saved event scopes
# point at that object, so a held popup cannot authenticate against a replacement.
sb_martinus_begin_coercive_generation = {
	if = {
		limit = {
			country_definition = cd:TRN
			NOT = { container_exists = sb_martinus_coercive_generation_state }
			NOT = { container_exists = sb_martinus_legal_union_generation_state }
			NOT = { has_variable = sb_martinus_legal_union_pending_var }
			NOT = { has_variable = sb_martinus_union_install_pending_var }
		}
		create_container = {
			name = sb_martinus_coercive_generation_state
			tags = { sb_story sb_martinus_coercive_generation }
			parent = root
			on_created = {
				set_variable = sb_martinus_generation_pending_var
				set_variable = sb_martinus_stage_010_pending_var
				set_variable = { name = sb_martinus_stage_010_queued_var days = 45 }
			}
		}
		if = {
			limit = { container_exists = sb_martinus_coercive_generation_state }
			container:sb_martinus_coercive_generation_state = { save_scope_as = sb_martinus_coercive_receipt_scope }
			set_variable = { name = sb_martinus_coercive_generation_scope value = scope:sb_martinus_coercive_receipt_scope }
			set_variable = sb_martinus_coercive_chain_pending_var
			trigger_event = { id = sb_martinus_confederation.010 days = 30 popup = yes }
		}
	}
}

# Reconcile only the current generation. Busy states retain the stage and rearm
# after their delivery lease expires; durable political invalidity retires it.
sb_martinus_reconcile_coercive_generation = {
	if = {
		limit = {
			country_definition = cd:TRN
			has_variable = sb_martinus_coercive_generation_scope
			NOT = { has_variable = sb_martinus_union_install_pending_var }
			var:sb_martinus_coercive_generation_scope ?= {
				OR = { has_variable = sb_martinus_generation_pending_var has_variable = sb_martinus_generation_active_var }
			}
		}
		if = {
			limit = { NOT = { sb_martinus_durable_pretorius_union_authority = yes } }
			sb_martinus_clear_coercive_chain_state = yes
		}
		else = {
			var:sb_martinus_coercive_generation_scope = {
				if = {
					limit = { has_variable = sb_martinus_lyd_response_pending_var NOT = { c:LYD ?= { country_definition = cd:LYD is_country_alive = yes is_subject = no } } }
					remove_variable = sb_martinus_lyd_response_pending_var
					if = { limit = { has_variable = sb_martinus_lyd_response_queued_var } remove_variable = sb_martinus_lyd_response_queued_var }
				}
				if = {
					limit = { has_variable = sb_martinus_zpb_response_pending_var NOT = { c:ZPB ?= { country_definition = cd:ZPB is_country_alive = yes is_subject = no } } }
					remove_variable = sb_martinus_zpb_response_pending_var
					if = { limit = { has_variable = sb_martinus_zpb_response_queued_var } remove_variable = sb_martinus_zpb_response_queued_var }
				}
				if = {
					limit = { has_variable = sb_martinus_nal_response_pending_var NOT = { c:NAL ?= { country_definition = cd:NAL is_country_alive = yes is_subject = no country_has_primary_culture = cu:boer } } }
					remove_variable = sb_martinus_nal_response_pending_var
					if = { limit = { has_variable = sb_martinus_nal_response_queued_var } remove_variable = sb_martinus_nal_response_queued_var }
				}
				save_scope_as = sb_martinus_coercive_receipt_scope
			}
			if = {
				limit = { sb_martinus_current_coercive_generation_authority = yes }
				if = {
					limit = { scope:sb_martinus_coercive_receipt_scope ?= { has_variable = sb_martinus_stage_010_pending_var NOT = { has_variable = sb_martinus_stage_010_queued_var } NOT = { has_variable = sb_martinus_stage_010_popup_receipt_var } } }
					scope:sb_martinus_coercive_receipt_scope = { set_variable = { name = sb_martinus_stage_010_queued_var days = 45 } }
					trigger_event = { id = sb_martinus_confederation.010 days = 1 popup = yes }
				}
				if = {
					limit = { scope:sb_martinus_coercive_receipt_scope ?= { has_variable = sb_martinus_stage_020_pending_var NOT = { has_variable = sb_martinus_stage_020_queued_var } NOT = { has_variable = sb_martinus_stage_020_popup_receipt_var } } }
					scope:sb_martinus_coercive_receipt_scope = { set_variable = { name = sb_martinus_stage_020_queued_var days = 45 } }
					c:ORA ?= { trigger_event = { id = sb_martinus_confederation.020 days = 1 popup = yes } }
				}
				if = {
					limit = { scope:sb_martinus_coercive_receipt_scope ?= { has_variable = sb_martinus_stage_021_pending_var NOT = { has_variable = sb_martinus_stage_021_queued_var } NOT = { has_variable = sb_martinus_stage_021_popup_receipt_var } } }
					scope:sb_martinus_coercive_receipt_scope = { set_variable = { name = sb_martinus_stage_021_queued_var days = 45 } }
					trigger_event = { id = sb_martinus_confederation.021 days = 1 popup = yes }
				}
				if = {
					limit = { scope:sb_martinus_coercive_receipt_scope ?= { has_variable = sb_martinus_lyd_response_pending_var NOT = { has_variable = sb_martinus_lyd_response_queued_var } NOT = { has_variable = sb_martinus_lyd_response_popup_receipt_var } } }
					scope:sb_martinus_coercive_receipt_scope = { set_variable = { name = sb_martinus_lyd_response_queued_var days = 45 } }
					c:LYD ?= { trigger_event = { id = sb_martinus_confederation.030 days = 1 popup = yes } }
				}
				if = {
					limit = { scope:sb_martinus_coercive_receipt_scope ?= { has_variable = sb_martinus_zpb_response_pending_var NOT = { has_variable = sb_martinus_zpb_response_queued_var } NOT = { has_variable = sb_martinus_zpb_response_popup_receipt_var } } }
					scope:sb_martinus_coercive_receipt_scope = { set_variable = { name = sb_martinus_zpb_response_queued_var days = 45 } }
					c:ZPB ?= { trigger_event = { id = sb_martinus_confederation.031 days = 1 popup = yes } }
				}
				if = {
					limit = { scope:sb_martinus_coercive_receipt_scope ?= { has_variable = sb_martinus_nal_response_pending_var NOT = { has_variable = sb_martinus_nal_response_queued_var } NOT = { has_variable = sb_martinus_nal_response_popup_receipt_var } } }
					scope:sb_martinus_coercive_receipt_scope = { set_variable = { name = sb_martinus_nal_response_queued_var days = 45 } }
					c:NAL ?= { trigger_event = { id = sb_martinus_confederation.032 days = 1 popup = yes } }
				}
				sb_martinus_maybe_queue_final_offer = yes
			}
		}
	}
}

# The final coercive choice waits for every requested ally reply. Invalid or lost
# recipients are acknowledged by the monthly watchdog before this is called.
sb_martinus_maybe_queue_final_offer = {
	if = {
		limit = {
			exists = scope:sb_martinus_coercive_receipt_scope
			c:TRN ?= {
				sb_martinus_current_pretorius_union_authority = yes
				NOT = { has_variable = sb_martinus_ambition_resolved_var }
				has_variable = sb_martinus_coercive_generation_scope
				var:sb_martinus_coercive_generation_scope = scope:sb_martinus_coercive_receipt_scope
				has_variable = sb_martinus_coercive_chain_active_var
			}
			scope:sb_martinus_coercive_receipt_scope ?= {
				has_variable = sb_martinus_generation_active_var
				NOT = { has_variable = sb_martinus_stage_010_pending_var }
				NOT = { has_variable = sb_martinus_stage_020_pending_var }
				NOT = { has_variable = sb_martinus_stage_021_pending_var }
				NOT = { has_variable = sb_martinus_lyd_response_pending_var }
				NOT = { has_variable = sb_martinus_zpb_response_pending_var }
				NOT = { has_variable = sb_martinus_nal_response_pending_var }
				NOT = { has_variable = sb_martinus_final_offer_queued_var }
				NOT = { has_variable = sb_martinus_final_offer_delivered_var }
				NOT = { has_variable = sb_martinus_generation_union_install_pending_var }
			}
		}
		scope:sb_martinus_coercive_receipt_scope = {
			if = { limit = { has_variable = sb_martinus_final_lyd_support_var } remove_variable = sb_martinus_final_lyd_support_var }
			if = { limit = { has_variable = sb_martinus_final_zpb_support_var } remove_variable = sb_martinus_final_zpb_support_var }
			if = { limit = { has_variable = sb_martinus_final_nal_support_var } remove_variable = sb_martinus_final_nal_support_var }
			set_variable = { name = sb_martinus_final_offer_queued_var days = 45 }
		}
		c:TRN = { trigger_event = { id = sb_martinus_confederation.040 days = 1 popup = no } }
	}
}

# A Presidential Union lasts only for the incumbent who held office when that
# autonomy level began.
sb_martinus_record_presidential_union_president = {
	if = {
		limit = { exists = ruler }

			name = sb_martinus_presidential_union_president_var
			value = ruler
		}
	}
}

sb_martinus_finalize_verified_presidential_union_with_ora = {
	if = {
		limit = {
			c:ORA ?= {
				is_country_alive = yes
				is_direct_subject_of = root
				is_subject_type = subject_type_sb_boer_presidential_union
			}
		}
		set_variable = sb_martinus_ambition_resolved_var
		set_variable = sb_martinus_legal_union_resolved_var
		set_variable = sb_martinus_presidential_union_active_var
		sb_martinus_record_presidential_union_president = yes
		c:ORA = {
			set_variable = sb_martinus_presidential_union_junior_var
			change_relations = { country = c:TRN value = 30 }
		}
		sb_story_humiliation_clear_martinus_state = yes
		sb_martinus_clear_legal_union_generation_state = yes
		sb_martinus_clear_coercive_chain_state = yes
		sb_martinus_clear_union_install_state = yes
	}
}

# `create_diplomatic_pact` has no return value. Proof is the exact installed pact;
# absence after the attempt leaves the durable, generation-bound receipt intact.
sb_martinus_create_presidential_union_with_ora = {
	if = {
		limit = { sb_martinus_current_union_completion_authority = yes }
		if = {
			limit = { NOT = { c:ORA ?= { is_direct_subject_of = root is_subject_type = subject_type_sb_boer_presidential_union } } }
			create_diplomatic_pact = { country = c:ORA type = sb_boer_presidential_union }
		}
		if = {
			limit = { c:ORA ?= { is_direct_subject_of = root is_subject_type = subject_type_sb_boer_presidential_union } }
			sb_martinus_finalize_verified_presidential_union_with_ora = yes
		}
	}
}

sb_martinus_begin_story_union_install = {

		limit = {

			has_variable = sb_martinus_story_humiliation_route_var
			has_variable = sb_martinus_story_play_scope
			sb_martinus_durable_story_union_install_authority = yes
			has_variable = sb_martinus_story_generation_scope
			var:sb_martinus_story_generation_scope ?= { has_variable = sb_martinus_generation_story_var }
		}

		set_variable = sb_martinus_union_install_pending_var

		sb_martinus_try_complete_pending_union_install = yes
	}
}

sb_martinus_try_complete_pending_union_install = {
	if = {
		limit = {
			country_definition = cd:TRN
			has_variable = sb_martinus_union_install_pending_var
			OR = {
				sb_martinus_current_story_union_install_generation = yes
				sb_martinus_current_coercive_union_install_generation = yes
				sb_martinus_current_legal_union_install_generation = yes
			}
		}
		if = {
			limit = { c:ORA ?= { is_direct_subject_of = root is_subject_type = subject_type_sb_boer_presidential_union } }
			sb_martinus_finalize_verified_presidential_union_with_ora = yes
		}
		else_if = {
			limit = { NOT = { sb_martinus_union_install_structurally_valid = yes } }
			if = {
				limit = { has_variable = sb_martinus_union_install_source_story_var }
				# Lost authority is cancellation, not a negotiated Martinus stand-down.
				sb_story_humiliation_clear_martinus_state = yes
				sb_martinus_clear_coercive_chain_state = yes
			}
			else_if = {
				limit = { has_variable = sb_martinus_union_install_source_legal_var }
				# An invalid generation must not manufacture the separate-republic result.
				sb_martinus_clear_legal_union_generation_state = yes
			}
			else = { sb_martinus_clear_coercive_chain_state = yes }
			sb_martinus_clear_union_install_state = yes
		}
		else_if = {
			limit = { sb_martinus_union_install_ready_now = yes }
			sb_martinus_create_presidential_union_with_ora = yes
		}
	}
}

		set_variable = sb_martinus_story_humiliation_route_var
		set_variable = { name = sb_martinus_story_generation_scope value = scope:sb_martinus_coercive_receipt_scope }
		# The lease is consumed only by the matching on-start callback. Backers and
		# the counter-demand are configured on that actual diplomatic-play root.

		set_variable = sb_martinus_include_natal_backer_var

			limit = { has_variable = sb_martinus_story_launch_lease_var }
			# Same-tick refusal is operational, not a terminal outcome. Roll back only
			# this exact launch lease and rearm the same generation's final offer.
			remove_variable = sb_martinus_story_launch_lease_var
			if = { limit = { has_variable = sb_martinus_include_natal_backer_var } remove_variable = sb_martinus_include_natal_backer_var }

			sb_martinus_maybe_queue_final_offer = yes
		}

			scope:sb_martinus_coercive_receipt_scope = {
				if = { limit = { has_variable = sb_martinus_generation_active_var } remove_variable = sb_martinus_generation_active_var }

# Custom Boer union types are meaningful only under a Boer-led direct overlord.
# Preserve ORA's autonomy level when a foreign overlord inherits the relationship.
sb_martinus_normalize_ora_foreign_union = {
	if = {
		limit = {
			country_definition = cd:ORA
			is_country_alive = yes
			is_subject = yes
			OR = {
				is_subject_type = subject_type_sb_boer_presidential_union
				is_subject_type = subject_type_sb_boer_confederal_partner
			}
			overlord = {
				NOT = { country_has_primary_culture = cu:boer }
			}
		}

		if = {
			limit = {
				is_subject_type = subject_type_sb_boer_presidential_union
			}
			if = {
				limit = {
					overlord = { is_country_type = unrecognized }
				}
				change_subject_type = subject_type_tributary
			}
			else = {
				change_subject_type = subject_type_protectorate
			}
		}
		else_if = {
			limit = {
				is_subject_type = subject_type_sb_boer_confederal_partner
			}
			if = {
				limit = {
					overlord = { is_country_type = unrecognized }
				}
				change_subject_type = subject_type_vassal
			}
			else = {
				change_subject_type = subject_type_puppet
			}
		}

		if = {
			limit = { has_variable = sb_martinus_confederal_relationship_v3_var }
			remove_variable = sb_martinus_confederal_relationship_v3_var
		}
		if = {
			limit = { has_variable = sb_martinus_confederal_partner_var }
			remove_variable = sb_martinus_confederal_partner_var
		}
		if = {
			limit = { has_variable = sb_martinus_presidential_union_junior_var }
			remove_variable = sb_martinus_presidential_union_junior_var
		}

		c:TRN ?= {
			if = {
				limit = { has_variable = sb_martinus_confederation_active_var }
				remove_variable = sb_martinus_confederation_active_var
			}
			if = {
				limit = { has_variable = sb_martinus_presidential_union_active_var }
				remove_variable = sb_martinus_presidential_union_active_var
			}
			if = {
				limit = { has_variable = sb_martinus_presidential_union_president_var }
				remove_variable = sb_martinus_presidential_union_president_var
			}
		}
	}
}

# Shared-ruler subject pacts can be retired automatically by the engine. Durable
# confederal markers remain authoritative, so repair any marked relationship.
sb_martinus_restore_confederal_relationship = {
	if = {
		limit = {
			country_definition = cd:TRN
			has_variable = sb_martinus_confederation_active_var
			has_law_or_variant = law_type:law_presidential_republic
			c:ORA ?= {
				is_country_alive = yes
				has_variable = sb_martinus_confederal_partner_var
				OR = {
					is_subject = no
					AND = {
						is_direct_subject_of = c:TRN
						is_subject_type = subject_type_sb_boer_confederal_partner
					}
				}
			}
		}
		if = {
			limit = { c:ORA ?= { is_subject = no } }
			end_truce = c:ORA
			create_diplomatic_pact = {
				country = c:ORA
				type = sb_boer_confederal_partner
			}
		}
		if = {
			limit = {
				c:ORA ?= {
					is_direct_subject_of = root
					is_subject_type = subject_type_sb_boer_confederal_partner
				}
			}
			c:ORA ?= { set_variable = sb_martinus_confederal_relationship_v3_var }
			sb_martinus_sync_ora_laws_to_trn = yes
		}
	}
}
