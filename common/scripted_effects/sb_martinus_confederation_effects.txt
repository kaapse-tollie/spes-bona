sb_martinus_ensure_trn_pretorius_cast = {

		limit = {
			country_definition = cd:TRN
			NOT = {
				any_scope_character = {
					has_template = TRN_marthinus_wessel_pretorius
				}
			}
		}
		create_character = {
			template = TRN_marthinus_wessel_pretorius
		}
	}
	random_scope_character = {
		limit = {
			has_template = TRN_marthinus_wessel_pretorius
			is_character_alive = yes
		}
		set_interest_group = c:TRN.ig:ig_landowners
	}
}

# A Presidential Union lasts only for the incumbent who held office when that
# autonomy level began.
sb_martinus_record_presidential_union_president = {
	if = {
		limit = { exists = ruler }
		set_variable = {
			name = sb_martinus_presidential_union_president_var
			value = ruler
		}
	}
}

sb_martinus_create_presidential_union_with_ora = {
	set_variable = sb_martinus_ambition_resolved_var
	set_variable = sb_martinus_presidential_union_active_var

		remove_variable = sb_martinus_coercive_chain_active_var

		limit = { has_variable = sb_martinus_coercive_chain_pending_var }
		remove_variable = sb_martinus_coercive_chain_pending_var

}

# Shared-ruler subject pacts are retired by the engine when the senior ruler
# changes. Recreate only the marked confederal relationship after that election.
sb_martinus_restore_confederal_relationship = {
	if = {
		limit = {
			country_definition = cd:TRN
			has_variable = sb_martinus_confederation_active_var
			has_law_or_variant = law_type:law_presidential_republic
			c:ORA ?= {
				is_country_alive = yes
				has_variable = sb_martinus_confederal_partner_var
				OR = {
					is_subject = no
					AND = {
						is_direct_subject_of = c:TRN
						is_subject_type = subject_type_sb_boer_confederal_partner
					}
				}
			}
		}
		if = {
			limit = { c:ORA ?= { is_subject = no } }
			end_truce = c:ORA
			create_diplomatic_pact = {
				country = c:ORA
				type = sb_boer_confederal_partner
			}
		}
		if = {
			limit = {
				c:ORA ?= {
					is_direct_subject_of = root
					is_subject_type = subject_type_sb_boer_confederal_partner
				}
			}
			c:ORA ?= { set_variable = sb_martinus_confederal_relationship_v3_var }
			sb_martinus_sync_ora_laws_to_trn = yes
		}
	}
}

# Generic autonomy actions change subject type without running the Martinus
# effects, so keep the union lifecycle variables aligned with the actual pact.
sb_martinus_reconcile_ora_union_status = {
	# One-time compatibility repair for saves made after the engine retired the
	# pact at a TRN election but before the joint-election handoff was added.
	if = {
		limit = {
			country_definition = cd:TRN
			has_variable = sb_martinus_confederation_active_var
			has_law_or_variant = law_type:law_presidential_republic
			c:ORA ?= {
				is_country_alive = yes
				has_variable = sb_martinus_confederal_partner_var
				NOT = { has_variable = sb_martinus_confederal_relationship_v3_var }
			}
		}
		sb_martinus_restore_confederal_relationship = yes
	}

	if = {
		limit = {
			country_definition = cd:TRN
			c:ORA ?= {
				is_country_alive = yes
				is_direct_subject_of = root
				is_subject_type = subject_type_sb_boer_confederal_partner
			}
		}
		if = {
			limit = { NOT = { has_variable = sb_martinus_confederation_active_var } }
			set_variable = sb_martinus_confederation_active_var
		}
		if = {
			limit = { has_variable = sb_martinus_presidential_union_active_var }
			remove_variable = sb_martinus_presidential_union_active_var
		}
		if = {
			limit = { has_variable = sb_martinus_presidential_union_president_var }
			remove_variable = sb_martinus_presidential_union_president_var
		}
		c:ORA ?= {
			if = {
				limit = { NOT = { has_variable = sb_martinus_confederal_partner_var } }
				set_variable = sb_martinus_confederal_partner_var
			}
			if = {
				limit = { has_variable = sb_martinus_presidential_union_junior_var }
				remove_variable = sb_martinus_presidential_union_junior_var
			}
			if = {
				limit = { NOT = { has_variable = sb_martinus_confederal_relationship_v3_var } }
				set_variable = sb_martinus_confederal_relationship_v3_var
			}
		}
	}
	else_if = {
		limit = {
			country_definition = cd:TRN
			c:ORA ?= {
				is_country_alive = yes
				is_direct_subject_of = root
				is_subject_type = subject_type_sb_boer_presidential_union
			}
		}
		if = {
			limit = { NOT = { has_variable = sb_martinus_presidential_union_active_var } }
			set_variable = sb_martinus_presidential_union_active_var
		}
		if = {
			limit = { has_variable = sb_martinus_confederation_active_var }
			remove_variable = sb_martinus_confederation_active_var
		}
		c:ORA ?= {
			if = {
				limit = { has_variable = sb_martinus_confederal_relationship_v3_var }
				remove_variable = sb_martinus_confederal_relationship_v3_var
			}
			if = {
				limit = { NOT = { has_variable = sb_martinus_presidential_union_junior_var } }
				set_variable = sb_martinus_presidential_union_junior_var
			}
			if = {
				limit = { has_variable = sb_martinus_confederal_partner_var }
				remove_variable = sb_martinus_confederal_partner_var
			}
		}
		if = {
			limit = { NOT = { has_variable = sb_martinus_presidential_union_president_var } }
			sb_martinus_record_presidential_union_president = yes
		}
	}
}

			set_variable = sb_martinus_natalia_presidential_union_active_var
		}
		c:NAL ?= {
			if = {
				limit = { NOT = { has_variable = sb_martinus_natalia_presidential_union_junior_var } }
				set_variable = sb_martinus_natalia_presidential_union_junior_var
			}
		}
		if = {
			limit = { NOT = { has_variable = sb_martinus_natalia_presidential_union_president_var } }
			sb_martinus_record_natalia_presidential_union_president = yes
		}
	}
	else = {
		if = {
			limit = { has_variable = sb_martinus_natalia_presidential_union_active_var }
			remove_variable = sb_martinus_natalia_presidential_union_active_var
		}
		if = {
			limit = { has_variable = sb_martinus_natalia_presidential_union_president_var }
			remove_variable = sb_martinus_natalia_presidential_union_president_var
		}
		c:NAL ?= {
			if = {
				limit = { has_variable = sb_martinus_natalia_presidential_union_junior_var }
				remove_variable = sb_martinus_natalia_presidential_union_junior_var
			}
		}
	}
}
