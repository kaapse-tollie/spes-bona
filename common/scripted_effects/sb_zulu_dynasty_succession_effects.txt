# Spes Bona - Zulu Succession and Collapse Effects
#
# These helpers are split out from `sb_zulu_dynasty_effects.txt` so the main
# file can stay focused on the dynastic-stability bar and monthly upkeep while
# this one owns the actual claimant preparation, ruler swaps, and collapse
# resolution.

sb_zulu_kill_current_heir = {
	if = {
		limit = {
			exists = heir
			heir = {
				is_character_alive = yes
			}
		}
		heir = {
			kill_character = {
				hidden = yes
			}
		}
	}
}

sb_zulu_assign_new_heir_scope = {
	set_heir = scope:new_heir_scope
	if = {
		limit = {
			scope:new_heir_scope ?= {
				is_in_void = yes
			}
		}
		hidden_effect = {
			scope:new_heir_scope ?= {
				free_character_from_void = yes
				transfer_character = prev
			}
		}
	}
	if = {
		limit = {
			scope:new_heir_scope ?= {
				is_in_exile_pool = yes
			}
		}
		scope:new_heir_scope ?= {
			transfer_character = prev
			set_variable = exile_to_ruler_var
			add_career_length = { months = 60 }
			hidden_effect = {
				free_from_exile_effect = yes
			}
		}
	}
	scope:new_heir_scope ?= {
		set_interest_group = c:ZUL.ig:ig_landowners
	}
}

sb_zulu_install_summoned_claimant_as_ruler = {
	scope:summoned_character_scope ?= {
		if = {
			limit = {
				is_in_void = yes
			}
			hidden_effect = {
				free_character_from_void = yes
				transfer_character = prev
			}
		}
		if = {
			limit = {
				is_in_exile_pool = yes
			}
			transfer_character = prev
			set_variable = exile_to_ruler_var
			add_career_length = {
				months = 60
			}
			hidden_effect = {
				free_from_exile_effect = yes
			}
		}
		if = {
			limit = {
				NOT = { has_role_of_type = politician }
			}
			add_character_role = politician
		}
		set_interest_group = c:ZUL.ig:ig_landowners
		hidden_effect = {
			cleanup_ruler_character_roles = yes
		}
		set_character_as_ruler = yes
	}
}

# Resolve the authored Gaza counter-war only from a frozen goal target that
# remains under Gaza ownership at the exact saved diplomatic-play close.
sb_gaza_resolve_exact_terminal = {
	if = {
		limit = {
			initiator ?= { has_variable = sb_gaza_zulu_play_scope var:sb_gaza_zulu_play_scope = root }
			c:GZA ?= { is_country_alive = yes }
			OR = {
				initiator ?= { has_variable = sb_gaza_east_transvaal_goal_added_var var:sb_gaza_east_transvaal_target_state_scope ?= { owner = c:GZA } }
				initiator ?= { has_variable = sb_gaza_zululand_goal_added_var var:sb_gaza_zululand_target_state_scope ?= { owner = c:GZA } }
			}
		}
		c:GZA = {
			if = {
				limit = { has_variable = sb_gaza_eastern_transvaal_claim_if_zulu_defeated_var NOT = { has_claim = s:STATE_EAST_TRANSVAAL } }
				s:STATE_EAST_TRANSVAAL = { add_claim = c:GZA }
			}
		}
	}
	c:GZA ?= {
		if = { limit = { has_variable = sb_gaza_claim_goal_accepted_pending_var } remove_variable = sb_gaza_claim_goal_accepted_pending_var }
		if = { limit = { has_variable = sb_gaza_eastern_transvaal_claim_if_zulu_defeated_var } remove_variable = sb_gaza_eastern_transvaal_claim_if_zulu_defeated_var }
	}
	initiator ?= {
		if = { limit = { has_variable = sb_gaza_east_transvaal_goal_added_var } remove_variable = sb_gaza_east_transvaal_goal_added_var }
		if = { limit = { has_variable = sb_gaza_east_transvaal_target_state_scope } remove_variable = sb_gaza_east_transvaal_target_state_scope }
		if = { limit = { has_variable = sb_gaza_zululand_goal_added_var } remove_variable = sb_gaza_zululand_goal_added_var }
		if = { limit = { has_variable = sb_gaza_zululand_target_state_scope } remove_variable = sb_gaza_zululand_target_state_scope }

# The dynasty JE chooses Mpande's immediate heir, but Vanilla resumes generic
# heir generation after that character dies. Preserve the selected line once:
# Mbuyazi's counterfactual line passes to its defined fictional successor,
# while Cetshwayo's line passes to Dinuzulu. Template-use checks distinguish a
# dead claimant from an initial assignment failure and prevent resurrection.
sb_zulu_reconcile_secured_mpande_line = {
	if = {
		limit = {
			OR = {
				NOT = { exists = heir }
				heir ?= {
					NOR = {
						has_template = $HEIR$
						has_template = $SUCCESSOR$
					}
				}
			}
		}
		if = {
			limit = {
				OR = {
					any_scope_character = {
						has_template = $HEIR$
						is_character_alive = yes
					}
					NOT = { is_template_used = $HEIR$ }
				}
			}
			sb_zulu_kill_current_heir = yes
			country_summon_heir_effect = { TEMPLATE = $HEIR$ }
		}
		else_if = {
			limit = {
				any_scope_character = {
					has_template = $SUCCESSOR$
					is_character_alive = yes
				}
			}
			sb_zulu_kill_current_heir = yes
			country_summon_heir_effect = { TEMPLATE = $SUCCESSOR$ }
			set_variable = { name = $SUCCESSOR_MARKER$ }
		}
		else_if = {
			limit = {
				NOT = { has_variable = $SUCCESSOR_MARKER$ }
				NOT = { is_template_used = $SUCCESSOR$ }
				game_date >= $SUCCESSOR_BIRTH_DATE$
			}
			sb_zulu_kill_current_heir = yes
			country_summon_heir_effect = { TEMPLATE = $SUCCESSOR$ }
			if = {
				limit = {
					exists = heir
					heir = { has_template = $SUCCESSOR$ }
				}
				set_variable = { name = $SUCCESSOR_MARKER$ }
			}
		}
		else_if = {
			limit = { is_template_used = $SUCCESSOR$ }
			set_variable = { name = $SUCCESSOR_MARKER$ }
		}
	}
	if = {
		limit = {
			NOT = { has_variable = $SUCCESSOR_MARKER$ }
			exists = heir
			heir = { has_template = $SUCCESSOR$ }
		}
		set_variable = { name = $SUCCESSOR_MARKER$ }
	}
}

		limit = { has_variable = sb_zulu_mpande_mbuyazi_heir_var }
		set_global_variable = sb_zulu_mbuyazi_succession_global_var
	}
	else_if = {
		limit = {
			has_variable = sb_zulu_mpande_cetshwayo_heir_var
			has_global_variable = sb_zulu_mbuyazi_succession_global_var
		}
		remove_global_variable = sb_zulu_mbuyazi_succession_global_var
	}
	sb_zulu_update_dynastic_stability_modifiers = yes
	sb_zulu_maybe_open_swazi_question = yes
	if = {
		limit = {
			has_variable = sb_zulu_mpande_succession_secured_var
			has_variable = sb_zulu_mpande_cetshwayo_heir_var
			NOT = { has_variable = sb_zulu_mpande_mbuyazi_heir_var }

			NOT = { has_variable = sb_zulu_mpande_withdrawal_fired_var }

				has_template = ZUL_mpande_zulu
			}
		}

	}

		limit = {
			NOT = { has_variable = sb_zulu_dynasty_secured_var }
		}
		if = {
			limit = {
				average_sol < average_expected_sol
			}
			sb_zulu_remove_dynastic_stability_1 = yes
		}

		# averages about 0.77 court events per year across the four-event pool.

			limit = {
				NOT = { has_variable = sb_zulu_court_200_cooldown_var }
			}
			random_list = {
				2 = {
					set_variable = {
						name = sb_zulu_court_200_cooldown_var
						months = 12
					}
					trigger_event = { id = sb_zulu_court.200 days = 7 popup = yes }
				}

			}
		}
		if = {
			limit = {
				NOT = { has_variable = sb_zulu_court_201_cooldown_var }
			}
			random_list = {
				2 = {
					set_variable = {
						name = sb_zulu_court_201_cooldown_var
						months = 12
					}
					trigger_event = { id = sb_zulu_court.201 days = 7 popup = yes }
				}
				98 = { }
			}
		}
		if = {
			limit = {
				NOT = { has_variable = sb_zulu_court_202_cooldown_var }
			}
			random_list = {
				2 = {
					set_variable = {
						name = sb_zulu_court_202_cooldown_var
						months = 12
					}
					trigger_event = { id = sb_zulu_court.202 days = 7 popup = yes }

				2 = {

						name = sb_zulu_court_203_cooldown_var

					}
					trigger_event = { id = sb_zulu_court.203 days = 7 popup = yes }

		if = {
			limit = {
				NOT = { has_modifier = sb_zulu_secured_the_throne }
			}
			add_modifier = { name = sb_zulu_secured_the_throne }
		}
	}
	if = {
		limit = {
			ruler = {
				has_template = ZUL_dingane
			}
		}
		if = {
			limit = {
				exists = heir
				heir = {
					NOT = { has_template = ZUL_uthumbu_claimed_son }
				}
			}
			heir = {
				kill_character = { hidden = yes }
			}
		}
		sb_zulu_prepare_uthumbu_heir = yes
	}
	else_if = {
		limit = {
			ruler = {
				has_template = ZUL_mpande_zulu
			}
		}
		if = {
			limit = {
				exists = heir
				heir = {
					NOT = { has_template = ZUL_cetshwayo_heir }
				}
			}
			heir = {
				kill_character = { hidden = yes }
			}
		}
		sb_zulu_prepare_cetshwayo_heir = yes
	}
	else = {
		sb_zulu_kill_current_heir = yes
		sb_zulu_prepare_random_good_heir = yes
	}
}
