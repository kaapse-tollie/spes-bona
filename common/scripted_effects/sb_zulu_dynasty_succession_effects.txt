sb_zulu_assign_new_heir_scope = {
	save_scope_as = sb_zulu_install_country
	# Temporarily release this court's CMF block for the authored appointment.
	sb_court_release_heir_blocker = yes
	set_heir = scope:new_heir_scope
	if = {
		limit = {
			scope:new_heir_scope ?= {
				is_in_void = yes
			}
		}
		hidden_effect = {
			scope:new_heir_scope ?= {
				free_character_from_void = yes
				transfer_character = prev
			}
		}
	}
	if = {
		limit = {
			scope:new_heir_scope ?= {
				is_in_exile_pool = yes
			}
		}
		scope:new_heir_scope ?= {
			transfer_character = prev
			set_variable = exile_to_ruler_var
			add_career_length = { months = 60 }
			hidden_effect = {
				free_from_exile_effect = yes
			}
		}
	}
	scope:new_heir_scope ?= {
		set_interest_group = scope:sb_zulu_install_country.ig:ig_landowners
	}
	sb_court_update_heir_blocker = yes
}

sb_zulu_install_summoned_claimant_as_ruler = {
	save_scope_as = sb_zulu_install_country
	scope:summoned_character_scope ?= {
		if = {
			limit = {
				is_in_void = yes
			}
			hidden_effect = {
				free_character_from_void = yes
				transfer_character = prev
			}
		}
		if = {
			limit = {
				is_in_exile_pool = yes
			}
			transfer_character = prev
			set_variable = exile_to_ruler_var
			add_career_length = {
				months = 60
			}
			hidden_effect = {
				free_from_exile_effect = yes
			}
		}
		if = {
			limit = {
				NOT = { has_role_of_type = politician }
			}
			add_character_role = politician
		}
		set_interest_group = scope:sb_zulu_install_country.ig:ig_landowners
		hidden_effect = {
			cleanup_ruler_character_roles = yes
		}
		set_character_as_ruler = yes
	}
}

# Called only by the bound play's terminal resolver. Reserve before scheduling;
# the Zulu event owns the political consequences, including for backdown.
sb_zulu_queue_swazi_campaign_outcome = {
 if = {
  limit = { NOR = { has_variable = sb_court_swazi_result_var has_variable = sb_court_swazi_outcome_pending_var } }
  save_scope_as = sb_court_country
  set_variable = { name = sb_court_swazi_outcome_pending_var value = $RESULT$ }
  if = {
   limit = { NOT = { has_variable = sb_court_swazi_ruler_scope } }
   ruler ?= { scope:sb_court_country = { set_variable = { name = sb_court_swazi_ruler_scope value = prev } } }
  }
  if = {
   limit = { var:sb_court_swazi_outcome_pending_var = 1 }
   trigger_event = { id = sb_swazi_frontier.097 days = 1 popup = yes }
  }
  else = { trigger_event = { id = sb_swazi_frontier.098 days = 1 popup = yes } }
 }
}

sb_zulu_apply_swazi_campaign_victory = {
 if = {
  limit = { NOT = { has_variable = sb_court_swazi_result_var } }
  sb_court_campaign_result = { CAMPAIGN = swazi OUTCOME = victory WON = yes DEFEAT_PREROGATIVE = -30 }

	s:STATE_EAST_TRANSVAAL = {
		add_claim = c:ZUL
	}
	# Preserve the pre-rebase post-Swazi policy: one AI-only dynamic-history
	# 50/50 Gaza follow-up decision after an actual Swazi campaign victory.
	if = {
		limit = {
			country_definition = cd:ZUL
			is_ai = yes
			sb_frontier_ai_scripting_enabled = yes
			sb_frontier_ai_behavior_strict_historical = no
			has_variable = sb_zulu_blood_river_victory_var
			ruler = { has_template = ZUL_dingane }
			c:GZA ?= { is_country_alive = yes is_ai = yes is_subject = no }
			NOT = { has_variable = sb_zulu_gaza_followup_resolved_var }
		}
		set_variable = sb_zulu_gaza_followup_resolved_var
		random_list = {
			50 = { set_variable = sb_zulu_gaza_followup_pending_var }
			50 = { }
		}
	}
 }
}

# Resolve the authored Gaza counter-war only from a frozen goal target that
# remains under Gaza ownership at the exact saved diplomatic-play close.
sb_gaza_resolve_exact_terminal = {
	if = {
		limit = {
			initiator ?= { has_variable = sb_gaza_zulu_play_scope var:sb_gaza_zulu_play_scope = root }
			c:GZA ?= { is_country_alive = yes }
			OR = {
				initiator ?= { has_variable = sb_gaza_east_transvaal_goal_added_var var:sb_gaza_east_transvaal_target_state_scope ?= { owner = c:GZA } }
				initiator ?= { has_variable = sb_gaza_zululand_goal_added_var var:sb_gaza_zululand_target_state_scope ?= { owner = c:GZA } }
			}
		}
		c:GZA = {
			if = {
				limit = { has_variable = sb_gaza_eastern_transvaal_claim_if_zulu_defeated_var NOT = { has_claim = s:STATE_EAST_TRANSVAAL } }
				s:STATE_EAST_TRANSVAAL = { add_claim = c:GZA }
			}
		}
	}
	c:GZA ?= {
		if = { limit = { has_variable = sb_gaza_claim_goal_accepted_pending_var } remove_variable = sb_gaza_claim_goal_accepted_pending_var }
		if = { limit = { has_variable = sb_gaza_eastern_transvaal_claim_if_zulu_defeated_var } remove_variable = sb_gaza_eastern_transvaal_claim_if_zulu_defeated_var }
	}
	initiator ?= {
		if = { limit = { has_variable = sb_gaza_east_transvaal_goal_added_var } remove_variable = sb_gaza_east_transvaal_goal_added_var }
		if = { limit = { has_variable = sb_gaza_east_transvaal_target_state_scope } remove_variable = sb_gaza_east_transvaal_target_state_scope }
		if = { limit = { has_variable = sb_gaza_zululand_goal_added_var } remove_variable = sb_gaza_zululand_goal_added_var }
		if = { limit = { has_variable = sb_gaza_zululand_target_state_scope } remove_variable = sb_gaza_zululand_target_state_scope }
		if = { limit = { has_variable = sb_gaza_zulu_play_scope } remove_variable = sb_gaza_zulu_play_scope }
		if = { limit = { has_variable = sb_gaza_zulu_launch_lease_var } remove_variable = sb_gaza_zulu_launch_lease_var }
	}
}
