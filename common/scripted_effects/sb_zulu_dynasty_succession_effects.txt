# Spes Bona - Zulu Succession and Collapse Effects
#
# These helpers are split out from `sb_zulu_dynasty_effects.txt` so the main
# file can stay focused on the dynastic-stability bar and monthly upkeep while
# this one owns the actual claimant preparation, ruler swaps, and collapse
# resolution.

sb_zulu_kill_current_heir = {
	if = {
		limit = {
			exists = heir
			heir = {
				is_character_alive = yes
			}
		}
		heir = {
			kill_character = {
				hidden = yes
			}
		}
	}
}

sb_zulu_assign_new_heir_scope = {
	scope:new_heir_scope ?= {
		if = {
			limit = {
				is_in_void = yes
			}
			hidden_effect = {
				free_character_from_void = yes
				transfer_character = prev
			}
		}
		if = {
			limit = {
				is_in_exile_pool = yes
			}
			transfer_character = prev
			set_variable = exile_to_ruler_var
			add_career_length = {
				months = 60
			}
			hidden_effect = {
				free_from_exile_effect = yes
			}
		}
	}
	set_heir = scope:new_heir_scope
	scope:new_heir_scope ?= {
		replace_character_roles = { add = { heir } remove_all = yes }
		set_interest_group = c:ZUL.ig:ig_armed_forces
	}
}

sb_zulu_prepare_uthumbu_heir = {
	if = {
		limit = {
			NOT = {
				any_scope_character = {
					has_template = ZUL_uthumbu_claimed_son
					is_character_alive = yes
				}
			}
		}
		create_character = {
			template = ZUL_uthumbu_claimed_son
			on_created = {
				save_scope_as = new_heir_scope
			}
		}
	}
	else = {
		random_scope_character = {
			limit = {
				has_template = ZUL_uthumbu_claimed_son
				is_character_alive = yes
			}
			save_scope_as = new_heir_scope
		}
	}
	sb_zulu_assign_new_heir_scope = yes
}

sb_zulu_install_summoned_claimant_as_ruler = {
	scope:summoned_character_scope ?= {
		if = {
			limit = {
				is_in_void = yes
			}
			hidden_effect = {
				free_character_from_void = yes
				transfer_character = prev
			}
		}
		if = {
			limit = {
				is_in_exile_pool = yes
			}
			transfer_character = prev
			set_variable = exile_to_ruler_var
			add_career_length = {
				months = 60
			}
			hidden_effect = {
				free_from_exile_effect = yes
			}
		}
		if = {
			limit = {
				NOT = { has_role_of_type = politician }
			}
			add_character_role = politician
		}
		set_interest_group = c:ZUL.ig:ig_armed_forces
		hidden_effect = {
			cleanup_ruler_character_roles = yes
		}
		set_character_as_ruler = yes
	}
}

sb_zulu_apply_swazi_campaign_victory = {
	if = { limit = { has_variable = sb_zulu_dynastic_collapse_pending_var } remove_variable = sb_zulu_dynastic_collapse_pending_var }
	sb_zulu_add_dynastic_stability_20 = yes
	sb_zulu_add_dynastic_stability_10 = yes
	random_scope_character = {
		limit = {
			has_template = ZUL_dingane
			is_character_alive = yes
		}
		if = { limit = { has_modifier = sb_dingane_humbled } remove_modifier = sb_dingane_humbled }
		if = { limit = { has_modifier = sb_dingane_checked_by_swazi } remove_modifier = sb_dingane_checked_by_swazi }
	}
	s:STATE_EAST_TRANSVAAL = {
		add_claim = root
	}
	trigger_event = { id = sb_swazi_frontier.097 days = 5 popup = yes }
}

sb_zulu_monthly_dynasty_housekeeping = {
	sb_zulu_update_dynastic_stability_modifiers = yes
	sb_zulu_maybe_open_swazi_question = yes
	if = {
		limit = {
			has_variable = sb_zulu_mpande_succession_secured_var
			has_variable = sb_zulu_mpande_cetshwayo_heir_var
			NOT = { has_variable = sb_zulu_mpande_mbuyazi_heir_var }
			NOT = { has_variable = sb_zulu_mpande_withdrawal_delay_var }
			NOT = { has_variable = sb_zulu_mpande_withdrawal_fired_var }
			ruler = {
				has_template = ZUL_mpande_zulu
			}
		}
		trigger_event = { id = sb_zulu_dynasty.094 days = 1 popup = yes }
	}
	if = {
		limit = {
			NOT = { has_variable = sb_zulu_dynasty_secured_var }
		}
		if = {
			limit = {
				average_sol < average_expected_sol
			}
			sb_zulu_remove_dynastic_stability_1 = yes
		}
		# 2% monthly per eligible court event with a 12-month per-event cooldown
		# averages about 0.77 court events per year across the four-event pool.
		if = {
			limit = {
				NOT = { has_variable = sb_zulu_court_200_cooldown_var }
			}
			random_list = {
				2 = {
					set_variable = {
						name = sb_zulu_court_200_cooldown_var
						months = 12
					}
					trigger_event = { id = sb_zulu_court.200 days = 7 popup = yes }
				}
				98 = { }
			}
		}
		if = {
			limit = {
				NOT = { has_variable = sb_zulu_court_201_cooldown_var }
			}
			random_list = {
				2 = {
					set_variable = {
						name = sb_zulu_court_201_cooldown_var
						months = 12
					}
					trigger_event = { id = sb_zulu_court.201 days = 7 popup = yes }
				}
				98 = { }
			}
		}
		if = {
			limit = {
				NOT = { has_variable = sb_zulu_court_202_cooldown_var }
			}
			random_list = {
				2 = {
					set_variable = {
						name = sb_zulu_court_202_cooldown_var
						months = 12
					}
					trigger_event = { id = sb_zulu_court.202 days = 7 popup = yes }
				}
				98 = { }
			}
		}
		if = {
			limit = {
				NOT = { has_variable = sb_zulu_court_203_cooldown_var }
			}
			random_list = {
				2 = {
					set_variable = {
						name = sb_zulu_court_203_cooldown_var
						months = 12
					}
					trigger_event = { id = sb_zulu_court.203 days = 7 popup = yes }
				}
				98 = { }
			}
		}
	}
}

sb_resolve_zulu_dynastic_collapse = {
	if = {
		limit = {
			has_variable = sb_zulu_dynastic_collapse_pending_var
		}
		remove_variable = sb_zulu_dynastic_collapse_pending_var
	}
	sb_zulu_disable_dynastic_claims = yes
	sb_zulu_clear_dynastic_stability_modifiers = yes
	sb_zulu_set_dynastic_stability_33 = yes
	capital = {
		add_radicals_in_state = {
			value = large_radicals
		}
	}
	if = {
		limit = {
			NOT = { has_modifier = sb_zulu_dynastic_crisis }
		}
		add_modifier = {
			name = sb_zulu_dynastic_crisis
			months = 120
		}
	}
	if = {
		limit = {
			ruler = {
				has_template = ZUL_dingane
			}
		}
		set_variable = sb_zulu_mpande_pretender_heir_pending_var
		trigger_event = { id = sb_zulu_dynasty.095 days = 7 popup = yes }
	}
	ruler = {
		kill_character = { hidden = yes }
	}
}

sb_resolve_zulu_dynasty_secured = {
	if = {
		limit = {
			has_modifier = sb_zulu_dynastic_crisis
		}
		remove_modifier = sb_zulu_dynastic_crisis
	}
	if = {
		limit = {
			has_variable = sb_zulu_dynastic_collapse_pending_var
		}
		remove_variable = sb_zulu_dynastic_collapse_pending_var
	}
	set_variable = sb_zulu_dynasty_secured_var
	sb_zulu_disable_dynastic_claims = yes
	sb_zulu_clear_dynastic_stability_modifiers = yes
	ruler = {
		if = {
			limit = {
				NOT = { has_modifier = sb_zulu_secured_the_throne }
			}
			add_modifier = { name = sb_zulu_secured_the_throne }
		}
	}
	if = {
		limit = {
			ruler = {
				has_template = ZUL_dingane
			}
		}
		if = {
			limit = {
				exists = heir
				heir = {
					NOT = { has_template = ZUL_uthumbu_claimed_son }
				}
			}
			heir = {
				kill_character = { hidden = yes }
			}
		}
		sb_zulu_prepare_uthumbu_heir = yes
	}
	else_if = {
		limit = {
			ruler = {
				has_template = ZUL_mpande_zulu
			}
		}
		if = {
			limit = {
				exists = heir
				heir = {
					NOT = { has_template = ZUL_cetshwayo_heir }
				}
			}
			heir = {
				kill_character = { hidden = yes }
			}
		}
		sb_zulu_prepare_cetshwayo_heir = yes
	}
	else = {
		sb_zulu_kill_current_heir = yes
		sb_zulu_prepare_random_good_heir = yes
	}
}
