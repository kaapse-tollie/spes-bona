# Country scope. Automatic challenges and contested accessions share one reserved crisis.
sb_court_check_claimant_challenge = {
 if = {
  limit = { sb_court_dynastic_can_challenge = yes }
  save_scope_as = sb_court_country
  ordered_in_list = {
   variable = sb_court_members
   limit = { sb_court_member_of_country = yes sb_court_organised_rival = yes NOT = { is_ruler_of_own_country = yes } }
   order_by = var:sb_court_current_weight_var position = 0 check_range_bounds = no
   scope:sb_court_country = {
    set_variable = { name = sb_court_accession_nominee_scope value = ruler }
    set_variable = { name = sb_court_accession_rival_scope value = prev }
    sb_court_reserve_dynastic_crisis = { REASON = 1 }
   }
  }
 }
}

# REASON: 1 living-ruler challenge, 2 succession, 3 exposed purge.
sb_court_reserve_dynastic_crisis = {
 save_scope_as = sb_court_country
 if = {
  limit = {
   sb_court_dynastic_enabled = yes sb_court_dynastic_country = yes
   sb_court_dynastic_conflict_free = yes
   NOT = { has_variable = sb_court_dynastic_crisis_var }
   NOT = { has_variable = sb_court_dynastic_selection_open_var }
   # Native disbanding is deferred. Do not revive a cancelled episode's movement.
   NOT = { any_political_movement = { is_political_movement_type = movement_sb_dynastic_test } }
   var:sb_court_accession_nominee_scope ?= { sb_court_dynastic_candidate = yes }
   var:sb_court_accession_rival_scope ?= { sb_court_dynastic_candidate = yes }
  }
  set_variable = sb_court_dynastic_crisis_var
  # Reserve the native concurrency flag; this does not start a coup or use its mechanics.
  set_variable = coup_event_active
  set_variable = sb_court_dynastic_owns_coup_lock_var
  set_variable = sb_court_accession_open_var
  remove_variable = sb_court_dynastic_resolved_var
  set_variable = { name = sb_court_dynastic_reason_var value = $REASON$ }
  set_variable = { name = sb_court_dynastic_nominee_scope value = var:sb_court_accession_nominee_scope }
  set_variable = { name = sb_court_dynastic_rival_scope value = var:sb_court_accession_rival_scope }
  set_variable = { name = sb_court_dynastic_original_law_var value = 1 }
  if = { limit = { sb_court_has_social_monarchy = yes } set_variable = { name = sb_court_dynastic_original_law_var value = 2 } }
  set_variable = { name = sb_court_dynastic_original_law_scope value = active_law:lawgroup_governance_principles.type }
  sb_court_snapshot_dynastic_coalitions = yes
  sb_court_snapshot_dynastic_roster = yes
  if = {
   limit = { sb_court_dynastic_pair_valid = yes sb_court_dynastic_pair_weight > 0 var:sb_court_dynastic_rival_weight_var > 0 }
   set_variable = sb_court_accession_popup_var
   set_variable = sb_court_dynastic_selection_open_var
   trigger_event = { id = sb_court_succession.004 popup = yes }
  }
  else = { sb_court_cancel_dynastic_crisis = yes }
 }
}

# A representative carries its aggregated house exactly once. Members follow that house;
# royal ancestry itself supplies no second constituency.
sb_court_snapshot_dynastic_coalitions = {
 save_scope_as = sb_court_country
 sb_court_calculate_backing = yes
 clear_variable_list = sb_court_dynastic_incumbent_members
 clear_variable_list = sb_court_dynastic_rival_members
 set_variable = { name = sb_court_dynastic_incumbent_weight_var value = 0 }
 set_variable = { name = sb_court_dynastic_rival_weight_var value = 0 }
 every_in_list = {
  variable = sb_court_members
  limit = { sb_court_member_of_country = yes has_variable = sb_court_representative_var has_variable = sb_court_represented_weight_var }
  if = {
   limit = { scope:sb_court_country = { var:sb_court_dynastic_reason_var = 2 } }
   if = {
    limit = {
     OR = {
      this = scope:sb_court_country.var:sb_court_dynastic_rival_scope
      AND = { NOT = { sb_court_royal_candidate = yes } var:sb_court_successor_pledge_scope ?= scope:sb_court_country.var:sb_court_dynastic_rival_scope }
     }
    }
    sb_court_snapshot_dynastic_house = { SIDE = rival }
   }
   else_if = {
    limit = {
     OR = {
      this = scope:sb_court_country.var:sb_court_dynastic_nominee_scope
      AND = { NOT = { sb_court_royal_candidate = yes } var:sb_court_successor_pledge_scope ?= scope:sb_court_country.var:sb_court_dynastic_nominee_scope }
     }
    }
    sb_court_snapshot_dynastic_house = { SIDE = incumbent }
   }
  }
  else = {
   if = { limit = { sb_court_member_endorsement > 0 } sb_court_snapshot_dynastic_house = { SIDE = incumbent } }
   else_if = {
    limit = { sb_court_member_endorsement < 0 var:sb_court_claimant_scope ?= scope:sb_court_country.var:sb_court_dynastic_rival_scope }
    sb_court_snapshot_dynastic_house = { SIDE = rival }
   }
  }
 }
 # A recognised minor has no adult following, but must still be placed on their own side.
 var:sb_court_dynastic_nominee_scope = { sb_court_snapshot_dynastic_member = { SIDE = incumbent } }
 var:sb_court_dynastic_rival_scope = { sb_court_snapshot_dynastic_member = { SIDE = rival } }
 set_variable = { name = sb_court_dynastic_share_var value = sb_court_dynastic_claimant_share }
 set_variable = { name = sb_court_dynastic_target_var value = sb_court_dynastic_target_fraction }
}
sb_court_snapshot_dynastic_house = {
 scope:sb_court_country = { change_variable = { name = sb_court_dynastic_$SIDE$_weight_var add = prev.var:sb_court_represented_weight_var } }
 if = { limit = { NOT = { this = scope:sb_court_country.var:sb_court_dynastic_nominee_scope } NOT = { this = scope:sb_court_country.var:sb_court_dynastic_rival_scope } } sb_court_snapshot_dynastic_member = { SIDE = $SIDE$ } }
 every_in_list = {
  variable = sb_court_house_members
  limit = { is_character_alive = yes NOT = { this = scope:sb_court_country.var:sb_court_dynastic_nominee_scope } NOT = { this = scope:sb_court_country.var:sb_court_dynastic_rival_scope } }
  sb_court_snapshot_dynastic_member = { SIDE = $SIDE$ }
 }
}
sb_court_snapshot_dynastic_member = {
 scope:sb_court_country = {
  if = {
   limit = { NOT = { is_target_in_variable_list = { name = sb_court_dynastic_$SIDE$_members target = prev } } }
   add_to_variable_list = { name = sb_court_dynastic_$SIDE$_members target = prev }
  }
 }
}

# Preserve the cast independently of both countries' live court lists. Children,
# advisers and third-party princes are not discarded with the defeated country.
sb_court_snapshot_dynastic_roster = {
 save_scope_as = sb_court_country
 clear_variable_list = sb_court_dynastic_roster
 every_in_list = {
  variable = sb_court_members
  limit = { is_character_alive = yes var:sb_court_home_scope ?= scope:sb_court_country }
  sb_court_snapshot_dynastic_character = yes
 }
 every_scope_character = {
  limit = { is_character_alive = yes }
  sb_court_snapshot_dynastic_character = yes
 }
}
sb_court_snapshot_dynastic_character = {
 if = {
  limit = { NOT = { scope:sb_court_country = { is_target_in_variable_list = { name = sb_court_dynastic_roster target = prev } } } }
  scope:sb_court_country = { add_to_variable_list = { name = sb_court_dynastic_roster target = prev } }
  if = { limit = { has_variable = sb_court_registered_var } set_variable = sb_court_dynastic_was_registered_var }
  sb_court_snapshot_dynastic_field = { FIELD = following_var }
  sb_court_snapshot_dynastic_field = { FIELD = alignment_var }
  sb_court_snapshot_dynastic_field = { FIELD = endorsement_var }
  sb_court_snapshot_dynastic_field = { FIELD = claimant_scope }
  sb_court_snapshot_dynastic_field = { FIELD = successor_pledge_scope }
  sb_court_snapshot_dynastic_field = { FIELD = council_promise_scope }
  sb_court_snapshot_dynastic_field = { FIELD = personal_council_scope }
  # Relative incumbent endorsements need the original sovereign's identity.
  if = { limit = { var:sb_court_endorsement_var ?= 1 } set_variable = { name = sb_court_dynastic_patron_scope value = scope:sb_court_country.ruler } }
  if = { limit = { owner ?= scope:sb_court_country } set_variable = sb_court_dynastic_local_var }
  if = { limit = { is_in_void = yes } set_variable = sb_court_dynastic_was_void_var }
  if = { limit = { is_in_exile_pool = yes } set_variable = sb_court_dynastic_was_exiled_var }
  # A pledged adviser with no numerical constituency is still part of the faction.
  # House members already follow their representative's frozen assignment.
  if = {
   limit = {
    NOT = { scope:sb_court_country = { is_target_in_variable_list = { name = sb_court_dynastic_incumbent_members target = prev } } }
    NOT = { scope:sb_court_country = { is_target_in_variable_list = { name = sb_court_dynastic_rival_members target = prev } } }
   }
   if = {
    limit = { scope:sb_court_country = { var:sb_court_dynastic_reason_var = 2 } }
    if = { limit = { var:sb_court_successor_pledge_scope ?= scope:sb_court_country.var:sb_court_dynastic_nominee_scope } sb_court_snapshot_dynastic_member = { SIDE = incumbent } }
    else_if = { limit = { var:sb_court_successor_pledge_scope ?= scope:sb_court_country.var:sb_court_dynastic_rival_scope } sb_court_snapshot_dynastic_member = { SIDE = rival } }
   }
   else = {
    if = { limit = { var:sb_court_endorsement_var ?= 1 } sb_court_snapshot_dynastic_member = { SIDE = incumbent } }
    else_if = { limit = { var:sb_court_endorsement_var ?= -1 var:sb_court_claimant_scope ?= scope:sb_court_country.var:sb_court_dynastic_rival_scope } sb_court_snapshot_dynastic_member = { SIDE = rival } }
   }
  }
 }
}
sb_court_snapshot_dynastic_field = {
 if = { limit = { has_variable = sb_court_$FIELD$ } set_variable = { name = sb_court_dynastic_saved_$FIELD$ value = var:sb_court_$FIELD$ } }
}
sb_court_restore_dynastic_field = {
 remove_variable = sb_court_$FIELD$
 if = { limit = { has_variable = sb_court_dynastic_saved_$FIELD$ } set_variable = { name = sb_court_$FIELD$ value = var:sb_court_dynastic_saved_$FIELD$ } }
}

# The human's chosen contender keeps the original country; the opposition becomes rebels.
sb_court_answer_dynastic_crisis = {
 save_scope_as = sb_court_country
 if = {
  limit = {
   sb_court_dynastic_enabled = yes sb_court_dynastic_pair_valid = yes
   NOT = { has_variable = sb_court_dynastic_test_pending_var }
   NOT = { has_variable = sb_court_dynastic_war_scope }
   scope:sb_court_nominee ?= var:sb_court_dynastic_nominee_scope
   scope:sb_court_rival ?= var:sb_court_dynastic_rival_scope
   sb_court_dynastic_conflict_free = yes
  }
  if = { limit = { always = $CLAIMANT$ } set_variable = sb_court_dynastic_back_claimant_var }
  else = { remove_variable = sb_court_dynastic_back_claimant_var }
  set_variable = { name = sb_court_dynastic_target_var value = sb_court_dynastic_target_fraction }
  sb_court_enable_recruitment_trial = yes
  sb_court_initialise_recruitment_trial = yes
  remove_variable = sb_court_dynastic_selection_open_var
  set_variable = sb_court_dynastic_test_pending_var
  # Preserve penalties from earlier revolutions; this personal contest must not
  # leave the native twenty-year defeated-IG penalty on either faction's groups.
  every_interest_group = {
   remove_variable = sb_court_dynastic_prior_defeat_var
   if = { limit = { has_modifier = revolution_lost_ig } set_variable = sb_court_dynastic_prior_defeat_var }
  }
  set_variable = sb_court_dynastic_igs_snapshotted_var
  sb_court_prepare_dynastic_forces = yes
  # No initial-enthusiasm refresh. This movement exists solely for its reserved conflict.
  create_political_movement = { type = movement_sb_dynastic_test }
  random_political_movement = {
   limit = { is_political_movement_type = movement_sb_dynastic_test }
   scope:sb_court_country = { set_variable = { name = sb_court_dynastic_movement_scope value = prev } }
   if = { limit = { owner = { var:sb_court_dynastic_original_law_var = 2 } } set_core_ideology = ideology_sb_contested_social_crown }
   add_modifier = { name = sb_court_claimant_mobilisation }
  }
  sb_court_apply_recruitment_trial = yes
  var:sb_court_dynastic_rival_scope = {
   if = {
    limit = { NOT = { has_role_of_type = agitator } }
    set_variable = sb_court_dynastic_added_agitator_var
    add_character_role = agitator
   }
  }
  sb_court_progress_dynastic_crisis = yes
  sb_court_queue_dynastic_poll = yes
 }
 else_if = { limit = { NOT = { has_variable = sb_court_dynastic_test_pending_var } NOT = { has_variable = sb_court_dynastic_war_scope } } sb_court_dismiss_dynastic_choice = yes }
}
sb_court_queue_dynastic_poll = {
 if = {
  limit = { has_variable = sb_court_dynastic_test_pending_var NOT = { has_variable = sb_court_dynastic_poll_queued_var } NOT = { has_variable = sb_court_recruitment_failed_var } }
  set_variable = sb_court_dynastic_poll_queued_var
  trigger_event = { id = sb_court_succession.991 days = 7 }
 }
}
sb_court_dismiss_dynastic_choice = {
 if = {
  limit = { NOT = { has_variable = sb_court_dynastic_crisis_var } }
  remove_variable = sb_court_dynastic_selection_open_var
 }
 else_if = {
  limit = { exists = var:sb_court_dynastic_nominee_scope exists = var:sb_court_dynastic_rival_scope }
  if = {
   limit = { scope:sb_court_nominee ?= var:sb_court_dynastic_nominee_scope scope:sb_court_rival ?= var:sb_court_dynastic_rival_scope }
   remove_variable = sb_court_dynastic_selection_open_var
   sb_court_cancel_dynastic_crisis = yes
  }
 }
}

# Country scope. Supply temporary IG sponsorship without converting leaders or the court.
# Repeated polls add nothing; ordinary political preferences return with crisis cleanup.
sb_court_reconcile_claimant_sponsor = {
 if = {
  limit = {
   has_variable = sb_court_dynastic_crisis_var
   has_variable = sb_court_dynastic_test_pending_var
   exists = var:sb_court_dynastic_movement_scope
  }
  every_interest_group = {
   if = {
    limit = { owner = { var:sb_court_dynastic_original_law_var = 2 } }
    if = { limit = { NOT = { has_ideology = ideology:ideology_sb_contested_social_crown } } add_ideology = ideology_sb_contested_social_crown }
   }
   else = {
    if = { limit = { NOT = { has_ideology = ideology:ideology_sb_contested_crown } } add_ideology = ideology_sb_contested_crown }
   }
   if = { limit = { NOT = { has_modifier = sb_court_claimant_armed_forces } } add_modifier = { name = sb_court_claimant_armed_forces } }
  }
 }
}

# Called weekly only for a reserved country. Never manipulate another movement's conflict.
sb_court_progress_dynastic_crisis = {
 save_scope_as = sb_court_country
 if = {
  limit = { has_variable = sb_court_dynastic_crisis_var NOT = { has_variable = sb_court_dynastic_war_scope } }
  if = { limit = { NOT = { sb_court_dynastic_pair_valid = yes } } sb_court_cancel_dynastic_crisis = yes }
  else_if = {
   limit = { has_variable = sb_court_dynastic_test_pending_var exists = var:sb_court_dynastic_movement_scope }
   sb_court_reconcile_claimant_sponsor = yes
   if = {
    limit = {
     var:sb_court_dynastic_rival_scope = { political_movement ?= scope:sb_court_country.var:sb_court_dynastic_movement_scope }
     var:sb_court_dynastic_movement_scope = { sb_court_recruitment_ready = yes }
     OR = {
      AND = { var:sb_court_dynastic_original_law_var = 1 var:sb_court_dynastic_movement_scope = { most_desired_law ?= law_type:law_sb_contested_crown } }
      AND = { var:sb_court_dynastic_original_law_var = 2 var:sb_court_dynastic_movement_scope = { most_desired_law ?= law_type:law_sb_contested_social_crown } }
     }
    }
    set_variable = sb_court_dynastic_demand_verified_var
    if = {
     limit = { NOT = { has_variable = sb_court_dynastic_advanced_var } }
     random_civil_war = {
      limit = { is_civil_war_type = revolution political_movement ?= scope:sb_court_country.var:sb_court_dynastic_movement_scope }
      scope:sb_court_country = {
       set_variable = { name = sb_court_dynastic_preparing_war_scope value = prev }
       set_variable = sb_court_dynastic_advanced_var
      }
      add_civil_war_progress = 1
     }
    }
   }
  }
  else_if = { limit = { has_variable = sb_court_dynastic_test_pending_var } sb_court_cancel_dynastic_crisis = yes }
 }
}

# Root is the original country and scope:target is the actual uprising country.
sb_court_track_dynastic_war = {
 save_scope_as = sb_court_country
 if = {
  limit = {
   has_variable = sb_court_dynastic_test_pending_var sb_court_dynastic_pair_valid = yes
   exists = scope:target exists = var:sb_court_dynastic_movement_scope
   scope:target = {
    civil_war_origin_country ?= scope:sb_court_country
    var:civil_war_type_var ?= movement_type:movement_sb_dynastic_test
   }
  }
  random_civil_war = {
   limit = { is_civil_war_type = revolution political_movement ?= scope:sb_court_country.var:sb_court_dynastic_movement_scope }
   scope:sb_court_country = { set_variable = { name = sb_court_dynastic_war_scope value = prev } }
  }
  if = {
   limit = { exists = var:sb_court_dynastic_war_scope }
   scope:target = { save_scope_as = sb_court_dynastic_rebels }
   save_scope_as = sb_court_dynastic_original
   set_variable = sb_court_selecting_head_var
   set_variable = { name = sb_court_dynastic_original_country_scope value = this }
   set_variable = { name = sb_court_dynastic_opponent_scope value = scope:sb_court_dynastic_rebels }
   if = {
    limit = { has_variable = sb_court_dynastic_back_claimant_var }
    set_variable = { name = sb_court_dynastic_candidate_scope value = var:sb_court_dynastic_rival_scope }
    set_variable = { name = sb_court_dynastic_opponent_candidate_scope value = var:sb_court_dynastic_nominee_scope }
    save_scope_as = sb_court_dynastic_rival_country
    scope:sb_court_dynastic_rebels = { save_scope_as = sb_court_dynastic_incumbent_country }
   }
   else = {
    set_variable = { name = sb_court_dynastic_candidate_scope value = var:sb_court_dynastic_nominee_scope }
    set_variable = { name = sb_court_dynastic_opponent_candidate_scope value = var:sb_court_dynastic_rival_scope }
    save_scope_as = sb_court_dynastic_incumbent_country
    scope:sb_court_dynastic_rebels = { save_scope_as = sb_court_dynastic_rival_country }
   }
   scope:sb_court_dynastic_rebels = { sb_court_copy_dynastic_record = yes }
   # Prerogative follows the contender, regardless of which country the player backs.
   scope:sb_court_dynastic_rival_country = {
    set_variable = { name = sb_prerogative_var value = 0 }
    set_variable = { name = sb_prerogative_last_change_var value = 0 }
   }
   # In a contested succession both princes begin new reigns; neither inherits
   # their predecessor's Prerogative. A challenged living incumbent retains it.
   if = {
    limit = { var:sb_court_dynastic_reason_var = 2 }
    scope:sb_court_dynastic_incumbent_country = {
     set_variable = { name = sb_prerogative_var value = 0 }
     set_variable = { name = sb_prerogative_last_change_var value = 0 }
    }
   }
   scope:sb_court_dynastic_rebels = { sb_court_copy_dynastic_firearms = yes }
   # Keep the same penalty on both countries even if native IG inheritance omits it.
   scope:sb_court_dynastic_rebels = {
    every_interest_group = {
     if = { limit = { NOT = { has_modifier = sb_court_claimant_armed_forces } } add_modifier = { name = sb_court_claimant_armed_forces } }
    }
   }
   every_in_list = {
    variable = sb_court_dynastic_roster
    scope:sb_court_dynastic_rebels = {
     if = { limit = { prev = { has_variable = sb_court_registered_var } NOT = { is_target_in_variable_list = { name = sb_court_members target = prev } } } add_to_variable_list = { name = sb_court_members target = prev } }
    }
    # Unpledged/third-party actors retain their commitments and the native physical allocation.
    if = { limit = { owner ?= scope:sb_court_dynastic_rebels } set_variable = { name = sb_court_home_scope value = scope:sb_court_dynastic_rebels } }
   }
   every_in_list = { variable = sb_court_dynastic_incumbent_members sb_court_place_dynastic_member = { COUNTRY = scope:sb_court_dynastic_incumbent_country } }
   every_in_list = { variable = sb_court_dynastic_rival_members sb_court_place_dynastic_member = { COUNTRY = scope:sb_court_dynastic_rival_country } }
   scope:sb_court_dynastic_incumbent_country = { sb_court_rebase_dynastic_side = { SIDE = incumbent } sb_court_set_dynastic_head = yes }
   scope:sb_court_dynastic_rival_country = { sb_court_rebase_dynastic_side = { SIDE = rival } sb_court_set_dynastic_head = yes }
   if = { limit = { var:sb_court_dynastic_reason_var = 2 } sb_court_retire_mpande = yes }
   scope:sb_court_dynastic_rebels = {
    if = { limit = { var:sb_court_dynastic_original_law_var = 2 } activate_law = law_type:law_sb_contested_social_crown }
    else = { activate_law = law_type:law_sb_contested_crown }
    sb_court_set_dynastic_head = yes
    remove_variable = sb_court_selecting_head_var
    sb_institutions_reconcile = yes
   }
   sb_court_rebuild_dynastic_forces = yes
   sb_court_reconcile_dynastic_supply = yes
   scope:sb_court_dynastic_rebels = { sb_court_reconcile_dynastic_supply = yes }
   # Helpers in the rebel country change temporary saved scopes; rebind our original actor.
   scope:sb_court_dynastic_original = { save_scope_as = sb_court_country }
   remove_variable = sb_court_selecting_head_var
   remove_variable = sb_court_dynastic_test_pending_var
   sb_institutions_reconcile = yes
  }
 }
}
