sb_boer_ai_economy_ora_trade_center = {
	if = {
		limit = {
			is_country_alive = yes
			is_ai = yes
			NOT = { any_scope_building = { is_building_type = building_trade_center } }
		}
		random_scope_state = {
			limit = { state_region = s:STATE_VRYSTAAT }
			create_building = {
				building = building_trade_center
				activate_production_methods = { "pm_trade_center_trade_quantity_normal" }
				add_ownership = {
					building = {
						type = building_trade_center
						country = c:ORA
						levels = 1
						region = "STATE_VRYSTAAT"
					}
				}
				reserves = 1
			}
		}
	}
}
