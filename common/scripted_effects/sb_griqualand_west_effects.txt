# Spes Bona - Kimberley and Griqualand West helpers

# Kimberley starts as a direct WBL building. Resource potential is delayed
# until the sovereignty chain resolves to avoid split Griqualand West leakage.

sb_griqualand_west_save_country_loc_scopes = {
	c:CAP ?= {
		if = {
			limit = { is_country_alive = yes }
			save_scope_as = sb_cap_scope
		}
	}
	c:ORA ?= {
		if = {
			limit = {
				OR = {
					is_country_alive = yes
					c:TRN ?= {
						is_country_alive = yes
						has_variable = sb_oranje_federated_into_trn_var
					}
				}
			}
			save_scope_as = sb_oranje_country_scope
		}
	}

	if = {
		limit = {
			c:TRN ?= {
				is_country_alive = yes
				OR = {
					has_variable = sb_oranje_federated_into_trn_var
					has_variable = sb_kimberley_oranje_proxy_var
					has_variable = sb_kimberley_ora_pressed_claim_var
					c:ORA ?= {
						is_country_alive = yes
						is_direct_subject_of = c:TRN
						OR = {
							is_subject_type = subject_type_sb_boer_presidential_union
							is_subject_type = subject_type_sb_boer_confederal_partner
						}
					}
				}
			}
		}
		c:TRN ?= { save_scope_as = sb_oranje_claimant_scope }
	}
	else_if = {
		limit = {
			c:ORA ?= { is_country_alive = yes }
		}
		c:ORA ?= { save_scope_as = sb_oranje_claimant_scope }
	}

	if = {
		limit = {
			c:WBL ?= { is_country_alive = yes }
		}
		c:WBL ?= { save_scope_as = sb_local_authority_scope }
	}
	else = {
		save_scope_as = sb_local_authority_scope
	}

	if = {
		limit = { has_variable = sb_griqualand_west_cap_annexation_demand_var }
		c:CAP ?= { save_scope_as = sb_demanding_claimant_scope }
	}
	else_if = {
		limit = { has_variable = sb_griqualand_west_trn_submission_demand_var }
		c:TRN ?= { save_scope_as = sb_demanding_claimant_scope }
	}
	else_if = {
		limit = { has_variable = sb_griqualand_west_ora_submission_demand_var }
		c:ORA ?= { save_scope_as = sb_demanding_claimant_scope }
	}
}

# Oranje remains the legal claimant when TRN arbitrates the dispute as its
# senior confederal partner. The TRN cleanup also repairs active old-save arcs.
sb_griqualand_west_grant_oranje_claim = {
	if = {
		limit = { c:ORA ?= { is_country_alive = yes } }
		c:ORA ?= {
			if = {
				limit = { NOT = { has_claim = s:STATE_GRIQUALAND_WEST } }
				s:STATE_GRIQUALAND_WEST = { add_claim = c:ORA }
			}
		}
		c:TRN ?= {
			if = {
				limit = { has_claim = s:STATE_GRIQUALAND_WEST }
				s:STATE_GRIQUALAND_WEST = { remove_claim = c:TRN }
			}
		}
	}
	else_if = {
		limit = {
			c:TRN ?= {
				is_country_alive = yes
				has_variable = sb_oranje_federated_into_trn_var
			}
		}
		c:TRN ?= {
			if = {
				limit = { NOT = { has_claim = s:STATE_GRIQUALAND_WEST } }
				s:STATE_GRIQUALAND_WEST = { add_claim = c:TRN }
			}
		}
	}
}

# Federation transfers Oranje's unresolved Kimberley standing and any subjects
# created by its diamond/corridor routes to the surviving Transvaal republic.
sb_griqualand_west_inherit_oranje_federation = {
	if = {
		limit = {
			country_definition = cd:TRN
			c:ORA ?= {
				is_country_alive = yes
				is_direct_subject_of = root
				is_subject_type = subject_type_sb_boer_confederal_partner
			}
		}
		set_variable = sb_oranje_federated_into_trn_var

		if = {
			limit = { c:ORA ?= { has_variable = sb_kimberley_ora_pressed_claim_var } }
			set_variable = sb_kimberley_ora_pressed_claim_var
		}
		if = {
			limit = {
				OR = {
					c:ORA ?= { has_variable = sb_kimberley_ora_claimant_var }
					AND = {
						has_global_variable = sb_kimberley_boer_victory_global_var
						OR = {
							c:ORA ?= { owns_entire_state_region = STATE_GRIQUALAND_WEST }
							c:WBL ?= {
								is_country_alive = yes
								OR = {
									is_direct_subject_of = c:ORA
									is_direct_subject_of = root
								}
							}
						}
					}
				}
			}
			set_variable = sb_kimberley_trn_claimant_var
		}
		if = {
			limit = { c:ORA ?= { has_claim = s:STATE_GRIQUALAND_WEST } }
			s:STATE_GRIQUALAND_WEST = { remove_claim = c:ORA }
			if = {
				limit = { NOT = { has_claim = s:STATE_GRIQUALAND_WEST } }
				s:STATE_GRIQUALAND_WEST = { add_claim = c:TRN }
			}
		}
		if = {
			limit = {
				has_global_variable = sb_kimberley_discovered_global_var
				NOT = { has_global_variable = sb_kimberley_remaining_potential_seeded_var }
			}
			set_variable = sb_kimberley_oranje_proxy_var
		}

		if = {
			limit = {
				c:WBL ?= {
					is_country_alive = yes
					is_direct_subject_of = c:ORA
				}
			}
			transfer_subject = c:WBL
		}
		if = {
			limit = {
				c:SGO ?= {
					is_country_alive = yes
					is_direct_subject_of = c:ORA
				}
			}
			transfer_subject = c:SGO
		}
		if = {
			limit = {
				has_global_variable = sb_bechuanaland_corridor_open_global_var
				NOT = { has_journal_entry = je_sb_bechuanaland_corridor }
			}
			add_contextless_journal_entry = je_sb_bechuanaland_corridor
			sb_bechuanaland_sync_influence_bar = yes
		}
	}
}

sb_kimberley_create_initial_diamond_mine = {
	if = {
		limit = {
			NOT = { has_building = building_diamond_mine }
		}
		create_building = {
			building = building_diamond_mine
			level = 1
			reserves = 1
			activate_production_methods = {
				"pm_picks_and_shovels_building_gold_mine"
				"pm_no_explosives"
				"pm_no_steam_automation"
				"pm_road_carts"
			}
		}
	}
}

sb_kimberley_seed_remaining_potential = {
	if = {
		limit = {
			NOT = { has_global_variable = sb_kimberley_remaining_potential_seeded_var }
		}
		set_global_variable = sb_kimberley_remaining_potential_seeded_var
		p:x68B5E8.state = {
			if = {
				limit = {
					NOT = { has_potential_resource = building_diamond_mine }
				}
				add_resource_potential = building_diamond_mine
				change_resource_potential = {
					type = building_diamond_mine
					amount = 18
				}
			}
			else = {
				change_resource_potential = {
					type = building_diamond_mine
					amount = 19
				}
			}
		}
	}
}

	c:CAP ?= {
		if = {
			limit = {
				is_country_alive = yes
				is_ai = yes
				owns_entire_state_region = STATE_GRIQUALAND_WEST
			}
			s:STATE_GRIQUALAND_WEST = {
				region_state:CAP ?= {

sb_kimberley_finalize_cap_annexation_route = {
	set_global_variable = sb_kimberley_cap_victory_global_var
	set_global_variable = sb_bechuanaland_corridor_watch_global_var
	c:WBL ?= {
		if = {
			limit = { is_country_alive = yes }
				c:CAP ?= {
					if = {
						limit = { is_country_alive = yes }
						if = {
							limit = { c:WBL ?= { is_subject = yes } }
							transfer_subject = c:WBL
						}
						annex = c:WBL
					}
			}
		}
	}
	hidden_effect = {
		sb_cap_ai_incorporate_griqualand_west = yes
		sb_kimberley_seed_remaining_potential = yes
		sb_griqualand_west_mark_resolved = yes
		sb_griqualand_west_clear_crisis_vars = yes
	}

	c:CAP ?= {
		if = {
			limit = { is_country_alive = yes }
			trigger_event = { id = sb_griqualand_west.260 days = 14 popup = yes }
		}
	}
}

sb_griqualand_west_finalize_refusal_contest = {
	if = {
		limit = { p:x68B5E8.state.owner = c:CAP }
		sb_kimberley_finalize_cap_annexation_route = yes
	}
	else_if = {
		limit = {
			p:x68B5E8.state.owner = c:WBL
			c:WBL ?= { is_subject_of = c:CAP }
		}
		sb_kimberley_finalize_cap_protectorate_route = yes
	}
	else_if = {
		limit = {
			p:x68B5E8.state.owner = c:WBL
			c:WBL ?= {
				OR = {
					is_subject_of = c:ORA
					is_subject_of = c:TRN
				}
			}
		}
		sb_kimberley_finalize_oranje_protectorate_route = yes
	}
	else_if = {
		limit = {
			OR = {
				p:x68B5E8.state.owner = c:ORA
				p:x68B5E8.state.owner = c:TRN
			}
		}
		sb_kimberley_finalize_direct_boer_route = yes
	}
	else_if = {
		limit = { p:x68B5E8.state.owner = c:WBL }
		sb_kimberley_finalize_wbl_diamond_mines_route = yes
	}
	else = {
		# A third-party intervention should not leave Kimberley permanently pending.
		sb_kimberley_seed_remaining_potential = yes
		sb_griqualand_west_mark_resolved = yes
