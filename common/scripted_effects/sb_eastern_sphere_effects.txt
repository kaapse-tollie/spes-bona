sb_gaza_monthly_random_event = {
	if = {
		limit = {
			has_journal_entry = je_sb_consolidating_gaza
			NOT = { has_variable = sb_gaza_event_cooldown_var }
		}
		random_list = {
			8 = {
				trigger = { NOT = { has_variable = sb_gaza_010_fired_var } }
				set_variable = { name = sb_gaza_event_cooldown_var months = 1 }
				trigger_event = { id = sb_gaza.010 days = 7 popup = yes }
			}
			5 = {
				trigger = { NOT = { has_variable = sb_gaza_020_fired_var } }
				set_variable = { name = sb_gaza_event_cooldown_var months = 1 }
				trigger_event = { id = sb_gaza.020 days = 7 popup = yes }
			}
			5 = {
				trigger = { NOT = { has_variable = sb_gaza_030_fired_var } }
				set_variable = { name = sb_gaza_event_cooldown_var months = 1 }
				trigger_event = { id = sb_gaza.030 days = 7 popup = yes }
			}
			5 = {
				trigger = {
					NOT = { has_variable = sb_gaza_040_fired_var }
					c:POR ?= {
						OR = {
							any_scope_state = { state_region = s:STATE_LOURENCO_MARQUES owner = c:POR }
							any_scope_state = { state_region = s:STATE_ZAMBEZIA owner = c:POR }
						}
					}
				}
				set_variable = { name = sb_gaza_event_cooldown_var months = 1 }
				trigger_event = { id = sb_gaza.040 days = 7 popup = yes }
			}
			77 = { } #
		}
	}
}

		}
	}
	c:MZQ ?= {
		set_strategy = ai_strategy_colonial_extraction
		remove_primary_culture = cu:british
		add_primary_culture = cu:portuguese

	every_primary_culture = {
		limit = {
			has_discrimination_trait_group = heritage_group_european

	}

sb_create_or_refresh_imperial_bind_trade_treaty = {
	# Existing treaties cannot be extended directly, so reuse them and refresh
	# the scheme marker rather than creating parallel trade-through treaties.
	if = {
		limit = {
			NOT = {
				any_scope_treaty = {
					binds = scope:sb_confed_owner
					any_scope_article = {
						has_type = transit_rights
						source_country = scope:sb_confed_owner
						target_country = scope:sb_bound_participant
					}
					any_scope_article = {
						has_type = goods_transfer
						input_goods = g:merchant_marine
						source_country = scope:sb_confed_owner
						target_country = scope:sb_bound_participant
					}
				}
			}
		}
		scope:sb_confed_owner = {
			if = {
				limit = {
					country_definition = cd:GBR
					scope:sb_bound_participant = {
						NOT = { sb_has_gbr_power_bloc_embassy = yes }
					}
				}
				create_treaty = {
					first_country = scope:sb_confed_owner
					second_country = scope:sb_bound_participant
					is_draft = no
					binding_period = { years = 5 }
					articles_to_create = {
						{ article = transit_rights source_country = scope:sb_confed_owner target_country = scope:sb_bound_participant }
						{ article = goods_transfer source_country = scope:sb_confed_owner target_country = scope:sb_bound_participant inputs = { { goods = g:merchant_marine } { quantity = 10 } } }
						{ article = host_power_bloc_embassy source_country = scope:sb_bound_participant target_country = scope:sb_confed_owner }
					}
				}
			}
			else = {
				create_treaty = {
					first_country = scope:sb_confed_owner
					second_country = scope:sb_bound_participant
					is_draft = no
