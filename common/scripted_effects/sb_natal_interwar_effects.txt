# Natal interwar, Indian indenture, and post-annexation Zululand settlement.

sb_natal_archive_zulu_firearms = {
	if = {
		limit = {
			country_definition = cd:ZUL
			c:NAL ?= {
				is_country_alive = yes
				has_variable = sb_natalia_british_colony_resolved_var
			}
		}
		c:NAL = {
			set_variable = { name = sb_natal_archived_firearms_progress_var value = 0 }
			set_variable = { name = sb_natal_archived_firearms_treaty_progress_var value = 0 }
			set_variable = { name = sb_natal_archived_firearms_industry_progress_var value = 0 }
			set_variable = { name = sb_natal_archived_firearms_tier_var value = 0 }
			if = {
				limit = { ROOT = { has_variable = sb_firearms_progress_months_var } }
				set_variable = {
					name = sb_natal_archived_firearms_progress_var
					value = ROOT.var:sb_firearms_progress_months_var
				}
			}
			if = {
				limit = { ROOT = { has_variable = sb_firearms_treaty_progress_months_var } }
				set_variable = {
					name = sb_natal_archived_firearms_treaty_progress_var
					value = ROOT.var:sb_firearms_treaty_progress_months_var
				}
			}
			if = {
				limit = { ROOT = { has_variable = sb_firearms_industry_progress_months_var } }
				set_variable = {
					name = sb_natal_archived_firearms_industry_progress_var
					value = ROOT.var:sb_firearms_industry_progress_months_var
				}
			}
			if = {
				limit = { ROOT = { has_variable = sb_firearms_applied_tier_var } }
				set_variable = {
					name = sb_natal_archived_firearms_tier_var
					value = ROOT.var:sb_firearms_applied_tier_var
				}
			}
			if = {
				limit = { ROOT = { has_variable = sb_firearms_modernized_var } }
				set_variable = sb_natal_archived_firearms_modernized_var
			}
			else_if = {
				limit = { has_variable = sb_natal_archived_firearms_modernized_var }
				remove_variable = sb_natal_archived_firearms_modernized_var
			}
			set_variable = sb_natal_zulu_firearms_archived_var
		}
	}
}

sb_natal_restore_archived_zulu_firearms = {
	if = {
		limit = {
			country_definition = cd:ZUL
			c:NAL ?= { has_variable = sb_natal_zulu_firearms_archived_var }
		}
		set_variable = {
			name = sb_firearms_progress_months_var
			value = c:NAL.var:sb_natal_archived_firearms_progress_var
		}
		set_variable = {
			name = sb_firearms_treaty_progress_months_var
			value = c:NAL.var:sb_natal_archived_firearms_treaty_progress_var
		}
		set_variable = {
			name = sb_firearms_industry_progress_months_var
			value = c:NAL.var:sb_natal_archived_firearms_industry_progress_var
		}
		if = {
			limit = { c:NAL = { has_variable = sb_natal_archived_firearms_modernized_var } }
			set_variable = sb_firearms_modernized_var
			sb_remove_iron_age_weapons_modifiers = yes
		}
		else = {
			if = {
				limit = { NOT = { has_journal_entry = je_sb_adopt_firearms } }
				add_journal_entry = { type = je_sb_adopt_firearms }
			}
			if = {
				limit = { c:NAL = { has_variable = sb_natal_archived_firearms_tier_var } }
				set_variable = {
					name = sb_firearms_applied_tier_var
					value = c:NAL.var:sb_natal_archived_firearms_tier_var
				}
			}
			sb_repair_iron_age_weapons_penalty = yes
			sb_sync_firearms_progress_bar = yes
		}
	}
}

sb_natal_select_indenture_origin_and_create_migration = {
	s:STATE_NATAL.region_state:NAL ?= {
		if = {
			limit = { state_is_eligible_as_mass_migration_target = yes }
			save_scope_as = sb_natal_indenture_target_scope
		}
	}
	random_list = {
		35 = {
			trigger = {
				c:GBR = {
					any_subject_or_below = {
						sb_natal_country_is_indian = yes
						any_scope_state = { any_scope_pop = { culture = cu:tamil } }
					}
				}
			}
			cu:tamil = { save_scope_as = sb_natal_indenture_culture_scope }
		}
		25 = {
			trigger = {
				c:GBR = {
					any_subject_or_below = {
						sb_natal_country_is_indian = yes
						any_scope_state = { any_scope_pop = { culture = cu:telegu } }
					}
				}
			}
			cu:telegu = { save_scope_as = sb_natal_indenture_culture_scope }
		}
		20 = {
			trigger = {
				c:GBR = {
					any_subject_or_below = {
						sb_natal_country_is_indian = yes
						any_scope_state = { any_scope_pop = { culture = cu:hindustani } }
					}
				}
			}
			cu:hindustani = { save_scope_as = sb_natal_indenture_culture_scope }
		}
		15 = {
			trigger = {
				c:GBR = {
					any_subject_or_below = {
						sb_natal_country_is_indian = yes
						any_scope_state = { any_scope_pop = { culture = cu:bihari } }
					}
				}
			}
			cu:bihari = { save_scope_as = sb_natal_indenture_culture_scope }
		}
		5 = {
			trigger = {
				c:GBR = {
					any_subject_or_below = {
						sb_natal_country_is_indian = yes
						any_scope_state = { any_scope_pop = { culture = cu:gujarati } }
					}
				}
			}
			cu:gujarati = { save_scope_as = sb_natal_indenture_culture_scope }
		}
	}
	if = {
		limit = {
			exists = scope:sb_natal_indenture_target_scope
			exists = scope:sb_natal_indenture_culture_scope
		}
		c:GBR = {
			random_subject_or_below = {
				limit = {
					sb_natal_country_is_indian = yes
					any_scope_state = {
						any_scope_pop = { culture = scope:sb_natal_indenture_culture_scope }
					}
				}
				save_scope_as = sb_natal_indenture_origin_scope
			}
		}
	}
	if = {
		limit = {
			exists = scope:sb_natal_indenture_target_scope
			exists = scope:sb_natal_indenture_culture_scope
			exists = scope:sb_natal_indenture_origin_scope
		}
		scope:sb_natal_indenture_target_scope = {
			create_mass_migration = {
				origin = scope:sb_natal_indenture_origin_scope
				culture = scope:sb_natal_indenture_culture_scope
			}
		}
		set_variable = sb_natal_indenture_completed_var
	}
}

sb_natal_ensure_gandhi_for_civil_rights = {
	if = {
		limit = {
			game_date >= 1893.1.1
			sb_natal_has_civil_rights_movement = yes
			NOT = { has_variable = sb_natal_civil_rights_resolved_var }
		}
		if = {
			limit = {
				any_country = {
					NOT = { sb_natal_country_is_indian = yes }
					any_scope_character = {
						has_template = BIC_mohandas_karamchand_gandhi
						is_character_alive = yes
					}
				}
				NOT = {
					any_scope_character = {
						has_template = BIC_mohandas_karamchand_gandhi
						is_character_alive = yes
					}
				}
			}
			random_country = {
				limit = {
					NOT = { sb_natal_country_is_indian = yes }
					any_scope_character = {
						has_template = BIC_mohandas_karamchand_gandhi
						is_character_alive = yes
					}
				}
				random_scope_character = {
					limit = {
						has_template = BIC_mohandas_karamchand_gandhi
						is_character_alive = yes
					}
					transfer_character = ROOT
				}
			}
		}
		else_if = {
			limit = {
				any_character_in_exile_pool = {
					has_template = BIC_mohandas_karamchand_gandhi
					is_character_alive = yes
				}
			}
			random_character_in_exile_pool = {
				limit = {
					has_template = BIC_mohandas_karamchand_gandhi
					is_character_alive = yes
				}
				transfer_character = ROOT
			}
		}
		else_if = {
			limit = {
				NOT = { has_global_variable = gandhi_spawn }
				NOT = {
					any_country = {
						any_scope_character = {
							has_template = BIC_mohandas_karamchand_gandhi
							is_character_alive = yes
						}
					}
				}
			}
			set_global_variable = gandhi_spawn
			create_character = { template = BIC_mohandas_karamchand_gandhi }
		}
	}
}

sb_natal_schedule_civil_rights_petition = {
	if = {
		limit = {
			sb_natal_has_responsible_government = yes
			has_law = law_type:law_racial_segregation
			NOT = { has_variable = sb_natal_civil_rights_petition_spent_var }
			NOT = { has_variable = sb_natal_civil_rights_petition_pending_var }
			any_political_movement = {
				is_political_movement_type = movement_sb_natal_civil_rights
				political_movement_radicalism >= 0.75
			}
		}
		set_variable = sb_natal_civil_rights_petition_spent_var
		set_variable = {
			name = sb_natal_civil_rights_petition_pending_var
			days = 120
		}
		if = {
			limit = {
				any_scope_character = {
					has_template = BIC_mohandas_karamchand_gandhi
					is_character_alive = yes
				}
			}
			set_variable = sb_natal_civil_rights_petition_requires_gandhi_var
		}
		trigger_event = { id = sb_natal_interwar.010 days = 90 popup = yes }
	}
}

sb_natal_cancel_invalid_civil_rights_petition = {
	if = {
		limit = {
			has_variable = sb_natal_civil_rights_petition_pending_var
			OR = {
				NOT = { sb_natal_has_responsible_government = yes }
				NOT = { sb_natal_has_civil_rights_movement = yes }
				AND = {
					has_variable = sb_natal_civil_rights_petition_requires_gandhi_var
					NOT = {
						any_scope_character = {
							has_template = BIC_mohandas_karamchand_gandhi
							is_character_alive = yes
						}
					}
				}
			}
		}
		remove_variable = sb_natal_civil_rights_petition_pending_var
		if = {
			limit = { has_variable = sb_natal_civil_rights_petition_requires_gandhi_var }
			remove_variable = sb_natal_civil_rights_petition_requires_gandhi_var
		}
	}
}

sb_natal_ensure_zulu_restoration_agitator = {
	if = {
		limit = {
			game_date >= 1884.1.1
			NOT = {
				any_scope_character = {
					has_template = ZUL_dinuzulu_heir
					is_character_alive = yes
				}
			}
		}
		if = {
			limit = {
				any_country = {
					any_scope_character = {
						has_template = ZUL_dinuzulu_heir
						is_character_alive = yes
					}
				}
			}
			set_global_variable = sb_dinuzulu_ever_created_global_var
			random_country = {
				limit = {
					any_scope_character = {
						has_template = ZUL_dinuzulu_heir
						is_character_alive = yes
					}
				}
				random_scope_character = {
					limit = {
						has_template = ZUL_dinuzulu_heir
						is_character_alive = yes
						NOT = { is_ruler_of_own_country = yes }
					}
					transfer_character = ROOT
					replace_character_roles = { add = { agitator } remove_all = yes }
				}
			}
		}
		else_if = {
			limit = { NOT = { has_global_variable = sb_dinuzulu_ever_created_global_var } }
			set_global_variable = sb_dinuzulu_ever_created_global_var
			create_character = {
				template = ZUL_dinuzulu_heir
				on_created = {
					replace_character_roles = { add = { agitator } remove_all = yes }
				}
			}
		}
	}
	if = {
		limit = {
			NOT = {
				any_scope_character = {
					is_character_alive = yes
					OR = {
						has_template = ZUL_dinuzulu_heir
						has_variable = sb_natal_generated_zulu_restoration_leader_var
					}
				}
			}
		}
		create_character = {
			culture = cu:zulu
			religion = rel:animist
			age = 30
			is_agitator = yes
			interest_group = ig_landowners
			ideology = ideology_ethno_nationalist
			traits = { ambitious inspirational_orator }
			on_created = { set_variable = sb_natal_generated_zulu_restoration_leader_var }
		}
	}
}

sb_natal_interwar_monthly_housekeeping = {
	if = {
		limit = {
			sb_natal_is_british_colony = yes
			has_variable = sb_klip_river_county_resolved_var
			NOT = { has_variable = sb_natal_indenture_completed_var }
			NOT = { has_variable = sb_natal_indenture_pending_var }
			sb_natal_british_network_has_indenture_origin = yes
			s:STATE_NATAL.region_state:NAL ?= {
				state_is_eligible_as_mass_migration_target = yes
			}
		}
		random_list = {
			1 = {
				set_variable = {
					name = sb_natal_indenture_pending_var
					days = 30
				}
				trigger_event = { id = sb_natal_interwar.001 days = 1 popup = yes }
			}
			99 = { }
		}
	}
	if = {
		limit = {
			sb_natal_is_british_colony = yes
			c:ZUL ?= { is_country_alive = yes }
		}
		c:ZUL = { sb_natal_archive_zulu_firearms = yes }
	}
	sb_natal_ensure_civil_rights_movement = yes
	sb_natal_ensure_gandhi_for_civil_rights = yes
	sb_natal_schedule_civil_rights_petition = yes
	sb_natal_cancel_invalid_civil_rights_petition = yes
	sb_natal_schedule_zulu_settlement = yes
	if = {
		limit = { sb_natal_has_zulu_restoration_movement = yes }
		sb_natal_ensure_zulu_restoration_agitator = yes
	}
	if = {
		limit = {
			sb_natal_has_zulu_restoration_movement = yes
			NOT = { has_variable = sb_natal_zulu_restoration_uprising_pending_var }
			NOT = { has_variable = sb_natal_zulu_restoration_resolved_var }
			NOT = { c:ZUL ?= { is_country_alive = yes } }
			sb_natal_owns_complete_northern_zulu_core = yes
			any_political_movement = {
				is_political_movement_type = movement_sb_zulu_restoration
				political_movement_radicalism >= 1
			}
		}
		sb_natal_begin_zulu_restoration_uprising = yes
	}
	if = {
		limit = {
			has_variable = sb_natal_zulu_arms_appeal_sent_var
			NOT = { has_variable = sb_natal_zulu_arms_appeal_wait_var }
			NOT = { has_variable = sb_natal_zulu_arms_appeal_resolved_var }
		}
		set_variable = sb_natal_zulu_arms_appeal_resolved_var
		set_variable = sb_natal_zulu_restoration_play_pending_var
	}
	if = {
		limit = { has_variable = sb_natal_zulu_restoration_play_pending_var }
		sb_natal_launch_zulu_restoration_play = yes
	}
	if = {
		limit = {
			has_variable = sb_natal_zulu_restoration_play_active_var
			is_at_war = no
			is_active_in_diplomatic_play = no
			c:ZUL ?= {
				is_country_alive = yes
				is_at_war = no
				is_active_in_diplomatic_play = no
			}
		}
		set_variable = sb_natal_zulu_restoration_play_pending_var
		sb_natal_launch_zulu_restoration_play = yes
	}
}
