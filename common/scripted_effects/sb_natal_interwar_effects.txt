# Natal interwar, Indian indenture, and post-annexation Zululand settlement.

sb_natal_archive_zulu_firearms = {
	if = {
		limit = {
			country_definition = cd:ZUL
			c:NAL ?= {
				is_country_alive = yes
				has_variable = sb_natalia_british_colony_resolved_var
			}
		}
		c:NAL = {
			set_variable = { name = sb_natal_archived_firearms_progress_var value = 0 }
			set_variable = { name = sb_natal_archived_firearms_treaty_progress_var value = 0 }
			set_variable = { name = sb_natal_archived_firearms_industry_progress_var value = 0 }
			set_variable = { name = sb_natal_archived_firearms_tier_var value = 0 }
			if = {
				limit = { ROOT = { has_variable = sb_firearms_progress_months_var } }
				set_variable = {
					name = sb_natal_archived_firearms_progress_var
					value = ROOT.var:sb_firearms_progress_months_var
				}
			}
			if = {
				limit = { ROOT = { has_variable = sb_firearms_treaty_progress_months_var } }
				set_variable = {
					name = sb_natal_archived_firearms_treaty_progress_var
					value = ROOT.var:sb_firearms_treaty_progress_months_var
				}
			}
			if = {
				limit = { ROOT = { has_variable = sb_firearms_industry_progress_months_var } }
				set_variable = {
					name = sb_natal_archived_firearms_industry_progress_var
					value = ROOT.var:sb_firearms_industry_progress_months_var
				}
			}
			if = {
				limit = { ROOT = { has_variable = sb_firearms_applied_tier_var } }
				set_variable = {
					name = sb_natal_archived_firearms_tier_var
					value = ROOT.var:sb_firearms_applied_tier_var
				}
			}
			if = {
				limit = { ROOT = { has_variable = sb_firearms_modernized_var } }
				set_variable = sb_natal_archived_firearms_modernized_var
			}
			else_if = {
				limit = { has_variable = sb_natal_archived_firearms_modernized_var }
				remove_variable = sb_natal_archived_firearms_modernized_var
			}
			set_variable = sb_natal_zulu_firearms_archived_var
		}
	}
}

sb_natal_restore_archived_zulu_firearms = {
	if = {
		limit = {
			country_definition = cd:ZUL
			c:NAL ?= { has_variable = sb_natal_zulu_firearms_archived_var }
		}
		set_variable = {
			name = sb_firearms_progress_months_var
			value = c:NAL.var:sb_natal_archived_firearms_progress_var
		}
		set_variable = {
			name = sb_firearms_treaty_progress_months_var
			value = c:NAL.var:sb_natal_archived_firearms_treaty_progress_var
		}
		set_variable = {
			name = sb_firearms_industry_progress_months_var
			value = c:NAL.var:sb_natal_archived_firearms_industry_progress_var
		}
		if = {
			limit = { c:NAL = { has_variable = sb_natal_archived_firearms_modernized_var } }
			set_variable = sb_firearms_modernized_var
			sb_remove_iron_age_weapons_modifiers = yes
		}
		else = {
			if = {
				limit = { NOT = { has_journal_entry = je_sb_adopt_firearms } }
				add_journal_entry = { type = je_sb_adopt_firearms }
			}
			if = {
				limit = { c:NAL = { has_variable = sb_natal_archived_firearms_tier_var } }
				set_variable = {
					name = sb_firearms_applied_tier_var
					value = c:NAL.var:sb_natal_archived_firearms_tier_var
				}
			}
			sb_repair_iron_age_weapons_penalty = yes
			sb_sync_firearms_progress_bar = yes
		}
	}
}

sb_natal_schedule_indenture_program = {
	if = {

sb_natal_cancel_pending_indenture_program = {
	if = {
		limit = { has_variable = sb_natal_indenture_pending_var }
		remove_variable = sb_natal_indenture_pending_var
	}
	if = {
		limit = { has_variable = sb_natal_indenture_wait_var }
		remove_variable = sb_natal_indenture_wait_var
	}
	if = {
		limit = { has_variable = sb_natal_indenture_retry_count_var }
		remove_variable = sb_natal_indenture_retry_count_var
	}
	if = {
		limit = { has_variable = sb_natal_indenture_opening_event_pending_var }
		remove_variable = sb_natal_indenture_opening_event_pending_var
	}
	set_variable = sb_natal_indenture_lapsed_var
}

sb_natal_update_pending_indenture_program = {
	if = {
		limit = { has_variable = sb_natal_indenture_pending_var }
		if = {
			limit = {
				OR = {
					NOT = { sb_natal_is_british_colony = yes }
					NOT = { sb_natal_bic_has_indenture_origin = yes }
					NOT = { sb_natal_has_shepstone_system = yes }
					has_law = law_type:law_closed_borders
				}
			}
			sb_natal_cancel_pending_indenture_program = yes
		}
		else_if = {
			limit = {
				NOT = { has_variable = sb_natal_indenture_wait_var }
				NOT = { has_variable = sb_natal_indenture_opening_event_pending_var }
			}
			if = {
				limit = { sb_natal_bic_has_indenture_opening_cohort = yes }
				set_variable = {
					name = sb_natal_indenture_opening_event_pending_var
					days = 30
				}
				trigger_event = { id = sb_natal_interwar.001 days = 1 popup = yes }
			}
			else_if = {
				limit = { var:sb_natal_indenture_retry_count_var < 5 }
				change_variable = { name = sb_natal_indenture_retry_count_var add = 1 }
				set_variable = { name = sb_natal_indenture_wait_var days = 365 }
			}
			else = { sb_natal_cancel_pending_indenture_program = yes }
		}
	}
}

sb_natal_transfer_indenture_cohort_from_state = {
	# move_partial_pop does not expose the split-off destination pop as a scope.
	# Stage the selected source pop under an absent culture so only the exact
	# transferred cohort can change profession after it reaches Natal.
	ordered_scope_pop = {
		limit = {
			culture = $CULTURE$
			is_pop_type = peasants
			total_size >= $SIZE$
		}
		order_by = total_size
		position = 0
		check_range_bounds = no
		change_pop_culture = {
			target = $STAGING_CULTURE$
			value = 1.00
		}
	}
	ordered_scope_pop = {
		limit = {
			culture = $STAGING_CULTURE$
			is_pop_type = peasants
			total_size >= $SIZE$
		}
		order_by = total_size
		position = 0
		check_range_bounds = no
		move_partial_pop = {
			state = scope:sb_natal_indenture_target_scope
			population = $SIZE$
		}
	}
	every_scope_pop = {
		limit = {
			culture = $STAGING_CULTURE$
			is_pop_type = peasants
		}
		change_pop_culture = {
			target = $CULTURE$
			value = 1.00
		}
	}
	scope:sb_natal_indenture_target_scope = {
		every_scope_pop = {
			limit = {
				culture = $STAGING_CULTURE$
				is_pop_type = peasants
			}
			change_poptype = $DESTINATION_POP_TYPE$
		}
		every_scope_pop = {
			limit = { culture = $STAGING_CULTURE$ }
			change_pop_culture = {
				target = $CULTURE$
				value = 1.00
			}
		}
	}
}

sb_natal_transfer_indenture_cohort = {
	s:STATE_NATAL.region_state:NAL ?= {
		save_temporary_scope_as = sb_natal_indenture_target_scope
	}
	c:BIC ?= {
		ordered_scope_state = {
			limit = {
				any_scope_pop = {
					culture = $CULTURE$
					is_pop_type = peasants
					total_size >= $SIZE$
				}
			}
			order_by = state_population
			position = 0
			check_range_bounds = no
			if = {
				limit = {
					NOT = { any_scope_pop = { culture = cu:promethean } }
					scope:sb_natal_indenture_target_scope = {
						NOT = { any_scope_pop = { culture = cu:promethean } }
					}
				}
				sb_natal_transfer_indenture_cohort_from_state = {
					CULTURE = $CULTURE$
					SIZE = $SIZE$
					STAGING_CULTURE = cu:promethean
					DESTINATION_POP_TYPE = $DESTINATION_POP_TYPE$
				}
			}
			else_if = {
				limit = {
					NOT = { any_scope_pop = { culture = cu:ainu } }
					scope:sb_natal_indenture_target_scope = {
						NOT = { any_scope_pop = { culture = cu:ainu } }
					}
				}
				sb_natal_transfer_indenture_cohort_from_state = {
					CULTURE = $CULTURE$
					SIZE = $SIZE$
					STAGING_CULTURE = cu:ainu
					DESTINATION_POP_TYPE = $DESTINATION_POP_TYPE$
				}
			}
			else = {
				sb_natal_transfer_indenture_cohort_from_state = {
					CULTURE = $CULTURE$
					SIZE = $SIZE$
					STAGING_CULTURE = cu:maori
					DESTINATION_POP_TYPE = $DESTINATION_POP_TYPE$
				}
			}
		}
	}
}

sb_natal_open_indenture_program = {
	if = {
		limit = {
			has_variable = sb_natal_indenture_pending_var
			sb_natal_indenture_program_is_valid = yes
			sb_natal_bic_has_indenture_opening_cohort = yes
		}
		remove_variable = sb_natal_indenture_pending_var
		if = {
			limit = { has_variable = sb_natal_indenture_wait_var }
			remove_variable = sb_natal_indenture_wait_var
		}
		if = {
			limit = { has_variable = sb_natal_indenture_retry_count_var }
			remove_variable = sb_natal_indenture_retry_count_var
		}
		if = {
			limit = { has_variable = sb_natal_indenture_opening_event_pending_var }
			remove_variable = sb_natal_indenture_opening_event_pending_var
		}
		set_variable = sb_natal_indenture_completed_var
		set_variable = sb_natal_indian_status_granted_var
		set_variable = { name = sb_natal_indenture_cycle_index_var value = 0 }
		set_variable = { name = sb_natal_indian_origin_population_var value = 0 }
		active_law:lawgroup_citizenship ?= {
			if = {
				limit = {
					NOT = {
						has_amendment = amendment_type:amendment_sb_natal_indian_indenture_status
					}
				}
				add_amendment = {
					type = amendment_sb_natal_indian_indenture_status
					sponsor = PREV.ig:ig_armed_forces
				}
			}
		}
		hidden_effect = {
			sb_natal_transfer_indenture_cohort = {
				CULTURE = cu:tamil
				SIZE = 342
				DESTINATION_POP_TYPE = pop_type:laborers
			}
		}
		if = {
			limit = { NOT = { has_journal_entry = je_sb_natal_indenture_program_v2 } }
			add_journal_entry = { type = je_sb_natal_indenture_program_v2 }
		}
		sb_natal_refresh_indian_origin_population = yes
		sb_natal_add_indenture_program_costs = yes
		sb_natal_start_indenture_cycle = yes
	}
}

sb_natal_refresh_indian_origin_population = {
	set_variable = { name = sb_natal_indian_origin_population_var value = 0 }
	s:STATE_NATAL.region_state:NAL ?= {
		save_temporary_scope_as = sb_natal_indian_population_state_scope
		set_variable = {
			name = sb_natal_indian_origin_population_temp_var
			value = {
				every_scope_pop = {
					limit = { sb_natal_pop_is_indian_origin = yes }
					add = total_size
				}
			}
		}
	}
	if = {
		limit = { exists = scope:sb_natal_indian_population_state_scope }
		set_variable = {
			name = sb_natal_indian_origin_population_var
			value = scope:sb_natal_indian_population_state_scope.var:sb_natal_indian_origin_population_temp_var
		}
		scope:sb_natal_indian_population_state_scope = {
			remove_variable = sb_natal_indian_origin_population_temp_var
		}
	}
}

sb_natal_add_indenture_program_costs = {
	# Old saves can expose this effect to the JE UI before the monthly population
	# refresh has created the cached value. Initialise it for real execution, and
	# keep the variable-dependent modifier behind an existence guard so tooltip
	# evaluation never tries to resolve an unset script value.
	if = {
		limit = { NOT = { has_variable = sb_natal_indian_origin_population_var } }
		hidden_effect = {
			set_variable = { name = sb_natal_indian_origin_population_var value = 0 }
		}
	}
	if = {
		limit = { has_modifier = sb_natal_indenture_program_costs }
		remove_modifier = sb_natal_indenture_program_costs
	}
	add_modifier = {
		name = sb_natal_indenture_program_costs
		multiplier = {
			value = 1
			if = {
				limit = { has_variable = sb_natal_indenture_intensified_var }
				multiply = 2
			}
		}
	}
	if = {
		limit = { has_modifier = sb_natal_indenture_program_scaled_costs }
		remove_modifier = sb_natal_indenture_program_scaled_costs
	}
	if = {
		limit = { has_variable = sb_natal_indian_origin_population_var }
		add_modifier = {
			name = sb_natal_indenture_program_scaled_costs
			# f(x) = x^2, where x is Indian-origin population / 100,000.
			multiplier = {
				value = var:sb_natal_indian_origin_population_var
				divide = 100000
				max = 1
				min = 0
				multiply = {
					value = var:sb_natal_indian_origin_population_var
					divide = 100000
					max = 1
					min = 0
				}
				if = {
					limit = { has_variable = sb_natal_indenture_intensified_var }
					multiply = 2
				}
			}
		}
	}
}

sb_natal_remove_indenture_program_costs = {
	if = {
		limit = { has_modifier = sb_natal_indenture_program_costs }
		remove_modifier = sb_natal_indenture_program_costs
	}
	if = {
		limit = { has_modifier = sb_natal_indenture_program_scaled_costs }
		remove_modifier = sb_natal_indenture_program_scaled_costs
	}
}

sb_natal_clear_indenture_cycle = {
	if = {
		limit = { has_variable = sb_natal_indenture_cycle_wait_var }
		remove_variable = sb_natal_indenture_cycle_wait_var
	}
	if = {
		limit = { has_variable = sb_natal_indenture_cycle_event_pending_var }
		remove_variable = sb_natal_indenture_cycle_event_pending_var
	}
}

sb_natal_start_indenture_cycle = {
	if = {
		limit = {
			has_journal_entry = je_sb_natal_indenture_program_v2
			sb_natal_indenture_program_is_valid = yes
			NOT = { has_variable = sb_natal_indenture_suspended_var }
			NOT = { has_variable = sb_natal_indenture_cycle_wait_var }
			NOT = { has_variable = sb_natal_indenture_cycle_event_pending_var }
		}
		# Expire the wait guard before even the shortest calendar month so the
		# scheduled event can resolve after a complete lease.
		set_variable = { name = sb_natal_indenture_cycle_wait_var days = 27 }
		set_variable = sb_natal_indenture_cycle_event_pending_var
		trigger_event = { id = sb_natal_interwar.002 months = 1 }
	}
}

sb_natal_process_indenture_cycle = {
	sb_natal_migrate_indenture_program_journal = yes
	if = {
		limit = {
			has_journal_entry = je_sb_natal_indenture_program_v2
			sb_natal_indenture_program_is_valid = yes
			NOT = { has_variable = sb_natal_indenture_suspended_var }
			has_variable = sb_natal_indenture_cycle_event_pending_var
			NOT = { has_variable = sb_natal_indenture_cycle_wait_var }
		}
		remove_variable = sb_natal_indenture_cycle_event_pending_var
		sb_natal_recruit_next_indenture_cohort = yes
		sb_natal_refresh_indian_origin_population = yes
		sb_natal_start_indenture_cycle = yes
	}
}

sb_natal_suspend_indenture_program = {

		limit = {

		sb_natal_remove_indenture_program_costs = yes

}

sb_natal_resume_indenture_program = {
	if = {
		limit = {

sb_natal_establish_natal_indian_community = {
	# Culture objects are serialized when a campaign is created. The guard keeps
	# checkpoints from before Natal Indian was defined from reporting false success.
	if = {
		limit = { exists = cu:natal_indian }
		if = {
			limit = {
				has_variable = sb_natal_indian_ethnogenesis_event_pending_var
				sb_natal_bic_has_indenture_cohort = { CULTURE = cu:gujarati SIZE = 1500 }
			}
			sb_natal_transfer_indenture_cohort = {
				CULTURE = cu:gujarati
				SIZE = 1500
				DESTINATION_POP_TYPE = pop_type:shopkeepers
			}
		}
		if = {
			limit = {
				NOT = { has_variable = sb_natal_indian_community_established_var }
				NOT = {
					s:STATE_NATAL.region_state:NAL ?= {
						any_scope_pop = { culture = cu:natal_indian }
					}
				}
			}
			s:STATE_NATAL.region_state:NAL ?= {
				every_scope_pop = {
					limit = { sb_natal_pop_is_indian_origin = yes }
					change_pop_culture = {
						target = cu:natal_indian
						value = 0.25
					}
				}
			}
		}
		if = {
			limit = {
				s:STATE_NATAL.region_state:NAL ?= {
					any_scope_pop = { culture = cu:natal_indian }
				}
				NOT = { has_variable = sb_natal_indian_ethnogenesis_conversion_applied_var }
			}
			set_variable = sb_natal_indian_ethnogenesis_conversion_applied_var
		}
		if = {
			limit = {
				has_variable = sb_natal_indian_ethnogenesis_conversion_applied_var
				s:STATE_NATAL = { NOT = { is_homeland = cu:natal_indian } }
			}
			s:STATE_NATAL = { add_homeland = cu:natal_indian }
		}
		if = {
			limit = {
				has_variable = sb_natal_indian_ethnogenesis_conversion_applied_var
				s:STATE_NATAL = { is_homeland = cu:natal_indian }
				s:STATE_NATAL.region_state:NAL ?= {
					any_scope_pop = { culture = cu:natal_indian }
				}
			}
			set_variable = sb_natal_indian_community_established_var
			set_global_variable = sb_natal_indian_ethnogenesis_completed_global_var
			if = {
				limit = { has_variable = sb_natal_indian_ethnogenesis_event_pending_var }
				remove_variable = sb_natal_indian_ethnogenesis_event_pending_var
			}
		}
	}
	else = {
		# Older checkpoints cannot acquire a culture missing from their serialized
		# registry. Clear the former false-success markers and stop monthly retries.
		if = {
			limit = { has_global_variable = sb_natal_indian_ethnogenesis_completed_global_var }
			remove_global_variable = sb_natal_indian_ethnogenesis_completed_global_var
		}
		if = {
			limit = { has_variable = sb_natal_indian_ethnogenesis_conversion_applied_var }
			remove_variable = sb_natal_indian_ethnogenesis_conversion_applied_var
		}
		if = {
			limit = { has_variable = sb_natal_indian_community_established_var }
			remove_variable = sb_natal_indian_community_established_var
		}
		if = {
			limit = { has_variable = sb_natal_indian_ethnogenesis_event_pending_var }
			remove_variable = sb_natal_indian_ethnogenesis_event_pending_var
		}
	}
}

sb_natal_update_indenture_program = {
	sb_natal_migrate_indenture_program_journal = yes
	if = {
		limit = { has_journal_entry = je_sb_natal_indenture_program_v2 }
		if = {
			limit = { NOT = { has_variable = sb_natal_indenture_cycle_index_var } }
			set_variable = { name = sb_natal_indenture_cycle_index_var value = 0 }
		}
		sb_natal_refresh_indian_origin_population = yes
		if = {
			limit = { sb_natal_indenture_program_is_valid = yes }
			if = {
				limit = {
					is_ai = yes
					NOT = { has_variable = sb_natal_indenture_suspended_var }
					scaled_debt >= 0.50
				}
				sb_natal_suspend_indenture_program = yes
			}
			else_if = {
				limit = {
					is_ai = yes
					has_variable = sb_natal_indenture_suspended_var
					bureaucracy >= 0
					net_fixed_income >= 0
					scaled_debt < 0.10
				}
				sb_natal_resume_indenture_program = yes
			}
			if = {
				limit = {
					is_ai = yes
					has_variable = sb_natal_indenture_intensified_var
				}
				# Baseline calibration: AI Natal recruits at the normal rate.
				remove_variable = sb_natal_indenture_intensified_var
				sb_natal_clear_indenture_cycle = yes
			}
			if = {
				limit = { NOT = { has_variable = sb_natal_indenture_suspended_var } }
				sb_natal_add_indenture_program_costs = yes
				sb_natal_start_indenture_cycle = yes
			}
			else = { sb_natal_remove_indenture_program_costs = yes }
		}
		else = {
			sb_natal_remove_indenture_program_costs = yes
			sb_natal_clear_indenture_cycle = yes
		}
	}
}

		change_variable = {
			name = sb_natal_zulu_aid_servicemen_ratio_var

		}
		clamp_variable = { name = sb_natal_zulu_aid_servicemen_ratio_var min = 0 max = 1 }
		every_scope_state = {
			every_scope_pop = {
				limit = {
					culture = cu:boer
					is_pop_type = soldiers
				}
				move_partial_pop = {
					state = scope:sb_natal_zulu_aid_servicemen_target
					population_ratio = { add = c:TRN.var:sb_natal_zulu_aid_servicemen_ratio_var }
				}
			}
		}
		remove_variable = sb_natal_zulu_aid_servicemen_total_var
		remove_variable = sb_natal_zulu_aid_servicemen_ratio_var
	}
