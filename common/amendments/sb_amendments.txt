amendment_sb_cape_qualified_franchise = {

	parent = law_cultural_exclusion

	allowed_laws = {
		law_cultural_exclusion
	}

		# Current Phase 1 target under Cultural Exclusion:
		# - Cape Creole and Griqua land in a clearly improved "second-class" bracket
		# - Khoisan/Bantu cultures improve materially without jumping straight to
		#   full acceptance before later reforms
		modifier = {
			country_acceptance_no_shared_heritage_trait_add = 10
		}

	possible = {
		country_definition = cd:CAP
		NOT = { has_variable = sb_cape_franchise_weakened_var }
	}

	would_sponsor = {
		is_interest_group_type = ig_intelligentsia
	}

	ai_will_revoke = {
		always = no
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 0.8
}

	}

	would_sponsor = {
		ROOT = {
			OR = {
				is_interest_group_type = ig_rural_folk

					scope:approval > 0

		cu:boer = {
			add_cultural_obsession = small_arms

		}
	}

	ai_will_revoke = {
		always = no
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment

amendment_sb_zulu_chiefdom_settlement = {

	parent = law_sb_imperial_administration

	allowed_laws = {
		law_sb_imperial_administration
	}

	possible = {
		country_definition = cd:NAL
		has_variable = sb_natal_zulu_chiefdom_settlement_var
	}

	would_sponsor = {
		is_interest_group_type = ig_armed_forces
	}

	on_activate = {
		set_variable = sb_natal_zulu_chiefdom_bureaucracy_guard_var
		set_variable = {
			name = sb_natal_zulu_chiefdom_dismantle_cooldown_var
			days = 1825
		}
		s:STATE_ZULULAND.region_state:NAL ?= {
			add_modifier = { name = sb_zulu_chiefdom_administration }
		}
	}

	on_deactivate = {
		if = {
			limit = { has_variable = sb_natal_zulu_chiefdom_bureaucracy_guard_var }
			remove_variable = sb_natal_zulu_chiefdom_bureaucracy_guard_var
		}
		if = {
			limit = { has_variable = sb_natal_zulu_chiefdom_dismantle_cooldown_var }
			remove_variable = sb_natal_zulu_chiefdom_dismantle_cooldown_var
		}
		s:STATE_ZULULAND.region_state:NAL ?= {
			if = {
				limit = { has_modifier = sb_zulu_chiefdom_administration }
				remove_modifier = sb_zulu_chiefdom_administration
			}
		}
	}

	ai_will_revoke = { always = no }

	can_repeal = { always = no }

	amendment_activism_multiplier = 0.5
}
