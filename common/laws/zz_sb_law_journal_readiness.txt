REPLACE:law_technocracy = {
	group = lawgroup_distribution_of_power

	icon = "gfx/interface/icons/law_icons/technocracy.dds"

	progressiveness = -50

	on_activate = {
		if = {
			limit = {
				has_law = law_type:law_elected_bureaucrats
			}
			deactivate_law = law_type:law_elected_bureaucrats
			activate_law = law_type:law_appointed_bureaucrats
		}
		scripted_effect_parties_disappearence = yes
		cleanup_voting_strength_modifiers = yes
		sb_add_technocracy_journal_when_ready = yes
	}

	modifier = {
		country_legitimacy_headofstate_add = 10
		country_legitimacy_govt_total_clout_add = 120
		country_legitimacy_govt_size_add = 2
		country_legitimacy_ideological_incoherence_mult = -0.1
		country_academics_pol_str_mult = 0.33
		country_engineers_pol_str_mult = 0.33
		country_officers_pol_str_mult = 0.33
		country_authority_add = 100
		country_max_companies_add = 1
		country_free_charters_add = 1
	}

	unlocking_technologies = {
		central_archives
	}

	can_enact = {
		trigger_if = {
			limit = {
				c:BIC ?= ROOT
				is_subject = yes
				NOT = { has_law = scope:law }
			}
			custom_tooltip = {
				text = BIC_not_able_to_change_distribution_of_power_tt
				has_variable = britain_granted_dop_permission
			}
		}
	}

	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_distribution_of_power_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}

	ai_impose_chance = {
		value = 0

		if = {
			limit = {
				has_law = law_type:law_technocracy
				"ai_ideological_opinion(scope:target_country)" >= 25 # Only want to do this on ideologically similar countries
				scope:target_country = { # Is their law different enough from what we want to impose?
					NOT = { has_law_or_variant = law_type:law_single_party_state }
				}
			}

			add = base_impose_law_weight
		}

		if = {
			limit = {
				scope:target_country = {
					c:BIC ?= ROOT
					OR = {
						has_journal_entry = je_british_dictate_plantations
						has_journal_entry = je_british_dictate_military
						has_journal_entry = je_british_dictate_universities
					}
				}
			}
			multiply = 0
		}
	}
}

REPLACE:law_single_party_state = {
	group = lawgroup_distribution_of_power

	icon = "gfx/interface/icons/law_icons/single_party_state.dds"

	progressiveness = 50

	can_enact = {
		custom_tooltip = {
			text = sps_government_party_tt
			any_interest_group = {
				is_in_government = yes
				has_party = yes
			}
		}
		any_interest_group = {
			is_in_government = yes
			party ?= {
				party_has_insurrectionary_member = no
			}
		}
		trigger_if = {
			limit = {
				c:BIC ?= ROOT
				is_subject = yes
				NOT = { has_law = scope:law }
			}
			custom_tooltip = {
				text = BIC_not_able_to_change_distribution_of_power_tt
				has_variable = britain_granted_dop_permission
			}
		}
	}

	unlocking_technologies = {
		political_agitation
	}

	modifier = {
		country_legitimacy_govt_total_votes_add = 100
		country_legitimacy_ideological_incoherence_mult = -0.25
		country_authority_add = 250
		country_voting_power_base_add = 50
		interest_group_in_government_attraction_mult = 0.20
		country_electoral_confidence_impact_mult = -1.0
	}

	on_activate = {
		custom_tooltip = {
			text = set_government_party_as_only_legal_party_tt
			if = {
				limit = {
					var:sps_dominant_party_ig_var ?= {
						has_party = yes
					}
				}
				set_only_legal_party_from_ig = var:sps_dominant_party_ig_var
			}
			else = {
				set_only_legal_party_from_ig = ruler.interest_group
			}
		}
		if = {
			limit = {
				has_variable = sps_dominant_party_ig_var
			}
			remove_variable = sps_dominant_party_ig_var
		}
		add_modifier = {
			name = modifier_single_party_consolidation
			days = 90
		}
		custom_tooltip = {
			text = sps_government_reshuffle_tt
			every_interest_group = {
				limit = {
					is_in_government = yes
					NOT = { #Removes all IGs from government that aren't in your party
						has_party = yes
					}
				}
				remove_ruling_interest_group = yes
			}
			every_interest_group = {
				limit = { #Adds all IGs that are in your party to the government
					is_in_government = no
					has_party = yes
				}
				add_ruling_interest_group = yes
			}
		}

		if = {
			limit = {
				has_law = law_type:law_monarchy
				has_variable = king_in_parliament_possible
				NOT = {
					has_journal_entry = je_king_in_parliament
				}
			}
			custom_tooltip = {
				text = king_in_parliament_tt
				sb_add_king_in_parliament_journal_when_ready = yes
			}
		}
		if = {
			limit = {
				game_date > 1836.1.1
				country_has_voting_franchise = yes
			}
			initialize_electoral_confidence = yes
		}
		call_election = {
			months = 3
		}
	}

	on_deactivate = {
		unset_only_legal_party = yes
		update_party_support = yes
		if = {
			limit = {
				has_variable = fatherland_front_var
			}
			remove_variable = fatherland_front_var
		}
	}

	ai_will_do = {
		OR = {
			AND = {
				exists = ruler
				ruler = {
					interest_group ?= {
						has_party = yes
					}
					law_stance = {
						law = law_type:law_single_party_state
						value > neutral
					}
				}
			}
			AND = {
				has_strategy = ai_strategy_consolidation_agenda
				has_governing_communist_or_socialist_party = yes
			}
		}
	}
	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_distribution_of_power_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			AND = { # modification of the standard trigger, only other single-party states countries can impose single-party states on their subjects
				has_law = scope:law
				can_impose_law_default = yes
			}
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}

		if = {
			limit = {
				owner ?= {
					has_journal_entry = je_zaibatsu
					has_variable = ai_zaibatsu_curtail
				}
			}
			add = 500
		}

	}

			has_monarchist_ideology = yes
		}
	}

				NOT = {
					has_variable = one_true_french_dynasty
				}
				has_dlc_feature = voice_of_the_people_content
				game_date > 1836.1.1
			}
			if = {
				limit = {
					has_variable = fra_divided_monarchists_freeze
				}
				remove_variable = fra_divided_monarchists_freeze
			}
			sb_add_divided_monarchists_journal_when_ready = yes

			if = {
				limit = {
					c:FRA ?= this
					NOT = {
						has_variable = one_true_french_dynasty
					}
					has_dlc_feature = voice_of_the_people_content

						ruler_is_bonapartist = yes
						ruler_is_orleanist = yes
						ruler_is_legitimist = yes
					}
				}
				fra_handle_republic_monarch_assignment = yes
			}
		}
		if = {
			limit = {
				has_dlc_feature = ip4_content
				c:SPA ?= this
				is_revolutionary = no
				is_secessionist = no
				game_date > 1836.1.1
			}
			add_modifier = modifier_regency
			custom_tooltip = {
				text = activate_provisional_government_tt
				sb_add_spain_provisional_government_journal_when_ready = yes
			}
		}
		if = {
			limit = {
				country_has_primary_culture = cu:vietnamese
			}
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_CAMBODIA
					}
				}
				random_scope_state = {
					limit = {
						state_region = s:STATE_CAMBODIA
					}
					STATE_CAMBODIA_state_name_assign = yes
				}
			}
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_LAOS
					}
				}
				random_scope_state = {
					limit = {
						state_region = s:STATE_LAOS
					}
					STATE_LAOS_state_name_assign = yes
				}
			}
		}
	}
	
	on_deactivate = {
		if = {
			limit = {
				AND = {
					has_law_or_variant = law_type:law_subjecthood
					NOR = {
						has_law = law_type:law_theocracy
						is_enacting_law = law_type:law_theocracy
					}
				}
			}
			custom_tooltip = because_subjecthood_is_active_tt
			if = {
				limit = {
					any_interest_group = {
						is_in_government = yes
						any_preferred_law = {
							is_citizenship_law = yes
							law_is_available = yes
							NOT = {
								type ?= law_type:law_subjecthood
							}
						}
					}
				}
				ordered_interest_group = {
					limit = {
						is_in_government = yes
						any_preferred_law = {
							is_citizenship_law = yes
							law_is_available = yes
							NOT = {
								type ?= law_type:law_subjecthood
							}
						}
					}
					order_by = ig_clout
					position = 0
					check_range_bounds = no
					ordered_preferred_law = {
						limit = {
							is_citizenship_law = yes
							law_is_available = yes
							NOT = {
								type ?= law_type:law_subjecthood
							}
						}
						position = 0
						check_range_bounds = no
						prev.owner = {
							activate_law = prev.type
						}
					}
				}
			}
			else = {
				activate_law = law_type:law_national_supremacy
			}
		}
		if = {
			limit = {
				NOR = {
					has_law = law_type:law_social_monarchy
					is_enacting_law = law_type:law_social_monarchy
				}
			}
			remove_regency_modifier = yes
		}
	}
	
	ai_enact_weight_modifier = { #Petitions

			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
	
	ai_impose_chance = {
		value = 0
		
		if = {
			limit = {
				has_law = law_type:law_monarchy

				scope:target_country = { # Is their law different enough from what we want to impose?
					NOT = { has_law_or_variant = law_type:law_theocracy }
				}

			if = {

					NOT = { has_strategy = ai_strategy_egalitarian_agenda }
				}
				multiply = 5
			}
			else = {
				multiply = 0.5 # AI is less likely to mess with governance principles unless dealing with council republics
			}
		}
		
		if = {
			limit = {
				scope:target_country = {
					c:BIC ?= this
					OR = {
						has_journal_entry = je_british_dictate_plantations
						has_journal_entry = je_british_dictate_military
						has_journal_entry = je_british_dictate_universities
					}
				}
			}
			multiply = 0
		}
	}
}
