REPLACE:state_trait_severe_malaria = {
	icon = "gfx/interface/icons/state_trait_icons/severe_malaria.dds"

	required_techs_for_colonization = { "malaria_prevention" }
	disabling_technologies = { "malaria_prevention" }

	modifier = {
		state_non_homeland_colony_growth_speed_mult = -0.95
		state_non_homeland_mortality_mult = 0.25
	}
}

# Vanilla provides only infrastructure; SB adds a small subsistence benefit for
# communities drawing food and exchange from the Zambezi river system.
REPLACE:state_trait_zambezi_river = {
	icon = "gfx/interface/icons/state_trait_icons/river.dds"

	modifier = {
		state_infrastructure_add = 15
		building_subsistence_output_mult = 0.1
	}
}

state_trait_sb_south_west_africa_administrative_barrier = {
	icon = "gfx/interface/icons/state_trait_icons/dry_climate.dds"

	required_techs_for_colonization = { "sb_namibian_coastal_access" }
	disabling_technologies = { "sb_namibian_coastal_access" }

	# Existing enclaves bypass the establishment gate, so block their frontier
	# colony growth until the scripted access technology is granted as well.
	modifier = {
		state_colony_growth_speed_mult = -100
	}
}

# Xhosa Resistance — Represents active Xhosa resistance to colonial encroachment
# in the Eastern Cape. This should hinder any colonizing power, not just CAP.
	# REMOVAL: The trait is stripped once XHO is gone, regardless of whether
	# the state ends up under CAP, ABY, or some
# unrelated outside power.
state_trait_sb_xhosa_resistance = {
	icon = "gfx/interface/icons/state_trait_icons/xhosa_resistance.dds"

	modifier = {
		state_colony_growth_speed_mult = -0.66
		state_infrastructure_mult = -0.05
		state_construction_mult = -0.05
		building_group_bg_agriculture_tax_mult = -0.05
		building_group_bg_ranching_tax_mult = -0.05
		building_mobilization_cost_mult = 0.05
	}
}
