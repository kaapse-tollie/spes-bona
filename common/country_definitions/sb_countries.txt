# Cape Colony — British-controlled colony at the southern tip of Africa.
# Historically governed from Cape Town, it was the main British possession in southern Africa from 1806.
# country_type = colonial because it is a subject of GBR at game start (not an independent recognized nation).
# tier = grand_principality keeps it above the Boer and Griqua republics while
# still below a fully sovereign kingdom-tier union.
# cultures = anglo_african: the local English-speaking colonial elite; Boer,
# Cape Creole, Griqua, and Xhosa pops are not primary cultures at start.
# Cross-ref: common/dynamic_country_names/sb_dynamic_names.txt overrides vanilla name to "Cape Colony" (priority 200).
