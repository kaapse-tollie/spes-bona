ai_strategy_sb_boer_frontier_republic = {
	icon = "gfx/interface/icons/ai_strategy_icons/colonial_expansion.dds"

	type = diplomatic

	undesirable_infamy_level = {
		value = 35
	}

	unacceptable_infamy_level = {
		value = 70
	}

	diplomatic_play_neutrality = {
		value = 5
	}

	diplomatic_play_boldness = {
		value = 45
	}

	recklessness = {
		value = 0.2
	}

	aggression = {
		value = 0.75

		if = {
			limit = {
				has_strategic_adjacency = scope:target_country
				scope:target_country = {
					any_scope_state = {
						has_claim_by = ROOT
					}
				}
			}
			add = 2.0
		}
	}

	building_group_weights = {
		bg_army = 1.25
	}

	secret_goal_weights = {
		protect = 0.5
		conquer = 2.0
	}

	secret_goal_scores = {
		conquer = {
			score = {
				if = {
					limit = {
						NOT = { scope:target_country = { is_subject_of = root } }
						has_strategic_adjacency = scope:target_country
						scope:target_country = {
							any_scope_state = {
								has_claim_by = ROOT
							}
						}
					}
					add = 300
				}
			}
		}
	}

	wargoal_weights = {
		conquer_state = 2.0
		return_state = 3.0
	}

	wargoal_scores = {
		conquer_state = {
			if = {
				limit = {
					scope:target_state = {
						has_claim_by = ROOT
					}
				}
				add = 500
			}
		}
		return_state = {
			if = {
				limit = {
					scope:target_state = {
						has_claim_by = ROOT
					}
				}
				add = 750
			}
		}
	}

	possible = {
		is_subject = no
		OR = {
			country_definition = cd:ORA
			country_definition = cd:TRN
			country_definition = cd:LYD
			country_definition = cd:ZPB
		}
		has_law_or_variant = law_type:law_frontier_colonization
	}

	weight = {
		value = 500
	}
}

	icon = "gfx/interface/icons/ai_strategy_icons/colonial_expansion.dds"

	type = political

	# Vanilla and major overhaul mods put institution investment priorities on
	# political strategies. Keep this separate from the diplomatic claim-presser
	# so the AI has an explicit, valid Colonial Affairs investment strategy. The
	# Boer tags are tiny and often have few competing institutions, so this needs
	# to be far stronger than vanilla's broad "modernizer" weights.

		institution_colonial_affairs = {
			value = 10000
		}
	}

	possible = {
		is_subject = no

			country_definition = cd:ORA
			country_definition = cd:TRN
			country_definition = cd:LYD
			country_definition = cd:ZPB
		}
		has_law_or_variant = law_type:law_frontier_colonization
	}

	weight = {
		value = 1000
	}
}

# Bounded support for the Portuguese colonial programme. It raises the desired
# fleet rather than creating ships, and pressures only Kongo-held North Angola.
ai_strategy_sb_portuguese_african_planning = {
	icon = "gfx/interface/icons/ai_strategy_icons/colonial_expansion.dds"

	type = diplomatic

	wanted_navy_size = { value = 6 }
	wanted_num_supply_ships = { value = 120 }

	building_group_weights = {
		bg_navy = 1.5
	}

	aggression = {
		value = 0
		if = {
			limit = { sb_pink_map_kongo_holds_northern_angola = yes }
			add = 1.0
		}
	}

	secret_goal_scores = {
		conquer = {
			trigger = { sb_pink_map_kongo_holds_northern_angola = yes }
			score = { value = 500 }
		}
	}

	possible = {
		is_ai = yes
		is_subject = no
		OR = {
			country_definition = cd:POR
			country_definition = cd:IBE
		}
		sb_pink_map_story_requires_naval_support = yes
	}

	weight = { value = 10000 }
}

# Defiance creates hostility only in the arbiter that actually rejected the
# Pink Map, and only until the Portuguese JE ends.
ai_strategy_sb_pink_map_defiance_hostility = {
	icon = "gfx/interface/icons/ai_strategy_icons/colonial_expansion.dds"

	type = diplomatic

	aggression = {
		value = 0
		if = {
			limit = {
				scope:target_country = {
					has_variable = sb_pink_map_defiance_active_var
					has_journal_entry = je_the_pink_map
				}
			}
			add = 25
		}
	}

	possible = {
		is_ai = yes
		has_variable = sb_pink_map_defiance_arbiter_var
		any_country = {
			has_variable = sb_pink_map_defiance_active_var
			has_journal_entry = je_the_pink_map
		}
	}

	weight = { value = 10000 }
}
