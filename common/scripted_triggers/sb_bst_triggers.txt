# Spes Bona - BST frontier and STA scripted triggers

sb_bst_holds_drakensberg = {
	any_scope_state = {
		state_region = s:STATE_DRAKENSBERG
	}
}

sb_bst_holds_vrystaat = {
	any_scope_state = {
		state_region = s:STATE_VRYSTAAT
	}
}

sb_bst_holds_frontier_land = {
	OR = {
		sb_bst_holds_drakensberg = yes
		sb_bst_holds_vrystaat = yes
	}
}

sb_ora_holds_vrystaat = {
	any_scope_state = {
		state_region = s:STATE_VRYSTAAT
	}
}

sb_bst_oranje_frontier_actor = {
	is_country_alive = yes
	OR = {
		country_has_primary_culture = cu:boer
		country_has_primary_culture = cu:anglo_african
	}
	NOT = { country_definition = cd:BST }
	sb_ora_holds_vrystaat = yes
}

sb_bst_any_oranje_frontier_actor_exists = {
	any_country = {
		sb_bst_oranje_frontier_actor = yes
	}
}

sb_bst_oranje_frontier_land_holder = {
	is_country_alive = yes
	OR = {
		country_has_primary_culture = cu:boer
		country_has_primary_culture = cu:anglo_african
	}
	NOT = { country_definition = cd:BST }
	sb_bst_holds_frontier_land = yes
}

sb_bst_any_oranje_frontier_land_holder_exists = {
	any_country = {
		sb_bst_oranje_frontier_land_holder = yes
	}
}

sb_bst_controls_only_drakensberg = {
	owns_entire_state_region = STATE_DRAKENSBERG
	NOT = {
		any_scope_state = {
			NOT = {
				state_region = s:STATE_DRAKENSBERG
			}
		}
	}
}

sb_bst_oranje_frontier_actor_can_be_raided = {
	sb_bst_oranje_frontier_actor = yes
	NOT = { has_war_with = c:BST }
	NOT = { is_diplomatic_play_participant_with = c:BST }
	NOT = { has_truce_with = c:BST }
}

sb_bst_frontier_raid_valid = {
	country_definition = cd:BST
	any_country = {
		sb_bst_oranje_frontier_actor_can_be_raided = yes
	}
}

sb_has_adjacent_british_protection_route = {
	any_scope_state = {
		any_neighbouring_state = {
			owner = {
				OR = {
					country_definition = cd:GBR
					is_subject_of = c:GBR
				}
			}
		}
	}
}

sb_cap_is_self_governing_colony = {
	OR = {
		is_subject_type = subject_type_dominion
		is_subject_type = subject_type_sb_dominion
		is_subject_type = subject_type_sb_responsible_colony
		is_subject_type = subject_type_sb_responsible_colony_monarchy
	}
}

sb_cap_has_bst_subject_for_disarmament = {
	country_definition = cd:CAP
	is_country_alive = yes
	sb_cap_is_self_governing_colony = yes
	c:BST ?= {
		is_country_alive = yes
		is_direct_subject_of = root
		is_subject_type = subject_type_puppet
	}
}
