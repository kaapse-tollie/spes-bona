# Contested Crown is an independent law, so native parent-family checks are insufficient.
sb_court_has_royal_law = {
 OR = {
  has_law_or_variant = law_type:law_monarchy
  sb_court_has_social_monarchy = yes
  has_law = law_type:law_sb_contested_crown
 }
}
sb_court_has_social_monarchy = {
 OR = { has_law_or_variant = law_type:law_social_monarchy has_law = law_type:law_sb_contested_social_crown }
}
sb_court_has_contested_crown = {
 OR = { has_law = law_type:law_sb_contested_crown has_law = law_type:law_sb_contested_social_crown }
}
# DP enabled normal claimant crises after the successful AI replay (28 October 2026).
sb_court_dynastic_enabled = { sb_institutions_profile_country = yes }
# Frozen backing decides the homeland; a tie retains Zululand for the incumbent.
sb_court_dynastic_rebels_stronger = {
 OR = {
  AND = {
   has_variable = sb_court_dynastic_back_claimant_var
   var:sb_court_dynastic_incumbent_weight_var >= var:sb_court_dynastic_rival_weight_var
  }
  AND = {
   NOT = { has_variable = sb_court_dynastic_back_claimant_var }
   var:sb_court_dynastic_rival_weight_var > var:sb_court_dynastic_incumbent_weight_var
  }
 }
}
sb_court_dynastic_conflict_free = {
 OR = { NOT = { has_variable = coup_event_active } has_variable = sb_court_dynastic_owns_coup_lock_var }
 NOT = { has_journal_entry = je_ip4_coup }
 NOT = { has_variable = sb_court_civil_war_var }
 NOT = { any_civil_war = { always = yes } }
}

}

  NOT = { var:sb_court_dynastic_nominee_scope = var:sb_court_dynastic_rival_scope }
 }
}
