# Spes Bona - Bechuanaland Corridor / Caprivi Crisis triggers
#
# Lifecycle predicates below are root-agnostic unless a country predicate says
# otherwise. The escalation sequence is exclusive:
# decision window -> queued crisis -> active war -> settlement/cleanup.
# No transition may start unless the corridor is open and unresolved.

sb_bechuanaland_has_no_british_security_treaty = {
	NOT = {
		any_scope_treaty = {
			binds = c:GBR
			any_scope_article = {
				OR = {
					has_type = alliance
					has_type = defensive_pact
				}
			}
		}
	}
}

sb_bechuanaland_swa_overlord_candidate = {
	is_country_alive = yes
	country_rank = rank_value:great_power
	NOT = { country_definition = cd:GBR }
	c:SWA ?= {
		is_country_alive = yes
		is_direct_subject_of = prev
	}
	sb_bechuanaland_has_no_british_security_treaty = yes
}

sb_bechuanaland_trn_controls_transvaal_core = {
	country_definition = cd:TRN
	is_country_alive = yes
	owns_entire_state_region = STATE_TRANSVAAL
	owns_entire_state_region = STATE_EAST_TRANSVAAL
	owns_entire_state_region = STATE_NORTHERN_TRANSVAAL
}

sb_bechuanaland_trn_is_sar_like = {
	country_definition = cd:TRN
	is_country_alive = yes
	OR = {
		AND = {
			NOT = { has_variable = sb_early_republic_var }
			OR = {
				has_variable = sb_martinus_presidential_union_active_var
				has_variable = sb_transvaal_unity_resolved_var
			}
		}
		sb_bechuanaland_trn_controls_transvaal_core = yes
	}
}

sb_bechuanaland_boer_actor = {
	NOT = { is_in_same_power_bloc = c:GBR }
	OR = {
		AND = {
			country_definition = cd:TRN
			is_country_alive = yes
			OR = {
				has_variable = sb_kimberley_trn_claimant_var
				AND = {
					has_variable = sb_oranje_federated_into_trn_var
					has_global_variable = sb_kimberley_boer_victory_global_var
				}
				sb_bechuanaland_trn_is_sar_like = yes
			}
		}
		AND = {
			country_definition = cd:ORA
			is_country_alive = yes
			has_variable = sb_kimberley_ora_claimant_var
		}
	}
}

sb_bechuanaland_primary_boer_actor_at_peace = {
	OR = {
		c:TRN ?= {
			sb_bechuanaland_boer_actor = yes
			is_at_war = no
		}
		AND = {
			NOT = { c:TRN ?= { sb_bechuanaland_boer_actor = yes } }
			c:ORA ?= {
				sb_bechuanaland_boer_actor = yes
				is_at_war = no
			}
		}
	}
}

sb_bechuanaland_is_primary_boer_actor = {
	sb_bechuanaland_boer_actor = yes
	OR = {
		country_definition = cd:TRN
		AND = {
			country_definition = cd:ORA
			NOT = { c:TRN ?= { sb_bechuanaland_boer_actor = yes } }
		}
	}
}

# Global terminal guard shared by decisions, queues, and delayed launch retries.
sb_bechuanaland_corridor_crisis_is_live = {
	has_global_variable = sb_bechuanaland_corridor_open_global_var
	NOT = { has_global_variable = sb_bechuanaland_corridor_resolved_global_var }
	NOT = { has_global_variable = sb_bechuanaland_british_victory_global_var }
	NOT = { has_global_variable = sb_bechuanaland_boer_swa_victory_global_var }
}

# Used by the escalation buttons before reserving a delayed decision window.
sb_bechuanaland_crisis_escalation_available = {
	sb_bechuanaland_corridor_crisis_is_live = yes
	NOT = {
		OR = {
			has_global_variable = sb_bechuanaland_crisis_decision_pending_global_var
			has_global_variable = sb_bechuanaland_unresolved_settlement_pending_global_var
			has_global_variable = sb_bechuanaland_crisis_escalation_pending_global_var
			has_global_variable = sb_bechuanaland_crisis_war_active_var
		}
	}
}

# Delayed Warren/Caprivi events must cancel when this reservation is no longer
# authoritative. Queuing a route consumes the reservation and sets escalation.
sb_bechuanaland_crisis_decision_window_is_valid = {
	sb_bechuanaland_corridor_crisis_is_live = yes
	has_global_variable = sb_bechuanaland_crisis_decision_pending_global_var
	NOT = { has_global_variable = sb_bechuanaland_crisis_escalation_pending_global_var }
	NOT = { has_global_variable = sb_bechuanaland_crisis_war_active_var }
}

sb_bechuanaland_warren_decision_window_is_valid = {
	sb_bechuanaland_crisis_decision_window_is_valid = yes
	has_global_variable = sb_bechuanaland_warren_decision_pending_global_var
	NOT = { has_global_variable = sb_bechuanaland_caprivi_decision_pending_global_var }
}

sb_bechuanaland_caprivi_decision_window_is_valid = {
	sb_bechuanaland_crisis_decision_window_is_valid = yes
	has_global_variable = sb_bechuanaland_caprivi_decision_pending_global_var
	NOT = { has_global_variable = sb_bechuanaland_warren_decision_pending_global_var }
	any_country = { sb_bechuanaland_swa_overlord_candidate = yes }
}

# Diplomatic-play scope. The final branch supports saves already running the
# pre-0.13.5 generic CAP-SGO annexation play; new launches use exact custom IDs.
sb_bechuanaland_is_corridor_diplomatic_play = {
	OR = {
		is_diplomatic_play_type = dp_sb_bechuanaland_direct_intervention_locked
		is_diplomatic_play_type = dp_sb_bechuanaland_proxy_intervention_locked
		is_diplomatic_play_type = dp_sb_bechuanaland_cap_sgo_return_locked
		is_diplomatic_play_type = dp_sb_bechuanaland_cap_dual_return_locked
		AND = {
			is_diplomatic_play_type = dp_annex_war
			initiator = c:CAP
			target = c:SGO
			has_global_variable = sb_bechuanaland_crisis_war_active_var
			c:CAP ?= { has_variable = sb_bechuanaland_british_war_side_var }
			c:SGO ?= { has_variable = sb_bechuanaland_boer_swa_war_side_var }
		}
	}
}

sb_bechuanaland_has_active_corridor_diplomatic_play = {
	any_country = {
		any_diplomatic_play = { sb_bechuanaland_is_corridor_diplomatic_play = yes }
	}
}

sb_bechuanaland_crisis_core_is_valid = {
	sb_bechuanaland_corridor_crisis_is_live = yes
	has_global_variable = sb_bechuanaland_crisis_swa_overlord_global_scope
	c:GBR ?= {
		is_country_alive = yes
	}
	c:CAP ?= {
		is_country_alive = yes
		is_subject_of = c:GBR
	}
	c:SWA ?= {
		is_country_alive = yes
	}
	global_var:sb_bechuanaland_crisis_swa_overlord_global_scope = {
		is_country_alive = yes
		c:SWA ?= { is_direct_subject_of = PREV }
	}
	OR = {
		NOT = { has_global_variable = sb_bechuanaland_crisis_boer_support_global_var }
		AND = {
			has_global_variable = sb_bechuanaland_boer_actor_global_scope
			global_var:sb_bechuanaland_boer_actor_global_scope = {
				is_country_alive = yes
				sb_bechuanaland_is_primary_boer_actor = yes
			}
		}
	}
}

sb_bechuanaland_crisis_launch_participants_ready = {
	sb_bechuanaland_crisis_core_is_valid = yes
	c:GBR ?= {
		is_at_war = no
		is_active_in_diplomatic_play = no
	}
	c:CAP ?= {
		is_at_war = no
		is_active_in_diplomatic_play = no
	}
	c:SWA ?= {
		is_at_war = no
		is_active_in_diplomatic_play = no
	}
	global_var:sb_bechuanaland_crisis_swa_overlord_global_scope = {
		is_at_war = no
		is_active_in_diplomatic_play = no
	}
	OR = {
		NOT = { has_global_variable = sb_bechuanaland_crisis_boer_support_global_var }
		NOT = {
			any_country = {
				has_variable = sb_bechuanaland_crisis_boer_network_var
				OR = {
					is_at_war = yes
					is_active_in_diplomatic_play = yes
				}
			}
		}
	}
	OR = {
		NOT = { has_global_variable = sb_bechuanaland_crisis_direct_route_global_var }
		NOT = {
			any_country = {
				has_variable = sb_bechuanaland_crisis_british_subject_target_var
				OR = {
					is_at_war = yes
					is_active_in_diplomatic_play = yes
				}
			}
		}
	}
}

		any_country = {
			is_country_alive = yes

				is_subject_type = subject_type_puppet

	NOT = { has_global_variable = sb_bechuanaland_corridor_open_global_var }
