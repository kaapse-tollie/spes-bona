# Spes Bona - Bechuanaland Corridor / Caprivi Crisis triggers
#
# Lifecycle predicates below are root-agnostic unless a country predicate says
# otherwise. The escalation sequence is exclusive:
# decision window -> queued crisis -> active war -> settlement/cleanup.
# No transition may start unless the corridor is open and unresolved.

sb_bechuanaland_has_no_british_security_treaty = {
	NOT = {
		any_scope_treaty = {
			binds = c:GBR
			any_scope_article = {
				OR = {
					has_type = alliance
					has_type = defensive_pact
				}
			}
		}
	}
}

sb_bechuanaland_swa_direct_overlord = {
	is_country_alive = yes
	NOT = { country_definition = cd:GBR }
	c:SWA ?= {
		is_country_alive = yes
		is_direct_subject_of = prev
	}
}

# A new sponsor must be a GP. A saved incumbent may retain sponsorship after
# rank loss while it still directly owns SWA and remains outside Britain's
# alliance/defensive-pact network.
sb_bechuanaland_swa_overlord_candidate = {
	sb_bechuanaland_swa_direct_overlord = yes
	country_rank = rank_value:great_power
	sb_bechuanaland_has_no_british_security_treaty = yes
}

sb_bechuanaland_swa_active_sponsor = {
	sb_bechuanaland_swa_direct_overlord = yes
	sb_bechuanaland_has_no_british_security_treaty = yes
	OR = {
		country_rank = rank_value:great_power
		AND = {
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_swa_overlord_scope }
			container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_swa_overlord_scope = { this = PREV }
		}
	}
}

sb_bechuanaland_cap_is_swa_overlord_subject = {

# Root-agnostic: either GBR directly owns SWA or its current direct overlord
# has an alliance/defensive-pact treaty with GBR.

	OR = {
		c:SWA ?= { is_country_alive = yes is_direct_subject_of = c:GBR }

			NOT = { sb_bechuanaland_has_no_british_security_treaty = yes }
		}
	}
}

# Diplomatic-play scope. The final branch supports saves already running the
# pre-0.13.5 generic CAP-SGO annexation play; new launches use exact custom IDs.

	OR = {
		is_diplomatic_play_type = dp_sb_bechuanaland_direct_intervention_locked
		is_diplomatic_play_type = dp_sb_bechuanaland_warren_intervention_locked
		is_diplomatic_play_type = dp_sb_bechuanaland_proxy_intervention_locked
		is_diplomatic_play_type = dp_sb_bechuanaland_proxy_namaqualand_locked
		is_diplomatic_play_type = dp_sb_bechuanaland_cap_sgo_return
		AND = {
			is_diplomatic_play_type = dp_annex_war
			initiator = c:CAP
			target = c:SGO
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_war_active_var }

			c:SGO ?= { sb_bechuanaland_is_boer_swa_war_side_member = yes }
		}
	}
}

			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_caprivi_route_var }

					container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_direct_route_var }

						is_direct_subject_of = c:GBR

					}
					c:SWA ?= { sb_bechuanaland_subject_can_be_transferred = yes }

						can_transfer_subject = c:CAP
					}
					c:GBR ?= { can_transfer_subject = c:SWA }
				}
			}
		}
		AND = {
			NOT = { container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_crisis_caprivi_route_var } }
			container:sb_bechuanaland_corridor_state = { has_variable = sb_bechuanaland_warren_defender_scope }
			container:sb_bechuanaland_corridor_state.var:sb_bechuanaland_warren_defender_scope = {
				is_country_alive = yes
				NOT = { is_subject_of = c:GBR }
