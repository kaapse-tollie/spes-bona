# Diplomatic-play scope. Merely being "involved" exposes interested countries
# that have not joined either side, so both the recipient and its player enemy
# must be committed participants in this specific play.
sb_frontier_play_has_committed_player_enemy = {
	any_scope_play_involved = {
		is_player = yes
		is_diplomatic_play_committed_participant = yes
		is_diplomatic_play_enemy_of = $RECIPIENT$
	}
}

# Diplomatic-play scope. Player-facing routing takes precedence and has no
# AI-History fallback when Player Challenge is disabled.
sb_frontier_play_artificial_assistance_enabled = {
	$RECIPIENT$ ?= {
		is_ai = yes
		is_country_alive = yes
	}
	any_scope_play_involved = {
		this = $RECIPIENT$
		is_diplomatic_play_committed_participant = yes
	}
	OR = {
		AND = {
			sb_frontier_play_has_committed_player_enemy = { RECIPIENT = $RECIPIENT$ }
			sb_frontier_player_challenge_enabled = yes
		}
		AND = {
			NOT = {
				sb_frontier_play_has_committed_player_enemy = { RECIPIENT = $RECIPIENT$ }
			}
			sb_frontier_ai_scripting_enabled = yes
		}
	}
}

# Country scope watchdog for permanent force floors and mobilization retries.
# It succeeds when at least one play in which ROOT is a committed participant
# independently qualifies under the two-rule matrix.
sb_frontier_artificial_assistance_enabled = {
	is_ai = yes
	any_diplomatic_play = {
		sb_frontier_play_artificial_assistance_enabled = { RECIPIENT = ROOT }
	}
}

# Country scope. Mzilikazi's Host is retained only for a player-controlled ORA.
# It is independent of both frontier-assistance rules and is withheld when ORA
# is AI-controlled.
sb_mtb_native_host_assistance_enabled = {
	country_definition = cd:MTB
	is_country_alive = yes
	is_ai = yes
	c:ORA ?= { is_country_alive = yes is_player = yes }
}

# Country scope for the secession coalition before its diplomatic play exists.
# NAL and GBR are the known opposing side, so the two assistance rules can be
# routed explicitly before a diplomatic-play scope is available.
sb_klip_river_coalition_player_challenge_enabled = {
	is_ai = yes
	OR = {
		c:NAL ?= { is_country_alive = yes is_player = yes }
		c:GBR ?= { is_country_alive = yes is_player = yes }
	}
	sb_frontier_player_challenge_enabled = yes
}

sb_klip_river_coalition_ai_history_enabled = {
	is_ai = yes
	c:NAL ?= { is_country_alive = yes is_ai = yes }
	c:GBR ?= { is_country_alive = yes is_ai = yes }
	sb_frontier_ai_scripting_enabled = yes
}

sb_klip_river_coalition_assistance_enabled = {
	OR = {
		sb_klip_river_coalition_player_challenge_enabled = yes
		sb_klip_river_coalition_ai_history_enabled = yes
	}
}

# Country scope for the punitive expedition before its diplomatic play exists.
sb_klip_river_punitive_assistance_enabled = {
	is_ai = yes
	OR = {
		AND = {
			OR = {
				c:NAL ?= { is_country_alive = yes is_player = yes }
				c:GBR ?= { is_country_alive = yes is_player = yes }
			}
			sb_frontier_player_challenge_enabled = yes
		}
		AND = {
			c:NAL ?= { is_country_alive = yes is_ai = yes }
			c:GBR ?= { is_country_alive = yes is_ai = yes }
			sb_frontier_ai_scripting_enabled = yes
		}
	}
}
