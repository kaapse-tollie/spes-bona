	country_definition = cd:GER
	has_journal_entry = je_sb_map_namibian_coast
	OR = {
		NOT = { has_variable = sb_nam_colonial_prestige_resolved_var }

	}
}

sb_nam_maps_offer_is_current = {
	sb_nam_german_survey_can_receive_maps = yes
	OR = {
		AND = {
			exists = scope:sb_nam_maps_survey
			je:je_sb_map_namibian_coast = scope:sb_nam_maps_survey
			scope:sb_nam_maps_survey = { var:sb_nam_maps_offer_var ?= 1 }
		}
		# Old queued events have only the country marker, cleared at expedition end.
		AND = {
			NOT = { exists = scope:sb_nam_maps_survey }
			has_variable = sb_nam_german_missionary_support_offered_var
			je:je_sb_map_namibian_coast = { NOT = { has_variable = sb_nam_maps_offer_var } }
		}
	}
}

sb_nam_can_pay_for_maps = {
	OR = {
		gold_reserves >= 100000
		AND = {
			is_ai = yes
			credit > 0
			sb_nam_maps_projected_principal <= { value = credit multiply = 0.30 }
		}
	}
}

# Country scope can be SWA or the British administration. Only Cape-held
# Walvis Bay can remain outside the administrator's territory.
sb_nam_controls_administration_territory = {
	owns_entire_state_region = STATE_HEREROLAND
	OR = {
		owns_entire_state_region = STATE_NAMAQUALAND
		AND = {
			p:x8031D0.state ?= { owner = c:CAP num_provinces = 1 }
			s:STATE_NAMAQUALAND = {
				NOT = {
					any_scope_state = {
						NOR = { owner = c:$ADMINISTRATOR$ this = p:x8031D0.state }
					}
				}
			}
		}
	}
}
