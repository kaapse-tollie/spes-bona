sb_nam_has_any_colonial_law = {
	OR = {
		has_law = law_type:law_frontier_colonization
		has_law = law_type:law_colonial_resettlement
		has_law = law_type:law_colonial_exploitation
	}
}

sb_nam_has_qualifying_overseas_colonial_law = {
	OR = {
		has_law = law_type:law_colonial_resettlement
		has_law = law_type:law_colonial_exploitation
	}
}

sb_nam_has_coastal_engineering_technologies = {
	has_technology_researched = floating_harbor
	has_technology_researched = civilizing_mission
}

sb_nam_has_physical_coastal_access = {
	any_scope_state = {
		owner = root
		num_provinces >= 1
		OR = {
			state_region = s:STATE_NORTHERN_CAPE
			state_region = s:STATE_CAPE_COLONY
			state_region = s:STATE_BOTSWANA
			state_region = s:STATE_SOUTH_ANGOLA
			state_region = s:STATE_NAMAQUALAND
			state_region = s:STATE_HEREROLAND
		}
	}
}

sb_nam_has_free_coastal_beachhead = {
	OR = {
		p:xDEAA3C.state.owner = { is_country_type = decentralized }
		p:xB04050.state.owner = { is_country_type = decentralized }
	}
}

sb_nam_has_available_expedition_commander = {
	any_scope_character = {
		is_busy = no
		is_ruler_of_own_country = no
		character_is_valid_for_events = yes
		OR = {
			has_role_of_type = general
			has_role_of_type = admiral
		}
		count >= 1
	}
}

sb_nam_controls_both_consolidation_ports = {
	p:xDEAA3C.state.owner = root
	p:xB04050.state.owner = root
}

sb_nam_has_british_southern_african_subject_admin = {
	OR = {
		c:CAP ?= {
			is_country_alive = yes
			is_subject = yes
			exists = top_overlord
			top_overlord = c:GBR
		}
		c:SAF ?= {
			is_country_alive = yes
			is_subject = yes
			exists = top_overlord
			top_overlord = c:GBR
		}
	}
}

sb_nam_can_run_consolidation_je = {
	sb_nam_controls_both_consolidation_ports = yes
	NOT = { is_country_type = decentralized }
	NOT = { country_definition = cd:SWA }
	NOT = { has_variable = sb_nam_consolidation_resolved_var }
	NOT = { capital = { region = sr:region_southern_africa } }
	NOT = {
		AND = {
			country_definition = cd:GBR
			sb_nam_has_british_southern_african_subject_admin = yes
		}
	}
	OR = {
		country_definition = cd:GBR
		sb_nam_has_any_colonial_law = yes
	}
}

sb_nam_can_delegate_british_namibia_to_saf = {
	country_definition = cd:GBR
	c:SAF ?= {
		is_country_alive = yes
		is_subject_of = root
	}
	OR = {
		any_scope_state = {
			state_region = s:STATE_NAMAQUALAND
			owner = root
			num_provinces >= 22
		}
		AND = {
			any_scope_state = {
				state_region = s:STATE_NAMAQUALAND
				owner = root
				num_provinces >= 21
			}
			c:SAF ?= {
				any_scope_state = {
					state_region = s:STATE_NAMAQUALAND
				}
			}
		}
	}
	OR = {
		any_scope_state = {
			state_region = s:STATE_HEREROLAND
			owner = root
			num_provinces >= 44
		}
		AND = {
			any_scope_state = {
				state_region = s:STATE_HEREROLAND
				owner = root
				num_provinces >= 43
			}
			c:SAF ?= {
				any_scope_state = {
					state_region = s:STATE_HEREROLAND
				}
			}
		}
	}
}
