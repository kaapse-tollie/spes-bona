# Political assent is frozen when a choice or exact war result creates the
# install receipt. Retries recheck only structural safety and transient idleness.
sb_martinus_union_install_structurally_valid = {
	country_definition = cd:TRN
	is_country_alive = yes
	is_subject = no
	OR = {
		AND = {
			has_variable = sb_martinus_union_install_source_story_var
			sb_martinus_durable_story_union_install_authority = yes
		}
		AND = {
			NOT = { has_variable = sb_martinus_union_install_source_story_var }
			sb_martinus_durable_pretorius_union_authority = yes
		}
	}
	OR = {
		NOT = { has_variable = sb_martinus_union_install_source_legal_var }
		sb_martinus_trn_has_east_transvaal_consolidation = yes
	}
	c:ORA ?= {
		country_definition = cd:ORA
		is_country_alive = yes
		OR = {
			is_subject = no
			AND = { is_direct_subject_of = c:TRN is_subject_type = subject_type_sb_boer_presidential_union }
		}
	}
}

sb_martinus_union_install_ready_now = {
	sb_martinus_union_install_structurally_valid = yes
	is_at_war = no
	is_active_in_diplomatic_play = no
	c:ORA ?= {
		OR = {
			AND = { is_at_war = no is_active_in_diplomatic_play = no }
			AND = { is_direct_subject_of = c:TRN is_subject_type = subject_type_sb_boer_presidential_union }
		}
	}
}
