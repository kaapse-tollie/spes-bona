# Shared trek / Transvaal ownership triggers.
#
# These triggers sit between the audited three-state Transvaal split and the
# higher-level JE/event logic so the content can ask for "west Transvaal core"
# or "Transvaal except the Swazi carve-out" without repeating province lists.

sb_can_use_frontier_colonization = {
	OR = {
		AND = {
			country_is_colonial_or_company = yes
			any_decentralized_country = {
				is_adjacent_to_country = root
			}
		}
		any_scope_state = {
			is_homeland_of_country_cultures = root
			any_neighbouring_state = {
				owner = {
					is_country_type = decentralized
				}
			}
		}
		AND = {
			country_has_primary_culture = cu:boer
			OR = {
				country_definition = cd:ORA
				country_definition = cd:TRN
				country_definition = cd:ZPB
				country_definition = cd:LYD
				country_definition = cd:NAL
				country_definition = cd:SGO
			}
		}
	}
}

sb_controls_west_transvaal_core = {
	owns_entire_state_region = STATE_TRANSVAAL
}

sb_has_west_transvaal_foothold = {
	OR = {
		p:xD0A010.state.owner = this
		p:x8BDC4D.state.owner = this
		p:xA22DC2.state.owner = this
		p:xD02190.state.owner = this
		p:x6C1786.state.owner = this
		p:x9A137B.state.owner = this
	}
}

sb_controls_east_transvaal_core = {
	owns_entire_state_region = STATE_EAST_TRANSVAAL
}

sb_no_east_transvaal_claim_holder = {
	NOT = {
		any_country = {
			has_claim = s:STATE_EAST_TRANSVAAL
		}
	}
}

sb_all_zululand_under_british_control = {
	s:STATE_ZULULAND = {
		any_scope_state = {
			count = all
			owner = {
				OR = {
					c:GBR ?= THIS
					is_subject_of = c:GBR
				}
			}
		}
	}
}

sb_east_transvaal_has_boer_settler_critical_mass = {
	any_scope_state = {
		state_region = s:STATE_EAST_TRANSVAAL
		culture_percent_state = {
			target = cu:boer
			value >= 0.15
		}
	}
}

sb_controls_transvaal_except_swazi = {
	sb_controls_west_transvaal_core = yes
	sb_controls_lydenburg_core = yes
	sb_controls_north_transvaal_core = yes
}
sb_transvaal_unity_readiness = {
	government_legitimacy >= 50
	ig:ig_intelligentsia ?= { is_in_government = no }
	ig:ig_trade_unions ?= { is_in_government = no }
	OR = {
		has_law = law_type:law_national_supremacy
		has_law = law_type:law_ethnostate
	}
	OR = {
		has_law_or_variant = law_type:law_state_religion
		has_law_or_variant = law_type:law_freedom_of_conscience
	}
	has_law = law_type:law_national_militia
	active_law:lawgroup_army_model ?= {
		has_amendment = amendment_type:amendment_sb_commando_corps
	}
	c:LYD ?= {
		is_country_alive = yes
		relations:root >= relations_threshold:amicable
	}
	root.gdp > {
		value = c:LYD.gdp
		multiply = 1.25
	}
}
