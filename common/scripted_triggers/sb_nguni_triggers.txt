# ROOT = CAP or ABY. A queued frontier war may only be consumed when the actual
# target and XHO can enter a new diplomatic play and no relevant truce blocks it.
sb_xhosa_frontier_war_target_is_launchable = {
	is_country_alive = yes
	is_at_war = no
	is_active_in_diplomatic_play = no
	c:XHO ?= {
		is_country_alive = yes
		is_at_war = no
		is_active_in_diplomatic_play = no
		NOT = { has_truce_with = ROOT }
		NOT = { has_truce_with = c:CAP }
		NOT = { has_truce_with = c:GBR }
	}
}

sb_nguni_conqueror_path_open = {
	sb_nguni_controls_core_eastern_sphere_land = yes
	OR = {
		AND = {
			country_definition = cd:ZUL
			NOT = { c:SWZ ?= { is_country_alive = yes } }
			NOT = { c:GZA ?= { is_country_alive = yes } }
		}
		AND = {
			country_definition = cd:SWZ
			NOT = { c:ZUL ?= { is_country_alive = yes } }
			NOT = { c:GZA ?= { is_country_alive = yes } }
		}
		AND = {
			country_definition = cd:GZA
			NOT = { c:ZUL ?= { is_country_alive = yes } }
			NOT = { c:SWZ ?= { is_country_alive = yes } }
		}
	}
}
