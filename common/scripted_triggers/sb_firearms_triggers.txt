# Countries in current Spes Bona scope that should pursue widespread access to firearms.
sb_zul_antiplayer_boosts_enabled = {
	# Disabled by default. Set this global variable to re-enable parked ZUL
	# anti-player / emergency balance boosts after the pop-rework balance settles.
	has_global_variable = sb_zul_antiplayer_boosts_enabled_var
}

sb_is_firearms_acquisition_country = {
	OR = {
		country_definition = cd:ZUL
		country_definition = cd:BST
		country_definition = cd:SWZ
		country_definition = cd:GZA
		country_definition = cd:NGN
	}
}

sb_is_zulu_dynasty_country = {
	OR = {
		country_definition = cd:ZUL
		AND = {
			country_definition = cd:NGN
			has_variable = sb_ngu_inherited_zul_content_var
		}
	}
}

sb_has_active_firearms_import_treaty = {
	any_scope_treaty = {
		binds = root
		any_scope_article = {
			has_type = goods_transfer
			input_goods = g:small_arms
			target_country = root
			source_country = {
				any_scope_state = {
					has_active_building = building_arms_industry
				}
			}
		}
	}
}

sb_has_qualifying_firearms_industry = {
	any_scope_building = {
		is_building_type = building_arms_industry
		occupancy >= 0.02
	}
}
