# Preserve Vanilla incorporation policy, with one narrow exception for AI Natal's
# scripted postwar settlement. The engine exposes no script effect that starts the
# ordinary incorporation interaction, so this lets the normal AI command do it.

sb_zululand_ai_should_incorporate = {
	state_region = s:STATE_ZULULAND
	owner = {
		country_definition = cd:NAL
		is_ai = yes
		OR = {
			has_variable = sb_zululand_incorporation_requested_var
			has_variable = sb_zululand_incorporation_started_var
		}
	}
}

REPLACE:ai_can_incorporate_state = {
	OR = {
		is_homeland_of_country_cultures = root
		"years_to_incorporate(root)" <= define:NAI|INCORPORATE_STATE_MAX_YEARS
		sb_zululand_ai_should_incorporate = yes
	}
}

REPLACE:ai_will_incorporate_state = {
	OR = {
		AND = {
			OR = {
				is_homeland_of_country_cultures = root
				"years_to_incorporate(root)" <= define:NAI|INCORPORATE_STATE_MAX_YEARS
			}
			state_population >= define:NAI|INCORPORATE_STATE_MIN_POPULATION
		}
		sb_zululand_ai_should_incorporate = yes
	}
}
