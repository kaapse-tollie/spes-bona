# Preserve the complete Vanilla 1.14.0 incorporation gates while giving the AI
# exact regional priorities selected for Natal, the Cape, and Oranje. The helper
# is state-scoped; root is the country evaluating incorporation.

sb_regional_ai_should_incorporate = {
	owner = root
	root = {
		is_ai = yes
	}
	OR = {
		AND = {
			root = { country_definition = cd:NAL }
			state_region = s:STATE_ZULULAND
		}
		AND = {
			root = { country_definition = cd:CAP }
			state_region = s:STATE_BECHUANALAND
		}
		AND = {
			root = { country_definition = cd:CAP }
			state_region = s:STATE_GRIQUALAND_WEST
		}
		AND = {
			root = { country_definition = cd:ORA }
			state_region = s:STATE_DRAKENSBERG
		}
	}
}

REPLACE:ai_can_incorporate_state = {
	OR = {
		AND = {
			OR = {
				is_homeland_of_country_cultures = root
				"years_to_incorporate(root)" <= define:NAI|INCORPORATE_STATE_MAX_YEARS
			}
		}
		sb_regional_ai_should_incorporate = yes
	}
}

REPLACE:ai_will_incorporate_state = {
	OR = {
		AND = {
			OR = {
				is_homeland_of_country_cultures = root
				"years_to_incorporate(root)" <= define:NAI|INCORPORATE_STATE_MAX_YEARS
			}
			state_population >= define:NAI|INCORPORATE_STATE_MIN_POPULATION
		}
		sb_regional_ai_should_incorporate = yes
	}
}

REPLACE:ai_colony_will_incorporate_state = {
	OR = {
		AND = {
			OR = {
				region = root.capital.region
				any_neighbouring_state = {
					owner = root
					is_incorporated = yes
				}
			}
			state_population >= define:NAI|INCORPORATE_STATE_MIN_POPULATION
		}
		sb_regional_ai_should_incorporate = yes
	}
}
