# Country scope. Only the two authored levy laws require their existing firearms completion.
sb_can_reform_customary_army = {
	OR = {
		has_variable = sb_firearms_modernized_var
		NOR = {
			has_law = law_type:law_sb_amabutho_system
			has_law = law_type:law_sb_chiefly_levy
		}
	}
}

# Country scope; a surviving authored constituency prevents a duplicate generic absolutist.
sb_uses_authored_zulu_movements = {
	OR = {
		AND = { sb_institutions_profile_country = yes has_variable = sb_institutions_active_var }
		any_political_movement = { is_political_movement_type = movement_sb_zulu_traditionalist }
	}
}
