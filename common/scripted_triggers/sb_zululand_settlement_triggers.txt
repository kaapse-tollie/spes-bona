# ROOT = NRP. Revalidates the saved diplomatic initiator and every state scope
# needed by the exact boundary play before a held choice can mutate anything.
sb_nrp_boundary_current_launch_authority = {
	country_definition = cd:NRP
	is_country_alive = yes
	is_at_war = no
	is_active_in_diplomatic_play = no
	has_variable = sb_nrp_boundary_initiator_scope
	var:sb_nrp_boundary_initiator_scope ?= {
		is_country_alive = yes
		is_at_war = no
		is_active_in_diplomatic_play = no
		aggressive_diplomatic_plays_permitted = yes
		OR = { this = c:NAL this = c:GBR }
	}
	exists = s:STATE_ZULULAND.region_state:NRP
	trigger_if = {
		limit = { has_variable = sb_nrp_direct_annexation_rebellion_var }
		c:NAL ?= { is_country_alive = yes }
		exists = s:STATE_ZULULAND.region_state:NAL
	}
	trigger_else = {
		global_var:sb_zululand_live_zul_scope ?= { is_country_alive = yes }
		global_var:sb_zululand_live_zul_scope ?= { any_scope_state = { state_region = s:STATE_ZULULAND } }
	}
	trigger_if = {
		limit = { is_direct_subject_of = c:TRN is_subject_type = subject_type_sb_boer_confederal_partner }
		c:TRN ?= { is_country_alive = yes is_subject = no is_at_war = no is_active_in_diplomatic_play = no }
		var:sb_nrp_boundary_initiator_scope ?= { NOT = { has_truce_with = c:TRN } }
	}
	trigger_else = {
		is_subject = no
		var:sb_nrp_boundary_initiator_scope ?= { NOT = { has_truce_with = c:NRP } }
	}
}
