# Spes Bona - postwar Zululand settlement and Nieuwe Republiek triggers

sb_zululand_british_postwar_owner = {
	OR = {
		this = c:GBR
		AND = {
			is_ai = yes
			is_subject_of = c:GBR
		}
	}
}

sb_zululand_under_british_postwar_control = {
	country_definition = cd:GBR
	c:NAL ?= {
		is_country_alive = yes
		is_subject_of = c:GBR
		has_variable = sb_natalia_british_colony_resolved_var
	}
	OR = {
		c:ZUL ?= {
			is_country_alive = yes
			sb_zululand_british_postwar_owner = yes
			owns_entire_state_region = STATE_ZULULAND
		}
		AND = {
			NOT = { c:ZUL ?= { is_country_alive = yes } }
			p:xBE6FEE.state.owner = {
				sb_zululand_british_postwar_owner = yes
				owns_entire_state_region = STATE_ZULULAND
			}
		}
	}
	NOT = {
		c:ZUL ?= {
			is_country_alive = yes
			is_direct_subject_of = c:NAL
		}
	}
}

sb_zululand_chiefdoms_situation_active = {
	has_global_variable = sb_zululand_chiefdoms_open_global_var
	NOT = { has_global_variable = sb_zululand_chiefdoms_resolved_global_var }
	container_exists = sb_zululand_chiefdoms_state
}

		is_direct_subject_of = c:NAL

	}
}

sb_nrp_boundary_demand_can_open = {
	country_definition = cd:NRP
	is_country_alive = yes
	is_at_war = no
	is_active_in_diplomatic_play = no
	has_variable = sb_nrp_boundary_pending_var
	NOT = { has_variable = sb_nrp_boundary_not_before_var }
	NOT = { has_variable = sb_nrp_boundary_event_open_var }
	sb_nrp_is_greater_republic = yes
	OR = {
		AND = {
			NOT = { has_variable = sb_nrp_direct_annexation_rebellion_var }
			exists = s:STATE_ZULULAND.region_state:ZUL
			c:ZUL ?= {
				is_country_alive = yes
				is_at_war = no
				is_active_in_diplomatic_play = no
			}
		}
		AND = {
			has_variable = sb_nrp_direct_annexation_rebellion_var
			NOT = { c:ZUL ?= { is_country_alive = yes } }
			exists = s:STATE_ZULULAND.region_state:NAL
		}
	}
	OR = {
		c:NAL ?= {
			sb_zululand_nal_is_self_governing = yes
			is_active_in_diplomatic_play = no
			NOT = { has_truce_with = c:NRP }
		}
		AND = {
			c:NAL ?= {
				is_country_alive = yes
				NOT = { sb_zululand_nal_is_self_governing = yes }
			}
			c:GBR ?= {
				is_country_alive = yes
				is_active_in_diplomatic_play = no
				NOT = { has_truce_with = c:NRP }
			}
		}
	}
}
