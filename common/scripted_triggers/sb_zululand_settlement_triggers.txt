# Spes Bona - postwar Zululand settlement and Nieuwe Republiek triggers

sb_zululand_british_postwar_owner = {
	NOT = { country_definition = cd:NAL }
	OR = {
		this = c:GBR
		AND = {
			is_ai = yes
			is_subject_of = c:GBR
		}
	}
}

sb_zululand_under_british_postwar_control = {
	country_definition = cd:GBR
	c:NAL ?= {
		is_country_alive = yes
		is_subject_of = c:GBR
		has_variable = sb_natalia_british_colony_resolved_var
		NOT = { has_variable = sb_klip_river_secession_settlement_pending_var }
	}
	OR = {
		c:ZUL ?= {
			is_country_alive = yes
			OR = {
				AND = { sb_zululand_british_postwar_owner = yes owns_entire_state_region = STATE_ZULULAND }
				AND = { sb_hamu_republic_founded = yes is_subject_of = c:GBR any_scope_state = { state_region = s:STATE_ZULULAND } }
			}
		}
		AND = {
			NOT = { c:ZUL ?= { is_country_alive = yes } }
			OR = {
				AND = {
					sb_hamu_republic_founded = yes
					any_in_list = { variable = sb_hamu_residual_land state.owner ?= { sb_zululand_british_postwar_owner = yes NOT = { country_definition = cd:NRP } } }
				}
				AND = {
					sb_hamu_republic_founded = no
					p:xBE6FEE.state.owner = { sb_zululand_british_postwar_owner = yes owns_entire_state_region = STATE_ZULULAND }
				}
			}
		}
	}
	NOT = {
		c:ZUL ?= {
			is_country_alive = yes
			is_direct_subject_of = c:NAL
		}
	}
}

# An unrelated imperial war can delay a new play, but cannot prevent a boundary letter.
sb_nrp_at_peace_on_boundary = {
	NOR = { has_war_with = c:NRP is_diplomatic_play_enemy_of = c:NRP }
	trigger_if = {
		limit = { c:NRP ?= { sb_nrp_boundary_defended_by_trn = yes } }
		NOR = { has_war_with = c:TRN is_diplomatic_play_enemy_of = c:TRN }
	}
}

sb_nrp_can_issue_boundary_demand = {
	is_country_alive = yes
	sb_nrp_at_peace_on_boundary = yes
	NOT = { has_truce_with = c:NRP }
	trigger_if = {
		limit = { c:NRP ?= { sb_nrp_boundary_defended_by_trn = yes } }
		NOT = { has_truce_with = c:TRN }
	}
	OR = {
		sb_zululand_nal_is_self_governing = yes
		AND = {
			country_definition = cd:GBR
			c:NAL ?= {
				is_country_alive = yes
				is_subject_of = c:GBR
				sb_zululand_nal_is_self_governing = no
				sb_nrp_at_peace_on_boundary = yes
			}
		}
	}
}

# NRP scope. Revalidate the initiator and territory before launching the boundary play.
sb_nrp_boundary_current_launch_authority = {
	country_definition = cd:NRP
	is_country_alive = yes
	is_at_war = no
	is_active_in_diplomatic_play = no
	has_variable = sb_nrp_boundary_initiator_scope
	var:sb_nrp_boundary_initiator_scope ?= {
		is_country_alive = yes
		is_at_war = no
		is_active_in_diplomatic_play = no
		aggressive_diplomatic_plays_permitted = yes
		OR = { this = c:NAL this = c:GBR }
	}
	exists = s:STATE_ZULULAND.region_state:NRP
	trigger_if = {
		limit = { has_variable = sb_nrp_direct_annexation_rebellion_var }
		c:NAL ?= { is_country_alive = yes }
		exists = s:STATE_ZULULAND.region_state:NAL
	}
	trigger_else = {
		global_var:sb_zululand_live_zul_scope ?= { is_country_alive = yes }
		global_var:sb_zululand_live_zul_scope ?= { any_scope_state = { state_region = s:STATE_ZULULAND } }
	}
	trigger_if = {
		limit = { sb_nrp_boundary_defended_by_trn = yes }
		c:TRN ?= { is_country_alive = yes is_subject = no is_at_war = no is_active_in_diplomatic_play = no }
		var:sb_nrp_boundary_initiator_scope ?= { NOT = { has_truce_with = c:TRN } }
	}
	trigger_else = {
		is_subject = no
		var:sb_nrp_boundary_initiator_scope ?= { NOT = { has_truce_with = c:NRP } }
	}
}
