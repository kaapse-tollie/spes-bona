# Spes Bona - mineral discovery and Griqualand triggers

# Public Griqualand West content should speak through the Cape when it exists. If CAP is
# gone, Britain inherits the opening lane directly.
sb_cap_has_northern_cape_foothold = {
	c:CAP ?= {
		is_country_alive = yes
		any_scope_state = {
			state_region = s:STATE_NORTHERN_CAPE
			num_provinces >= 1
		}
	}
}

sb_can_discover_kimberley_diamonds = {
	NOT = { has_global_variable = sb_kimberley_discovered_global_var }
	NOT = { has_global_variable = sb_kimberley_discovery_pending_var }
	OR = {
		p:x68B5E8.state.owner = {
			has_technology_researched = mechanical_tools
		}
		c:CAP ?= {
			is_country_alive = yes
			has_technology_researched = dynamite
		}
		c:ORA ?= {
			is_country_alive = yes
			has_technology_researched = nitroglycerin
		}
		c:TRN ?= {
			is_country_alive = yes
			has_variable = sb_oranje_federated_into_trn_var
			has_technology_researched = dynamite
		}
	}
}

sb_cap_has_liberal_franchise_route = {
	OR = {
		has_variable = sb_cape_liberal_resolved_var
		has_modifier = sb_cape_liberal_victory
		has_modifier = sb_cape_frontier_compromise
		AND = {
			has_law = law_type:law_cultural_exclusion
			active_law:lawgroup_citizenship ?= {
				sb_has_cape_qualified_franchise_amendment = yes
			}
		}
		has_law_or_variant = law_type:law_multicultural
		has_law = law_type:law_sb_non_racialism
	}
}

sb_cap_is_full_liberal_cape_creole_route = {
	OR = {
		country_has_primary_culture = cu:cape_creole
		active_law:lawgroup_citizenship ?= {
			sb_has_cape_qualified_franchise_amendment = yes
		}
	}
}

sb_cap_is_responsible_or_dominion = {
	is_subject = yes
	OR = {
		is_subject_type = subject_type_dominion
		is_subject_type = subject_type_sb_dominion
		is_subject_type = subject_type_sb_responsible_colony
		is_subject_type = subject_type_sb_responsible_colony_monarchy
	}
}

sb_eastern_cape_frontier_is_unsettled = {
	c:XHO ?= {
		is_country_alive = yes
		any_scope_state = {
			state_region = s:STATE_EASTERN_CAPE
			num_provinces >= 1
		}
	}
}

sb_cap_can_form_gle = {
	country_definition = cd:CAP
	is_country_alive = yes
	OR = {
		NOT = { is_subject = yes }
		sb_cap_is_responsible_or_dominion = yes
	}
	NOT = { has_variable = sb_griqualand_east_resolved_var }
	NOT = { has_variable = sb_griqualand_east_pending_var }
	NOT = { exists = c:GLE }
	NOT = { exists = c:ABY }
	NOT = { c:PHL ?= { is_country_alive = yes } }
	sb_eastern_cape_frontier_is_unsettled = yes
	OR = {
		active_law:lawgroup_citizenship ?= {
			sb_has_cape_qualified_franchise_amendment = yes
		}
		country_has_primary_culture = cu:cape_creole
	}
}

sb_cap_can_integrate_gle = {
	country_definition = cd:CAP
	is_country_alive = yes
	sb_cap_is_full_liberal_cape_creole_route = yes
	NOT = {
		c:XHO ?= {
			any_scope_state = {
				state_region = s:STATE_EASTERN_CAPE
				num_provinces >= 1
			}
		}
	}
	c:GLE ?= {
		is_country_alive = yes
		is_subject_of = root
		liberty_desire <= 25
		relations:root >= 25
	}
}
