# Joining-rule country scope; both sides must match the recorded native conflict.
# The existing Chiefdoms pact must also pass DP's automatic-participation replay.
sb_chiefs_foreign_join_allowed = {
 trigger_if = {
  limit = {
   sb_zululand_chiefdoms_situation_active = yes
   sb_chiefs_native_mode = yes
   exists = container:sb_zululand_chiefdoms_state.var:sb_chiefs_origin_scope
   exists = container:sb_zululand_chiefdoms_state.var:sb_chiefs_rebels_scope
   exists = container:sb_zululand_chiefdoms_state.var:sb_chiefs_war_scope
   exists = scope:target_country
   exists = scope:enemy_country
  }
  NOT = {
   AND = {
    OR = {
     AND = {
      scope:target_country = container:sb_zululand_chiefdoms_state.var:sb_chiefs_origin_scope
      scope:enemy_country = container:sb_zululand_chiefdoms_state.var:sb_chiefs_rebels_scope
     }
     AND = {
      scope:target_country = container:sb_zululand_chiefdoms_state.var:sb_chiefs_rebels_scope
      scope:enemy_country = container:sb_zululand_chiefdoms_state.var:sb_chiefs_origin_scope
     }
    }
    NOT = {
     OR = {
      this = container:sb_zululand_chiefdoms_state.var:sb_chiefs_origin_scope
      this = container:sb_zululand_chiefdoms_state.var:sb_chiefs_rebels_scope
     }
    }
   }
  }
 }
 trigger_else = { always = yes }
}
