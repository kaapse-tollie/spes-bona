# War-goal holder scope. The holder and target stay the founding parties even
# when an intervening power takes leadership. A backdown need never create a war.
sb_court_bound_take_power_demand = {
	is_country_alive = yes
	exists = scope:diplomatic_play
	var:sb_court_reunion_play_scope ?= scope:diplomatic_play
	var:sb_court_reunion_origin_scope ?= scope:target_country
	OR = {
		var:sb_court_dynastic_opponent_scope ?= scope:target_country
		var:sb_court_civil_war_var ?= scope:target_country
	}
	scope:target_country ?= { country_definition = cd:ZUL is_country_alive = yes }
}
