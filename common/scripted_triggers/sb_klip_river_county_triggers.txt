# Exact pre-create checks. Receipts and launchers both call these because an option can be held.
# Natal is a British colony. These scripted plays bypass diplomatic status;
# ordinary permission to start plays must not gate the colonial actor.
sb_klip_river_secession_preflight = {
	country_definition = cd:NAL
	is_country_alive = yes
	is_at_war = no
	is_active_in_diplomatic_play = no
	NOT = { has_variable = sb_klip_river_secession_play_scope }
	p:x279045.state.owner = c:NAL
	c:KLR ?= {
		is_country_alive = yes
		is_at_war = no
		is_active_in_diplomatic_play = no
		p:xBBCA32.state.owner = this
		p:xDE0EDE.state.owner = this
		p:x552449.state.owner = this
		NOT = { has_war_with = c:NAL }
	}
	c:ZUL ?= {
		is_country_alive = yes
		is_subject = no
		aggressive_diplomatic_plays_permitted = yes
		is_at_war = no
		is_active_in_diplomatic_play = no
		NOT = { has_war_with = c:NAL }
		NOT = { has_war_with = c:KLR }
		NOT = { any_diplomatic_play = { initiator = c:ZUL target = c:NAL is_diplomatic_play_type = dp_sb_klip_river_secession } }
	}
}
