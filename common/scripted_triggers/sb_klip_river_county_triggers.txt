# A delivery receipt is a unique container object, never a timed country bit.
# The event's saved receipt scope must equal NAL's currently live generation.
sb_klip_river_secession_bound_delivery_receipt = {
	exists = scope:sb_klip_river_secession_receipt_scope
	country_definition = cd:NAL
	is_country_alive = yes
	has_variable = sb_klip_river_secession_delivery_pending_var
	has_variable = sb_klip_river_secession_generation_scope
	var:sb_klip_river_secession_generation_scope = scope:sb_klip_river_secession_receipt_scope
	container_exists = sb_klip_river_secession_generation_state
	scope:sb_klip_river_secession_receipt_scope ?= { has_variable = sb_klip_river_secession_generation_pending_var }
}

	has_variable = sb_klip_river_punitive_generation_scope

# Permanent invalidity is distinct from busy war/play occupancy. It cancels this
# exact receipt rather than silently rearming it against a later route.

	country_definition = cd:NAL
	OR = {
		NOT = { aggressive_diplomatic_plays_permitted = yes }
		NOT = { p:x279045.state.owner = root }
		NOT = { c:KLR ?= { is_country_alive = yes is_subject = no p:xBBCA32.state.owner = this p:xDE0EDE.state.owner = this p:x552449.state.owner = this } }
		NOT = { c:ZUL ?= { is_country_alive = yes is_subject = no aggressive_diplomatic_plays_permitted = yes } }
	}
}
