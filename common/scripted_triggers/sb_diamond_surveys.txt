# State scope, including when a foreign country proposes construction here.
sb_diamond_owner_has_survey_technology = {
	trigger_if = {
		limit = { state_region.var:sb_diamond_gate ?= flag:dynamite }
		owner = { has_technology_researched = dynamite }
	}
	trigger_else_if = {
		limit = { state_region.var:sb_diamond_gate ?= flag:pumpjacks }
		owner = { has_technology_researched = pumpjacks }
	}
	trigger_else_if = {
		limit = { state_region.var:sb_diamond_gate ?= flag:pneumatic_tools }
		owner = { has_technology_researched = pneumatic_tools }
	}
	trigger_else = { always = no }
}
